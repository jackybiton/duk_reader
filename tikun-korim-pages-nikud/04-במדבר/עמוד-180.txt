 1  וּבֵין הָאֱמֹרִי׃ עַל־כֵּן יֵאָמַר בְּסֵפֶר מִלְחֲמֹת יְהוָה
 2   אֶת־וָהֵב בְּסוּפָה וְאֶת־הַנְּחָלִים אַרְנוֹן׃ וְאֶשֶׁד הַנְּחָלִים
 3   אֲשֶׁר נָטָה לְשֶׁבֶת עָר וְנִשְׁעַן לִגְבוּל מוֹאָב׃ וּמִשָּׁם
 4   בְּאֵרָה הִוא הַבְּאֵר אֲשֶׁר אָמַר יְהוָה לְמֹשֶׁה אֱסֹף אֶת־
 5  הָעָם וְאֶתְּנָה לָהֶם מָיִם׃ [ס]             אָז יָשִׁיר
 6   יִשְׂרָאֵל אֶת־הַשִּׁירָה הַזֹּאת עֲלִי בְאֵר עֱנוּ־לָהּ׃ בְּאֵר
 7   חֲפָרוּהָ שָׂרִים כָּרוּהָ נְדִיבֵי הָעָם בִּמְחֹקֵק בְּמִשְׁעֲנֹתָם
 8   וּמִמִּדְבָּר מַתָּנָה׃ וּמִמַּתָּנָה נַחֲלִיאֵל וּמִנַּחֲלִיאֵל בָּמוֹת׃
 9   וּמִבָּמוֹת הַגַּיְא אֲשֶׁר בִּשְׂדֵה מוֹאָב רֹאשׁ הַפִּסְגָּה
10   וְנִשְׁקָפָה עַל־פְּנֵי הַיְשִׁימֹן׃ [פ]
11    וַיִּשְׁלַח יִשְׂרָאֵל מַלְאָכִים אֶל־סִיחֹן מֶלֶךְ־הָאֱמֹרִי
12   לֵאמֹר׃ אֶעְבְּרָה בְאַרְצֶךָ לֹא נִטֶּה בְּשָׂדֶה וּבְכֶרֶם לֹא
13   נִשְׁתֶּה מֵי בְאֵר בְּדֶרֶךְ הַמֶּלֶךְ נֵלֵךְ עַד אֲשֶׁר־נַעֲבֹר
14   גְּבֻלֶךָ׃ וְלֹא־נָתַן סִיחֹן אֶת־יִשְׂרָאֵל עֲבֹר בִּגְבֻלוֹ וַיֶּאֱסֹף
15   סִיחֹן אֶת־כָּל־עַמּוֹ וַיֵּצֵא לִקְרַאת יִשְׂרָאֵל הַמִּדְבָּרָה
16   וַיָּבֹא יָהְצָה וַיִּלָּחֶם בְּיִשְׂרָאֵל׃ וַיַּכֵּהוּ יִשְׂרָאֵל לְפִי־
17  חָרֶב וַיִּירַשׁ אֶת־אַרְצוֹ מֵאַרְנֹן עַד־יַבֹּק עַד־בְּנֵי עַמּוֹן
18   כִּי עַז גְּבוּל בְּנֵי עַמּוֹן׃ וַיִּקַּח יִשְׂרָאֵל אֵת כָּל־הֶעָרִים
19   הָאֵלֶּה וַיֵּשֶׁב יִשְׂרָאֵל בְּכָל־עָרֵי הָאֱמֹרִי בְּחֶשְׁבּוֹן
20   וּבְכָל־בְּנֹתֶיהָ׃ כִּי חֶשְׁבּוֹן עִיר סִיחֹן מֶלֶךְ הָאֱמֹרִי
21   הִוא וְהוּא נִלְחַם בְּמֶלֶךְ מוֹאָב הָרִאשׁוֹן וַיִּקַּח אֶת־
22  כָּל־אַרְצוֹ מִיָּדוֹ עַד־אַרְנֹן׃ עַל־כֵּן יֹאמְרוּ הַמֹּשְׁלִים
23   בֹּאוּ חֶשְׁבּוֹן תִּבָּנֶה וְתִכּוֹנֵן עִיר סִיחוֹן׃ כִּי־אֵשׁ יָצְאָה
24   מֵחֶשְׁבּוֹן לֶהָבָה מִקִּרְיַת סִיחֹן אָכְלָה עָר מוֹאָב
25   בַּעֲלֵי בָּמוֹת אַרְנֹן׃ אוֹי־לְךָ מוֹאָב אָבַדְתָּ עַם־כְּמוֹשׁ
26   נָתַן בָּנָיו פְּלֵיטִם וּבְנֹתָיו בַּשְּׁבִית לְמֶלֶךְ אֱמֹרִי סִיחוֹן׃
27   וַנִּירָם אָבַד חֶשְׁבּוֹן עַד־דִּיבֹן וַנַּשִּׁים עַד־נֹפַח אֲשֶׁר
28   עַד־מֵידְבָא׃ וַיֵּשֶׁב יִשְׂרָאֵל בְּאֶרֶץ הָאֱמֹרִי׃ וַיִּשְׁלַח
29   מֹשֶׁה לְרַגֵּל אֶת־יַעְזֵר וַיִּלְכְּדוּ בְּנֹתֶיהָ וַיּוֹרֶשׁ אֶת־
30  הָאֱמֹרִי אֲשֶׁר־שָׁם׃ וַיִּפְנוּ וַיַּעֲלוּ דֶּרֶךְ הַבָּשָׁן וַיֵּצֵא
31   עוֹג מֶלֶךְ־הַבָּשָׁן לִקְרָאתָם הוּא וְכָל־עַמּוֹ לַמִּלְחָמָה
32   אֶדְרֶעִי׃  וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה אַל־תִּירָא אֹתוֹ כִּי
33   בְיָדְךָ נָתַתִּי אֹתוֹ וְאֶת־כָּל־עַמּוֹ וְאֶת־אַרְצוֹ וְעָשִׂיתָ
34   לּוֹ כַּאֲשֶׁר עָשִׂיתָ לְסִיחֹן מֶלֶךְ הָאֱמֹרִי אֲשֶׁר יוֹשֵׁב
35   בְּחֶשְׁבּוֹן׃ וַיַּכּוּ אֹתוֹ וְאֶת־בָּנָיו וְאֶת־כָּל־עַמּוֹ עַד־
36  בִּלְתִּי הִשְׁאִיר־לוֹ שָׂרִיד וַיִּירְשׁוּ אֶת־אַרְצוֹ׃ וַיִּסְעוּ
37   בְּנֵי יִשְׂרָאֵל וַיַּחֲנוּ בְּעַרְבוֹת מוֹאָב מֵעֵבֶר לְיַרְדֵּן
38   יְרֵחוֹ׃ [ס]            וַיַּרְא בָּלָק בֶּן־צִפּוֹר אֵת
39   כָּל־אֲשֶׁר־עָשָׂה יִשְׂרָאֵל לָאֱמֹרִי׃ וַיָּגָר מוֹאָב מִפְּנֵי
40   הָעָם מְאֹד כִּי רַב־הוּא וַיָּקָץ מוֹאָב מִפְּנֵי בְּנֵי יִשְׂרָאֵל׃
41   וַיֹּאמֶר מוֹאָב אֶל־זִקְנֵי מִדְיָן עַתָּה יְלַחֲכוּ הַקָּהָל
42   אֶת־כָּל־סְבִיבֹתֵינוּ כִּלְחֹךְ הַשּׁוֹר אֵת יֶרֶק הַשָּׂדֶה
