 1  וַיַּקְרִיבוּ הַנְּשִׂאִים אֵת חֲנֻכַּת הַמִּזְבֵּחַ בְּיוֹם
 2   הִמָּשַׁח אֹתוֹ וַיַּקְרִיבוּ הַנְּשִׂיאִם אֶת־קָרְבָּנָם לִפְנֵי
 3   הַמִּזְבֵּחַ׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה נָשִׂיא אֶחָד לַיּוֹם
 4   נָשִׂיא אֶחָד לַיּוֹם יַקְרִיבוּ אֶת־קָרְבָּנָם לַחֲנֻכַּת
 5   הַמִּזְבֵּחַ׃ [ס]             וַיְהִי
 6   הַמַּקְרִיב בַּיּוֹם הָרִאשׁוֹן אֶת־קָרְבָּנוֹ נַחְשׁוֹן בֶּן־
 7  עַמִּינָדָב לְמַטֵּה יְהוּדָה׃ וְקָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת
 8   שְׁלֹשִׁים וּמֵאָה מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים
 9   שֶׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה
10   בַשֶּׁמֶן לְמִנְחָה׃ כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה
11   קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר אַיִל אֶחָד כֶּבֶשׂ־אֶחָד
12   בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים אֶחָד לְחַטָּאת׃ וּלְזֶבַח
13   הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם חֲמִשָּׁה עַתּוּדִים
14   חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה קָרְבַּן נַחְשׁוֹן
15   בֶּן־עַמִּינָדָב׃ [פ]
16   בַּיּוֹם הַשֵּׁנִי הִקְרִיב נְתַנְאֵל בֶּן־צוּעָר נְשִׂיא יִשָּׂשכָר׃
17   הִקְרִב אֶת־קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים
18   וּמֵאָה מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל
19   בְּשֶׁקֶל הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה
20   בַשֶּׁמֶן לְמִנְחָה׃ כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה
21   קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר אַיִל אֶחָד כֶּבֶשׂ־אֶחָד
22   בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים אֶחָד לְחַטָּאת׃ וּלְזֶבַח
23   הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם חֲמִשָּׁה עַתֻּדִים
24   חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה קָרְבַּן נְתַנְאֵל
25   בֶּן־צוּעָר׃ [פ]
26   בַּיּוֹם הַשְּׁלִישִׁי נָשִׂיא לִבְנֵי זְבוּלֻן אֱלִיאָב בֶּן־
27  חֵלֹן׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים וּמֵאָה
28   מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל בְּשֶׁקֶל
29   הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה בַשֶּׁמֶן
30   לְמִנְחָה׃ כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה קְטֹרֶת׃
31   פַּר אֶחָד בֶּן־בָּקָר אַיִל אֶחָד כֶּבֶשׂ־אֶחָד בֶּן־
32  שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים אֶחָד לְחַטָּאת׃ וּלְזֶבַח
33   הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם חֲמִשָּׁה עַתֻּדִים
34   חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה קָרְבַּן אֱלִיאָב
35   בֶּן־חֵלֹן׃ [פ]
36   בַּיּוֹם הָרְבִיעִי נָשִׂיא לִבְנֵי רְאוּבֵן אֱלִיצוּר בֶּן־
37  שְׁדֵיאוּר׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים
38   וּמֵאָה מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל
39   בְּשֶׁקֶל הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה
40   בַשֶּׁמֶן לְמִנְחָה׃ כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה
41   קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר אַיִל אֶחָד כֶּבֶשׂ־אֶחָד
42   בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים אֶחָד לְחַטָּאת׃
