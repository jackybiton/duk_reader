חומש במדבר · עמוד 193 · 42 שורות
1וְאֶלְעָזָרהַכֹּהֵןוְכָל־נְשִׂיאֵיהָעֵדָהלִקְרָאתָםאֶל־
2מִחוּץלַמַּחֲנֶה׃וַיִּקְצֹףמֹשֶׁהעַלפְּקוּדֵיהֶחָיִלשָׂרֵי
3הָאֲלָפִיםוְשָׂרֵיהַמֵּאוֹתהַבָּאִיםמִצְּבָאהַמִּלְחָמָה׃
4וַיֹּאמֶראֲלֵיהֶםמֹשֶׁההַחִיִּיתֶםכָּל־נְקֵבָה׃הֵןהֵנָּההָיוּ
5לִבְנֵייִשְׂרָאֵלבִּדְבַרבִּלְעָםלִמְסָר־מַעַלבַּיהוָהעַל־
6דְּבַר־פְּעוֹרוַתְּהִיהַמַּגֵּפָהבַּעֲדַתיְהוָה׃וְעַתָּההִרְגוּ
7כָל־זָכָרבַּטָּףוְכָל־אִשָּׁהיֹדַעַתאִישׁלְמִשְׁכַּבזָכָר
8הֲרֹגוּ׃וְכֹלהַטַּףבַּנָּשִׁיםאֲשֶׁרלֹא־יָדְעוּמִשְׁכַּב
9זָכָרהַחֲיוּלָכֶם׃וְאַתֶּםחֲנוּמִחוּץלַמַּחֲנֶהשִׁבְעַת
10יָמִיםכֹּלהֹרֵגנֶפֶשׁוְכֹלנֹגֵעַבֶּחָלָלתִּתְחַטְּאוּבַּיּוֹם
11הַשְּׁלִישִׁיוּבַיּוֹםהַשְּׁבִיעִיאַתֶּםוּשְׁבִיכֶם׃וְכָל־בֶּגֶד
12וְכָל־כְּלִי־עוֹרוְכָל־מַעֲשֵׂהעִזִּיםוְכָל־כְּלִי־עֵץ
13תִּתְחַטָּאוּ׃וַיֹּאמֶר אֶלְעָזָר הַכֹּהֵן
14אֶל־אַנְשֵׁיהַצָּבָאהַבָּאִיםלַמִּלְחָמָהזֹאתחֻקַּתהַתּוֹרָה
15אֲשֶׁר־צִוָּהיְהוָהאֶת־מֹשֶׁה׃אַךְאֶת־הַזָּהָבוְאֶת־הַכָּסֶף
16אֶת־הַנְּחֹשֶׁתאֶת־הַבַּרְזֶלאֶת־הַבְּדִילוְאֶת־הָעֹפָרֶת׃
17כָּל־דָּבָראֲשֶׁר־יָבֹאבָאֵשׁתַּעֲבִירוּבָאֵשׁוְטָהֵראַךְ
18בְּמֵינִדָּהיִתְחַטָּאוְכֹלאֲשֶׁרלֹא־יָבֹאבָּאֵשׁתַּעֲבִירוּ
19בַמָּיִם׃וְכִבַּסְתֶּםבִּגְדֵיכֶםבַּיּוֹםהַשְּׁבִיעִיוּטְהַרְתֶּם
20וְאַחַר תָּבֹאוּ אֶל־הַמַּחֲנֶה׃וַיֹּאמֶר
21יְהוָהאֶל־מֹשֶׁהלֵּאמֹר׃שָׂאאֵתרֹאשׁמַלְקוֹחַ
22הַשְּׁבִיבָּאָדָםוּבַבְּהֵמָהאַתָּהוְאֶלְעָזָרהַכֹּהֵןוְרָאשֵׁי
23אֲבוֹתהָעֵדָה׃וְחָצִיתָאֶת־הַמַּלְקוֹחַבֵּיןתֹּפְשֵׂי
24הַמִּלְחָמָההַיֹּצְאִיםלַצָּבָאוּבֵיןכָּל־הָעֵדָה׃וַהֲרֵמֹתָ
25מֶכֶסלַיהוָהמֵאֵתאַנְשֵׁיהַמִּלְחָמָההַיֹּצְאִיםלַצָּבָא
26אֶחָדנֶפֶשׁמֵחֲמֵשׁהַמֵּאוֹתמִן־הָאָדָםוּמִן־הַבָּקָר
27וּמִן־הַחֲמֹרִיםוּמִן־הַצֹּאן׃מִמַּחֲצִיתָםתִּקָּחוּוְנָתַתָּה
28לְאֶלְעָזָרהַכֹּהֵןתְּרוּמַתיְהוָה׃וּמִמַּחֲצִתבְּנֵי־יִשְׂרָאֵל
29תִּקַּחאֶחָדאָחֻזמִן־הַחֲמִשִּׁיםמִן־הָאָדָםמִן־הַבָּקָר
30מִן־הַחֲמֹרִיםוּמִן־הַצֹּאןמִכָּל־הַבְּהֵמָהוְנָתַתָּהאֹתָם
31לַלְוִיִּםשֹׁמְרֵימִשְׁמֶרֶתמִשְׁכַּןיְהוָה׃וַיַּעַשׂמֹשֶׁה
32וְאֶלְעָזָרהַכֹּהֵןכַּאֲשֶׁרצִוָּהיְהוָהאֶת־מֹשֶׁה׃וַיְהִי
33הַמַּלְקוֹחַיֶתֶרהַבָּזאֲשֶׁרבָּזְזוּעַםהַצָּבָאצֹאןשֵׁשׁ־
34מֵאוֹתאֶלֶףוְשִׁבְעִיםאֶלֶףוַחֲמֵשֶׁתאֲלָפִים׃וּבָקָר
35שְׁנַיִםוְשִׁבְעִיםאָלֶף׃וַחֲמֹרִיםאֶחָדוְשִׁשִּׁיםאָלֶף׃
36וְנֶפֶשׁאָדָםמִן־הַנָּשִׁיםאֲשֶׁרלֹא־יָדְעוּמִשְׁכַּבזָכָר
37כָּל־נֶפֶשׁשְׁנַיִםוּשְׁלֹשִׁיםאָלֶף׃וַתְּהִיהַמֶּחֱצָהחֵלֶק
38הַיֹּצְאִיםבַּצָּבָאמִסְפַּרהַצֹּאןשְׁלֹשׁ־מֵאוֹתאֶלֶף
39וּשְׁלֹשִׁיםאֶלֶףוְשִׁבְעַתאֲלָפִיםוַחֲמֵשׁמֵאוֹת׃וַיְהִי
40הַמֶּכֶסלַיהוָהמִן־הַצֹּאןשֵׁשׁמֵאוֹתחָמֵשׁוְשִׁבְעִים׃
41וְהַבָּקָרשִׁשָּׁהוּשְׁלֹשִׁיםאָלֶףוּמִכְסָםלַיהוָהשְׁנַיִם
42וְשִׁבְעִים׃וַחֲמֹרִיםשְׁלֹשִׁיםאֶלֶףוַחֲמֵשׁמֵאוֹת
