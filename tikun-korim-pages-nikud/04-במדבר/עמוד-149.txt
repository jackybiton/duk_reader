 1  וַיִּקַּח מֹשֶׁה וְאַהֲרֹן אֵת הָאֲנָשִׁים הָאֵלֶּה אֲשֶׁר
 2   נִקְּבוּ בְּשֵׁמֹת׃ וְאֵת כָּל־הָעֵדָה הִקְהִילוּ בְּאֶחָד
 3   לַחֹדֶשׁ הַשֵּׁנִי וַיִּתְיַלְדוּ עַל־מִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
 4   בְּמִסְפַּר שֵׁמוֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה לְגֻלְגְּלֹתָם׃
 5   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה וַיִּפְקְדֵם בְּמִדְבַּר
 6   סִינָי׃♢ [ס]              וַיִּהְיוּ בְנֵי־רְאוּבֵן בְּכֹר
 7   יִשְׂרָאֵל תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם בְּמִסְפַּר
 8   שֵׁמוֹת לְגֻלְגְּלֹתָם כָּל־זָכָר מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה
 9   כֹּל יֹצֵא צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה רְאוּבֵן שִׁשָּׁה
10   וְאַרְבָּעִים אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ [פ]
11   לִבְנֵי שִׁמְעוֹן תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
12   פְּקֻדָיו בְּמִסְפַּר שֵׁמוֹת לְגֻלְגְּלֹתָם כָּל־זָכָר מִבֶּן
13   עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא צָבָא׃ פְּקֻדֵיהֶם
14   לְמַטֵּה שִׁמְעוֹן תִּשְׁעָה וַחֲמִשִּׁים אֶלֶף וּשְׁלֹשׁ
15   מֵאוֹת׃ [פ]
16   לִבְנֵי גָד תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
17   בְּמִסְפַּר שֵׁמוֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
18   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה גָד חֲמִשָּׁה וְאַרְבָּעִים אֶלֶף
19   וְשֵׁשׁ מֵאוֹת וַחֲמִשִּׁים׃ [פ]
20   לִבְנֵי יְהוּדָה תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
21   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
22   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה יְהוּדָה אַרְבָּעָה וְשִׁבְעִים
23   אֶלֶף וְשֵׁשׁ מֵאוֹת׃ [פ]
24   לִבְנֵי יִשָּׂשכָר תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
25   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
26   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה יִשָּׂשכָר אַרְבָּעָה וַחֲמִשִּׁים
27   אֶלֶף וְאַרְבַּע מֵאוֹת׃ [פ]
28   לִבְנֵי זְבוּלֻן תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
29   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
30   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה זְבוּלֻן שִׁבְעָה וַחֲמִשִּׁים
31   אֶלֶף וְאַרְבַּע מֵאוֹת׃ [פ]
32   לִבְנֵי יוֹסֵף לִבְנֵי אֶפְרַיִם תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית
33   אֲבֹתָם בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה
34   כֹּל יֹצֵא צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה אֶפְרָיִם אַרְבָּעִים
35   אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ [פ]
36   לִבְנֵי מְנַשֶּׁה תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
37   בְּמִסְפַּר שֵׁמוֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל
38   יֹצֵא צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה מְנַשֶּׁה שְׁנַיִם וּשְׁלֹשִׁים
39   אֶלֶף וּמָאתָיִם׃ [פ]
40   לִבְנֵי בִנְיָמִן תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
41   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
42   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה בִנְיָמִן חֲמִשָּׁה וּשְׁלֹשִׁים אֶלֶף
