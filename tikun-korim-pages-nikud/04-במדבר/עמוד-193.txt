 1  וְאֶלְעָזָר הַכֹּהֵן וְכָל־נְשִׂיאֵי הָעֵדָה לִקְרָאתָם אֶל־
 2  מִחוּץ לַמַּחֲנֶה׃ וַיִּקְצֹף מֹשֶׁה עַל פְּקוּדֵי הֶחָיִל שָׂרֵי
 3   הָאֲלָפִים וְשָׂרֵי הַמֵּאוֹת הַבָּאִים מִצְּבָא הַמִּלְחָמָה׃
 4   וַיֹּאמֶר אֲלֵיהֶם מֹשֶׁה הַחִיִּיתֶם כָּל־נְקֵבָה׃ הֵן הֵנָּה הָיוּ
 5   לִבְנֵי יִשְׂרָאֵל בִּדְבַר בִּלְעָם לִמְסָר־מַעַל בַּיהוָה עַל־
 6  דְּבַר־פְּעוֹר וַתְּהִי הַמַּגֵּפָה בַּעֲדַת יְהוָה׃ וְעַתָּה הִרְגוּ
 7   כָל־זָכָר בַּטָּף וְכָל־אִשָּׁה יֹדַעַת אִישׁ לְמִשְׁכַּב זָכָר
 8   הֲרֹגוּ׃ וְכֹל הַטַּף בַּנָּשִׁים אֲשֶׁר לֹא־יָדְעוּ מִשְׁכַּב
 9   זָכָר הַחֲיוּ לָכֶם׃ וְאַתֶּם חֲנוּ מִחוּץ לַמַּחֲנֶה שִׁבְעַת
10   יָמִים כֹּל הֹרֵג נֶפֶשׁ וְכֹל  נֹגֵעַ בֶּחָלָל תִּתְחַטְּאוּ בַּיּוֹם
11   הַשְּׁלִישִׁי וּבַיּוֹם הַשְּׁבִיעִי אַתֶּם וּשְׁבִיכֶם׃ וְכָל־בֶּגֶד
12   וְכָל־כְּלִי־עוֹר וְכָל־מַעֲשֵׂה עִזִּים וְכָל־כְּלִי־עֵץ
13   תִּתְחַטָּאוּ׃ [ס]             וַיֹּאמֶר אֶלְעָזָר הַכֹּהֵן
14   אֶל־אַנְשֵׁי הַצָּבָא הַבָּאִים לַמִּלְחָמָה זֹאת חֻקַּת הַתּוֹרָה
15   אֲשֶׁר־צִוָּה יְהוָה אֶת־מֹשֶׁה׃ אַךְ אֶת־הַזָּהָב וְאֶת־הַכָּסֶף
16   אֶת־הַנְּחֹשֶׁת אֶת־הַבַּרְזֶל אֶת־הַבְּדִיל וְאֶת־הָעֹפָרֶת׃
17   כָּל־דָּבָר אֲשֶׁר־יָבֹא בָאֵשׁ תַּעֲבִירוּ בָאֵשׁ וְטָהֵר אַךְ
18   בְּמֵי נִדָּה יִתְחַטָּא וְכֹל אֲשֶׁר לֹא־יָבֹא בָּאֵשׁ תַּעֲבִירוּ
19   בַמָּיִם׃ וְכִבַּסְתֶּם בִּגְדֵיכֶם בַּיּוֹם הַשְּׁבִיעִי וּטְהַרְתֶּם
20   וְאַחַר תָּבֹאוּ אֶל־הַמַּחֲנֶה׃ [ס]              וַיֹּאמֶר
21   יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ שָׂא אֵת רֹאשׁ מַלְקוֹחַ
22   הַשְּׁבִי בָּאָדָם וּבַבְּהֵמָה אַתָּה וְאֶלְעָזָר הַכֹּהֵן וְרָאשֵׁי
23   אֲבוֹת הָעֵדָה׃ וְחָצִיתָ אֶת־הַמַּלְקוֹחַ בֵּין תֹּפְשֵׂי
24   הַמִּלְחָמָה הַיֹּצְאִים לַצָּבָא וּבֵין כָּל־הָעֵדָה׃ וַהֲרֵמֹתָ
25   מֶכֶס לַיהוָה מֵאֵת אַנְשֵׁי הַמִּלְחָמָה הַיֹּצְאִים לַצָּבָא
26   אֶחָד נֶפֶשׁ מֵחֲמֵשׁ הַמֵּאוֹת מִן־הָאָדָם וּמִן־הַבָּקָר
27   וּמִן־הַחֲמֹרִים וּמִן־הַצֹּאן׃ מִמַּחֲצִיתָם תִּקָּחוּ וְנָתַתָּה
28   לְאֶלְעָזָר הַכֹּהֵן תְּרוּמַת יְהוָה׃ וּמִמַּחֲצִת בְּנֵי־יִשְׂרָאֵל
29   תִּקַּח  אֶחָד  אָחֻז מִן־הַחֲמִשִּׁים מִן־הָאָדָם מִן־הַבָּקָר
30   מִן־הַחֲמֹרִים וּמִן־הַצֹּאן מִכָּל־הַבְּהֵמָה וְנָתַתָּה אֹתָם
31   לַלְוִיִּם שֹׁמְרֵי מִשְׁמֶרֶת מִשְׁכַּן יְהוָה׃ וַיַּעַשׂ מֹשֶׁה
32   וְאֶלְעָזָר הַכֹּהֵן כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃ וַיְהִי
33   הַמַּלְקוֹחַ יֶתֶר הַבָּז אֲשֶׁר בָּזְזוּ עַם הַצָּבָא צֹאן שֵׁשׁ־
34  מֵאוֹת אֶלֶף וְשִׁבְעִים אֶלֶף וַחֲמֵשֶׁת אֲלָפִים׃ וּבָקָר
35   שְׁנַיִם וְשִׁבְעִים אָלֶף׃ וַחֲמֹרִים אֶחָד וְשִׁשִּׁים אָלֶף׃
36   וְנֶפֶשׁ אָדָם מִן־הַנָּשִׁים אֲשֶׁר לֹא־יָדְעוּ מִשְׁכַּב זָכָר
37   כָּל־נֶפֶשׁ שְׁנַיִם וּשְׁלֹשִׁים אָלֶף׃ וַתְּהִי הַמֶּחֱצָה חֵלֶק
38   הַיֹּצְאִים בַּצָּבָא מִסְפַּר הַצֹּאן שְׁלֹשׁ־מֵאוֹת אֶלֶף
39   וּשְׁלֹשִׁים אֶלֶף וְשִׁבְעַת אֲלָפִים וַחֲמֵשׁ מֵאוֹת׃ וַיְהִי
40   הַמֶּכֶס לַיהוָה מִן־הַצֹּאן שֵׁשׁ מֵאוֹת חָמֵשׁ וְשִׁבְעִים׃
41   וְהַבָּקָר שִׁשָּׁה וּשְׁלֹשִׁים אָלֶף וּמִכְסָם לַיהוָה שְׁנַיִם
42   וְשִׁבְעִים׃ וַחֲמֹרִים שְׁלֹשִׁים אֶלֶף וַחֲמֵשׁ מֵאוֹת
