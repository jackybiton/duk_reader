 1  וְעָבַר עַצְמֹנָה׃ וְנָסַב הַגְּבוּל מֵעַצְמוֹן נַחְלָה מִצְרָיִם
 2   וְהָיוּ תוֹצְאֹתָיו הַיָּמָּה׃ וּגְבוּל יָם וְהָיָה לָכֶם הַיָּם
 3   הַגָּדוֹל וּגְבוּל זֶה־יִהְיֶה לָכֶם גְּבוּל יָם׃ וְזֶה־יִהְיֶה לָכֶם
 4   גְּבוּל צָפוֹן מִן־הַיָּם הַגָּדֹל תְּתָאוּ לָכֶם הֹר הָהָר׃
 5   מֵהֹר הָהָר תְּתָאוּ לְבֹא חֲמָת וְהָיוּ תּוֹצְאֹת הַגְּבֻל
 6   צְדָדָה׃ וְיָצָא הַגְּבֻל זִפְרֹנָה וְהָיוּ תוֹצְאֹתָיו חֲצַר עֵינָן
 7   זֶה־יִהְיֶה לָכֶם גְּבוּל צָפוֹן׃ וְהִתְאַוִּיתֶם לָכֶם לִגְבוּל
 8   קֵדְמָה מֵחֲצַר עֵינָן שְׁפָמָה׃ וְיָרַד הַגְּבֻל מִשְּׁפָם
 9   הָרִבְלָה מִקֶּדֶם לָעָיִן וְיָרַד הַגְּבֻל וּמָחָה עַל־כֶּתֶף
10   יָם־כִּנֶּרֶת קֵדְמָה׃ וְיָרַד הַגְּבוּל הַיַּרְדֵּנָה וְהָיוּ תוֹצְאֹתָיו
11   יָם הַמֶּלַח זֹאת תִּהְיֶה לָכֶם הָאָרֶץ לִגְבֻלֹתֶיהָ סָבִיב׃
12   וַיְצַו מֹשֶׁה אֶת־בְּנֵי יִשְׂרָאֵל לֵאמֹר זֹאת הָאָרֶץ
13   אֲשֶׁר תִּתְנַחֲלוּ אֹתָהּ בְּגוֹרָל אֲשֶׁר צִוָּה יְהוָה לָתֵת
14   לְתִשְׁעַת הַמַּטּוֹת וַחֲצִי הַמַּטֶּה׃ כִּי לָקְחוּ מַטֵּה בְנֵי
15   הָראוּבֵנִי לְבֵית אֲבֹתָם וּמַטֵּה בְנֵי־הַגָּדִי לְבֵית
16   אֲבֹתָם וַחֲצִי מַטֵּה מְנַשֶּׁה לָקְחוּ נַחֲלָתָם׃ שְׁנֵי
17   הַמַּטּוֹת וַחֲצִי הַמַּטֶּה לָקְחוּ נַחֲלָתָם מֵעֵבֶר לְיַרְדֵּן
18   יְרֵחוֹ קֵדְמָה מִזְרָחָה׃ [פ]
19    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ אֵלֶּה שְׁמוֹת הָאֲנָשִׁים
20   אֲשֶׁר־יִנְחֲלוּ לָכֶם אֶת־הָאָרֶץ אֶלְעָזָר הַכֹּהֵן וִיהוֹשֻׁעַ
21   בִּן־נוּן׃ וְנָשִׂיא אֶחָד נָשִׂיא אֶחָד מִמַּטֶּה תִּקְחוּ לִנְחֹל
22   אֶת־הָאָרֶץ׃ וְאֵלֶּה שְׁמוֹת הָאֲנָשִׁים לְמַטֵּה יְהוּדָה
23   כָּלֵב בֶּן־יְפֻנֶּה׃ וּלְמַטֵּה בְּנֵי שִׁמְעוֹן שְׁמוּאֵל בֶּן־
24  עַמִּיהוּד׃ לְמַטֵּה בִנְיָמִן אֱלִידָד בֶּן־כִּסְלוֹן׃ וּלְמַטֵּה
25   בְנֵי־דָן נָשִׂיא בֻּקִּי בֶּן־יָגְלִי׃ לִבְנֵי יוֹסֵף לְמַטֵּה בְנֵי־
26  מְנַשֶּׁה נָשִׂיא חַנִּיאֵל בֶּן־אֵפֹד׃ וּלְמַטֵּה בְנֵי־אֶפְרַיִם
27   נָשִׂיא קְמוּאֵל בֶּן־שִׁפְטָן׃ וּלְמַטֵּה בְנֵי־זְבוּלֻן נָשִׂיא
28   אֱלִיצָפָן בֶּן־פַּרְנָךְ׃ וּלְמַטֵּה בְנֵי־יִשָּׂשכָר נָשִׂיא
29   פַּלְטִיאֵל בֶּן־עַזָּן׃ וּלְמַטֵּה בְנֵי־אָשֵׁר נָשִׂיא אֲחִיהוּד
30   בֶּן־שְׁלֹמִי׃ וּלְמַטֵּה בְנֵי־נַפְתָּלִי נָשִׂיא פְּדַהְאֵל בֶּן־
31  עַמִּיהוּד׃ אֵלֶּה אֲשֶׁר צִוָּה יְהוָה לְנַחֵל אֶת־בְּנֵי־
32  יִשְׂרָאֵל בְּאֶרֶץ כְּנָעַן׃ [פ]
33    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה בְּעַרְבֹת מוֹאָב עַל־יַרְדֵּן
34   יְרֵחוֹ לֵאמֹר׃ צַו אֶת־בְּנֵי יִשְׂרָאֵל וְנָתְנוּ לַלְוִיִּם מִנַּחֲלַת
35   אֲחֻזָּתָם עָרִים לָשָׁבֶת וּמִגְרָשׁ לֶעָרִים סְבִיבֹתֵיהֶם
36   תִּתְּנוּ לַלְוִיִּם׃ וְהָיוּ הֶעָרִים לָהֶם לָשָׁבֶת וּמִגְרְשֵׁיהֶם
37   יִהְיוּ לִבְהֶמְתָּם וְלִרְכֻשָׁם וּלְכֹל חַיָּתָם׃ וּמִגְרְשֵׁי
38   הֶעָרִים אֲשֶׁר תִּתְּנוּ לַלְוִיִּם מִקִּיר הָעִיר וָחוּצָה אֶלֶף
39   אַמָּה סָבִיב׃ וּמַדֹּתֶם מִחוּץ לָעִיר אֶת־פְּאַת־קֵדְמָה
40   אַלְפַּיִם בָּאַמָּה וְאֶת־פְּאַת־נֶגֶב אַלְפַּיִם בָּאַמָּה וְאֶת־
41  פְּאַת־יָם  אַלְפַּיִם בָּאַמָּה וְאֵת פְּאַת צָפוֹן אַלְפַּיִם
42   בָּאַמָּה וְהָעִיר בַּתָּוֶךְ זֶה יִהְיֶה לָהֶם מִגְרְשֵׁי הֶעָרִים׃
