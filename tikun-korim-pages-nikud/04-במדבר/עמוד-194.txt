 1  וּמִכְסָם לַיהוָה אֶחָד וְשִׁשִּׁים׃ וְנֶפֶשׁ אָדָם שִׁשָּׁה
 2   עָשָׂר אָלֶף וּמִכְסָם לַיהוָה שְׁנַיִם וּשְׁלֹשִׁים נָפֶשׁ׃
 3   וַיִּתֵּן מֹשֶׁה אֶת־מֶכֶס תְּרוּמַת יְהוָה לְאֶלְעָזָר הַכֹּהֵן
 4   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃  וּמִמַּחֲצִית בְּנֵי יִשְׂרָאֵל
 5   אֲשֶׁר חָצָה מֹשֶׁה מִן־הָאֲנָשִׁים הַצֹּבְאִים׃ וַתְּהִי מֶחֱצַת
 6   הָעֵדָה מִן־הַצֹּאן שְׁלֹשׁ־מֵאוֹת אֶלֶף וּשְׁלֹשִׁים אֶלֶף
 7   שִׁבְעַת אֲלָפִים וַחֲמֵשׁ מֵאוֹת׃ וּבָקָר שִׁשָּׁה וּשְׁלֹשִׁים
 8   אָלֶף׃ וַחֲמֹרִים שְׁלֹשִׁים אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ וְנֶפֶשׁ
 9   אָדָם שִׁשָּׁה עָשָׂר אָלֶף׃ וַיִּקַּח מֹשֶׁה מִמַּחֲצִת בְּנֵי־
10  יִשְׂרָאֵל אֶת־הָאָחֻז אֶחָד מִן־הַחֲמִשִּׁים מִן־הָאָדָם
11   וּמִן־הַבְּהֵמָה וַיִּתֵּן אֹתָם לַלְוִיִּם שֹׁמְרֵי מִשְׁמֶרֶת
12   מִשְׁכַּן יְהוָה כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃ וַיִּקְרְבוּ
13   אֶל־מֹשֶׁה הַפְּקֻדִים אֲשֶׁר לְאַלְפֵי הַצָּבָא שָׂרֵי
14   הָאֲלָפִים וְשָׂרֵי הַמֵּאוֹת׃ וַיֹּאמְרוּ אֶל־מֹשֶׁה עֲבָדֶיךָ
15   נָשְׂאוּ אֶת־רֹאשׁ אַנְשֵׁי הַמִּלְחָמָה אֲשֶׁר בְּיָדֵנוּ וְלֹא־
16  נִפְקַד מִמֶּנּוּ אִישׁ׃ וַנַּקְרֵב אֶת־קָרְבַּן יְהוָה אִישׁ אֲשֶׁר
17   מָצָא כְלִי־זָהָב אֶצְעָדָה וְצָמִיד טַבַּעַת עָגִיל וְכוּמָז
18   לְכַפֵּר עַל־נַפְשֹׁתֵינוּ לִפְנֵי יְהוָה׃ וַיִּקַּח מֹשֶׁה וְאֶלְעָזָר
19   הַכֹּהֵן אֶת־הַזָּהָב מֵאִתָּם כֹּל כְּלִי מַעֲשֶׂה׃ וַיְהִי  כָּל־
20  זְהַב הַתְּרוּמָה אֲשֶׁר הֵרִימוּ לַיהוָה שִׁשָּׁה עָשָׂר
21   אֶלֶף שְׁבַע־מֵאוֹת וַחֲמִשִּׁים שָׁקֶל מֵאֵת שָׂרֵי
22   הָאֲלָפִים וּמֵאֵת שָׂרֵי הַמֵּאוֹת׃ אַנְשֵׁי הַצָּבָא בָּזְזוּ
23   אִישׁ לוֹ׃ וַיִּקַּח מֹשֶׁה וְאֶלְעָזָר הַכֹּהֵן אֶת־הַזָּהָב מֵאֵת
24   שָׂרֵי הָאֲלָפִים וְהַמֵּאוֹת וַיָּבִאוּ אֹתוֹ אֶל־אֹהֶל מוֹעֵד
25   זִכָּרוֹן לִבְנֵי־יִשְׂרָאֵל לִפְנֵי יְהוָה׃ [פ]
26    וּמִקְנֶה  רַב הָיָה לִבְנֵי רְאוּבֵן וְלִבְנֵי־גָד עָצוּם מְאֹד
27   וַיִּרְאוּ אֶת־אֶרֶץ יַעְזֵר וְאֶת־אֶרֶץ גִּלְעָד וְהִנֵּה הַמָּקוֹם
28   מְקוֹם מִקְנֶה׃ וַיָּבֹאוּ בְנֵי־גָד וּבְנֵי רְאוּבֵן וַיֹּאמְרוּ
29   אֶל־מֹשֶׁה וְאֶל־אֶלְעָזָר הַכֹּהֵן וְאֶל־נְשִׂיאֵי הָעֵדָה
30   לֵאמֹר׃ עֲטָרוֹת וְדִיבֹן וְיַעְזֵר וְנִמְרָה וְחֶשְׁבּוֹן
31   וְאֶלְעָלֵה וּשְׂבָם וּנְבוֹ וּבְעֹן׃ הָאָרֶץ אֲשֶׁר הִכָּה יְהוָה
32   לִפְנֵי עֲדַת יִשְׂרָאֵל אֶרֶץ מִקְנֶה הִוא וְלַעֲבָדֶיךָ
33   מִקְנֶה׃ [ס]             וַיֹּאמְרוּ אִם־
34  מָצָאנוּ חֵן בְּעֵינֶיךָ יֻתַּן אֶת־הָאָרֶץ הַזֹּאת לַעֲבָדֶיךָ
35   לַאֲחֻזָּה אַל־תַּעֲבִרֵנוּ אֶת־הַיַּרְדֵּן׃ וַיֹּאמֶר מֹשֶׁה לִבְנֵי־
36  גָד וְלִבְנֵי רְאוּבֵן הַאַחֵיכֶם יָבֹאוּ לַמִּלְחָמָה וְאַתֶּם
37   תֵּשְׁבוּ פֹה׃ וְלָמָּה תְנִיאוּן אֶת־לֵב בְּנֵי יִשְׂרָאֵל
38   מֵעֲבֹר אֶל־הָאָרֶץ אֲשֶׁר־נָתַן לָהֶם יְהוָה׃ כֹּה עָשׂוּ
39   אֲבֹתֵיכֶם בְּשָׁלְחִי אֹתָם מִקָּדֵשׁ בַּרְנֵעַ לִרְאוֹת אֶת־
40  הָאָרֶץ׃ וַיַּעֲלוּ עַד־נַחַל אֶשְׁכּוֹל וַיִּרְאוּ אֶת־הָאָרֶץ
41   וַיָּנִיאוּ אֶת־לֵב בְּנֵי יִשְׂרָאֵל לְבִלְתִּי־בֹא אֶל־הָאָרֶץ
42   אֲשֶׁר־נָתַן לָהֶם יְהוָה׃ וַיִּחַר־אַף יְהוָה בַּיּוֹם הַהוּא
