 1  וְאַתְּ כִּי שָׂטִית תַּחַת אִישֵׁךְ וְכִי נִטְמֵאת וַיִּתֵּן אִישׁ
 2   בָּךְ אֶת־שְׁכָבְתּוֹ מִבַּלְעֲדֵי אִישֵׁךְ׃ וְהִשְׁבִּיעַ הַכֹּהֵן
 3   אֶת־הָאִשָּׁה בִּשְׁבֻעַת הָאָלָה וְאָמַר הַכֹּהֵן לָאִשָּׁה
 4   יִתֵּן יְהוָה אוֹתָךְ לְאָלָה וְלִשְׁבֻעָה בְּתוֹךְ עַמֵּךְ בְּתֵת
 5   יְהוָה אֶת־יְרֵכֵךְ נֹפֶלֶת וְאֶת־בִּטְנֵךְ צָבָה׃ וּבָאוּ הַמַּיִם
 6   הַמְאָרְרִים הָאֵלֶּה בְּמֵעַיִךְ לַצְבּוֹת בֶּטֶן וְלַנְפִּל יָרֵךְ
 7   וְאָמְרָה הָאִשָּׁה אָמֵן  אָמֵן׃ וְכָתַב אֶת־הָאָלֹת הָאֵלֶּה
 8   הַכֹּהֵן בַּסֵּפֶר וּמָחָה אֶל־מֵי הַמָּרִים׃ וְהִשְׁקָה אֶת־
 9  הָאִשָּׁה אֶת־מֵי הַמָּרִים הַמְאָרְרִים וּבָאוּ בָהּ הַמַּיִם
10   הַמְאָרְרִים לְמָרִים׃ וְלָקַח הַכֹּהֵן מִיַּד הָאִשָּׁה אֵת
11   מִנְחַת הַקְּנָאֹת וְהֵנִיף אֶת־הַמִּנְחָה לִפְנֵי יְהוָה וְהִקְרִיב
12   אֹתָהּ אֶל־הַמִּזְבֵּחַ׃ וְקָמַץ הַכֹּהֵן מִן־הַמִּנְחָה אֶת־
13  אַזְכָּרָתָהּ וְהִקְטִיר הַמִּזְבֵּחָה וְאַחַר יַשְׁקֶה אֶת־הָאִשָּׁה
14   אֶת־הַמָּיִם׃ וְהִשְׁקָהּ אֶת־הַמַּיִם וְהָיְתָה אִם־נִטְמְאָה
15   וַתִּמְעֹל מַעַל בְּאִישָׁהּ וּבָאוּ בָהּ הַמַּיִם הַמְאָרְרִים
16   לְמָרִים וְצָבְתָה בִטְנָהּ וְנָפְלָה יְרֵכָהּ וְהָיְתָה הָאִשָּׁה
17   לְאָלָה בְּקֶרֶב עַמָּהּ׃ וְאִם־לֹא נִטְמְאָה הָאִשָּׁה וּטְהֹרָה
18   הִוא וְנִקְּתָה וְנִזְרְעָה זָרַע׃ זֹאת תּוֹרַת הַקְּנָאֹת אֲשֶׁר
19   תִּשְׂטֶה אִשָּׁה תַּחַת אִישָׁהּ וְנִטְמָאָה׃ אוֹ אִישׁ אֲשֶׁר
20   תַּעֲבֹר עָלָיו רוּחַ קִנְאָה וְקִנֵּא אֶת־אִשְׁתּוֹ וְהֶעֱמִיד
21   אֶת־הָאִשָּׁה לִפְנֵי יְהוָה וְעָשָׂה לָהּ הַכֹּהֵן אֵת כָּל־
22  הַתּוֹרָה הַזֹּאת׃ וְנִקָּה הָאִישׁ מֵעָו‍ֹן וְהָאִשָּׁה הַהִוא
23   תִּשָּׂא אֶת־עֲו‍ֹנָהּ׃ [פ]
24   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
25   וְאָמַרְתָּ אֲלֵהֶם אִישׁ אוֹ־אִשָּׁה כִּי יַפְלִא לִנְדֹּר
26   נֶדֶר נָזִיר לְהַזִּיר לַיהוָה׃ מִיַּיִן וְשֵׁכָר יַזִּיר חֹמֶץ יַיִן
27   וְחֹמֶץ שֵׁכָר לֹא יִשְׁתֶּה וְכָל־מִשְׁרַת עֲנָבִים לֹא
28   יִשְׁתֶּה וַעֲנָבִים לַחִים וִיבֵשִׁים לֹא יֹאכֵל׃ כֹּל יְמֵי
29   נִזְרוֹ מִכֹּל אֲשֶׁר יֵעָשֶׂה מִגֶּפֶן הַיַּיִן מֵחַרְצַנִּים וְעַד־
30  זָג לֹא יֹאכֵל׃ כָּל־יְמֵי נֶדֶר נִזְרוֹ תַּעַר לֹא־יַעֲבֹר
31   עַל־רֹאשׁוֹ עַד־מְלֹאת הַיָּמִם אֲשֶׁר־יַזִּיר לַיהוָה
32   קָדֹשׁ יִהְיֶה גַּדֵּל פֶּרַע שְׂעַר רֹאשׁוֹ׃ כָּל־יְמֵי הַזִּירוֹ
33   לַיהוָה עַל־נֶפֶשׁ מֵת לֹא יָבֹא׃ לְאָבִיו וּלְאִמּוֹ לְאָחִיו
34   וּלְאַחֹתוֹ לֹא־יִטַּמָּא לָהֶם בְּמֹתָם כִּי נֵזֶר אֱלֹהָיו עַל־
35  רֹאשׁוֹ׃ כֹּל יְמֵי נִזְרוֹ קָדֹשׁ הוּא לַיהוָה׃ וְכִי־יָמוּת
36   מֵת עָלָיו בְּפֶתַע פִּתְאֹם וְטִמֵּא רֹאשׁ נִזְרוֹ וְגִלַּח
37   רֹאשׁוֹ בְּיוֹם טָהֳרָתוֹ בַּיּוֹם הַשְּׁבִיעִי יְגַלְּחֶנּוּ׃ וּבַיּוֹם
38   הַשְּׁמִינִי יָבִא שְׁתֵּי תֹרִים אוֹ שְׁנֵי בְּנֵי יוֹנָה אֶל־
39  הַכֹּהֵן אֶל־פֶּתַח אֹהֶל מוֹעֵד׃ וְעָשָׂה הַכֹּהֵן אֶחָד
40   לְחַטָּאת וְאֶחָד לְעֹלָה וְכִפֶּר עָלָיו מֵאֲשֶׁר חָטָא
41   עַל־הַנָּפֶשׁ וְקִדַּשׁ אֶת־רֹאשׁוֹ בַּיּוֹם הַהוּא׃ וְהִזִּיר
42   לַיהוָה אֶת־יְמֵי נִזְרוֹ וְהֵבִיא כֶּבֶשׂ בֶּן־שְׁנָתוֹ לְאָשָׁם
