 1  וְאֶת־מַמְלֶכֶת עוֹג מֶלֶךְ הַבָּשָׁן הָאָרֶץ לְעָרֶיהָ
 2   בִּגְבֻלֹת עָרֵי הָאָרֶץ סָבִיב׃ וַיִּבְנוּ בְנֵי־גָד אֶת־דִּיבֹן
 3   וְאֶת־עֲטָרֹת וְאֵת עֲרֹעֵר׃ וְאֶת־עַטְרֹת שׁוֹפָן וְאֶת־
 4  יַעְזֵר וְיָגְבְּהָה׃ וְאֶת־בֵּית נִמְרָה וְאֶת־בֵּית הָרָן עָרֵי
 5   מִבְצָר וְגִדְרֹת צֹאן׃ וּבְנֵי רְאוּבֵן בָּנוּ אֶת־חֶשְׁבּוֹן
 6   וְאֶת־אֶלְעָלֵא וְאֵת קִרְיָתָיִם׃ וְאֶת־נְבוֹ וְאֶת־בַּעַל מְעוֹן
 7   מוּסַבֹּת שֵׁם וְאֶת־שִׂבְמָה וַיִּקְרְאוּ בְשֵׁמֹת אֶת־שְׁמוֹת
 8   הֶעָרִים אֲשֶׁר בָּנוּ׃ וַיֵּלְכוּ בְּנֵי מָכִיר בֶּן־מְנַשֶּׁה גִּלְעָדָה
 9   וַיִּלְכְּדֻהָ וַיּוֹרֶשׁ אֶת־הָאֱמֹרִי אֲשֶׁר־בָּהּ׃  וַיִּתֵּן מֹשֶׁה
10   אֶת־הַגִּלְעָד לְמָכִיר בֶּן־מְנַשֶּׁה וַיֵּשֶׁב בָּהּ׃ וְיָאִיר בֶּן־
11  מְנַשֶּׁה הָלַךְ וַיִּלְכֹּד אֶת־חַוּ͏ֹתֵיהֶם וַיִּקְרָא אֶתְהֶן
12   חַוּ͏ֹת יָאִיר׃ וְנֹבַח הָלַךְ וַיִּלְכֹּד אֶת־קְנָת וְאֶת־בְּנֹתֶיהָ
13   וַיִּקְרָא לָה נֹבַח בִּשְׁמוֹ׃ [פ]
14  אֵלֶּה מַסְעֵי בְנֵי־יִשְׂרָאֵל אֲשֶׁר יָצְאוּ מֵאֶרֶץ מִצְרַיִם
15   לְצִבְאֹתָם בְּיַד־מֹשֶׁה וְאַהֲרֹן׃ וַיִּכְתֹּב מֹשֶׁה אֶת־
16  מוֹצָאֵיהֶם לְמַסְעֵיהֶם עַל־פִּי יְהוָה וְאֵלֶּה מַסְעֵיהֶם
17   לְמוֹצָאֵיהֶם׃ וַיִּסְעוּ מֵרַעְמְסֵס בַּחֹדֶשׁ הָרִאשׁוֹן
18   בַּחֲמִשָּׁה עָשָׂר יוֹם לַחֹדֶשׁ הָרִאשׁוֹן מִמָּחֳרַת
19   הַפֶּסַח יָצְאוּ בְנֵי־יִשְׂרָאֵל בְּיָד רָמָה לְעֵינֵי כָּל־
20  מִצְרָיִם׃♢ וּמִצְרַיִם מְקַבְּרִים אֵת אֲשֶׁר הִכָּה יְהוָה
21   בָּהֶם כָּל־בְּכוֹר וּבֵאלֹהֵיהֶם עָשָׂה יְהוָה שְׁפָטִים׃
22   וַיִּסְעוּ בְנֵי־יִשְׂרָאֵל מֵרַעְמְסֵס וַיַּחֲנוּ בְּסֻכֹּת׃ וַיִּסְעוּ
23   מִסֻּכֹּת וַיַּחֲנוּ בְאֵתָם אֲשֶׁר בִּקְצֵה הַמִּדְבָּר׃♢ וַיִּסְעוּ
24   מֵאֵתָם וַיָּשָׁב עַל־פִּי הַחִירֹת אֲשֶׁר עַל־פְּנֵי בַּעַל
25   צְפוֹן וַיַּחֲנוּ לִפְנֵי מִגְדֹּל׃ וַיִּסְעוּ מִפְּנֵי הַחִירֹת וַיַּעַבְרוּ
26   בְתוֹךְ־הַיָּם הַמִּדְבָּרָה וַיֵּלְכוּ דֶּרֶךְ שְׁלֹשֶׁת יָמִים
27   בְּמִדְבַּר אֵתָם וַיַּחֲנוּ בְּמָרָה׃ וַיִּסְעוּ מִמָּרָה וַיָּבֹאוּ
28   אֵילִמָה וּבְאֵילִם שְׁתֵּים עֶשְׂרֵה עֵינֹת מַיִם וְשִׁבְעִים
29   תְּמָרִים וַיַּחֲנוּ־שָׁם׃ וַיִּסְעוּ מֵאֵילִם וַיַּחֲנוּ עַל־יַם־סוּף׃
30    וַיִּסְעוּ מִיַּם־סוּף וַיַּחֲנוּ בְּמִדְבַּר־סִין׃ וַיִּסְעוּ מִמִּדְבַּר־
31  סִין וַיַּחֲנוּ בְּדָפְקָה׃ וַיִּסְעוּ מִדָּפְקָה וַיַּחֲנוּ בְּאָלוּשׁ׃
32   וַיִּסְעוּ מֵאָלוּשׁ וַיַּחֲנוּ בִּרְפִידִם וְלֹא־הָיָה שָׁם מַיִם
33   לָעָם לִשְׁתּוֹת׃ וַיִּסְעוּ מֵרְפִידִם וַיַּחֲנוּ בְּמִדְבַּר סִינָי׃♢
34   וַיִּסְעוּ מִמִּדְבַּר סִינָי וַיַּחֲנוּ בְּקִבְרֹת הַתַּאֲוָה׃ וַיִּסְעוּ
35   מִקִּבְרֹת הַתַּאֲוָה וַיַּחֲנוּ בַּחֲצֵרֹת׃ וַיִּסְעוּ מֵחֲצֵרֹת וַיַּחֲנוּ
36   בְּרִתְמָה׃ וַיִּסְעוּ מֵרִתְמָה וַיַּחֲנוּ בְּרִמֹּן פָּרֶץ׃ וַיִּסְעוּ
37   מֵרִמֹּן פָּרֶץ וַיַּחֲנוּ בְּלִבְנָה׃ וַיִּסְעוּ מִלִּבְנָה וַיַּחֲנוּ בְּרִסָּה׃
38   וַיִּסְעוּ מֵרִסָּה וַיַּחֲנוּ בִּקְהֵלָתָה׃ וַיִּסְעוּ מִקְּהֵלָתָה וַיַּחֲנוּ
39   בְּהַר־שָׁפֶר׃ וַיִּסְעוּ מֵהַר־שָׁפֶר וַיַּחֲנוּ בַּחֲרָדָה׃ וַיִּסְעוּ
40   מֵחֲרָדָה וַיַּחֲנוּ בְּמַקְהֵלֹת׃ וַיִּסְעוּ מִמַּקְהֵלֹת וַיַּחֲנוּ
41   בְּתָחַת׃ וַיִּסְעוּ מִתָּחַת וַיַּחֲנוּ בְּתָרַח׃ וַיִּסְעוּ מִתָּרַח
42   וַיַּחֲנוּ בְּמִתְקָה׃ וַיִּסְעוּ מִמִּתְקָה וַיַּחֲנוּ בְּחַשְׁמֹנָה׃
