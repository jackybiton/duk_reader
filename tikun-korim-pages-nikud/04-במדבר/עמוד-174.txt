 1  וִידַעְתֶּם כִּי נִאֲצוּ הָאֲנָשִׁים הָאֵלֶּה אֶת־יְהוָה׃ וַיְהִי
 2   כְּכַלֹּתוֹ לְדַבֵּר אֵת כָּל־הַדְּבָרִים הָאֵלֶּה וַתִּבָּקַע
 3   הָאֲדָמָה אֲשֶׁר תַּחְתֵּיהֶם׃ וַתִּפְתַּח הָאָרֶץ אֶת־פִּיהָ
 4   וַתִּבְלַע אֹתָם וְאֶת־בָּתֵּיהֶם וְאֵת כָּל־הָאָדָם אֲשֶׁר
 5   לְקֹרַח וְאֵת כָּל־הָרְכוּשׁ׃ וַיֵּרְדוּ הֵם וְכָל־אֲשֶׁר
 6   לָהֶם חַיִּים שְׁאֹלָה וַתְּכַס עֲלֵיהֶם הָאָרֶץ וַיֹּאבְדוּ
 7   מִתּוֹךְ הַקָּהָל׃ וְכָל־יִשְׂרָאֵל אֲשֶׁר סְבִיבֹתֵיהֶם נָסוּ
 8   לְקֹלָם כִּי אָמְרוּ פֶּן־תִּבְלָעֵנוּ הָאָרֶץ׃ וְאֵשׁ יָצְאָה
 9   מֵאֵת יְהוָה וַתֹּאכַל אֵת הַחֲמִשִּׁים וּמָאתַיִם אִישׁ
10   מַקְרִיבֵי הַקְּטֹרֶת׃ [ס]             וַיְדַבֵּר
11   יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ אֱמֹר אֶל־אֶלְעָזָר בֶּן־אַהֲרֹן
12   הַכֹּהֵן וְיָרֵם אֶת־הַמַּחְתֹּת מִבֵּין הַשְּׂרֵפָה וְאֶת־הָאֵשׁ
13   זְרֵה־הָלְאָה כִּי קָדֵשׁוּ׃ אֵת מַחְתּוֹת הַחַטָּאִים הָאֵלֶּה
14   בְּנַפְשֹׁתָם וְעָשׂוּ אֹתָם רִקֻּעֵי פַחִים צִפּוּי לַמִּזְבֵּחַ כִּי־
15  הִקְרִיבֻם לִפְנֵי־יְהוָה וַיִּקְדָּשׁוּ וְיִהְיוּ לְאוֹת לִבְנֵי
16   יִשְׂרָאֵל׃ וַיִּקַּח אֶלְעָזָר הַכֹּהֵן אֵת מַחְתּוֹת הַנְּחֹשֶׁת
17   אֲשֶׁר הִקְרִיבוּ הַשְּׂרֻפִים וַיְרַקְּעוּם צִפּוּי לַמִּזְבֵּחַ׃
18   זִכָּרוֹן לִבְנֵי יִשְׂרָאֵל לְמַעַן אֲשֶׁר לֹא־יִקְרַב אִישׁ
19   זָר אֲשֶׁר לֹא מִזֶּרַע אַהֲרֹן הוּא לְהַקְטִיר קְטֹרֶת
20   לִפְנֵי יְהוָה וְלֹא־יִהְיֶה כְקֹרַח וְכַעֲדָתוֹ כַּאֲשֶׁר דִּבֶּר
21   יְהוָה בְּיַד־מֹשֶׁה לוֹ׃ [פ]
22   וַיִּלֹּנוּ כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל מִמָּחֳרָת עַל־מֹשֶׁה וְעַל־
23  אַהֲרֹן לֵאמֹר אַתֶּם הֲמִתֶּם אֶת־עַם יְהוָה׃ וַיְהִי בְּהִקָּהֵל
24   הָעֵדָה עַל־מֹשֶׁה וְעַל־אַהֲרֹן וַיִּפְנוּ אֶל־אֹהֶל מוֹעֵד
25   וְהִנֵּה כִסָּהוּ הֶעָנָן וַיֵּרָא כְּבוֹד יְהוָה׃ וַיָּבֹא מֹשֶׁה וְאַהֲרֹן
26   אֶל־פְּנֵי אֹהֶל מוֹעֵד׃ [ס]              וַיְדַבֵּר יְהוָה
27   אֶל־מֹשֶׁה לֵּאמֹר׃ הֵרֹמּוּ מִתּוֹךְ הָעֵדָה הַזֹּאת וַאֲכַלֶּה
28   אֹתָם כְּרָגַע וַיִּפְּלוּ עַל־פְּנֵיהֶם׃ וַיֹּאמֶר מֹשֶׁה אֶל־אַהֲרֹן
29   קַח אֶת־הַמַּחְתָּה וְתֶן־עָלֶיהָ אֵשׁ מֵעַל הַמִּזְבֵּחַ וְשִׂים
30   קְטֹרֶת וְהוֹלֵךְ מְהֵרָה אֶל־הָעֵדָה וְכַפֵּר עֲלֵיהֶם כִּי־
31  יָצָא הַקֶּצֶף מִלִּפְנֵי יְהוָה הֵחֵל הַנָּגֶף׃ וַיִּקַּח אַהֲרֹן
32   כַּאֲשֶׁר  דִּבֶּר מֹשֶׁה וַיָּרָץ אֶל־תּוֹךְ הַקָּהָל וְהִנֵּה הֵחֵל
33   הַנֶּגֶף בָּעָם וַיִּתֵּן אֶת־הַקְּטֹרֶת וַיְכַפֵּר עַל־הָעָם׃ וַיַּעֲמֹד
34   בֵּין־הַמֵּתִים וּבֵין הַחַיִּים וַתֵּעָצַר הַמַּגֵּפָה׃ וַיִּהְיוּ הַמֵּתִים
35   בַּמַּגֵּפָה אַרְבָּעָה עָשָׂר אֶלֶף וּשְׁבַע מֵאוֹת מִלְּבַד
36   הַמֵּתִים עַל־דְּבַר־קֹרַח׃ וַיָּשָׁב אַהֲרֹן אֶל־מֹשֶׁה אֶל־
37  פֶּתַח אֹהֶל מוֹעֵד וְהַמַּגֵּפָה נֶעֱצָרָה׃ [פ]
38    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר  אֶל־
39  בְּנֵי יִשְׂרָאֵל וְקַח מֵאִתָּם מַטֶּה מַטֶּה לְבֵית אָב מֵאֵת
40   כָּל־נְשִׂיאֵהֶם לְבֵית אֲבֹתָם שְׁנֵים עָשָׂר מַטּוֹת אִישׁ
41   אֶת־שְׁמוֹ תִּכְתֹּב עַל־מַטֵּהוּ׃ וְאֵת שֵׁם אַהֲרֹן תִּכְתֹּב
42   עַל־מַטֵּה לֵוִי כִּי מַטֶּה אֶחָד לְרֹאשׁ בֵּית אֲבוֹתָם׃
