 1  וּמָאתָיִם׃ [ס]             בְּנֵי גָד לְמִשְׁפְּחֹתָם
 2   לִצְפוֹן מִשְׁפַּחַת הַצְּפוֹנִי לְחַגִּי מִשְׁפַּחַת הַחַגִּי לְשׁוּנִי
 3   מִשְׁפַּחַת הַשּׁוּנִי׃ לְאָזְנִי מִשְׁפַּחַת הָאָזְנִי לְעֵרִי
 4   מִשְׁפַּחַת הָעֵרִי׃ לַאֲרוֹד מִשְׁפַּחַת הָאֲרוֹדִי לְאַרְאֵלִי
 5   מִשְׁפַּחַת הָאַרְאֵלִי׃ אֵלֶּה מִשְׁפְּחֹת בְּנֵי־גָד לִפְקֻדֵיהֶם
 6   אַרְבָּעִים אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ [ס]             בְּנֵי
 7   יְהוּדָה עֵר וְאוֹנָן וַיָּמָת עֵר וְאוֹנָן בְּאֶרֶץ כְּנָעַן׃ וַיִּהְיוּ
 8   בְנֵי־יְהוּדָה לְמִשְׁפְּחֹתָם לְשֵׁלָה מִשְׁפַּחַת הַשֵּׁלָנִי
 9   לְפֶרֶץ מִשְׁפַּחַת הַפַּרְצִי לְזֶרַח מִשְׁפַּחַת הַזַּרְחִי׃ וַיִּהְיוּ
10   בְנֵי־פֶרֶץ לְחֶצְרֹן מִשְׁפַּחַת הַחֶצְרֹנִי לְחָמוּל מִשְׁפַּחַת
11   הֶחָמוּלִי׃ אֵלֶּה מִשְׁפְּחֹת יְהוּדָה לִפְקֻדֵיהֶם שִׁשָּׁה
12   וְשִׁבְעִים אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ [ס]             בְּנֵי
13   יִשָּׂשכָר לְמִשְׁפְּחֹתָם תּוֹלָע מִשְׁפַּחַת הַתּוֹלָעִי
14   לְפֻוָה מִשְׁפַּחַת הַפּוּנִי׃ לְיָשׁוּב מִשְׁפַּחַת הַיָּשֻׁבִי
15   לְשִׁמְרֹן מִשְׁפַּחַת הַשִּׁמְרֹנִי׃ אֵלֶּה מִשְׁפְּחֹת
16   יִשָּׂשכָר לִפְקֻדֵיהֶם אַרְבָּעָה וְשִׁשִּׁים אֶלֶף וּשְׁלֹשׁ
17   מֵאוֹת׃ [ס]             בְּנֵי
18   זְבוּלֻן לְמִשְׁפְּחֹתָם לְסֶרֶד מִשְׁפַּחַת הַסַּרְדִּי לְאֵלוֹן
19   מִשְׁפַּחַת הָאֵלֹנִי לְיַחְלְאֵל מִשְׁפַּחַת הַיַּחְלְאֵלִי׃ אֵלֶּה
20   מִשְׁפְּחֹת הַזְּבוּלֹנִי לִפְקֻדֵיהֶם שִׁשִּׁים אֶלֶף וַחֲמֵשׁ
21   מֵאוֹת׃ [ס]             בְּנֵי יוֹסֵף לְמִשְׁפְּחֹתָם
22   מְנַשֶּׁה וְאֶפְרָיִם׃ בְּנֵי מְנַשֶּׁה לְמָכִיר מִשְׁפַּחַת הַמָּכִירִי
23   וּמָכִיר הוֹלִיד אֶת־גִּלְעָד לְגִלְעָד מִשְׁפַּחַת הַגִּלְעָדִי׃
24   אֵלֶּה בְּנֵי גִלְעָד אִיעֶזֶר מִשְׁפַּחַת הָאִיעֶזְרִי לְחֵלֶק
25   מִשְׁפַּחַת הַחֶלְקִי׃ וְאַשְׂרִיאֵל מִשְׁפַּחַת הָאַשְׂרִאֵלִי
26   וְשֶׁכֶם מִשְׁפַּחַת הַשִּׁכְמִי׃ וּשְׁמִידָע מִשְׁפַּחַת
27   הַשְּׁמִידָעִי וְחֵפֶר מִשְׁפַּחַת הַחֶפְרִי׃ וּצְלָפְחָד בֶּן־
28  חֵפֶר לֹא־הָיוּ לוֹ בָּנִים כִּי אִם־בָּנוֹת וְשֵׁם בְּנוֹת
29   צְלָפְחָד מַחְלָה וְנֹעָה חָגְלָה מִלְכָּה וְתִרְצָה׃ אֵלֶּה
30   מִשְׁפְּחֹת מְנַשֶּׁה וּפְקֻדֵיהֶם שְׁנַיִם וַחֲמִשִּׁים אֶלֶף
31   וּשְׁבַע מֵאוֹת׃ [ס]             אֵלֶּה בְנֵי־אֶפְרַיִם
32   לְמִשְׁפְּחֹתָם לְשׁוּתֶלַח מִשְׁפַּחַת הַשֻּׁתַלְחִי לְבֶכֶר
33   מִשְׁפַּחַת הַבַּכְרִי לְתַחַן מִשְׁפַּחַת הַתַּחֲנִי׃ וְאֵלֶּה בְּנֵי
34   שׁוּתָלַח לְעֵרָן מִשְׁפַּחַת הָעֵרָנִי׃ אֵלֶּה מִשְׁפְּחֹת בְּנֵי־
35  אֶפְרַיִם לִפְקֻדֵיהֶם שְׁנַיִם וּשְׁלֹשִׁים אֶלֶף וַחֲמֵשׁ מֵאוֹת
36   אֵלֶּה בְנֵי־יוֹסֵף לְמִשְׁפְּחֹתָם׃ [ס]             בְּנֵי
37   בִנְיָמִן לְמִשְׁפְּחֹתָם לְבֶלַע מִשְׁפַּחַת הַבַּלְעִי לְאַשְׁבֵּל
38   מִשְׁפַּחַת הָאַשְׁבֵּלִי לַאֲחִירָם מִשְׁפַּחַת הָאֲחִירָמִי׃
39   לִשְׁפוּפָם מִשְׁפַּחַת הַשּׁוּפָמִי לְחוּפָם מִשְׁפַּחַת
40   הַחוּפָמִי׃ וַיִּהְיוּ בְנֵי־בֶלַע אַרְדְּ וְנַעֲמָן מִשְׁפַּחַת
41   הָאַרְדִּי לְנַעֲמָן מִשְׁפַּחַת הַנַּעֲמִי׃ אֵלֶּה בְנֵי־בִנְיָמִן
42   לְמִשְׁפְּחֹתָם וּפְקֻדֵיהֶם חֲמִשָּׁה וְאַרְבָּעִים אֶלֶף
