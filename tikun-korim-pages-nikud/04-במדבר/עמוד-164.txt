 1  וְעָשָׂה פֶסַח לַיהוָה כְּחֻקַּת הַפֶּסַח וּכְמִשְׁפָּטוֹ
 2   כֵּן יַעֲשֶׂה חֻקָּה אַחַת יִהְיֶה לָכֶם וְלַגֵּר וּלְאֶזְרַח
 3   הָאָרֶץ׃ [ס]              וּבְיוֹם הָקִים אֶת־
 4  הַמִּשְׁכָּן כִּסָּה הֶעָנָן אֶת־הַמִּשְׁכָּן לְאֹהֶל הָעֵדֻת
 5   וּבָעֶרֶב יִהְיֶה עַל־הַמִּשְׁכָּן כְּמַרְאֵה־אֵשׁ עַד־בֹּקֶר׃
 6   כֵּן יִהְיֶה תָמִיד הֶעָנָן יְכַסֶּנּוּ וּמַרְאֵה־אֵשׁ לָיְלָה׃
 7   וּלְפִי הֵעָלוֹת הֶעָנָן מֵעַל הָאֹהֶל וְאַחֲרֵי־כֵן יִסְעוּ
 8   בְּנֵי יִשְׂרָאֵל וּבִמְקוֹם אֲשֶׁר יִשְׁכָּן־שָׁם הֶעָנָן שָׁם
 9   יַחֲנוּ בְּנֵי יִשְׂרָאֵל׃ עַל־פִּי יְהוָה יִסְעוּ בְּנֵי יִשְׂרָאֵל
10   וְעַל־פִּי יְהוָה יַחֲנוּ כָּל־יְמֵי אֲשֶׁר יִשְׁכֹּן הֶעָנָן עַל־
11  הַמִּשְׁכָּן יַחֲנוּ׃ וּבְהַאֲרִיךְ הֶעָנָן עַל־הַמִּשְׁכָּן יָמִים
12   רַבִּים וְשָׁמְרוּ בְנֵי־יִשְׂרָאֵל אֶת־מִשְׁמֶרֶת יְהוָה
13   וְלֹא יִסָּעוּ׃ וְיֵשׁ אֲשֶׁר יִהְיֶה הֶעָנָן יָמִים מִסְפָּר עַל־
14  הַמִּשְׁכָּן עַל־פִּי יְהוָה יַחֲנוּ וְעַל־פִּי יְהוָה יִסָּעוּ׃
15   וְיֵשׁ אֲשֶׁר־יִהְיֶה הֶעָנָן מֵעֶרֶב עַד־בֹּקֶר וְנַעֲלָה
16   הֶעָנָן בַּבֹּקֶר וְנָסָעוּ אוֹ יוֹמָם וָלַיְלָה וְנַעֲלָה הֶעָנָן
17   וְנָסָעוּ׃ אוֹ־יֹמַיִם אוֹ־חֹדֶשׁ אוֹ־יָמִים בְּהַאֲרִיךְ הֶעָנָן
18   עַל־הַמִּשְׁכָּן לִשְׁכֹּן עָלָיו יַחֲנוּ בְנֵי־יִשְׂרָאֵל וְלֹא
19   יִסָּעוּ וּבְהֵעָלֹתוֹ יִסָּעוּ׃ עַל־פִּי יְהוָה יַחֲנוּ וְעַל־פִּי
20   יְהוָה יִסָּעוּ אֶת־מִשְׁמֶרֶת יְהוָה שָׁמָרוּ עַל־פִּי יְהוָה
21   בְּיַד־מֹשֶׁה׃ [פ]
22   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ עֲשֵׂה לְךָ שְׁתֵּי
23   חֲצוֹצְרֹת כֶּסֶף מִקְשָׁה תַּעֲשֶׂה אֹתָם וְהָיוּ לְךָ
24   לְמִקְרָא הָעֵדָה וּלְמַסַּע אֶת־הַמַּחֲנוֹת׃ וְתָקְעוּ בָּהֵן
25   וְנוֹעֲדוּ אֵלֶיךָ כָּל־הָעֵדָה אֶל־פֶּתַח אֹהֶל מוֹעֵד׃ וְאִם־
26  בְּאַחַת יִתְקָעוּ וְנוֹעֲדוּ אֵלֶיךָ הַנְּשִׂיאִים רָאשֵׁי אַלְפֵי
27   יִשְׂרָאֵל׃ וּתְקַעְתֶּם תְּרוּעָה וְנָסְעוּ הַמַּחֲנוֹת הַחֹנִים
28   קֵדְמָה׃ וּתְקַעְתֶּם תְּרוּעָה שֵׁנִית וְנָסְעוּ הַמַּחֲנוֹת
29   הַחֹנִים תֵּימָנָה תְּרוּעָה יִתְקְעוּ לְמַסְעֵיהֶם׃ וּבְהַקְהִיל
30   אֶת־הַקָּהָל תִּתְקְעוּ וְלֹא תָרִיעוּ׃ וּבְנֵי אַהֲרֹן הַכֹּהֲנִים
31   יִתְקְעוּ בַּחֲצֹצְרוֹת וְהָיוּ לָכֶם לְחֻקַּת עוֹלָם לְדֹרֹתֵיכֶם׃
32   וְכִי־תָבֹאוּ מִלְחָמָה בְּאַרְצְכֶם עַל־הַצַּר הַצֹּרֵר אֶתְכֶם
33   וַהֲרֵעֹתֶם בַּחֲצֹצְרֹת וְנִזְכַּרְתֶּם לִפְנֵי יְהוָה אֱלֹהֵיכֶם
34   וְנוֹשַׁעְתֶּם מֵאֹיְבֵיכֶם׃ וּבְיוֹם שִׂמְחַתְכֶם וּבְמוֹעֲדֵיכֶם
35   וּבְרָאשֵׁי חָדְשֵׁיכֶם וּתְקַעְתֶּם בַּחֲצֹצְרֹת עַל עֹלֹתֵיכֶם
36   וְעַל זִבְחֵי שַׁלְמֵיכֶם וְהָיוּ לָכֶם לְזִכָּרוֹן לִפְנֵי אֱלֹהֵיכֶם
37   אֲנִי יְהוָה אֱלֹהֵיכֶם׃ [פ]
38    וַיְהִי בַּשָּׁנָה הַשֵּׁנִית בַּחֹדֶשׁ הַשֵּׁנִי בְּעֶשְׂרִים בַּחֹדֶשׁ
39   נַעֲלָה הֶעָנָן מֵעַל מִשְׁכַּן הָעֵדֻת׃ וַיִּסְעוּ בְנֵי־יִשְׂרָאֵל
40   לְמַסְעֵיהֶם מִמִּדְבַּר סִינָי וַיִּשְׁכֹּן הֶעָנָן בְּמִדְבַּר
41   פָּארָן׃ וַיִּסְעוּ בָּרִאשֹׁנָה עַל־פִּי יְהוָה בְּיַד־מֹשֶׁה׃
42   וַיִּסַּע דֶּגֶל מַחֲנֵה בְנֵי־יְהוּדָה בָּרִאשֹׁנָה לְצִבְאֹתָם
