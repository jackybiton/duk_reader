 1  וַיֵּרֶד הָעֲמָלֵקִי וְהַכְּנַעֲנִי הַיֹּשֵׁב בָּהָר הַהוּא וַיַּכּוּם
 2   וַיַּכְּתוּם עַד־הַחָרְמָה׃ [פ]
 3   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
 4   וְאָמַרְתָּ אֲלֵהֶם כִּי תָבֹאוּ אֶל־אֶרֶץ מוֹשְׁבֹתֵיכֶם
 5   אֲשֶׁר אֲנִי נֹתֵן לָכֶם׃ וַעֲשִׂיתֶם אִשֶּׁה לַיהוָה עֹלָה אוֹ־
 6  זֶבַח לְפַלֵּא־נֶדֶר אוֹ בִנְדָבָה אוֹ בְּמֹעֲדֵיכֶם לַעֲשׂוֹת
 7   רֵיחַ נִיחֹחַ לַיהוָה מִן־הַבָּקָר אוֹ מִן־הַצֹּאן׃ וְהִקְרִיב
 8   הַמַּקְרִיב קָרְבָּנוֹ לַיהוָה מִנְחָה סֹלֶת עִשָּׂרוֹן בָּלוּל
 9   בִּרְבִעִית הַהִין שָׁמֶן׃ וְיַיִן לַנֶּסֶךְ רְבִיעִית הַהִין
10   תַּעֲשֶׂה עַל־הָעֹלָה אוֹ לַזָּבַח לַכֶּבֶשׂ הָאֶחָד׃ אוֹ
11   לָאַיִל תַּעֲשֶׂה מִנְחָה סֹלֶת שְׁנֵי עֶשְׂרֹנִים בְּלוּלָה
12   בַשֶּׁמֶן שְׁלִשִׁית הַהִין׃ וְיַיִן לַנֶּסֶךְ שְׁלִשִׁית הַהִין
13   תַּקְרִיב רֵיחַ־נִיחֹחַ לַיהוָה׃  וְכִי־תַעֲשֶׂה בֶן־בָּקָר עֹלָה
14   אוֹ־זָבַח לְפַלֵּא־נֶדֶר אוֹ־שְׁלָמִים לַיהוָה׃ וְהִקְרִיב
15   עַל־בֶּן־הַבָּקָר מִנְחָה סֹלֶת שְׁלֹשָׁה עֶשְׂרֹנִים בָּלוּל
16   בַּשֶּׁמֶן חֲצִי הַהִין׃ וְיַיִן תַּקְרִיב לַנֶּסֶךְ חֲצִי הַהִין
17   אִשֵּׁה רֵיחַ־נִיחֹחַ לַיהוָה׃ כָּכָה יֵעָשֶׂה לַשּׁוֹר הָאֶחָד
18   אוֹ לָאַיִל הָאֶחָד אוֹ־לַשֶּׂה בַכְּבָשִׂים אוֹ בָעִזִּים׃
19   כַּמִּסְפָּר אֲשֶׁר תַּעֲשׂוּ כָּכָה תַּעֲשׂוּ לָאֶחָד כְּמִסְפָּרָם׃
20   כָּל־הָאֶזְרָח יַעֲשֶׂה־כָּכָה אֶת־אֵלֶּה לְהַקְרִיב אִשֵּׁה
21   רֵיחַ־נִיחֹחַ לַיהוָה׃ וְכִי־יָגוּר אִתְּכֶם גֵּר אוֹ אֲשֶׁר־
22  בְּתוֹכְכֶם לְדֹרֹתֵיכֶם וְעָשָׂה אִשֵּׁה רֵיחַ־נִיחֹחַ לַיהוָה
23   כַּאֲשֶׁר תַּעֲשׂוּ כֵּן יַעֲשֶׂה׃ הַקָּהָל חֻקָּה אַחַת לָכֶם
24   וְלַגֵּר הַגָּר חֻקַּת עוֹלָם לְדֹרֹתֵיכֶם כָּכֶם כַּגֵּר יִהְיֶה
25   לִפְנֵי יְהוָה׃ תּוֹרָה אַחַת וּמִשְׁפָּט אֶחָד יִהְיֶה לָכֶם
26   וְלַגֵּר הַגָּר אִתְּכֶם׃ [פ]
27    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי
28   יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם בְּבֹאֲכֶם אֶל־הָאָרֶץ אֲשֶׁר
29   אֲנִי מֵבִיא אֶתְכֶם שָׁמָּה׃ וְהָיָה בַּאֲכָלְכֶם מִלֶּחֶם
30   הָאָרֶץ תָּרִימוּ תְרוּמָה לַיהוָה׃ רֵאשִׁית עֲרִסֹתֵכֶם
31   חַלָּה תָּרִימוּ תְרוּמָה כִּתְרוּמַת גֹּרֶן כֵּן תָּרִימוּ
32   אֹתָהּ׃ מֵרֵאשִׁית עֲרִסֹתֵיכֶם תִּתְּנוּ לַיהוָה תְּרוּמָה
33   לְדֹרֹתֵיכֶם׃ [ס]             וְכִי תִשְׁגּוּ וְלֹא
34   תַעֲשׂוּ אֵת כָּל־הַמִּצְו‍ֹת הָאֵלֶּה אֲשֶׁר־דִּבֶּר יְהוָה
35   אֶל־מֹשֶׁה׃ אֵת כָּל־אֲשֶׁר צִוָּה יְהוָה אֲלֵיכֶם בְּיַד־
36  מֹשֶׁה מִן־הַיּוֹם אֲשֶׁר צִוָּה יְהוָה וָהָלְאָה לְדֹרֹתֵיכֶם׃
37   וְהָיָה אִם מֵעֵינֵי הָעֵדָה נֶעֶשְׂתָה לִשְׁגָגָה וְעָשׂוּ
38   כָל־הָעֵדָה פַּר בֶּן־בָּקָר אֶחָד לְעֹלָה לְרֵיחַ נִיחֹחַ
39   לַיהוָה וּמִנְחָתוֹ וְנִסְכּוֹ כַּמִּשְׁפָּט וּשְׂעִיר־עִזִּים אֶחָד
40   לְחַטָּת׃ וְכִפֶּר הַכֹּהֵן עַל־כָּל־עֲדַת בְּנֵי יִשְׂרָאֵל
41   וְנִסְלַח לָהֶם כִּי־שְׁגָגָה הִוא וְהֵם הֵבִיאוּ אֶת־קָרְבָּנָם
42   אִשֶּׁה לַיהוָה וְחַטָּאתָם לִפְנֵי יְהוָה עַל־שִׁגְגָתָם׃
