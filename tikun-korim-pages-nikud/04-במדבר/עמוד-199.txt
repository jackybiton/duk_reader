 1  וְאֵת הֶעָרִים אֲשֶׁר תִּתְּנוּ לַלְוִיִּם אֵת שֵׁשׁ־עָרֵי
 2   הַמִּקְלָט אֲשֶׁר תִּתְּנוּ לָנֻס שָׁמָּה הָרֹצֵחַ וַעֲלֵיהֶם
 3   תִּתְּנוּ אַרְבָּעִים וּשְׁתַּיִם עִיר׃ כָּל־הֶעָרִים אֲשֶׁר תִּתְּנוּ
 4   לַלְוִיִּם אַרְבָּעִים וּשְׁמֹנֶה עִיר אֶתְהֶן וְאֶת־מִגְרְשֵׁיהֶן׃
 5   וְהֶעָרִים אֲשֶׁר תִּתְּנוּ מֵאֲחֻזַּת בְּנֵי־יִשְׂרָאֵל מֵאֵת הָרַב
 6   תַּרְבּוּ וּמֵאֵת הַמְעַט תַּמְעִיטוּ אִישׁ כְּפִי נַחֲלָתוֹ אֲשֶׁר
 7   יִנְחָלוּ יִתֵּן מֵעָרָיו לַלְוִיִּם׃ [פ]
 8    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
 9   וְאָמַרְתָּ אֲלֵהֶם כִּי אַתֶּם עֹבְרִים אֶת־הַיַּרְדֵּן אַרְצָה
10   כְּנָעַן׃ וְהִקְרִיתֶם לָכֶם עָרִים עָרֵי מִקְלָט תִּהְיֶינָה
11   לָכֶם וְנָס שָׁמָּה רֹצֵחַ מַכֵּה־נֶפֶשׁ בִּשְׁגָגָה׃ וְהָיוּ לָכֶם
12   הֶעָרִים לְמִקְלָט מִגֹּאֵל וְלֹא יָמוּת הָרֹצֵחַ עַד־עָמְדוֹ
13   לִפְנֵי הָעֵדָה לַמִּשְׁפָּט׃ וְהֶעָרִים אֲשֶׁר תִּתֵּנוּ שֵׁשׁ־עָרֵי
14   מִקְלָט תִּהְיֶינָה לָכֶם׃ אֵת  שְׁלֹשׁ הֶעָרִים תִּתְּנוּ מֵעֵבֶר
15   לַיַּרְדֵּן וְאֵת שְׁלֹשׁ הֶעָרִים תִּתְּנוּ בְּאֶרֶץ כְּנָעַן עָרֵי
16   מִקְלָט תִּהְיֶינָה׃ לִבְנֵי יִשְׂרָאֵל וְלַגֵּר וְלַתּוֹשָׁב בְּתוֹכָם
17   תִּהְיֶינָה שֵׁשׁ־הֶעָרִים הָאֵלֶּה לְמִקְלָט לָנוּס שָׁמָּה
18   כָּל־מַכֵּה־נֶפֶשׁ בִּשְׁגָגָה׃ וְאִם־בִּכְלִי בַרְזֶל  הִכָּהוּ וַיָּמֹת
19   רֹצֵחַ הוּא מוֹת יוּמַת הָרֹצֵחַ׃ וְאִם בְּאֶבֶן יָד אֲשֶׁר־
20  יָמוּת בָּהּ הִכָּהוּ וַיָּמֹת רֹצֵחַ הוּא מוֹת יוּמַת הָרֹצֵחַ׃ אוֹ
21   בִּכְלִי עֵץ־יָד אֲשֶׁר־יָמוּת בּוֹ הִכָּהוּ וַיָּמֹת רֹצֵחַ הוּא
22   מוֹת יוּמַת הָרֹצֵחַ׃ גֹּאֵל הַדָּם הוּא יָמִית אֶת־הָרֹצֵחַ
23   בְּפִגְעוֹ־בוֹ הוּא יְמִתֶנּוּ׃ וְאִם־בְּשִׂנְאָה יֶהְדֳּפֶנּוּ אוֹ־
24  הִשְׁלִיךְ עָלָיו בִּצְדִיָּה וַיָּמֹת׃ אוֹ בְאֵיבָה הִכָּהוּ בְיָדוֹ
25   וַיָּמֹת מוֹת־יוּמַת הַמַּכֶּה רֹצֵחַ הוּא גֹּאֵל הַדָּם יָמִית
26   אֶת־הָרֹצֵחַ בְּפִגְעוֹ־בוֹ׃ וְאִם־בְּפֶתַע בְּלֹא־אֵיבָה הֲדָפוֹ
27   אוֹ־הִשְׁלִיךְ עָלָיו כָּל־כְּלִי בְּלֹא צְדִיָּה׃ אוֹ בְכָל־אֶבֶן
28   אֲשֶׁר־יָמוּת בָּהּ בְּלֹא רְאוֹת וַיַּפֵּל עָלָיו וַיָּמֹת וְהוּא
29   לֹא־אוֹיֵב לוֹ וְלֹא מְבַקֵּשׁ רָעָתוֹ׃ וְשָׁפְטוּ הָעֵדָה בֵּין
30   הַמַּכֶּה וּבֵין גֹּאֵל הַדָּם עַל הַמִּשְׁפָּטִים הָאֵלֶּה׃ וְהִצִּילוּ
31   הָעֵדָה אֶת־הָרֹצֵחַ מִיַּד גֹּאֵל הַדָּם וְהֵשִׁיבוּ אֹתוֹ
32   הָעֵדָה אֶל־עִיר מִקְלָטוֹ אֲשֶׁר־נָס שָׁמָּה וְיָשַׁב בָּהּ
33   עַד־מוֹת הַכֹּהֵן הַגָּדֹל אֲשֶׁר־מָשַׁח אֹתוֹ בְּשֶׁמֶן הַקֹּדֶשׁ׃
34   וְאִם־יָצֹא יֵצֵא הָרֹצֵחַ אֶת־גְּבוּל עִיר מִקְלָטוֹ אֲשֶׁר
35   יָנוּס שָׁמָּה׃ וּמָצָא אֹתוֹ גֹּאֵל הַדָּם מִחוּץ לִגְבוּל עִיר
36   מִקְלָטוֹ וְרָצַח גֹּאֵל הַדָּם אֶת־הָרֹצֵחַ אֵין לוֹ דָּם׃ כִּי
37   בְעִיר מִקְלָטוֹ יֵשֵׁב עַד־מוֹת הַכֹּהֵן הַגָּדֹל וְאַחֲרֵי מוֹת
38   הַכֹּהֵן הַגָּדֹל יָשׁוּב הָרֹצֵחַ אֶל־אֶרֶץ אֲחֻזָּתוֹ׃ וְהָיוּ אֵלֶּה
39   לָכֶם לְחֻקַּת מִשְׁפָּט לְדֹרֹתֵיכֶם בְּכֹל מוֹשְׁבֹתֵיכֶם׃
40   כָּל־מַכֵּה־נֶפֶשׁ לְפִי עֵדִים יִרְצַח אֶת־הָרֹצֵחַ וְעֵד
41   אֶחָד לֹא־יַעֲנֶה בְנֶפֶשׁ לָמוּת׃ וְלֹא־תִקְחוּ כֹפֶר
42   לְנֶפֶשׁ רֹצֵחַ אֲשֶׁר־הוּא רָשָׁע לָמוּת כִּי־מוֹת יוּמָת׃
