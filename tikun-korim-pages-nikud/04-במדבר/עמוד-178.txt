 1  וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִתּוֹךְ הַקָּהָל כִּי אֶת־מִקְדַּשׁ
 2   יְהוָה טִמֵּא מֵי נִדָּה לֹא־זֹרַק עָלָיו טָמֵא הוּא׃ וְהָיְתָה
 3   לָהֶם לְחֻקַּת עוֹלָם וּמַזֵּה מֵי־הַנִּדָּה יְכַבֵּס בְּגָדָיו
 4   וְהַנֹּגֵעַ בְּמֵי הַנִּדָּה יִטְמָא עַד־הָעָרֶב׃ וְכֹל אֲשֶׁר־
 5  יִגַּע־בּוֹ הַטָּמֵא יִטְמָא וְהַנֶּפֶשׁ הַנֹּגַעַת תִּטְמָא עַד־
 6  הָעָרֶב׃ [פ]
 7   וַיָּבֹאוּ בְנֵי־יִשְׂרָאֵל כָּל־הָעֵדָה מִדְבַּר־צִן בַּחֹדֶשׁ
 8   הָרִאשׁוֹן וַיֵּשֶׁב הָעָם בְּקָדֵשׁ וַתָּמָת שָׁם מִרְיָם
 9   וַתִּקָּבֵר שָׁם׃ וְלֹא־הָיָה מַיִם לָעֵדָה וַיִּקָּהֲלוּ עַל־מֹשֶׁה
10   וְעַל־אַהֲרֹן׃ וַיָּרֶב הָעָם עִם־מֹשֶׁה וַיֹּאמְרוּ לֵאמֹר
11   וְלוּ גָוַעְנוּ בִּגְוַע אַחֵינוּ לִפְנֵי יְהוָה׃ וְלָמָה הֲבֵאתֶם
12   אֶת־קְהַל יְהוָה אֶל־הַמִּדְבָּר הַזֶּה לָמוּת שָׁם אֲנַחְנוּ
13   וּבְעִירֵנוּ׃ וְלָמָה הֶעֱלִיתֻנוּ מִמִּצְרַיִם לְהָבִיא אֹתָנוּ
14   אֶל־הַמָּקוֹם הָרָע הַזֶּה לֹא  מְקוֹם זֶרַע וּתְאֵנָה וְגֶפֶן
15   וְרִמּוֹן וּמַיִם אַיִן לִשְׁתּוֹת׃ וַיָּבֹא מֹשֶׁה וְאַהֲרֹן מִפְּנֵי
16   הַקָּהָל אֶל־פֶּתַח אֹהֶל מוֹעֵד וַיִּפְּלוּ עַל־פְּנֵיהֶם וַיֵּרָא
17   כְבוֹד־יְהוָה אֲלֵיהֶם׃ [פ]
18    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ קַח אֶת־הַמַּטֶּה
19   וְהַקְהֵל אֶת־הָעֵדָה אַתָּה וְאַהֲרֹן אָחִיךָ וְדִבַּרְתֶּם
20   אֶל־הַסֶּלַע לְעֵינֵיהֶם וְנָתַן מֵימָיו וְהוֹצֵאתָ לָהֶם
21   מַיִם מִן־הַסֶּלַע וְהִשְׁקִיתָ אֶת־הָעֵדָה וְאֶת־בְּעִירָם׃
22   וַיִּקַּח מֹשֶׁה אֶת־הַמַּטֶּה מִלִּפְנֵי יְהוָה כַּאֲשֶׁר צִוָּהוּ׃
23   וַיַּקְהִלוּ מֹשֶׁה וְאַהֲרֹן אֶת־הַקָּהָל אֶל־פְּנֵי הַסָּלַע
24   וַיֹּאמֶר לָהֶם שִׁמְעוּ־נָא הַמֹּרִים הֲמִן־הַסֶּלַע הַזֶּה
25   נוֹצִיא לָכֶם מָיִם׃ וַיָּרֶם מֹשֶׁה אֶת־יָדוֹ וַיַּךְ אֶת־
26  הַסֶּלַע בְּמַטֵּהוּ פַּעֲמָיִם וַיֵּצְאוּ מַיִם רַבִּים וַתֵּשְׁתְּ
27   הָעֵדָה וּבְעִירָם׃ [ס]             וַיֹּאמֶר
28   יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן יַעַן לֹא־הֶאֱמַנְתֶּם בִּי
29   לְהַקְדִּישֵׁנִי לְעֵינֵי בְּנֵי יִשְׂרָאֵל לָכֵן לֹא תָבִיאוּ אֶת־
30  הַקָּהָל הַזֶּה אֶל־הָאָרֶץ אֲשֶׁר־נָתַתִּי לָהֶם׃ הֵמָּה מֵי
31   מְרִיבָה אֲשֶׁר־רָבוּ בְנֵי־יִשְׂרָאֵל אֶת־יְהוָה וַיִּקָּדֵשׁ
32   בָּם׃ [ס]              וַיִּשְׁלַח מֹשֶׁה מַלְאָכִים
33   מִקָּדֵשׁ אֶל־מֶלֶךְ אֱדוֹם כֹּה אָמַר אָחִיךָ יִשְׂרָאֵל
34   אַתָּה יָדַעְתָּ אֵת כָּל־הַתְּלָאָה אֲשֶׁר מְצָאָתְנוּ׃ וַיֵּרְדוּ
35   אֲבֹתֵינוּ מִצְרַיְמָה וַנֵּשֶׁב בְּמִצְרַיִם יָמִים רַבִּים וַיָּרֵעוּ
36   לָנוּ מִצְרַיִם וְלַאֲבֹתֵינוּ׃ וַנִּצְעַק אֶל־יְהוָה וַיִּשְׁמַע קֹלֵנוּ
37   וַיִּשְׁלַח מַלְאָךְ וַיֹּצִאֵנוּ מִמִּצְרָיִם וְהִנֵּה אֲנַחְנוּ בְקָדֵשׁ
38   עִיר קְצֵה גְבוּלֶךָ׃ נַעְבְּרָה־נָּא בְאַרְצֶךָ לֹא נַעֲבֹר
39   בְּשָׂדֶה וּבְכֶרֶם וְלֹא נִשְׁתֶּה מֵי בְאֵר דֶּרֶךְ הַמֶּלֶךְ
40   נֵלֵךְ לֹא נִטֶּה יָמִין וּשְׂמֹאול עַד אֲשֶׁר־נַעֲבֹר גְּבֻלֶךָ׃
41   וַיֹּאמֶר אֵלָיו אֱדוֹם לֹא תַעֲבֹר בִּי פֶּן־בַּחֶרֶב אֵצֵא
42   לִקְרָאתֶךָ׃ וַיֹּאמְרוּ אֵלָיו בְּנֵי־יִשְׂרָאֵל בַּמְסִלָּה נַעֲלֶה
