 1  וְאַרְבַּע מֵאוֹת׃ [פ]
 2   לִבְנֵי דָן תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם בְּמִסְפַּר
 3   שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא צָבָא׃
 4   פְּקֻדֵיהֶם לְמַטֵּה דָן שְׁנַיִם וְשִׁשִּׁים אֶלֶף וּשְׁבַע
 5   מֵאוֹת׃ [פ]
 6   לִבְנֵי אָשֵׁר תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
 7   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
 8   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה אָשֵׁר אֶחָד וְאַרְבָּעִים אֶלֶף
 9   וַחֲמֵשׁ מֵאוֹת׃ [פ]
10   בְּנֵי נַפְתָּלִי תּוֹלְדֹתָם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם
11   בְּמִסְפַּר שֵׁמֹת מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כֹּל יֹצֵא
12   צָבָא׃ פְּקֻדֵיהֶם לְמַטֵּה נַפְתָּלִי שְׁלֹשָׁה וַחֲמִשִּׁים
13   אֶלֶף וְאַרְבַּע מֵאוֹת׃ [פ]
14   אֵלֶּה הַפְּקֻדִים אֲשֶׁר פָּקַד מֹשֶׁה וְאַהֲרֹן וּנְשִׂיאֵי
15   יִשְׂרָאֵל שְׁנֵים עָשָׂר אִישׁ אִישׁ־אֶחָד לְבֵית־אֲבֹתָיו
16   הָיוּ׃ וַיִּהְיוּ כָּל־פְּקוּדֵי בְנֵי־יִשְׂרָאֵל לְבֵית אֲבֹתָם מִבֶּן
17   עֶשְׂרִים שָׁנָה וָמַעְלָה כָּל־יֹצֵא צָבָא בְּיִשְׂרָאֵל׃ וַיִּהְיוּ
18   כָּל־הַפְּקֻדִים שֵׁשׁ־מֵאוֹת אֶלֶף וּשְׁלֹשֶׁת אֲלָפִים
19   וַחֲמֵשׁ מֵאוֹת וַחֲמִשִּׁים׃ וְהַלְוִיִּם לְמַטֵּה אֲבֹתָם לֹא
20   הָתְפָּקְדוּ בְּתוֹכָם׃ [פ]
21   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ אַךְ אֶת־מַטֵּה לֵוִי לֹא
22   תִפְקֹד וְאֶת־רֹאשָׁם לֹא תִשָּׂא בְּתוֹךְ בְּנֵי
23   יִשְׂרָאֵל׃ וְאַתָּה הַפְקֵד אֶת־הַלְוִיִּם עַל־מִשְׁכַּן הָעֵדֻת
24   וְעַל כָּל־כֵּלָיו וְעַל כָּל־אֲשֶׁר־לוֹ הֵמָּה יִשְׂאוּ אֶת־
25  הַמִּשְׁכָּן וְאֶת־כָּל־כֵּלָיו וְהֵם יְשָׁרְתֻהוּ וְסָבִיב
26   לַמִּשְׁכָּן יַחֲנוּ׃ וּבִנְסֹעַ הַמִּשְׁכָּן יוֹרִידוּ אֹתוֹ הַלְוִיִּם
27   וּבַחֲנֹת הַמִּשְׁכָּן יָקִימוּ אֹתוֹ הַלְוִיִּם וְהַזָּר הַקָּרֵב
28   יוּמָת׃ וְחָנוּ בְּנֵי יִשְׂרָאֵל אִישׁ עַל־מַחֲנֵהוּ וְאִישׁ
29   עַל־דִּגְלוֹ לְצִבְאֹתָם׃ וְהַלְוִיִּם יַחֲנוּ סָבִיב לְמִשְׁכַּן
30   הָעֵדֻת וְלֹא־יִהְיֶה קֶצֶף עַל־עֲדַת בְּנֵי יִשְׂרָאֵל
31   וְשָׁמְרוּ הַלְוִיִּם אֶת־מִשְׁמֶרֶת מִשְׁכַּן הָעֵדוּת׃ וַיַּעֲשׂוּ
32   בְּנֵי יִשְׂרָאֵל כְּכֹל אֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה כֵּן
33   עָשׂוּ׃ [פ]
34    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ אִישׁ עַל־
35  דִּגְלוֹ בְאֹתֹת לְבֵית אֲבֹתָם יַחֲנוּ בְּנֵי יִשְׂרָאֵל מִנֶּגֶד
36   סָבִיב לְאֹהֶל־מוֹעֵד יַחֲנוּ׃ וְהַחֹנִים קֵדְמָה מִזְרָחָה דֶּגֶל
37   מַחֲנֵה יְהוּדָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי יְהוּדָה נַחְשׁוֹן
38   בֶּן־עַמִּינָדָב׃ וּצְבָאוֹ וּפְקֻדֵיהֶם אַרְבָּעָה וְשִׁבְעִים
39   אֶלֶף וְשֵׁשׁ מֵאוֹת׃ וְהַחֹנִים עָלָיו מַטֵּה יִשָּׂשכָר
40   וְנָשִׂיא לִבְנֵי יִשָּׂשכָר נְתַנְאֵל בֶּן־צוּעָר׃ וּצְבָאוֹ
41   וּפְקֻדָיו אַרְבָּעָה וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע מֵאוֹת׃ מַטֵּה
42   זְבוּלֻן וְנָשִׂיא לִבְנֵי זְבוּלֻן אֱלִיאָב בֶּן־חֵלֹן׃ וּצְבָאוֹ
