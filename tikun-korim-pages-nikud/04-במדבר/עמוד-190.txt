 1  וְהִקְרַבְתֶּם עוֹלָה לְרֵיחַ נִיחֹחַ לַיהוָה פָּרִים בְּנֵי־בָקָר
 2   שְׁנַיִם אַיִל אֶחָד שִׁבְעָה כְבָשִׂים בְּנֵי שָׁנָה׃ וּמִנְחָתָם
 3   סֹלֶת בְּלוּלָה בַשָּׁמֶן שְׁלֹשָׁה עֶשְׂרֹנִים לַפָּר הָאֶחָד
 4   שְׁנֵי עֶשְׂרֹנִים לָאַיִל הָאֶחָד׃ עִשָּׂרוֹן עִשָּׂרוֹן לַכֶּבֶשׂ
 5   הָאֶחָד לְשִׁבְעַת הַכְּבָשִׂים׃ שְׂעִיר עִזִּים אֶחָד לְכַפֵּר
 6   עֲלֵיכֶם׃ מִלְּבַד עֹלַת הַתָּמִיד וּמִנְחָתוֹ תַּעֲשׂוּ תְּמִימִם
 7   יִהְיוּ־לָכֶם וְנִסְכֵּיהֶם׃ [פ]
 8   וּבַחֹדֶשׁ הַשְּׁבִיעִי בְּאֶחָד לַחֹדֶשׁ מִקְרָא־קֹדֶשׁ
 9   יִהְיֶה לָכֶם כָּל־מְלֶאכֶת עֲבֹדָה לֹא תַעֲשׂוּ יוֹם
10   תְּרוּעָה יִהְיֶה לָכֶם׃ וַעֲשִׂיתֶם עֹלָה לְרֵיחַ נִיחֹחַ
11   לַיהוָה פַּר בֶּן־בָּקָר אֶחָד אַיִל אֶחָד כְּבָשִׂים בְּנֵי־
12  שָׁנָה שִׁבְעָה תְּמִימִם׃ וּמִנְחָתָם סֹלֶת בְּלוּלָה
13   בַשָּׁמֶן שְׁלֹשָׁה עֶשְׂרֹנִים לַפָּר שְׁנֵי עֶשְׂרֹנִים
14   לָאָיִל׃ וְעִשָּׂרוֹן אֶחָד לַכֶּבֶשׂ הָאֶחָד לְשִׁבְעַת
15   הַכְּבָשִׂים׃ וּשְׂעִיר־עִזִּים אֶחָד חַטָּאת לְכַפֵּר עֲלֵיכֶם׃
16   מִלְּבַד עֹלַת הַחֹדֶשׁ וּמִנְחָתָהּ וְעֹלַת הַתָּמִיד
17   וּמִנְחָתָהּ וְנִסְכֵּיהֶם כְּמִשְׁפָּטָם לְרֵיחַ נִיחֹחַ אִשֶּׁה
18   לַיהוָה׃ [ס]             וּבֶעָשׂוֹר לַחֹדֶשׁ
19   הַשְּׁבִיעִי הַזֶּה מִקְרָא־קֹדֶשׁ יִהְיֶה לָכֶם וְעִנִּיתֶם
20   אֶת־נַפְשֹׁתֵיכֶם כָּל־מְלָאכָה לֹא תַעֲשׂוּ׃ וְהִקְרַבְתֶּם
21   עֹלָה לַיהוָה רֵיחַ נִיחֹחַ פַּר בֶּן־בָּקָר אֶחָד אַיִל
22   אֶחָד כְּבָשִׂים בְּנֵי־שָׁנָה שִׁבְעָה תְּמִימִם יִהְיוּ לָכֶם׃
23   וּמִנְחָתָם סֹלֶת בְּלוּלָה בַשָּׁמֶן שְׁלֹשָׁה עֶשְׂרֹנִים
24   לַפָּר שְׁנֵי עֶשְׂרֹנִים לָאַיִל הָאֶחָד׃ עִשָּׂרוֹן עִשָּׂרוֹן
25   לַכֶּבֶשׂ הָאֶחָד לְשִׁבְעַת הַכְּבָשִׂים׃ שְׂעִיר־עִזִּים
26   אֶחָד חַטָּאת מִלְּבַד חַטַּאת הַכִּפֻּרִים וְעֹלַת הַתָּמִיד
27   וּמִנְחָתָהּ וְנִסְכֵּיהֶם׃ [ס]               וּבַחֲמִשָּׁה
28   עָשָׂר יוֹם לַחֹדֶשׁ הַשְּׁבִיעִי מִקְרָא־קֹדֶשׁ יִהְיֶה לָכֶם
29   כָּל־מְלֶאכֶת עֲבֹדָה לֹא תַעֲשׂוּ וְחַגֹּתֶם חַג
30   לַיהוָה שִׁבְעַת יָמִים׃ וְהִקְרַבְתֶּם עֹלָה אִשֵּׁה רֵיחַ
31   נִיחֹחַ לַיהוָה פָּרִים בְּנֵי־בָקָר שְׁלֹשָׁה עָשָׂר אֵילִם
32   שְׁנָיִם כְּבָשִׂים בְּנֵי־שָׁנָה אַרְבָּעָה עָשָׂר תְּמִימִם
33   יִהְיוּ׃ וּמִנְחָתָם סֹלֶת בְּלוּלָה בַשָּׁמֶן שְׁלֹשָׁה עֶשְׂרֹנִים
34   לַפָּר הָאֶחָד לִשְׁלֹשָׁה עָשָׂר פָּרִים שְׁנֵי עֶשְׂרֹנִים
35   לָאַיִל הָאֶחָד לִשְׁנֵי הָאֵילִם׃ וְעִשָּׂרוֹן עִשָּׂרוֹן
36   לַכֶּבֶשׂ הָאֶחָד לְאַרְבָּעָה עָשָׂר כְּבָשִׂים׃ וּשְׂעִיר־
37  עִזִּים אֶחָד חַטָּאת מִלְּבַד עֹלַת הַתָּמִיד מִנְחָתָהּ
38   וְנִסְכָּהּ׃ [ס]             וּבַיּוֹם הַשֵּׁנִי פָּרִים
39   בְּנֵי־בָקָר שְׁנֵים עָשָׂר אֵילִם שְׁנָיִם כְּבָשִׂים בְּנֵי־
40  שָׁנָה אַרְבָּעָה עָשָׂר תְּמִימִם׃ וּמִנְחָתָם וְנִסְכֵּיהֶם
41   לַפָּרִים לָאֵילִם וְלַכְּבָשִׂים בְּמִסְפָּרָם כַּמִּשְׁפָּט׃
42   וּשְׂעִיר־עִזִּים אֶחָד חַטָּאת מִלְּבַד עֹלַת הַתָּמִיד
