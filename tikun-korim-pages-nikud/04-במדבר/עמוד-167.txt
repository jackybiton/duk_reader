 1  וַיֹּאמַר אֶלְדָּד וּמֵידָד מִתְנַבְּאִים בַּמַּחֲנֶה׃ וַיַּעַן יְהוֹשֻׁעַ
 2   בִּן־נוּן מְשָׁרֵת מֹשֶׁה מִבְּחֻרָיו וַיֹּאמַר אֲדֹנִי מֹשֶׁה
 3   כְּלָאֵם׃ וַיֹּאמֶר לוֹ מֹשֶׁה הַמְקַנֵּא אַתָּה לִי וּמִי יִתֵּן
 4   כָּל־עַם יְהוָה נְבִיאִים כִּי־יִתֵּן יְהוָה אֶת־רוּחוֹ עֲלֵיהֶם׃
 5    וַיֵּאָסֵף מֹשֶׁה אֶל־הַמַּחֲנֶה הוּא וְזִקְנֵי יִשְׂרָאֵל׃ וְרוּחַ
 6   נָסַע  מֵאֵת יְהוָה וַיָּגָז שַׂלְוִים מִן־הַיָּם וַיִּטֹּשׁ עַל־
 7  הַמַּחֲנֶה כְּדֶרֶךְ יוֹם כֹּה וּכְדֶרֶךְ יוֹם כֹּה סְבִיבוֹת
 8   הַמַּחֲנֶה וּכְאַמָּתַיִם עַל־פְּנֵי הָאָרֶץ׃ וַיָּקָם הָעָם כָּל־
 9  הַיּוֹם הַהוּא וְכָל־הַלַּיְלָה וְכֹל  יוֹם הַמָּחֳרָת וַיַּאַסְפוּ
10   אֶת־הַשְּׂלָו הַמַּמְעִיט אָסַף עֲשָׂרָה חֳמָרִים וַיִּשְׁטְחוּ
11   לָהֶם שָׁטוֹחַ סְבִיבוֹת הַמַּחֲנֶה׃ הַבָּשָׂר עוֹדֶנּוּ בֵּין
12   שִׁנֵּיהֶם טֶרֶם יִכָּרֵת וְאַף יְהוָה חָרָה בָעָם וַיַּךְ יְהוָה
13   בָּעָם מַכָּה רַבָּה מְאֹד׃ וַיִּקְרָא אֶת־שֵׁם־הַמָּקוֹם
14   הַהוּא קִבְרוֹת הַתַּאֲוָה כִּי־שָׁם קָבְרוּ אֶת־הָעָם
15   הַמִּתְאַוִּים׃ מִקִּבְרוֹת הַתַּאֲוָה נָסְעוּ הָעָם חֲצֵרוֹת
16   וַיִּהְיוּ בַּחֲצֵרוֹת׃ [פ]
17   וַתְּדַבֵּר מִרְיָם וְאַהֲרֹן בְּמֹשֶׁה עַל־אֹדוֹת הָאִשָּׁה
18   הַכֻּשִׁית אֲשֶׁר לָקָח כִּי־אִשָּׁה כֻשִׁית לָקָח׃ וַיֹּאמְרוּ
19   הֲרַק אַךְ־בְּמֹשֶׁה דִּבֶּר יְהוָה הֲלֹא גַּם־בָּנוּ דִבֵּר
20   וַיִּשְׁמַע יְהוָה׃ וְהָאִישׁ מֹשֶׁה עָנָו מְאֹד מִכֹּל הָאָדָם
21   אֲשֶׁר עַל־פְּנֵי הָאֲדָמָה׃ [ס]             וַיֹּאמֶר
22   יְהוָה פִּתְאֹם אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן וְאֶל־מִרְיָם צְאוּ
23   שְׁלָשְׁתְּכֶם אֶל־אֹהֶל מוֹעֵד וַיֵּצְאוּ שְׁלָשְׁתָּם׃ וַיֵּרֶד
24   יְהוָה בְּעַמּוּד עָנָן וַיַּעֲמֹד פֶּתַח הָאֹהֶל וַיִּקְרָא אַהֲרֹן
25   וּמִרְיָם וַיֵּצְאוּ שְׁנֵיהֶם׃ וַיֹּאמֶר שִׁמְעוּ־נָא דְבָרָי אִם־
26  יִהְיֶה נְבִיאֲכֶם יְהוָה בַּמַּרְאָה אֵלָיו אֶתְוַדָּע בַּחֲלוֹם
27   אֲדַבֶּר־בּוֹ׃ לֹא־כֵן עַבְדִּי מֹשֶׁה בְּכָל־בֵּיתִי נֶאֱמָן הוּא׃
28   פֶּה אֶל־פֶּה אֲדַבֶּר־בּוֹ וּמַרְאֶה וְלֹא בְחִידֹת וּתְמֻנַת
29   יְהוָה יַבִּיט וּמַדּוּעַ לֹא יְרֵאתֶם לְדַבֵּר בְּעַבְדִּי בְמֹשֶׁה׃
30   וַיִּחַר־אַף יְהוָה בָּם וַיֵּלַךְ׃ וְהֶעָנָן סָר מֵעַל הָאֹהֶל
31   וְהִנֵּה מִרְיָם מְצֹרַעַת כַּשָּׁלֶג וַיִּפֶן אַהֲרֹן אֶל־מִרְיָם
32   וְהִנֵּה מְצֹרָעַת׃ וַיֹּאמֶר אַהֲרֹן אֶל־מֹשֶׁה בִּי אֲדֹנִי
33   אַל־נָא תָשֵׁת עָלֵינוּ חַטָּאת אֲשֶׁר נוֹאַלְנוּ וַאֲשֶׁר
34   חָטָאנוּ׃ אַל־נָא תְהִי כַּמֵּת אֲשֶׁר בְּצֵאתוֹ מֵרֶחֶם אִמּוֹ
35   וַיֵּאָכֵל חֲצִי בְשָׂרוֹ׃ וַיִּצְעַק מֹשֶׁה אֶל־יְהוָה לֵאמֹר
36   אֵל נָא רְפָא נָא לָהּ׃ [פ]
37    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה וְאָבִיהָ יָרֹק יָרַק בְּפָנֶיהָ
38   הֲלֹא תִכָּלֵם שִׁבְעַת יָמִים תִּסָּגֵר שִׁבְעַת יָמִים
39   מִחוּץ לַמַּחֲנֶה וְאַחַר תֵּאָסֵף׃ וַתִּסָּגֵר מִרְיָם מִחוּץ
40   לַמַּחֲנֶה שִׁבְעַת יָמִים וְהָעָם לֹא נָסַע עַד־הֵאָסֵף
41   מִרְיָם׃ וְאַחַר נָסְעוּ הָעָם מֵחֲצֵרוֹת וַיַּחֲנוּ בְּמִדְבַּר
42   פָּארָן׃[פ]
