 1  וְשֵׁשׁ מֵאוֹת שֹׁמְרֵי מִשְׁמֶרֶת הַקֹּדֶשׁ׃ מִשְׁפְּחֹת
 2   בְּנֵי־קְהָת יַחֲנוּ עַל יֶרֶךְ הַמִּשְׁכָּן תֵּימָנָה׃ וּנְשִׂיא
 3   בֵית־אָב לְמִשְׁפְּחֹת הַקְּהָתִי אֱלִיצָפָן בֶּן־עֻזִּיאֵל׃
 4   וּמִשְׁמַרְתָּם הָאָרֹן וְהַשֻּׁלְחָן וְהַמְּנֹרָה וְהַמִּזְבְּחֹת
 5   וּכְלֵי הַקֹּדֶשׁ אֲשֶׁר יְשָׁרְתוּ בָּהֶם וְהַמָּסָךְ וְכֹל
 6   עֲבֹדָתוֹ׃ וּנְשִׂיא נְשִׂיאֵי הַלֵּוִי אֶלְעָזָר בֶּן־אַהֲרֹן הַכֹּהֵן
 7   פְּקֻדַּת שֹׁמְרֵי מִשְׁמֶרֶת הַקֹּדֶשׁ׃ לִמְרָרִי מִשְׁפַּחַת
 8   הַמַּחְלִי וּמִשְׁפַּחַת הַמּוּשִׁי אֵלֶּה הֵם מִשְׁפְּחֹת מְרָרִי׃
 9   וּפְקֻדֵיהֶם בְּמִסְפַּר כָּל־זָכָר מִבֶּן־חֹדֶשׁ וָמָעְלָה
10   שֵׁשֶׁת אֲלָפִים וּמָאתָיִם׃ וּנְשִׂיא בֵית־אָב לְמִשְׁפְּחֹת
11   מְרָרִי צוּרִיאֵל בֶּן־אֲבִיחָיִל עַל יֶרֶךְ הַמִּשְׁכָּן יַחֲנוּ
12   צָפֹנָה׃ וּפְקֻדַּת מִשְׁמֶרֶת בְּנֵי מְרָרִי קַרְשֵׁי הַמִּשְׁכָּן
13   וּבְרִיחָיו וְעַמֻּדָיו וַאֲדָנָיו וְכָל־כֵּלָיו וְכֹל עֲבֹדָתוֹ׃
14   וְעַמֻּדֵי הֶחָצֵר סָבִיב וְאַדְנֵיהֶם וִיתֵדֹתָם וּמֵיתְרֵיהֶם׃
15   וְהַחֹנִים לִפְנֵי הַמִּשְׁכָּן קֵדְמָה לִפְנֵי אֹהֶל־מוֹעֵד
16   מִזְרָחָה מֹשֶׁה  וְאַהֲרֹן וּבָנָיו שֹׁמְרִים מִשְׁמֶרֶת
17   הַמִּקְדָּשׁ לְמִשְׁמֶרֶת בְּנֵי יִשְׂרָאֵל וְהַזָּר הַקָּרֵב יוּמָת׃
18   כָּל־פְּקוּדֵי הַלְוִיִּם אֲשֶׁר פָּקַד מֹשֶׁה וְאַהֲרֹן עַל־פִּי
19   יְהוָה לְמִשְׁפְּחֹתָם כָּל־זָכָר מִבֶּן־חֹדֶשׁ וָמַעְלָה
20   שְׁנַיִם וְעֶשְׂרִים אָלֶף׃ [ס]              וַיֹּאמֶר
21   יְהוָה אֶל־מֹשֶׁה פְּקֹד כָּל־בְּכֹר זָכָר לִבְנֵי יִשְׂרָאֵל
22   מִבֶּן־חֹדֶשׁ וָמָעְלָה וְשָׂא אֵת מִסְפַּר שְׁמֹתָם׃
23   וְלָקַחְתָּ אֶת־הַלְוִיִּם לִי אֲנִי יְהוָה תַּחַת כָּל־בְּכֹר
24   בִּבְנֵי יִשְׂרָאֵל וְאֵת בֶּהֱמַת הַלְוִיִּם תַּחַת כָּל־בְּכוֹר
25   בְּבֶהֱמַת בְּנֵי יִשְׂרָאֵל׃ וַיִּפְקֹד מֹשֶׁה כַּאֲשֶׁר צִוָּה
26   יְהוָה אֹתוֹ אֶת־כָּל־בְּכוֹר בִּבְנֵי יִשְׂרָאֵל׃ וַיְהִי כָל־
27  בְּכוֹר זָכָר בְּמִסְפַּר שֵׁמֹת מִבֶּן־חֹדֶשׁ וָמַעְלָה
28   לִפְקֻדֵיהֶם שְׁנַיִם וְעֶשְׂרִים אֶלֶף שְׁלֹשָׁה וְשִׁבְעִים
29   וּמָאתָיִם׃ [פ]
30   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ קַח אֶת־הַלְוִיִּם
31   תַּחַת כָּל־בְּכוֹר בִּבְנֵי יִשְׂרָאֵל וְאֶת־בֶּהֱמַת הַלְוִיִּם
32   תַּחַת בְּהֶמְתָּם וְהָיוּ־לִי הַלְוִיִּם אֲנִי יְהוָה׃ וְאֵת פְּדוּיֵי
33   הַשְּׁלֹשָׁה וְהַשִּׁבְעִים וְהַמָּאתָיִם הָעֹדְפִים עַל־
34  הַלְוִיִּם מִבְּכוֹר בְּנֵי יִשְׂרָאֵל׃ וְלָקַחְתָּ חֲמֵשֶׁת חֲמֵשֶׁת
35   שְׁקָלִים לַגֻּלְגֹּלֶת בְּשֶׁקֶל הַקֹּדֶשׁ תִּקָּח עֶשְׂרִים
36   גֵּרָה הַשָּׁקֶל׃ וְנָתַתָּה הַכֶּסֶף לְאַהֲרֹן וּלְבָנָיו פְּדוּיֵי
37   הָעֹדְפִים בָּהֶם׃ וַיִּקַּח מֹשֶׁה אֵת כֶּסֶף הַפִּדְיוֹם
38   מֵאֵת הָעֹדְפִים עַל פְּדוּיֵי הַלְוִיִּם׃ מֵאֵת בְּכוֹר בְּנֵי
39   יִשְׂרָאֵל לָקַח אֶת־הַכָּסֶף חֲמִשָּׁה וְשִׁשִּׁים וּשְׁלֹשׁ
40   מֵאוֹת וָאֶלֶף בְּשֶׁקֶל הַקֹּדֶשׁ׃ וַיִּתֵּן מֹשֶׁה אֶת־כֶּסֶף
41   הַפְּדֻיִם לְאַהֲרֹן וּלְבָנָיו עַל־פִּי יְהוָה כַּאֲשֶׁר צִוָּה
42   יְהוָה אֶת־מֹשֶׁה׃ [פ]
