 1  וְזֶה־לְּךָ תְּרוּמַת מַתָּנָם לְכָל־תְּנוּפֹת בְּנֵי יִשְׂרָאֵל
 2   לְךָ נְתַתִּים וּלְבָנֶיךָ וְלִבְנֹתֶיךָ אִתְּךָ לְחָק־עוֹלָם כָּל־
 3  טָהוֹר בְּבֵיתְךָ יֹאכַל אֹתוֹ׃ כֹּל חֵלֶב יִצְהָר וְכָל־חֵלֶב
 4   תִּירוֹשׁ וְדָגָן רֵאשִׁיתָם אֲשֶׁר־יִתְּנוּ לַיהוָה לְךָ נְתַתִּים׃
 5   בִּכּוּרֵי כָּל־אֲשֶׁר בְּאַרְצָם אֲשֶׁר־יָבִיאוּ לַיהוָה לְךָ
 6   יִהְיֶה כָּל־טָהוֹר בְּבֵיתְךָ יֹאכְלֶנּוּ׃ כָּל־חֵרֶם בְּיִשְׂרָאֵל
 7   לְךָ יִהְיֶה׃ כָּל־פֶּטֶר רֶחֶם לְכָל־בָּשָׂר אֲשֶׁר־יַקְרִיבוּ
 8   לַיהוָה בָּאָדָם וּבַבְּהֵמָה יִהְיֶה־לָּךְ אַךְ  פָּדֹה תִפְדֶּה
 9   אֵת בְּכוֹר הָאָדָם וְאֵת בְּכוֹר־הַבְּהֵמָה הַטְּמֵאָה
10   תִּפְדֶּה׃ וּפְדוּיָו מִבֶּן־חֹדֶשׁ תִּפְדֶּה בְּעֶרְכְּךָ כֶּסֶף
11   חֲמֵשֶׁת שְׁקָלִים בְּשֶׁקֶל הַקֹּדֶשׁ עֶשְׂרִים גֵּרָה הוּא׃
12   אַךְ בְּכוֹר־שׁוֹר אוֹ־בְכוֹר כֶּשֶׂב אוֹ־בְכוֹר עֵז לֹא
13   תִפְדֶּה קֹדֶשׁ הֵם אֶת־דָּמָם תִּזְרֹק עַל־הַמִּזְבֵּחַ וְאֶת־
14  חֶלְבָּם תַּקְטִיר אִשֶּׁה לְרֵיחַ נִיחֹחַ לַיהוָה׃ וּבְשָׂרָם
15   יִהְיֶה־לָּךְ כַּחֲזֵה הַתְּנוּפָה וּכְשׁוֹק הַיָּמִין לְךָ יִהְיֶה׃
16   כֹּל  תְּרוּמֹת הַקֳּדָשִׁים אֲשֶׁר יָרִימוּ בְנֵי־יִשְׂרָאֵל
17   לַיהוָה נָתַתִּי לְךָ וּלְבָנֶיךָ וְלִבְנֹתֶיךָ אִתְּךָ לְחָק־עוֹלָם
18   בְּרִית מֶלַח עוֹלָם הִוא לִפְנֵי יְהוָה לְךָ וּלְזַרְעֲךָ
19   אִתָּךְ׃ וַיֹּאמֶר יְהוָה אֶל־אַהֲרֹן בְּאַרְצָם לֹא תִנְחָל
20   וְחֵלֶק לֹא־יִהְיֶה לְךָ בְּתוֹכָם אֲנִי חֶלְקְךָ וְנַחֲלָתְךָ
21   בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׃ [ס]              וְלִבְנֵי
22   לֵוִי הִנֵּה נָתַתִּי כָּל־מַעֲשֵׂר בְּיִשְׂרָאֵל לְנַחֲלָה חֵלֶף
23   עֲבֹדָתָם אֲשֶׁר־הֵם עֹבְדִים אֶת־עֲבֹדַת אֹהֶל מוֹעֵד׃
24   וְלֹא־יִקְרְבוּ עוֹד בְּנֵי יִשְׂרָאֵל אֶל־אֹהֶל מוֹעֵד
25   לָשֵׂאת חֵטְא לָמוּת׃ וְעָבַד הַלֵּוִי הוּא אֶת־עֲבֹדַת
26   אֹהֶל מוֹעֵד וְהֵם יִשְׂאוּ עֲו‍ֹנָם חֻקַּת עוֹלָם לְדֹרֹתֵיכֶם
27   וּבְתוֹךְ בְּנֵי יִשְׂרָאֵל לֹא יִנְחֲלוּ נַחֲלָה׃ כִּי אֶת־מַעְשַׂר
28   בְּנֵי־יִשְׂרָאֵל אֲשֶׁר יָרִימוּ לַיהוָה תְּרוּמָה נָתַתִּי
29   לַלְוִיִּם לְנַחֲלָה עַל־כֵּן אָמַרְתִּי לָהֶם בְּתוֹךְ בְּנֵי
30   יִשְׂרָאֵל לֹא יִנְחֲלוּ נַחֲלָה׃ [פ]
31   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ וְאֶל־הַלְוִיִּם תְּדַבֵּר
32   וְאָמַרְתָּ אֲלֵהֶם כִּי־תִקְחוּ מֵאֵת בְּנֵי־יִשְׂרָאֵל אֶת־
33  הַמַּעֲשֵׂר אֲשֶׁר נָתַתִּי לָכֶם מֵאִתָּם בְּנַחֲלַתְכֶם
34   וַהֲרֵמֹתֶם מִמֶּנּוּ תְּרוּמַת יְהוָה מַעֲשֵׂר מִן־הַמַּעֲשֵׂר׃
35   וְנֶחְשַׁב לָכֶם תְּרוּמַתְכֶם כַּדָּגָן מִן־הַגֹּרֶן וְכַמְלֵאָה
36   מִן־הַיָּקֶב׃ כֵּן תָּרִימוּ גַם־אַתֶּם תְּרוּמַת יְהוָה מִכֹּל
37   מַעְשְׂרֹתֵיכֶם אֲשֶׁר תִּקְחוּ מֵאֵת בְּנֵי יִשְׂרָאֵל
38   וּנְתַתֶּם מִמֶּנּוּ אֶת־תְּרוּמַת יְהוָה לְאַהֲרֹן הַכֹּהֵן׃
39   מִכֹּל מַתְּנֹתֵיכֶם תָּרִימוּ אֵת כָּל־תְּרוּמַת יְהוָה מִכָּל־
40  חֶלְבּוֹ אֶת־מִקְדְּשׁוֹ מִמֶּנּוּ׃  וְאָמַרְתָּ אֲלֵהֶם בַּהֲרִימְכֶם
41   אֶת־חֶלְבּוֹ מִמֶּנּוּ וְנֶחְשַׁב לַלְוִיִּם כִּתְבוּאַת גֹּרֶן
42   וְכִתְבוּאַת יָקֶב׃ וַאֲכַלְתֶּם אֹתוֹ בְּכָל־מָקוֹם אַתֶּם
