 1  וּבֵיתְכֶם כִּי־שָׂכָר הוּא לָכֶם חֵלֶף עֲבֹדַתְכֶם בְּאֹהֶל
 2   מוֹעֵד׃ וְלֹא־תִשְׂאוּ עָלָיו חֵטְא בַּהֲרִימְכֶם אֶת־חֶלְבּוֹ
 3   מִמֶּנּוּ וְאֶת־קָדְשֵׁי בְנֵי־יִשְׂרָאֵל לֹא תְחַלְּלוּ וְלֹא
 4   תָמוּתוּ׃ [פ]
 5  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ זֹאת
 6   חֻקַּת הַתּוֹרָה אֲשֶׁר־צִוָּה יְהוָה לֵאמֹר דַּבֵּר  אֶל־בְּנֵי
 7   יִשְׂרָאֵל וְיִקְחוּ אֵלֶיךָ פָרָה אֲדֻמָּה תְּמִימָה אֲשֶׁר
 8   אֵין־בָּהּ מוּם אֲשֶׁר לֹא־עָלָה עָלֶיהָ עֹל׃ וּנְתַתֶּם
 9   אֹתָהּ אֶל־אֶלְעָזָר הַכֹּהֵן וְהוֹצִיא אֹתָהּ אֶל־מִחוּץ
10   לַמַּחֲנֶה וְשָׁחַט אֹתָהּ לְפָנָיו׃ וְלָקַח אֶלְעָזָר הַכֹּהֵן
11   מִדָּמָהּ בְּאֶצְבָּעוֹ וְהִזָּה אֶל־נֹכַח פְּנֵי אֹהֶל־מוֹעֵד
12   מִדָּמָהּ שֶׁבַע פְּעָמִים׃ וְשָׂרַף אֶת־הַפָּרָה לְעֵינָיו
13   אֶת־עֹרָהּ וְאֶת־בְּשָׂרָהּ וְאֶת־דָּמָהּ עַל־פִּרְשָׁהּ יִשְׂרֹף׃
14   וְלָקַח הַכֹּהֵן עֵץ אֶרֶז וְאֵזוֹב וּשְׁנִי תוֹלָעַת וְהִשְׁלִיךְ
15   אֶל־תּוֹךְ שְׂרֵפַת הַפָּרָה׃♢ וְכִבֶּס בְּגָדָיו הַכֹּהֵן וְרָחַץ
16   בְּשָׂרוֹ בַּמַּיִם וְאַחַר יָבֹא אֶל־הַמַּחֲנֶה וְטָמֵא הַכֹּהֵן
17   עַד־הָעָרֶב׃ וְהַשֹּׂרֵף אֹתָהּ יְכַבֵּס בְּגָדָיו בַּמַּיִם וְרָחַץ
18   בְּשָׂרוֹ בַּמָּיִם וְטָמֵא עַד־הָעָרֶב׃ וְאָסַף  אִישׁ טָהוֹר
19   אֵת אֵפֶר הַפָּרָה וְהִנִּיחַ מִחוּץ לַמַּחֲנֶה בְּמָקוֹם
20   טָהוֹר וְהָיְתָה לַעֲדַת בְּנֵי־יִשְׂרָאֵל לְמִשְׁמֶרֶת לְמֵי
21   נִדָּה חַטָּאת הִוא׃♢ וְכִבֶּס הָאֹסֵף אֶת־אֵפֶר הַפָּרָה
22   אֶת־בְּגָדָיו וְטָמֵא עַד־הָעָרֶב וְהָיְתָה לִבְנֵי יִשְׂרָאֵל
23   וְלַגֵּר הַגָּר בְּתוֹכָם לְחֻקַּת עוֹלָם׃ הַנֹּגֵעַ בְּמֵת לְכָל־
24  נֶפֶשׁ אָדָם וְטָמֵא שִׁבְעַת יָמִים׃ הוּא יִתְחַטָּא־בוֹ
25   בַּיּוֹם הַשְּׁלִישִׁי וּבַיּוֹם הַשְּׁבִיעִי יִטְהָר וְאִם־לֹא
26   יִתְחַטָּא בַּיּוֹם הַשְּׁלִישִׁי וּבַיּוֹם הַשְּׁבִיעִי לֹא יִטְהָר׃
27   כָּל־הַנֹּגֵעַ בְּמֵת בְּנֶפֶשׁ הָאָדָם אֲשֶׁר־יָמוּת וְלֹא
28   יִתְחַטָּא אֶת־מִשְׁכַּן יְהוָה טִמֵּא וְנִכְרְתָה הַנֶּפֶשׁ
29   הַהִוא מִיִּשְׂרָאֵל כִּי מֵי נִדָּה לֹא־זֹרַק עָלָיו טָמֵא
30   יִהְיֶה עוֹד טֻמְאָתוֹ בוֹ׃ זֹאת הַתּוֹרָה אָדָם כִּי־יָמוּת
31   בְּאֹהֶל כָּל־הַבָּא אֶל־הָאֹהֶל וְכָל־אֲשֶׁר בָּאֹהֶל
32   יִטְמָא שִׁבְעַת יָמִים׃ וְכֹל כְּלִי פָתוּחַ אֲשֶׁר אֵין־
33  צָמִיד פָּתִיל עָלָיו טָמֵא הוּא׃ וְכֹל אֲשֶׁר־יִגַּע עַל־
34  פְּנֵי הַשָּׂדֶה בַּחֲלַל־חֶרֶב אוֹ בְמֵת אוֹ־בְעֶצֶם אָדָם
35   אוֹ בְקָבֶר יִטְמָא שִׁבְעַת יָמִים׃ וְלָקְחוּ לַטָּמֵא מֵעֲפַר
36   שְׂרֵפַת הַחַטָּאת וְנָתַן עָלָיו מַיִם חַיִּים אֶל־כֶּלִי׃♢
37    וְלָקַח אֵזוֹב וְטָבַל בַּמַּיִם אִישׁ טָהוֹר וְהִזָּה עַל־
38  הָאֹהֶל וְעַל־כָּל־הַכֵּלִים וְעַל־הַנְּפָשׁוֹת אֲשֶׁר הָיוּ־
39  שָׁם וְעַל־הַנֹּגֵעַ בַּעֶצֶם אוֹ בֶחָלָל אוֹ בַמֵּת אוֹ בַקָּבֶר׃
40   וְהִזָּה הַטָּהֹר עַל־הַטָּמֵא בַּיּוֹם הַשְּׁלִישִׁי וּבַיּוֹם
41   הַשְּׁבִיעִי וְחִטְּאוֹ בַּיּוֹם הַשְּׁבִיעִי וְכִבֶּס בְּגָדָיו וְרָחַץ
42   בַּמַּיִם וְטָהֵר בָּעָרֶב׃ וְאִישׁ אֲשֶׁר־יִטְמָא וְלֹא יִתְחַטָּא
