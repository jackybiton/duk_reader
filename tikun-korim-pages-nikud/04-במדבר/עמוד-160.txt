 1  וּלְזֶבַח הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם חֲמִשָּׁה עַתֻּדִים
 2   חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה קָרְבַּן אֱלִיצוּר
 3   בֶּן־שְׁדֵיאוּר׃ [פ]
 4   בַּיּוֹם הַחֲמִישִׁי נָשִׂיא לִבְנֵי שִׁמְעוֹן שְׁלֻמִיאֵל בֶּן־
 5  צוּרִישַׁדָּי׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים וּמֵאָה
 6   מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל בְּשֶׁקֶל
 7   הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה בַשֶּׁמֶן לְמִנְחָה׃
 8   כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר
 9   אַיִל אֶחָד כֶּבֶשׂ־אֶחָד בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים
10   אֶחָד לְחַטָּאת׃ וּלְזֶבַח הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם
11   חֲמִשָּׁה עַתֻּדִים חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה
12   קָרְבַּן שְׁלֻמִיאֵל בֶּן־צוּרִישַׁדָּי׃ [פ]
13    בַּיּוֹם הַשִּׁשִּׁי נָשִׂיא לִבְנֵי גָד אֶלְיָסָף בֶּן־דְּעוּאֵל׃
14   קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים וּמֵאָה מִשְׁקָלָהּ
15   מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ
16   שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה בַשֶּׁמֶן לְמִנְחָה׃ כַּף
17   אַחַת עֲשָׂרָה זָהָב מְלֵאָה קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר
18   אַיִל אֶחָד כֶּבֶשׂ־אֶחָד בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים
19   אֶחָד לְחַטָּאת׃ וּלְזֶבַח הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם
20   חֲמִשָּׁה עַתֻּדִים חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה
21   זֶה קָרְבַּן אֶלְיָסָף בֶּן־דְּעוּאֵל׃ [פ]
22   בַּיּוֹם הַשְּׁבִיעִי נָשִׂיא לִבְנֵי אֶפְרָיִם אֱלִישָׁמָע בֶּן־
23  עַמִּיהוּד׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים וּמֵאָה
24   מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל בְּשֶׁקֶל
25   הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה בַשֶּׁמֶן לְמִנְחָה׃
26   כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר
27   אַיִל אֶחָד כֶּבֶשׂ־אֶחָד בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים
28   אֶחָד לְחַטָּאת׃ וּלְזֶבַח הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם
29   חֲמִשָּׁה עַתֻּדִים חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה
30   קָרְבַּן אֱלִישָׁמָע בֶּן־עַמִּיהוּד׃ [פ]
31   בַּיּוֹם הַשְּׁמִינִי נָשִׂיא לִבְנֵי מְנַשֶּׁה גַּמְלִיאֵל בֶּן־
32  פְּדָהצוּר׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים
33   וּמֵאָה מִשְׁקָלָהּ מִזְרָק אֶחָד כֶּסֶף שִׁבְעִים שֶׁקֶל
34   בְּשֶׁקֶל הַקֹּדֶשׁ שְׁנֵיהֶם  מְלֵאִים סֹלֶת בְּלוּלָה
35   בַשֶּׁמֶן לְמִנְחָה׃ כַּף אַחַת עֲשָׂרָה זָהָב מְלֵאָה
36   קְטֹרֶת׃ פַּר אֶחָד בֶּן־בָּקָר אַיִל אֶחָד כֶּבֶשׂ־אֶחָד
37   בֶּן־שְׁנָתוֹ לְעֹלָה׃ שְׂעִיר־עִזִּים אֶחָד לְחַטָּאת׃ וּלְזֶבַח
38   הַשְּׁלָמִים בָּקָר שְׁנַיִם אֵילִם חֲמִשָּׁה עַתֻּדִים
39   חֲמִשָּׁה כְּבָשִׂים בְּנֵי־שָׁנָה חֲמִשָּׁה זֶה קָרְבַּן
40   גַּמְלִיאֵל בֶּן־פְּדָהצוּר׃ [פ]
41   בַּיּוֹם הַתְּשִׁיעִי נָשִׂיא לִבְנֵי בִנְיָמִן אֲבִידָן בֶּן־
42  גִּדְעֹנִי׃ קָרְבָּנוֹ קַעֲרַת־כֶּסֶף אַחַת שְׁלֹשִׁים
