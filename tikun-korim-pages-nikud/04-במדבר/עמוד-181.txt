 1  וּבָלָק בֶּן־צִפּוֹר מֶלֶךְ לְמוֹאָב בָּעֵת הַהִוא׃♢ וַיִּשְׁלַח
 2   מַלְאָכִים אֶל־בִּלְעָם בֶּן־בְּעֹר פְּתוֹרָה אֲשֶׁר עַל־
 3  הַנָּהָר אֶרֶץ בְּנֵי־עַמּוֹ לִקְרֹא־לוֹ לֵאמֹר הִנֵּה עַם
 4   יָצָא מִמִּצְרַיִם הִנֵּה כִסָּה אֶת־עֵין הָאָרֶץ וְהוּא יֹשֵׁב
 5   מִמֻּלִי׃ וְעַתָּה לְכָה־נָּא אָרָה־לִּי אֶת־הָעָם הַזֶּה כִּי־
 6  עָצוּם הוּא מִמֶּנִּי אוּלַי אוּכַל נַכֶּה־בּוֹ וַאֲגָרְשֶׁנּוּ מִן־
 7  הָאָרֶץ כִּי יָדַעְתִּי אֵת אֲשֶׁר־תְּבָרֵךְ מְבֹרָךְ וַאֲשֶׁר
 8   תָּאֹר יוּאָר׃ וַיֵּלְכוּ זִקְנֵי מוֹאָב וְזִקְנֵי מִדְיָן וּקְסָמִים
 9   בְּיָדָם וַיָּבֹאוּ אֶל־בִּלְעָם וַיְדַבְּרוּ אֵלָיו דִּבְרֵי בָלָק׃♢
10   וַיֹּאמֶר אֲלֵיהֶם לִינוּ פֹה הַלַּיְלָה וַהֲשִׁבֹתִי אֶתְכֶם
11   דָּבָר כַּאֲשֶׁר יְדַבֵּר יְהוָה אֵלָי וַיֵּשְׁבוּ שָׂרֵי־מוֹאָב
12   עִם־בִּלְעָם׃ וַיָּבֹא אֱלֹהִים אֶל־בִּלְעָם וַיֹּאמֶר מִי
13   הָאֲנָשִׁים הָאֵלֶּה עִמָּךְ׃ וַיֹּאמֶר בִּלְעָם אֶל־הָאֱלֹהִים
14   בָּלָק בֶּן־צִפֹּר מֶלֶךְ מוֹאָב שָׁלַח אֵלָי׃ הִנֵּה הָעָם
15   הַיֹּצֵא מִמִּצְרַיִם וַיְכַס אֶת־עֵין הָאָרֶץ עַתָּה לְכָה
16   קָבָה־לִּי אֹתוֹ אוּלַי אוּכַל לְהִלָּחֶם בּוֹ וְגֵרַשְׁתִּיו׃
17   וַיֹּאמֶר אֱלֹהִים אֶל־בִּלְעָם לֹא תֵלֵךְ עִמָּהֶם לֹא
18   תָאֹר אֶת־הָעָם כִּי בָרוּךְ הוּא׃♢  וַיָּקָם בִּלְעָם בַּבֹּקֶר
19   וַיֹּאמֶר אֶל־שָׂרֵי בָלָק לְכוּ אֶל־אַרְצְכֶם כִּי מֵאֵן
20   יְהוָה לְתִתִּי לַהֲלֹךְ עִמָּכֶם׃ וַיָּקוּמוּ שָׂרֵי מוֹאָב וַיָּבֹאוּ
21   אֶל־בָּלָק וַיֹּאמְרוּ מֵאֵן בִּלְעָם הֲלֹךְ עִמָּנוּ׃ וַיֹּסֶף עוֹד
22   בָּלָק שְׁלֹחַ שָׂרִים רַבִּים וְנִכְבָּדִים מֵאֵלֶּה׃ וַיָּבֹאוּ
23   אֶל־בִּלְעָם וַיֹּאמְרוּ לוֹ כֹּה אָמַר בָּלָק בֶּן־צִפּוֹר אַל־
24  נָא תִמָּנַע מֵהֲלֹךְ אֵלָי׃ כִּי־כַבֵּד אֲכַבֶּדְךָ מְאֹד וְכֹל
25   אֲשֶׁר־תֹּאמַר אֵלַי אֶעֱשֶׂה וּלְכָה־נָּא קָבָה־לִּי אֵת
26   הָעָם הַזֶּה׃ וַיַּעַן בִּלְעָם וַיֹּאמֶר אֶל־עַבְדֵי בָלָק אִם־
27  יִתֶּן־לִי בָלָק מְלֹא בֵיתוֹ כֶּסֶף וְזָהָב לֹא אוּכַל לַעֲבֹר
28   אֶת־פִּי יְהוָה אֱלֹהָי לַעֲשׂוֹת קְטַנָּה אוֹ גְדוֹלָה׃ וְעַתָּה
29   שְׁבוּ נָא בָזֶה גַּם־אַתֶּם הַלָּיְלָה וְאֵדְעָה מַה־יֹּסֵף יְהוָה
30   דַּבֵּר עִמִּי׃ וַיָּבֹא אֱלֹהִים  אֶל־בִּלְעָם לַיְלָה וַיֹּאמֶר
31   לוֹ אִם־לִקְרֹא לְךָ בָּאוּ הָאֲנָשִׁים קוּם לֵךְ אִתָּם
32   וְאַךְ אֶת־הַדָּבָר אֲשֶׁר־אֲדַבֵּר אֵלֶיךָ אֹתוֹ תַעֲשֶׂה׃
33    וַיָּקָם בִּלְעָם בַּבֹּקֶר וַיַּחֲבֹשׁ אֶת־אֲתֹנוֹ וַיֵּלֶךְ עִם־
34  שָׂרֵי מוֹאָב׃ וַיִּחַר־אַף אֱלֹהִים כִּי־הוֹלֵךְ הוּא וַיִּתְיַצֵּב
35   מַלְאַךְ יְהוָה בַּדֶּרֶךְ לְשָׂטָן לוֹ וְהוּא רֹכֵב עַל־אֲתֹנוֹ
36   וּשְׁנֵי נְעָרָיו עִמּוֹ׃ וַתֵּרֶא הָאָתוֹן אֶת־מַלְאַךְ יְהוָה
37   נִצָּב בַּדֶּרֶךְ וְחַרְבּוֹ שְׁלוּפָה בְּיָדוֹ וַתֵּט הָאָתוֹן מִן־
38  הַדֶּרֶךְ וַתֵּלֶךְ בַּשָּׂדֶה וַיַּךְ בִּלְעָם אֶת־הָאָתוֹן לְהַטֹּתָהּ
39   הַדָּרֶךְ׃ וַיַּעֲמֹד מַלְאַךְ יְהוָה בְּמִשְׁעוֹל הַכְּרָמִים
40   גָּדֵר מִזֶּה וְגָדֵר מִזֶּה׃ וַתֵּרֶא הָאָתוֹן אֶת־מַלְאַךְ יְהוָה
41   וַתִּלָּחֵץ אֶל־הַקִּיר וַתִּלְחַץ אֶת־רֶגֶל בִּלְעָם אֶל־
42  הַקִּיר וַיֹּסֶף לְהַכֹּתָהּ׃ וַיּוֹסֶף מַלְאַךְ־יְהוָה עֲבוֹר
