 1  וַיִּסְעוּ מֵחַשְׁמֹנָה וַיַּחֲנוּ בְּמֹסֵרוֹת׃ וַיִּסְעוּ מִמֹּסֵרוֹת
 2   וַיַּחֲנוּ בִּבְנֵי יַעֲקָן׃ וַיִּסְעוּ מִבְּנֵי יַעֲקָן וַיַּחֲנוּ בְּחֹר הַגִּדְגָּד׃
 3   וַיִּסְעוּ מֵחֹר הַגִּדְגָּד וַיַּחֲנוּ בְּיָטְבָתָה׃ וַיִּסְעוּ מִיָּטְבָתָה
 4   וַיַּחֲנוּ בְּעַבְרֹנָה׃ וַיִּסְעוּ מֵעַבְרֹנָה וַיַּחֲנוּ בְּעֶצְיֹן גָּבֶר׃
 5   וַיִּסְעוּ מֵעֶצְיֹן גָּבֶר וַיַּחֲנוּ בְמִדְבַּר־צִן הִוא קָדֵשׁ׃
 6   וַיִּסְעוּ מִקָּדֵשׁ וַיַּחֲנוּ בְּהֹר הָהָר בִּקְצֵה אֶרֶץ אֱדוֹם׃
 7   וַיַּעַל אַהֲרֹן הַכֹּהֵן אֶל־הֹר הָהָר עַל־פִּי יְהוָה
 8   וַיָּמָת שָׁם בִּשְׁנַת הָאַרְבָּעִים לְצֵאת בְּנֵי־יִשְׂרָאֵל
 9   מֵאֶרֶץ מִצְרַיִם בַּחֹדֶשׁ הַחֲמִישִׁי בְּאֶחָד לַחֹדֶשׁ׃
10   וְאַהֲרֹן בֶּן־שָׁלֹשׁ וְעֶשְׂרִים וּמְאַת שָׁנָה בְּמֹתוֹ
11   בְּהֹר הָהָר׃ [ס]             וַיִּשְׁמַע הַכְּנַעֲנִי
12   מֶלֶךְ עֲרָד וְהוּא־יֹשֵׁב בַּנֶּגֶב בְּאֶרֶץ כְּנָעַן בְּבֹא בְּנֵי
13   יִשְׂרָאֵל׃ וַיִּסְעוּ מֵהֹר הָהָר וַיַּחֲנוּ בְּצַלְמֹנָה׃ וַיִּסְעוּ
14   מִצַּלְמֹנָה וַיַּחֲנוּ בְּפוּנֹן׃ וַיִּסְעוּ מִפּוּנֹן וַיַּחֲנוּ בְּאֹבֹת׃
15   וַיִּסְעוּ מֵאֹבֹת וַיַּחֲנוּ בְּעִיֵּי הָעֲבָרִים בִּגְבוּל מוֹאָב׃
16   וַיִּסְעוּ מֵעִיִּים וַיַּחֲנוּ בְּדִיבֹן גָּד׃ וַיִּסְעוּ מִדִּיבֹן גָּד וַיַּחֲנוּ
17   בְּעַלְמֹן דִּבְלָתָיְמָה׃ וַיִּסְעוּ מֵעַלְמֹן דִּבְלָתָיְמָה וַיַּחֲנוּ
18   בְּהָרֵי הָעֲבָרִים לִפְנֵי נְבוֹ׃ וַיִּסְעוּ מֵהָרֵי הָעֲבָרִים
19   וַיַּחֲנוּ בְּעַרְבֹת מוֹאָב עַל יַרְדֵּן יְרֵחוֹ׃ וַיַּחֲנוּ עַל־
20  הַיַּרְדֵּן מִבֵּית הַיְשִׁמֹת עַד אָבֵל הַשִּׁטִּים בְּעַרְבֹת
21   מוֹאָב׃ [ס]              וַיְדַבֵּר יְהוָה אֶל־
22  מֹשֶׁה בְּעַרְבֹת מוֹאָב עַל־יַרְדֵּן יְרֵחוֹ לֵאמֹר׃ דַּבֵּר
23   אֶל־בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם כִּי אַתֶּם עֹבְרִים
24   אֶת־הַיַּרְדֵּן אֶל־אֶרֶץ כְּנָעַן׃ וְהוֹרַשְׁתֶּם אֶת־כָּל־יֹשְׁבֵי
25   הָאָרֶץ מִפְּנֵיכֶם וְאִבַּדְתֶּם אֵת כָּל־מַשְׂכִּיֹּתָם וְאֵת כָּל־
26  צַלְמֵי מַסֵּכֹתָם תְּאַבֵּדוּ וְאֵת כָּל־בָּמוֹתָם תַּשְׁמִידוּ׃
27   וְהוֹרַשְׁתֶּם אֶת־הָאָרֶץ וִישַׁבְתֶּם־בָּהּ כִּי לָכֶם נָתַתִּי
28   אֶת־הָאָרֶץ לָרֶשֶׁת אֹתָהּ׃ וְהִתְנַחַלְתֶּם אֶת־הָאָרֶץ
29   בְּגוֹרָל לְמִשְׁפְּחֹתֵיכֶם לָרַב תַּרְבּוּ אֶת־נַחֲלָתוֹ וְלַמְעַט
30   תַּמְעִיט אֶת־נַחֲלָתוֹ אֶל אֲשֶׁר־יֵצֵא לוֹ שָׁמָּה הַגּוֹרָל
31   לוֹ יִהְיֶה לְמַטּוֹת אֲבֹתֵיכֶם תִּתְנֶחָלוּ׃ וְאִם־לֹא תוֹרִישׁוּ
32   אֶת־יֹשְׁבֵי הָאָרֶץ מִפְּנֵיכֶם וְהָיָה אֲשֶׁר תּוֹתִירוּ מֵהֶם
33   לְשִׂכִּים בְּעֵינֵיכֶם וְלִצְנִינִם בְּצִדֵּיכֶם וְצָרְרוּ אֶתְכֶם
34   עַל־הָאָרֶץ אֲשֶׁר אַתֶּם יֹשְׁבִים בָּהּ׃ וְהָיָה כַּאֲשֶׁר
35   דִּמִּיתִי לַעֲשׂוֹת לָהֶם אֶעֱשֶׂה לָכֶם׃ [פ]
36   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ צַו אֶת־בְּנֵי יִשְׂרָאֵל
37   וְאָמַרְתָּ אֲלֵהֶם כִּי־אַתֶּם בָּאִים אֶל־הָאָרֶץ כְּנָעַן
38   זֹאת הָאָרֶץ אֲשֶׁר תִּפֹּל לָכֶם בְּנַחֲלָה אֶרֶץ כְּנַעַן
39   לִגְבֻלֹתֶיהָ׃ וְהָיָה לָכֶם פְּאַת־נֶגֶב מִמִּדְבַּר־צִן עַל־יְדֵי
40   אֱדוֹם וְהָיָה לָכֶם גְּבוּל נֶגֶב מִקְצֵה יָם־הַמֶּלַח קֵדְמָה׃
41   וְנָסַב לָכֶם הַגְּבוּל מִנֶּגֶב לְמַעֲלֵה עַקְרַבִּים וְעָבַר צִנָה
42   וְהָיוּ תּוֹצְאֹתָיו מִנֶּגֶב לְקָדֵשׁ בַּרְנֵעַ וְיָצָא חֲצַר־אַדָּר
