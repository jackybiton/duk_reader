 1  וְהָאֲנָשִׁים אֲשֶׁר־עָלוּ עִמּוֹ אָמְרוּ לֹא נוּכַל לַעֲלוֹת
 2   אֶל־הָעָם כִּי־חָזָק הוּא מִמֶּנּוּ׃ וַיֹּצִיאוּ דִּבַּת הָאָרֶץ
 3   אֲשֶׁר תָּרוּ אֹתָהּ אֶל־בְּנֵי יִשְׂרָאֵל לֵאמֹר הָאָרֶץ
 4   אֲשֶׁר עָבַרְנוּ בָהּ לָתוּר אֹתָהּ אֶרֶץ אֹכֶלֶת יוֹשְׁבֶיהָ
 5   הִוא וְכָל־הָעָם אֲשֶׁר־רָאִינוּ בְתוֹכָהּ אַנְשֵׁי מִדּוֹת׃
 6   וְשָׁם רָאִינוּ אֶת־הַנְּפִילִים בְּנֵי עֲנָק מִן־הַנְּפִלִים
 7   וַנְּהִי בְעֵינֵינוּ כַּחֲגָבִים וְכֵן הָיִינוּ בְּעֵינֵיהֶם׃ וַתִּשָּׂא
 8   כָּל־הָעֵדָה וַיִּתְּנוּ אֶת־קוֹלָם וַיִּבְכּוּ הָעָם בַּלַּיְלָה
 9   הַהוּא׃ וַיִּלֹּנוּ עַל־מֹשֶׁה וְעַל־אַהֲרֹן כֹּל בְּנֵי יִשְׂרָאֵל
10   וַיֹּאמְרוּ אֲלֵהֶם כָּל־הָעֵדָה לוּ־מַתְנוּ בְּאֶרֶץ מִצְרַיִם
11   אוֹ בַּמִּדְבָּר הַזֶּה לוּ־מָתְנוּ׃ וְלָמָה יְהוָה מֵבִיא אֹתָנוּ
12   אֶל־הָאָרֶץ הַזֹּאת לִנְפֹּל בַּחֶרֶב נָשֵׁינוּ וְטַפֵּנוּ יִהְיוּ
13   לָבַז הֲלוֹא טוֹב לָנוּ שׁוּב מִצְרָיְמָה׃ וַיֹּאמְרוּ אִישׁ
14   אֶל־אָחִיו נִתְּנָה רֹאשׁ וְנָשׁוּבָה מִצְרָיְמָה׃ וַיִּפֹּל מֹשֶׁה
15   וְאַהֲרֹן עַל־פְּנֵיהֶם לִפְנֵי כָּל־קְהַל עֲדַת בְּנֵי יִשְׂרָאֵל׃
16   וִיהוֹשֻׁעַ בִּן־נוּן וְכָלֵב בֶּן־יְפֻנֶּה מִן־הַתָּרִים אֶת־הָאָרֶץ
17   קָרְעוּ בִּגְדֵיהֶם׃ וַיֹּאמְרוּ אֶל־כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל
18   לֵאמֹר הָאָרֶץ אֲשֶׁר עָבַרְנוּ בָהּ לָתוּר אֹתָהּ טוֹבָה
19   הָאָרֶץ מְאֹד מְאֹד׃  אִם־חָפֵץ בָּנוּ יְהוָה וְהֵבִיא אֹתָנוּ
20   אֶל־הָאָרֶץ הַזֹּאת וּנְתָנָהּ לָנוּ אֶרֶץ אֲשֶׁר־הִוא זָבַת
21   חָלָב וּדְבָשׁ׃ אַךְ בַּיהוָה אַל־תִּמְרֹדוּ וְאַתֶּם אַל־תִּירְאוּ
22   אֶת־עַם הָאָרֶץ כִּי לַחְמֵנוּ הֵם סָר צִלָּם מֵעֲלֵיהֶם
23   וַיהוָה אִתָּנוּ אַל־תִּירָאֻם׃ וַיֹּאמְרוּ כָּל־הָעֵדָה לִרְגּוֹם
24   אֹתָם בָּאֲבָנִים וּכְבוֹד יְהוָה נִרְאָה בְּאֹהֶל מוֹעֵד
25   אֶל־כָּל־בְּנֵי יִשְׂרָאֵל׃ [פ]
26   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה עַד־אָנָה יְנַאֲצֻנִי הָעָם הַזֶּה
27   וְעַד־אָנָה לֹא־יַאֲמִינוּ בִי בְּכֹל הָאֹתוֹת אֲשֶׁר עָשִׂיתִי
28   בְּקִרְבּוֹ׃ אַכֶּנּוּ בַדֶּבֶר וְאוֹרִשֶׁנּוּ וְאֶעֱשֶׂה אֹתְךָ לְגוֹי־
29  גָּדוֹל וְעָצוּם מִמֶּנּוּ׃ וַיֹּאמֶר מֹשֶׁה אֶל־יְהוָה וְשָׁמְעוּ
30   מִצְרַיִם כִּי־הֶעֱלִיתָ בְכֹחֲךָ אֶת־הָעָם הַזֶּה מִקִּרְבּוֹ׃
31   וְאָמְרוּ אֶל־יוֹשֵׁב הָאָרֶץ הַזֹּאת שָׁמְעוּ כִּי־אַתָּה יְהוָה
32   בְּקֶרֶב הָעָם הַזֶּה אֲשֶׁר־עַיִן בְּעַיִן נִרְאָה  אַתָּה יְהוָה
33   וַעֲנָנְךָ עֹמֵד עֲלֵהֶם וּבְעַמֻּד עָנָן אַתָּה הֹלֵךְ לִפְנֵיהֶם
34   יוֹמָם וּבְעַמּוּד אֵשׁ לָיְלָה׃ וְהֵמַתָּה אֶת־הָעָם הַזֶּה
35   כְּאִישׁ אֶחָד וְאָמְרוּ הַגּוֹיִם אֲשֶׁר־שָׁמְעוּ אֶת־שִׁמְעֲךָ
36   לֵאמֹר׃ מִבִּלְתִּי יְכֹלֶת יְהוָה לְהָבִיא אֶת־הָעָם הַזֶּה
37   אֶל־הָאָרֶץ אֲשֶׁר־נִשְׁבַּע לָהֶם וַיִּשְׁחָטֵם בַּמִּדְבָּר׃
38   וְעַתָּה יִגְדַּל־נָא כֹּחַ אֲדֹנָי כַּאֲשֶׁר דִּבַּרְתָּ לֵאמֹר׃
39   יְהוָה אֶרֶךְ אַפַּיִם וְרַב־חֶסֶד נֹשֵׂא עָו‍ֹן וָפָשַׁע וְנַקֵּה
40   לֹא יְנַקֶּה פֹּקֵד עֲו‍ֹן אָבוֹת עַל־בָּנִים עַל־שִׁלֵּשִׁים
41   וְעַל־רִבֵּעִים׃ סְלַח־נָא לַעֲו‍ֹן הָעָם הַזֶּה כְּגֹדֶל חַסְדֶּךָ
42   וְכַאֲשֶׁר נָשָׂאתָה לָעָם הַזֶּה מִמִּצְרַיִם וְעַד־הֵנָּה׃
