 1  וְעָשׂוּ אֹתוֹ עֻגוֹת וְהָיָה טַעְמוֹ כְּטַעַם לְשַׁד הַשָּׁמֶן׃
 2   וּבְרֶדֶת הַטַּל עַל־הַמַּחֲנֶה לָיְלָה יֵרֵד הַמָּן עָלָיו׃
 3   וַיִּשְׁמַע מֹשֶׁה אֶת־הָעָם בֹּכֶה לְמִשְׁפְּחֹתָיו אִישׁ
 4   לְפֶתַח אָהֳלוֹ וַיִּחַר־אַף יְהוָה מְאֹד וּבְעֵינֵי מֹשֶׁה רָע׃
 5   וַיֹּאמֶר מֹשֶׁה אֶל־יְהוָה לָמָה הֲרֵעֹתָ לְעַבְדֶּךָ וְלָמָּה
 6   לֹא־מָצָתִי חֵן בְּעֵינֶיךָ לָשׂוּם אֶת־מַשָּׂא כָּל־הָעָם
 7   הַזֶּה עָלָי׃ הֶאָנֹכִי הָרִיתִי אֵת כָּל־הָעָם הַזֶּה אִם־אָנֹכִי
 8   יְלִדְתִּיהוּ כִּי־תֹאמַר אֵלַי שָׂאֵהוּ בְחֵיקֶךָ כַּאֲשֶׁר
 9   יִשָּׂא הָאֹמֵן אֶת־הַיֹּנֵק עַל הָאֲדָמָה אֲשֶׁר נִשְׁבַּעְתָּ
10   לַאֲבֹתָיו׃ מֵאַיִן לִי בָּשָׂר לָתֵת לְכָל־הָעָם הַזֶּה כִּי־
11  יִבְכּוּ עָלַי לֵאמֹר תְּנָה־לָּנוּ בָשָׂר וְנֹאכֵלָה׃ לֹא־אוּכַל
12   אָנֹכִי לְבַדִּי לָשֵׂאת אֶת־כָּל־הָעָם הַזֶּה כִּי כָבֵד מִמֶּנִּי׃
13   וְאִם־כָּכָה  אַתְּ־עֹשֶׂה לִּי הָרְגֵנִי נָא הָרֹג אִם־מָצָאתִי
14   חֵן בְּעֵינֶיךָ וְאַל־אֶרְאֶה בְּרָעָתִי׃ [פ]
15   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה אֶסְפָה־לִּי שִׁבְעִים אִישׁ
16   מִזִּקְנֵי יִשְׂרָאֵל אֲשֶׁר יָדַעְתָּ כִּי־הֵם זִקְנֵי הָעָם
17   וְשֹׁטְרָיו וְלָקַחְתָּ אֹתָם אֶל־אֹהֶל מוֹעֵד וְהִתְיַצְּבוּ
18   שָׁם עִמָּךְ׃ וְיָרַדְתִּי וְדִבַּרְתִּי עִמְּךָ שָׁם וְאָצַלְתִּי מִן־
19  הָרוּחַ אֲשֶׁר עָלֶיךָ וְשַׂמְתִּי עֲלֵיהֶם וְנָשְׂאוּ אִתְּךָ
20   בְּמַשָּׂא הָעָם וְלֹא־תִשָּׂא אַתָּה לְבַדֶּךָ׃ וְאֶל־הָעָם
21   תֹּאמַר הִתְקַדְּשׁוּ לְמָחָר וַאֲכַלְתֶּם בָּשָׂר כִּי בְּכִיתֶם
22   בְּאָזְנֵי יְהוָה לֵאמֹר מִי יַאֲכִלֵנוּ בָּשָׂר כִּי־טוֹב לָנוּ
23   בְּמִצְרָיִם וְנָתַן יְהוָה לָכֶם בָּשָׂר וַאֲכַלְתֶּם׃ לֹא יוֹם
24   אֶחָד תֹּאכְלוּן וְלֹא יוֹמָיִם וְלֹא  חֲמִשָּׁה יָמִים וְלֹא
25   עֲשָׂרָה יָמִים וְלֹא עֶשְׂרִים יוֹם׃ עַד  חֹדֶשׁ יָמִים
26   עַד אֲשֶׁר־יֵצֵא מֵאַפְּכֶם וְהָיָה לָכֶם לְזָרָא יַעַן כִּי־
27  מְאַסְתֶּם אֶת־יְהוָה אֲשֶׁר בְּקִרְבְּכֶם וַתִּבְכּוּ לְפָנָיו
28   לֵאמֹר לָמָּה זֶּה יָצָאנוּ מִמִּצְרָיִם׃ וַיֹּאמֶר מֹשֶׁה שֵׁשׁ־
29  מֵאוֹת אֶלֶף רַגְלִי הָעָם אֲשֶׁר אָנֹכִי בְּקִרְבּוֹ וְאַתָּה
30   אָמַרְתָּ בָּשָׂר אֶתֵּן לָהֶם וְאָכְלוּ חֹדֶשׁ יָמִים׃ הֲצֹאן
31   וּבָקָר יִשָּׁחֵט לָהֶם וּמָצָא לָהֶם אִם אֶת־כָּל־דְּגֵי הַיָּם
32   יֵאָסֵף לָהֶם וּמָצָא לָהֶם׃ [פ]
33   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה הֲיַד יְהוָה תִּקְצָר עַתָּה
34   תִרְאֶה הֲיִקְרְךָ דְבָרִי אִם־לֹא׃ וַיֵּצֵא מֹשֶׁה וַיְדַבֵּר
35   אֶל־הָעָם אֵת דִּבְרֵי יְהוָה וַיֶּאֱסֹף שִׁבְעִים אִישׁ
36   מִזִּקְנֵי הָעָם וַיַּעֲמֵד אֹתָם סְבִיבֹת הָאֹהֶל׃ וַיֵּרֶד יְהוָה
37   בֶּעָנָן וַיְדַבֵּר אֵלָיו וַיָּאצֶל מִן־הָרוּחַ אֲשֶׁר עָלָיו
38   וַיִּתֵּן עַל־שִׁבְעִים אִישׁ הַזְּקֵנִים וַיְהִי כְּנוֹחַ עֲלֵיהֶם
39   הָרוּחַ וַיִּתְנַבְּאוּ וְלֹא יָסָפוּ׃ וַיִּשָּׁאֲרוּ שְׁנֵי־אֲנָשִׁים
40   בַּמַּחֲנֶה שֵׁם הָאֶחָד  אֶלְדָּד וְשֵׁם הַשֵּׁנִי מֵידָד
41   וַתָּנַח עֲלֵהֶם הָרוּחַ וְהֵמָּה בַּכְּתֻבִים וְלֹא יָצְאוּ
42   הָאֹהֱלָה וַיִּתְנַבְּאוּ בַּמַּחֲנֶה׃ וַיָּרָץ הַנַּעַר וַיַּגֵּד לְמֹשֶׁה
