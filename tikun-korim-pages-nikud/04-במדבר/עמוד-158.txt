 1  וְהַיָּמִים הָרִאשֹׁנִים יִפְּלוּ כִּי טָמֵא נִזְרוֹ׃ וְזֹאת תּוֹרַת
 2   הַנָּזִיר בְּיוֹם מְלֹאת יְמֵי נִזְרוֹ יָבִיא אֹתוֹ אֶל־פֶּתַח
 3   אֹהֶל מוֹעֵד׃ וְהִקְרִיב אֶת־קָרְבָּנוֹ לַיהוָה כֶּבֶשׂ בֶּן־
 4  שְׁנָתוֹ תָמִים אֶחָד לְעֹלָה וְכַבְשָׂה אַחַת בַּת־שְׁנָתָהּ
 5   תְּמִימָה לְחַטָּאת וְאַיִל־אֶחָד תָּמִים לִשְׁלָמִים׃ וְסַל
 6   מַצּוֹת סֹלֶת חַלֹּת בְּלוּלֹת בַּשֶּׁמֶן וּרְקִיקֵי מַצּוֹת
 7   מְשֻׁחִים בַּשָּׁמֶן וּמִנְחָתָם וְנִסְכֵּיהֶם׃ וְהִקְרִיב הַכֹּהֵן
 8   לִפְנֵי יְהוָה וְעָשָׂה אֶת־חַטָּאתוֹ וְאֶת־עֹלָתוֹ׃ וְאֶת־הָאַיִל
 9   יַעֲשֶׂה זֶבַח שְׁלָמִים לַיהוָה עַל סַל הַמַּצּוֹת וְעָשָׂה
10   הַכֹּהֵן אֶת־מִנְחָתוֹ וְאֶת־נִסְכּוֹ׃ וְגִלַּח הַנָּזִיר פֶּתַח אֹהֶל
11   מוֹעֵד אֶת־רֹאשׁ נִזְרוֹ וְלָקַח אֶת־שְׂעַר רֹאשׁ נִזְרוֹ
12   וְנָתַן עַל־הָאֵשׁ אֲשֶׁר־תַּחַת זֶבַח הַשְּׁלָמִים׃ וְלָקַח
13   הַכֹּהֵן אֶת־הַזְּרֹעַ בְּשֵׁלָה מִן־הָאַיִל וְחַלַּת מַצָּה אַחַת
14   מִן־הַסַּל וּרְקִיק מַצָּה אֶחָד וְנָתַן עַל־כַּפֵּי הַנָּזִיר
15   אַחַר הִתְגַּלְּחוֹ אֶת־נִזְרוֹ׃ וְהֵנִיף אוֹתָם הַכֹּהֵן  תְּנוּפָה
16   לִפְנֵי יְהוָה קֹדֶשׁ הוּא לַכֹּהֵן עַל חֲזֵה הַתְּנוּפָה וְעַל
17   שׁוֹק הַתְּרוּמָה וְאַחַר יִשְׁתֶּה הַנָּזִיר יָיִן׃ זֹאת תּוֹרַת
18   הַנָּזִיר אֲשֶׁר יִדֹּר קָרְבָּנוֹ לַיהוָה עַל־נִזְרוֹ מִלְּבַד
19   אֲשֶׁר־תַּשִּׂיג יָדוֹ כְּפִי נִדְרוֹ אֲשֶׁר יִדֹּר כֵּן יַעֲשֶׂה
20   עַל תּוֹרַת נִזְרוֹ׃ [פ]
21   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־אַהֲרֹן
22   וְאֶל־בָּנָיו לֵאמֹר כֹּה תְבָרְכוּ אֶת־בְּנֵי יִשְׂרָאֵל
23   אָמוֹר לָהֶם׃ [ס]             יְבָרֶכְךָ יְהוָה
24   וְיִשְׁמְרֶךָ׃ [ס]             יָאֵר יְהוָה  פָּנָיו אֵלֶיךָ
25   וִיחֻנֶּךָּ׃ [ס]             יִשָּׂא יְהוָה  פָּנָיו אֵלֶיךָ וְיָשֵׂם
26   לְךָ שָׁלוֹם׃ [ס]             וְשָׂמוּ אֶת־שְׁמִי עַל־
27  בְּנֵי יִשְׂרָאֵל וַאֲנִי אֲבָרְכֵם׃ [ס]              וַיְהִי
28   בְּיוֹם כַּלּוֹת מֹשֶׁה לְהָקִים אֶת־הַמִּשְׁכָּן וַיִּמְשַׁח
29   אֹתוֹ וַיְקַדֵּשׁ אֹתוֹ וְאֶת־כָּל־כֵּלָיו וְאֶת־הַמִּזְבֵּחַ וְאֶת־
30  כָּל־כֵּלָיו וַיִּמְשָׁחֵם וַיְקַדֵּשׁ אֹתָם׃ וַיַּקְרִיבוּ נְשִׂיאֵי
31   יִשְׂרָאֵל רָאשֵׁי בֵּית אֲבֹתָם הֵם נְשִׂיאֵי הַמַּטֹּת הֵם
32   הָעֹמְדִים עַל־הַפְּקֻדִים׃ וַיָּבִיאוּ אֶת־קָרְבָּנָם לִפְנֵי יְהוָה
33   שֵׁשׁ־עֶגְלֹת צָב וּשְׁנֵי עָשָׂר בָּקָר עֲגָלָה עַל־שְׁנֵי
34   הַנְּשִׂאִים וְשׁוֹר לְאֶחָד וַיַּקְרִיבוּ אוֹתָם לִפְנֵי הַמִּשְׁכָּן׃
35   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ קַח מֵאִתָּם וְהָיוּ לַעֲבֹד
36   אֶת־עֲבֹדַת אֹהֶל מוֹעֵד וְנָתַתָּה אוֹתָם אֶל־הַלְוִיִּם
37   אִישׁ כְּפִי עֲבֹדָתוֹ׃ וַיִּקַּח מֹשֶׁה אֶת־הָעֲגָלֹת וְאֶת־
38  הַבָּקָר וַיִּתֵּן אוֹתָם אֶל־הַלְוִיִּם׃ אֵת  שְׁתֵּי הָעֲגָלוֹת וְאֵת
39   אַרְבַּעַת הַבָּקָר נָתַן לִבְנֵי גֵרְשׁוֹן כְּפִי עֲבֹדָתָם׃ וְאֵת
40   אַרְבַּע הָעֲגָלֹת וְאֵת שְׁמֹנַת הַבָּקָר נָתַן לִבְנֵי מְרָרִי
41   כְּפִי עֲבֹדָתָם בְּיַד אִיתָמָר בֶּן־אַהֲרֹן הַכֹּהֵן׃ וְלִבְנֵי
42   קְהָת לֹא נָתָן כִּי־עֲבֹדַת הַקֹּדֶשׁ עֲלֵהֶם בַּכָּתֵף יִשָּׂאוּ׃
