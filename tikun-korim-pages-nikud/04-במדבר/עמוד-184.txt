 1  מַה־טֹּבוּ אֹהָלֶיךָ יַעֲקֹב מִשְׁכְּנֹתֶיךָ יִשְׂרָאֵל׃ כִּנְחָלִים
 2   נִטָּיוּ כְּגַנֹּת עֲלֵי נָהָר כַּאֲהָלִים נָטַע יְהוָה כַּאֲרָזִים
 3   עֲלֵי־מָיִם׃ יִזַּל־מַיִם מִדָּלְיָו וְזַרְעוֹ בְּמַיִם רַבִּים וְיָרֹם
 4   מֵאֲגַג מַלְכּוֹ וְתִנַּשֵּׂא מַלְכֻתוֹ׃ אֵל מוֹצִיאוֹ מִמִּצְרַיִם
 5   כְּתוֹעֲפֹת רְאֵם לוֹ יֹאכַל גּוֹיִם צָרָיו וְעַצְמֹתֵיהֶם יְגָרֵם
 6   וְחִצָּיו יִמְחָץ׃ כָּרַע שָׁכַב כַּאֲרִי וּכְלָבִיא מִי יְקִימֶנּוּ
 7   מְבָרְכֶיךָ בָרוּךְ וְאֹרְרֶיךָ אָרוּר׃ וַיִּחַר־אַף בָּלָק אֶל־
 8  בִּלְעָם וַיִּסְפֹּק אֶת־כַּפָּיו וַיֹּאמֶר בָּלָק אֶל־בִּלְעָם לָקֹב
 9   אֹיְבַי קְרָאתִיךָ וְהִנֵּה בֵּרַכְתָּ בָרֵךְ זֶה שָׁלֹשׁ פְּעָמִים׃
10   וְעַתָּה בְּרַח־לְךָ אֶל־מְקוֹמֶךָ אָמַרְתִּי כַּבֵּד אֲכַבֶּדְךָ
11   וְהִנֵּה מְנָעֲךָ יְהוָה מִכָּבוֹד׃ וַיֹּאמֶר בִּלְעָם אֶל־בָּלָק
12   הֲלֹא גַּם אֶל־מַלְאָכֶיךָ אֲשֶׁר־שָׁלַחְתָּ אֵלַי דִּבַּרְתִּי
13   לֵאמֹר׃ אִם־יִתֶּן־לִי בָלָק מְלֹא בֵיתוֹ כֶּסֶף וְזָהָב לֹא
14   אוּכַל לַעֲבֹר אֶת־פִּי יְהוָה לַעֲשׂוֹת טוֹבָה אוֹ רָעָה
15   מִלִּבִּי אֲשֶׁר־יְדַבֵּר יְהוָה אֹתוֹ אֲדַבֵּר׃  וְעַתָּה הִנְנִי
16   הוֹלֵךְ לְעַמִּי לְכָה אִיעָצְךָ אֲשֶׁר יַעֲשֶׂה הָעָם הַזֶּה
17   לְעַמְּךָ בְּאַחֲרִית הַיָּמִים׃ וַיִּשָּׂא מְשָׁלוֹ וַיֹּאמַר נְאֻם
18   בִּלְעָם בְּנוֹ בְעֹר וּנְאֻם הַגֶּבֶר שְׁתֻם הָעָיִן׃ נְאֻם שֹׁמֵעַ
19   אִמְרֵי־אֵל וְיֹדֵעַ דַּעַת עֶלְיוֹן מַחֲזֵה שַׁדַּי יֶחֱזֶה נֹפֵל
20   וּגְלוּי עֵינָיִם׃ אֶרְאֶנּוּ וְלֹא עַתָּה אֲשׁוּרֶנּוּ וְלֹא קָרוֹב
21   דָּרַךְ כּוֹכָב מִיַּעֲקֹב וְקָם שֵׁבֶט מִיִּשְׂרָאֵל וּמָחַץ
22   פַּאֲתֵי מוֹאָב וְקַרְקַר כָּל־בְּנֵי־שֵׁת׃ וְהָיָה אֱדוֹם יְרֵשָׁה
23   וְהָיָה יְרֵשָׁה שֵׂעִיר אֹיְבָיו וְיִשְׂרָאֵל עֹשֶׂה חָיִל׃ וְיֵרְדְּ
24   מִיַּעֲקֹב וְהֶאֱבִיד שָׂרִיד מֵעִיר׃ וַיַּרְא אֶת־עֲמָלֵק וַיִּשָּׂא
25   מְשָׁלוֹ וַיֹּאמַר רֵאשִׁית גּוֹיִם עֲמָלֵק וְאַחֲרִיתוֹ עֲדֵי
26   אֹבֵד׃ וַיַּרְא אֶת־הַקֵּינִי וַיִּשָּׂא מְשָׁלוֹ וַיֹּאמַר אֵיתָן
27   מוֹשָׁבֶךָ וְשִׂים בַּסֶּלַע קִנֶּךָ׃ כִּי אִם־יִהְיֶה לְבָעֵר קָיִן
28   עַד־מָה אַשּׁוּר תִּשְׁבֶּךָּ׃ וַיִּשָּׂא מְשָׁלוֹ וַיֹּאמַר אוֹי מִי
29   יִחְיֶה מִשֻּׂמוֹ אֵל׃ וְצִים מִיַּד כִּתִּים וְעִנּוּ אַשּׁוּר וְעִנּוּ־
30  עֵבֶר וְגַם־הוּא עֲדֵי אֹבֵד׃ וַיָּקָם בִּלְעָם וַיֵּלֶךְ וַיָּשָׁב
31   לִמְקֹמוֹ וְגַם־בָּלָק הָלַךְ לְדַרְכּוֹ׃ [פ]
32   וַיֵּשֶׁב יִשְׂרָאֵל בַּשִּׁטִּים וַיָּחֶל הָעָם לִזְנוֹת אֶל־בְּנוֹת
33   מוֹאָב׃ וַתִּקְרֶאןָ לָעָם לְזִבְחֵי אֱלֹהֵיהֶן וַיֹּאכַל הָעָם
34   וַיִּשְׁתַּחֲווּ לֵאלֹהֵיהֶן׃ וַיִּצָּמֶד יִשְׂרָאֵל לְבַעַל פְּעוֹר
35   וַיִּחַר־אַף יְהוָה בְּיִשְׂרָאֵל׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה
36   קַח אֶת־כָּל־רָאשֵׁי הָעָם וְהוֹקַע אוֹתָם לַיהוָה נֶגֶד
37   הַשָּׁמֶשׁ וְיָשֹׁב חֲרוֹן אַף־יְהוָה מִיִּשְׂרָאֵל׃ וַיֹּאמֶר
38   מֹשֶׁה אֶל־שֹׁפְטֵי יִשְׂרָאֵל הִרְגוּ אִישׁ אֲנָשָׁיו
39   הַנִּצְמָדִים לְבַעַל פְּעוֹר׃ וְהִנֵּה אִישׁ מִבְּנֵי יִשְׂרָאֵל
40   בָּא וַיַּקְרֵב אֶל־אֶחָיו אֶת־הַמִּדְיָנִית לְעֵינֵי מֹשֶׁה
41   וּלְעֵינֵי כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל וְהֵמָּה בֹכִים פֶּתַח
42   אֹהֶל מוֹעֵד׃  וַיַּרְא פִּינְחָס בֶּן־אֶלְעָזָר בֶּן־אַהֲרֹן הַכֹּהֵן
