 1  וְהָיָה הָאִישׁ אֲשֶׁר־יִבְחַר יְהוָה הוּא הַקָּדוֹשׁ רַב־
 2  לָכֶם בְּנֵי לֵוִי׃♢ וַיֹּאמֶר מֹשֶׁה אֶל־קֹרַח שִׁמְעוּ־נָא בְּנֵי
 3   לֵוִי׃ הַמְעַט מִכֶּם כִּי־הִבְדִּיל אֱלֹהֵי יִשְׂרָאֵל אֶתְכֶם
 4   מֵעֲדַת יִשְׂרָאֵל לְהַקְרִיב אֶתְכֶם אֵלָיו לַעֲבֹד אֶת־
 5  עֲבֹדַת מִשְׁכַּן יְהוָה וְלַעֲמֹד לִפְנֵי הָעֵדָה לְשָׁרְתָם׃
 6   וַיַּקְרֵב אֹתְךָ וְאֶת־כָּל־אַחֶיךָ בְנֵי־לֵוִי אִתָּךְ וּבִקַּשְׁתֶּם
 7   גַּם־כְּהֻנָּה׃ לָכֵן אַתָּה וְכָל־עֲדָתְךָ הַנֹּעָדִים עַל־יְהוָה
 8   וְאַהֲרֹן מַה־הוּא כִּי תַלִּינוּ עָלָיו׃ וַיִּשְׁלַח מֹשֶׁה
 9   לִקְרֹא לְדָתָן וְלַאֲבִירָם בְּנֵי אֱלִיאָב וַיֹּאמְרוּ לֹא
10   נַעֲלֶה׃ הַמְעַט כִּי הֶעֱלִיתָנוּ מֵאֶרֶץ זָבַת חָלָב וּדְבַשׁ
11   לַהֲמִיתֵנוּ בַּמִּדְבָּר כִּי־תִשְׂתָּרֵר עָלֵינוּ גַּם־הִשְׂתָּרֵר׃♢
12    אַף לֹא אֶל־אֶרֶץ זָבַת חָלָב וּדְבַשׁ הֲבִיאֹתָנוּ וַתִּתֶּן־
13  לָנוּ נַחֲלַת שָׂדֶה וָכָרֶם הַעֵינֵי הָאֲנָשִׁים הָהֵם תְּנַקֵּר
14   לֹא נַעֲלֶה׃ וַיִּחַר לְמֹשֶׁה מְאֹד וַיֹּאמֶר אֶל־יְהוָה אַל־
15  תֵּפֶן אֶל־מִנְחָתָם לֹא חֲמוֹר אֶחָד מֵהֶם נָשָׂאתִי וְלֹא
16   הֲרֵעֹתִי אֶת־אַחַד מֵהֶם׃ וַיֹּאמֶר מֹשֶׁה אֶל־קֹרַח אַתָּה
17   וְכָל־עֲדָתְךָ הֱיוּ לִפְנֵי יְהוָה אַתָּה וָהֵם וְאַהֲרֹן מָחָר׃
18   וּקְחוּ  אִישׁ מַחְתָּתוֹ וּנְתַתֶּם עֲלֵיהֶם קְטֹרֶת וְהִקְרַבְתֶּם
19   לִפְנֵי יְהוָה אִישׁ מַחְתָּתוֹ חֲמִשִּׁים וּמָאתַיִם מַחְתֹּת
20   וְאַתָּה וְאַהֲרֹן אִישׁ מַחְתָּתוֹ׃ וַיִּקְחוּ אִישׁ מַחְתָּתוֹ וַיִּתְּנוּ
21   עֲלֵיהֶם אֵשׁ וַיָּשִׂימוּ עֲלֵיהֶם קְטֹרֶת וַיַּעַמְדוּ פֶּתַח
22   אֹהֶל מוֹעֵד וּמֹשֶׁה וְאַהֲרֹן׃ וַיַּקְהֵל עֲלֵיהֶם קֹרַח אֶת־
23  כָּל־הָעֵדָה אֶל־פֶּתַח אֹהֶל מוֹעֵד וַיֵּרָא כְבוֹד־יְהוָה
24   אֶל־כָּל־הָעֵדָה׃ [ס]              וַיְדַבֵּר יְהוָה
25   אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ הִבָּדְלוּ מִתּוֹךְ הָעֵדָה
26   הַזֹּאת וַאֲכַלֶּה אֹתָם כְּרָגַע׃ וַיִּפְּלוּ עַל־פְּנֵיהֶם וַיֹּאמְרוּ
27   אֵל אֱלֹהֵי הָרוּחֹת לְכָל־בָּשָׂר הָאִישׁ אֶחָד יֶחֱטָא
28   וְעַל כָּל־הָעֵדָה תִּקְצֹף׃ [ס]             וַיְדַבֵּר
29   יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־הָעֵדָה לֵאמֹר
30   הֵעָלוּ מִסָּבִיב לְמִשְׁכַּן־קֹרַח דָּתָן וַאֲבִירָם׃ וַיָּקָם
31   מֹשֶׁה וַיֵּלֶךְ אֶל־דָּתָן וַאֲבִירָם וַיֵּלְכוּ אַחֲרָיו זִקְנֵי
32   יִשְׂרָאֵל׃ וַיְדַבֵּר אֶל־הָעֵדָה לֵאמֹר סוּרוּ נָא מֵעַל
33   אָהֳלֵי הָאֲנָשִׁים הָרְשָׁעִים הָאֵלֶּה וְאַל־תִּגְּעוּ בְּכָל־
34  אֲשֶׁר לָהֶם פֶּן־תִּסָּפוּ בְּכָל־חַטֹּאתָם׃ וַיֵּעָלוּ מֵעַל
35   מִשְׁכַּן־קֹרַח דָּתָן וַאֲבִירָם מִסָּבִיב וְדָתָן וַאֲבִירָם
36   יָצְאוּ נִצָּבִים פֶּתַח אָהֳלֵיהֶם וּנְשֵׁיהֶם וּבְנֵיהֶם
37   וְטַפָּם׃ וַיֹּאמֶר מֹשֶׁה בְּזֹאת תֵּדְעוּן כִּי־יְהוָה שְׁלָחַנִי
38   לַעֲשׂוֹת אֵת כָּל־הַמַּעֲשִׂים הָאֵלֶּה כִּי־לֹא מִלִּבִּי׃
39   אִם־כְּמוֹת כָּל־הָאָדָם יְמֻתוּן אֵלֶּה וּפְקֻדַּת כָּל־
40  הָאָדָם יִפָּקֵד עֲלֵיהֶם לֹא יְהוָה שְׁלָחָנִי׃ וְאִם־בְּרִיאָה
41   יִבְרָא יְהוָה וּפָצְתָה הָאֲדָמָה אֶת־פִּיהָ וּבָלְעָה
42   אֹתָם וְאֶת־כָּל־אֲשֶׁר לָהֶם וְיָרְדוּ חַיִּים שְׁאֹלָה
