 1  וְהָיָה הַשָּׂדֶה בְּצֵאתוֹ בַיֹּבֵל קֹדֶשׁ לַיהוָה כִּשְׂדֵה
 2   הַחֵרֶם לַכֹּהֵן תִּהְיֶה אֲחֻזָּתוֹ׃  וְאִם אֶת־שְׂדֵה מִקְנָתוֹ
 3   אֲשֶׁר לֹא מִשְּׂדֵה אֲחֻזָּתוֹ יַקְדִּישׁ לַיהוָה׃ וְחִשַּׁב־לוֹ
 4   הַכֹּהֵן אֵת מִכְסַת הָעֶרְכְּךָ עַד שְׁנַת הַיֹּבֵל וְנָתַן אֶת־
 5  הָעֶרְכְּךָ בַּיּוֹם הַהוּא קֹדֶשׁ לַיהוָה׃ בִּשְׁנַת הַיּוֹבֵל
 6   יָשׁוּב הַשָּׂדֶה לַאֲשֶׁר קָנָהוּ מֵאִתּוֹ לַאֲשֶׁר־לוֹ אֲחֻזַּת
 7   הָאָרֶץ׃ וְכָל־עֶרְכְּךָ יִהְיֶה בְּשֶׁקֶל הַקֹּדֶשׁ עֶשְׂרִים
 8   גֵּרָה יִהְיֶה הַשָּׁקֶל׃ אַךְ־בְּכוֹר אֲשֶׁר־יְבֻכַּר לַיהוָה
 9   בִּבְהֵמָה לֹא־יַקְדִּישׁ אִישׁ אֹתוֹ אִם־שׁוֹר אִם־שֶׂה
10   לַיהוָה הוּא׃ וְאִם בַּבְּהֵמָה הַטְּמֵאָה וּפָדָה בְעֶרְכֶּךָ
11   וְיָסַף חֲמִשִׁתוֹ עָלָיו וְאִם־לֹא יִגָּאֵל וְנִמְכַּר בְּעֶרְכֶּךָ׃
12   אַךְ־כָּל־חֵרֶם אֲשֶׁר יַחֲרִם אִישׁ לַיהוָה מִכָּל־אֲשֶׁר־
13  לוֹ מֵאָדָם וּבְהֵמָה וּמִשְּׂדֵה אֲחֻזָּתוֹ לֹא יִמָּכֵר וְלֹא
14   יִגָּאֵל כָּל־חֵרֶם קֹדֶשׁ־קָדָשִׁים הוּא לַיהוָה׃  כָּל־חֵרֶם
15   אֲשֶׁר יָחֳרַם מִן־הָאָדָם לֹא יִפָּדֶה מוֹת יוּמָת׃ וְכָל־
16  מַעְשַׂר הָאָרֶץ מִזֶּרַע הָאָרֶץ מִפְּרִי הָעֵץ לַיהוָה הוּא
17   קֹדֶשׁ לַיהוָה׃ וְאִם־גָּאֹל יִגְאַל אִישׁ מִמַּעַשְׂרוֹ חֲמִשִׁיתוֹ
18   יֹסֵף עָלָיו׃  וְכָל־מַעְשַׂר בָּקָר וָצֹאן כֹּל אֲשֶׁר־יַעֲבֹר
19   תַּחַת הַשָּׁבֶט הָעֲשִׂירִי יִהְיֶה־קֹּדֶשׁ לַיהוָה׃ לֹא יְבַקֵּר
20   בֵּין־טוֹב לָרַע וְלֹא יְמִירֶנּוּ וְאִם־הָמֵר יְמִירֶנּוּ וְהָיָה־הוּא
21   וּתְמוּרָתוֹ יִהְיֶה־קֹּדֶשׁ לֹא יִגָּאֵל׃ אֵלֶּה הַמִּצְו‍ֹת אֲשֶׁר
22   צִוָּה יְהוָה אֶת־מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל בְּהַר סִינָי׃

27  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה בְּמִדְבַּר סִינַי בְּאֹהֶל מוֹעֵד
28   בְּאֶחָד לַחֹדֶשׁ הַשֵּׁנִי בַּשָּׁנָה הַשֵּׁנִית לְצֵאתָם מֵאֶרֶץ
29   מִצְרַיִם לֵאמֹר׃ שְׂאוּ אֶת־רֹאשׁ כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל
30   לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם בְּמִסְפַּר שֵׁמוֹת כָּל־זָכָר
31   לְגֻלְגְּלֹתָם׃ מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה כָּל־יֹצֵא צָבָא
32   בְּיִשְׂרָאֵל תִּפְקְדוּ אֹתָם לְצִבְאֹתָם אַתָּה וְאַהֲרֹן׃ וְאִתְּכֶם
33   יִהְיוּ אִישׁ אִישׁ לַמַּטֶּה אִישׁ רֹאשׁ לְבֵית־אֲבֹתָיו
34   הוּא׃♢ וְאֵלֶּה שְׁמוֹת הָאֲנָשִׁים אֲשֶׁר יַעַמְדוּ אִתְּכֶם
35   לִרְאוּבֵן אֱלִיצוּר בֶּן־שְׁדֵיאוּר׃ לְשִׁמְעוֹן שְׁלֻמִיאֵל
36   בֶּן־צוּרִישַׁדָּי׃ לִיהוּדָה נַחְשׁוֹן בֶּן־עַמִּינָדָב׃ לְיִשָּׂשכָר
37   נְתַנְאֵל בֶּן־צוּעָר׃ לִזְבוּלֻן אֱלִיאָב בֶּן־חֵלֹן׃ לִבְנֵי יוֹסֵף
38   לְאֶפְרַיִם אֱלִישָׁמָע בֶּן־עַמִּיהוּד לִמְנַשֶּׁה גַּמְלִיאֵל בֶּן־
39  פְּדָהצוּר׃ לְבִנְיָמִן אֲבִידָן בֶּן־גִּדְעֹנִי׃ לְדָן אֲחִיעֶזֶר
40   בֶּן־עַמִּישַׁדָּי׃ לְאָשֵׁר פַּגְעִיאֵל בֶּן־עָכְרָן׃ לְגָד אֶלְיָסָף בֶּן־
41  דְּעוּאֵל׃ לְנַפְתָּלִי אֲחִירַע בֶּן־עֵינָן׃ אֵלֶּה קְרוּאֵי הָעֵדָה
42   נְשִׂיאֵי מַטּוֹת אֲבוֹתָם רָאשֵׁי אַלְפֵי יִשְׂרָאֵל הֵם׃♢
