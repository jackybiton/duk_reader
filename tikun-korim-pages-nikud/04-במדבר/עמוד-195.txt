 1  וַיִּשָּׁבַע לֵאמֹר׃ אִם־יִרְאוּ הָאֲנָשִׁים הָעֹלִים מִמִּצְרַיִם
 2   מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה אֵת הָאֲדָמָה אֲשֶׁר
 3   נִשְׁבַּעְתִּי לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב כִּי לֹא־מִלְאוּ
 4   אַחֲרָי׃ בִּלְתִּי כָּלֵב בֶּן־יְפֻנֶּה הַקְּנִזִּי וִיהוֹשֻׁעַ בִּן־נוּן
 5   כִּי מִלְאוּ אַחֲרֵי יְהוָה׃ וַיִּחַר־אַף יְהוָה בְּיִשְׂרָאֵל
 6   וַיְנִעֵם בַּמִּדְבָּר אַרְבָּעִים שָׁנָה עַד־תֹּם כָּל־הַדּוֹר
 7   הָעֹשֶׂה הָרַע בְּעֵינֵי יְהוָה׃ וְהִנֵּה קַמְתֶּם תַּחַת
 8   אֲבֹתֵיכֶם תַּרְבּוּת אֲנָשִׁים חַטָּאִים לִסְפּוֹת עוֹד עַל
 9   חֲרוֹן אַף־יְהוָה אֶל־יִשְׂרָאֵל׃ כִּי תְשׁוּבֻן מֵאַחֲרָיו
10   וְיָסַף עוֹד לְהַנִּיחוֹ בַּמִּדְבָּר וְשִׁחַתֶּם לְכָל־הָעָם
11   הַזֶּה׃ [ס]             וַיִּגְּשׁוּ אֵלָיו
12   וַיֹּאמְרוּ גִּדְרֹת צֹאן נִבְנֶה לְמִקְנֵנוּ פֹּה וְעָרִים לְטַפֵּנוּ׃
13   וַאֲנַחְנוּ נֵחָלֵץ חֻשִׁים לִפְנֵי בְּנֵי יִשְׂרָאֵל עַד אֲשֶׁר
14   אִם־הֲבִיאֹנֻם אֶל־מְקוֹמָם וְיָשַׁב טַפֵּנוּ בְּעָרֵי הַמִּבְצָר
15   מִפְּנֵי יֹשְׁבֵי הָאָרֶץ׃ לֹא נָשׁוּב אֶל־בָּתֵּינוּ עַד הִתְנַחֵל
16   בְּנֵי יִשְׂרָאֵל אִישׁ נַחֲלָתוֹ׃ כִּי לֹא נִנְחַל אִתָּם מֵעֵבֶר
17   לַיַּרְדֵּן וָהָלְאָה כִּי בָאָה נַחֲלָתֵנוּ אֵלֵינוּ מֵעֵבֶר
18   הַיַּרְדֵּן מִזְרָחָה׃ [פ]
19    וַיֹּאמֶר אֲלֵיהֶם מֹשֶׁה אִם־תַּעֲשׂוּן אֶת־הַדָּבָר הַזֶּה
20   אִם־תֵּחָלְצוּ לִפְנֵי יְהוָה לַמִּלְחָמָה׃ וְעָבַר לָכֶם כָּל־
21  חָלוּץ אֶת־הַיַּרְדֵּן לִפְנֵי יְהוָה עַד הוֹרִישׁוֹ אֶת־אֹיְבָיו
22   מִפָּנָיו׃ וְנִכְבְּשָׁה הָאָרֶץ לִפְנֵי יְהוָה וְאַחַר תָּשֻׁבוּ
23   וִהְיִיתֶם נְקִיִּם מֵיְהוָה וּמִיִּשְׂרָאֵל וְהָיְתָה הָאָרֶץ הַזֹּאת
24   לָכֶם לַאֲחֻזָּה לִפְנֵי יְהוָה׃ וְאִם־לֹא תַעֲשׂוּן כֵּן הִנֵּה
25   חֲטָאתֶם לַיהוָה וּדְעוּ חַטַּאתְכֶם אֲשֶׁר תִּמְצָא אֶתְכֶם׃
26   בְּנוּ־לָכֶם עָרִים לְטַפְּכֶם וּגְדֵרֹת לְצֹנַאֲכֶם וְהַיֹּצֵא
27   מִפִּיכֶם תַּעֲשׂוּ׃ וַיֹּאמֶר בְּנֵי־גָד וּבְנֵי רְאוּבֵן אֶל־מֹשֶׁה
28   לֵאמֹר עֲבָדֶיךָ יַעֲשׂוּ כַּאֲשֶׁר אֲדֹנִי מְצַוֶּה׃ טַפֵּנוּ נָשֵׁינוּ
29   מִקְנֵנוּ וְכָל־בְּהֶמְתֵּנוּ יִהְיוּ־שָׁם בְּעָרֵי הַגִּלְעָד׃ וַעֲבָדֶיךָ
30   יַעַבְרוּ כָּל־חֲלוּץ צָבָא לִפְנֵי יְהוָה לַמִּלְחָמָה כַּאֲשֶׁר
31   אֲדֹנִי דֹּבֵר׃ וַיְצַו לָהֶם מֹשֶׁה אֵת אֶלְעָזָר הַכֹּהֵן וְאֵת
32   יְהוֹשֻׁעַ בִּן־נוּן וְאֶת־רָאשֵׁי אֲבוֹת הַמַּטּוֹת לִבְנֵי
33   יִשְׂרָאֵל׃ וַיֹּאמֶר מֹשֶׁה אֲלֵהֶם אִם־יַעַבְרוּ בְנֵי־גָד
34   וּבְנֵי־רְאוּבֵן  אִתְּכֶם אֶת־הַיַּרְדֵּן כָּל־חָלוּץ לַמִּלְחָמָה
35   לִפְנֵי יְהוָה וְנִכְבְּשָׁה הָאָרֶץ לִפְנֵיכֶם וּנְתַתֶּם לָהֶם
36   אֶת־אֶרֶץ הַגִּלְעָד לַאֲחֻזָּה׃ וְאִם־לֹא יַעַבְרוּ חֲלוּצִים
37   אִתְּכֶם וְנֹאחֲזוּ בְתֹכְכֶם בְּאֶרֶץ כְּנָעַן׃ וַיַּעֲנוּ בְנֵי־גָד
38   וּבְנֵי רְאוּבֵן לֵאמֹר אֵת אֲשֶׁר דִּבֶּר יְהוָה אֶל־עֲבָדֶיךָ
39   כֵּן נַעֲשֶׂה׃ נַחְנוּ נַעֲבֹר חֲלוּצִים לִפְנֵי יְהוָה אֶרֶץ
40   כְּנָעַן וְאִתָּנוּ אֲחֻזַּת נַחֲלָתֵנוּ מֵעֵבֶר לַיַּרְדֵּן׃ וַיִּתֵּן
41   לָהֶם  מֹשֶׁה לִבְנֵי־גָד וְלִבְנֵי רְאוּבֵן וְלַחֲצִי  שֵׁבֶט
42   מְנַשֶּׁה בֶן־יוֹסֵף אֶת־מַמְלֶכֶת סִיחֹן מֶלֶךְ הָאֱמֹרִי
