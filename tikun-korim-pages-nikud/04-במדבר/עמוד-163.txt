 1  וּלְבָנָיו מִתּוֹךְ בְּנֵי יִשְׂרָאֵל לַעֲבֹד אֶת־עֲבֹדַת בְּנֵי־
 2  יִשְׂרָאֵל בְּאֹהֶל מוֹעֵד וּלְכַפֵּר עַל־בְּנֵי יִשְׂרָאֵל וְלֹא
 3   יִהְיֶה בִּבְנֵי יִשְׂרָאֵל נֶגֶף בְּגֶשֶׁת בְּנֵי־יִשְׂרָאֵל אֶל־
 4  הַקֹּדֶשׁ׃ וַיַּעַשׂ מֹשֶׁה וְאַהֲרֹן וְכָל־עֲדַת בְּנֵי־יִשְׂרָאֵל
 5   לַלְוִיִּם כְּכֹל אֲשֶׁר־צִוָּה יְהוָה אֶת־מֹשֶׁה לַלְוִיִּם כֵּן־
 6  עָשׂוּ לָהֶם בְּנֵי יִשְׂרָאֵל׃ וַיִּתְחַטְּאוּ הַלְוִיִּם וַיְכַבְּסוּ
 7   בִּגְדֵיהֶם וַיָּנֶף אַהֲרֹן אֹתָם תְּנוּפָה לִפְנֵי יְהוָה וַיְכַפֵּר
 8   עֲלֵיהֶם אַהֲרֹן לְטַהֲרָם׃ וְאַחֲרֵי־כֵן בָּאוּ הַלְוִיִּם לַעֲבֹד
 9   אֶת־עֲבֹדָתָם בְּאֹהֶל מוֹעֵד לִפְנֵי אַהֲרֹן וְלִפְנֵי בָנָיו
10   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה עַל־הַלְוִיִּם כֵּן עָשׂוּ
11   לָהֶם׃ [ס]             וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה
12   לֵּאמֹר׃ זֹאת אֲשֶׁר לַלְוִיִּם מִבֶּן חָמֵשׁ וְעֶשְׂרִים
13   שָׁנָה וָמַעְלָה יָבוֹא לִצְבֹא צָבָא בַּעֲבֹדַת אֹהֶל
14   מוֹעֵד׃ וּמִבֶּן חֲמִשִּׁים שָׁנָה יָשׁוּב מִצְּבָא הָעֲבֹדָה
15   וְלֹא יַעֲבֹד עוֹד׃ וְשֵׁרֵת אֶת־אֶחָיו בְּאֹהֶל מוֹעֵד
16   לִשְׁמֹר מִשְׁמֶרֶת וַעֲבֹדָה לֹא יַעֲבֹד כָּכָה תַּעֲשֶׂה
17   לַלְוִיִּם בְּמִשְׁמְרֹתָם׃ [פ]
18    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה בְמִדְבַּר־סִינַי בַּשָּׁנָה הַשֵּׁנִית
19   לְצֵאתָם מֵאֶרֶץ מִצְרַיִם בַּחֹדֶשׁ הָרִאשׁוֹן לֵאמֹר׃
20   וְיַעֲשׂוּ בְנֵי־יִשְׂרָאֵל אֶת־הַפָּסַח בְּמוֹעֲדוֹ׃ בְּאַרְבָּעָה
21   עָשָׂר־יוֹם בַּחֹדֶשׁ הַזֶּה בֵּין הָעַרְבַּיִם תַּעֲשׂוּ אֹתוֹ
22   בְּמֹעֲדוֹ כְּכָל־חֻקֹּתָיו וּכְכָל־מִשְׁפָּטָיו תַּעֲשׂוּ אֹתוֹ׃
23   וַיְדַבֵּר מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל לַעֲשֹׂת הַפָּסַח׃ וַיַּעֲשׂוּ
24   אֶת־הַפֶּסַח בָּרִאשׁוֹן בְּאַרְבָּעָה עָשָׂר יוֹם לַחֹדֶשׁ
25   בֵּין הָעַרְבַּיִם בְּמִדְבַּר סִינָי כְּכֹל אֲשֶׁר צִוָּה יְהוָה אֶת־
26  מֹשֶׁה כֵּן עָשׂוּ בְּנֵי יִשְׂרָאֵל׃ וַיְהִי אֲנָשִׁים אֲשֶׁר הָיוּ
27   טְמֵאִים לְנֶפֶשׁ אָדָם וְלֹא־יָכְלוּ לַעֲשֹׂת־הַפֶּסַח בַּיּוֹם
28   הַהוּא וַיִּקְרְבוּ לִפְנֵי מֹשֶׁה וְלִפְנֵי אַהֲרֹן בַּיּוֹם הַהוּא׃
29   וַיֹּאמְרוּ הָאֲנָשִׁים הָהֵמָּה אֵלָיו אֲנַחְנוּ טְמֵאִים לְנֶפֶשׁ
30   אָדָם לָמָּה נִגָּרַע לְבִלְתִּי הַקְרִיב אֶת־קָרְבַּן יְהוָה
31   בְּמֹעֲדוֹ בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׃ וַיֹּאמֶר אֲלֵהֶם מֹשֶׁה עִמְדוּ
32   וְאֶשְׁמְעָה מַה־יְצַוֶּה יְהוָה לָכֶם׃ [פ]
33   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
34   לֵאמֹר אִישׁ אִישׁ כִּי־יִהְיֶה־טָמֵא  לָנֶפֶשׁ אוֹ בְדֶרֶךְ
35   רְחֹקָה לָכֶם אוֹ לְדֹרֹתֵיכֶם וְעָשָׂה פֶסַח לַיהוָה׃
36   בַּחֹדֶשׁ הַשֵּׁנִי בְּאַרְבָּעָה עָשָׂר יוֹם בֵּין הָעַרְבַּיִם
37   יַעֲשׂוּ אֹתוֹ עַל־מַצּוֹת וּמְרֹרִים יֹאכְלֻהוּ׃ לֹא־יַשְׁאִירוּ
38   מִמֶּנּוּ עַד־בֹּקֶר וְעֶצֶם לֹא יִשְׁבְּרוּ־בוֹ כְּכָל־חֻקַּת
39   הַפֶּסַח יַעֲשׂוּ אֹתוֹ׃ וְהָאִישׁ אֲשֶׁר־הוּא טָהוֹר וּבְדֶרֶךְ
40   לֹא־הָיָה וְחָדַל לַעֲשׂוֹת הַפֶּסַח וְנִכְרְתָה הַנֶּפֶשׁ
41   הַהִוא מֵעַמֶּיהָ כִּי  קָרְבַּן יְהוָה לֹא הִקְרִיב בְּמֹעֲדוֹ
42   חֶטְאוֹ יִשָּׂא הָאִישׁ הַהוּא׃ וְכִי־יָגוּר אִתְּכֶם גֵּר
