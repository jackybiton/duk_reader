 1  וְהִנַּחְתָּם בְּאֹהֶל מוֹעֵד לִפְנֵי הָעֵדוּת אֲשֶׁר אִוָּעֵד
 2   לָכֶם שָׁמָּה׃ וְהָיָה הָאִישׁ אֲשֶׁר אֶבְחַר־בּוֹ מַטֵּהוּ
 3   יִפְרָח וַהֲשִׁכֹּתִי מֵעָלַי אֶת־תְּלֻנּוֹת בְּנֵי יִשְׂרָאֵל אֲשֶׁר
 4   הֵם מַלִּינִם עֲלֵיכֶם׃ וַיְדַבֵּר מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל
 5   וַיִּתְּנוּ אֵלָיו  כָּל־נְשִׂיאֵיהֶם מַטֶּה לְנָשִׂיא אֶחָד מַטֶּה
 6   לְנָשִׂיא אֶחָד לְבֵית אֲבֹתָם שְׁנֵים עָשָׂר מַטּוֹת וּמַטֵּה
 7   אַהֲרֹן בְּתוֹךְ מַטּוֹתָם׃ וַיַּנַּח מֹשֶׁה אֶת־הַמַּטֹּת לִפְנֵי
 8   יְהוָה בְּאֹהֶל הָעֵדֻת׃ וַיְהִי מִמָּחֳרָת וַיָּבֹא מֹשֶׁה אֶל־
 9  אֹהֶל הָעֵדוּת וְהִנֵּה פָּרַח מַטֵּה־אַהֲרֹן לְבֵית לֵוִי
10   וַיֹּצֵא פֶרַח וַיָּצֵץ צִיץ וַיִּגְמֹל שְׁקֵדִים׃ וַיֹּצֵא מֹשֶׁה
11   אֶת־כָּל־הַמַּטֹּת מִלִּפְנֵי יְהוָה אֶל־כָּל־בְּנֵי יִשְׂרָאֵל
12   וַיִּרְאוּ וַיִּקְחוּ אִישׁ מַטֵּהוּ׃ [פ]
13    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה הָשֵׁב אֶת־מַטֵּה אַהֲרֹן לִפְנֵי
14   הָעֵדוּת לְמִשְׁמֶרֶת לְאוֹת לִבְנֵי־מֶרִי וּתְכַל תְּלוּנֹּתָם
15   מֵעָלַי וְלֹא יָמֻתוּ׃ וַיַּעַשׂ מֹשֶׁה כַּאֲשֶׁר צִוָּה יְהוָה
16   אֹתוֹ כֵּן עָשָׂה׃ [פ]
17   וַיֹּאמְרוּ בְּנֵי יִשְׂרָאֵל אֶל־מֹשֶׁה לֵאמֹר הֵן גָּוַעְנוּ
18   אָבַדְנוּ כֻּלָּנוּ אָבָדְנוּ׃ כֹּל הַקָּרֵב  הַקָּרֵב אֶל־מִשְׁכַּן
19   יְהוָה יָמוּת הַאִם תַּמְנוּ לִגְו͏ֹעַ׃ [ס]             וַיֹּאמֶר
20   יְהוָה אֶל־אַהֲרֹן אַתָּה וּבָנֶיךָ וּבֵית־אָבִיךָ אִתָּךְ תִּשְׂאוּ
21   אֶת־עֲו‍ֹן הַמִּקְדָּשׁ וְאַתָּה וּבָנֶיךָ אִתָּךְ תִּשְׂאוּ אֶת־
22  עֲו‍ֹן כְּהֻנַּתְכֶם׃ וְגַם אֶת־אַחֶיךָ מַטֵּה לֵוִי שֵׁבֶט אָבִיךָ
23   הַקְרֵב אִתָּךְ וְיִלָּווּ עָלֶיךָ וִישָׁרְתוּךָ וְאַתָּה וּבָנֶיךָ אִתָּךְ
24   לִפְנֵי אֹהֶל הָעֵדֻת׃ וְשָׁמְרוּ מִשְׁמַרְתְּךָ וּמִשְׁמֶרֶת כָּל־
25  הָאֹהֶל אַךְ אֶל־כְּלֵי הַקֹּדֶשׁ וְאֶל־הַמִּזְבֵּחַ לֹא יִקְרָבוּ
26   וְלֹא־יָמֻתוּ גַם־הֵם גַּם־אַתֶּם׃ וְנִלְווּ עָלֶיךָ וְשָׁמְרוּ אֶת־
27  מִשְׁמֶרֶת אֹהֶל מוֹעֵד לְכֹל עֲבֹדַת הָאֹהֶל וְזָר לֹא־
28  יִקְרַב אֲלֵיכֶם׃ וּשְׁמַרְתֶּם אֵת מִשְׁמֶרֶת הַקֹּדֶשׁ
29   וְאֵת מִשְׁמֶרֶת הַמִּזְבֵּחַ וְלֹא־יִהְיֶה עוֹד קֶצֶף עַל־
30  בְּנֵי יִשְׂרָאֵל׃ וַאֲנִי הִנֵּה לָקַחְתִּי אֶת־אֲחֵיכֶם הַלְוִיִּם
31   מִתּוֹךְ בְּנֵי יִשְׂרָאֵל לָכֶם מַתָּנָה נְתֻנִים לַיהוָה לַעֲבֹד
32   אֶת־עֲבֹדַת אֹהֶל מוֹעֵד׃ וְאַתָּה וּבָנֶיךָ אִתְּךָ תִּשְׁמְרוּ
33   אֶת־כְּהֻנַּתְכֶם לְכָל־דְּבַר הַמִּזְבֵּחַ וּלְמִבֵּית לַפָּרֹכֶת
34   וַעֲבַדְתֶּם עֲבֹדַת מַתָּנָה אֶתֵּן אֶת־כְּהֻנַּתְכֶם וְהַזָּר
35   הַקָּרֵב יוּמָת׃ [פ]
36   וַיְדַבֵּר יְהוָה אֶל־אַהֲרֹן וַאֲנִי הִנֵּה נָתַתִּי לְךָ אֶת־
37  מִשְׁמֶרֶת תְּרוּמֹתָי לְכָל־קָדְשֵׁי בְנֵי־יִשְׂרָאֵל לְךָ
38   נְתַתִּים לְמָשְׁחָה וּלְבָנֶיךָ לְחָק־עוֹלָם׃ זֶה־יִהְיֶה לְךָ
39   מִקֹּדֶשׁ הַקֳּדָשִׁים מִן־הָאֵשׁ כָּל־קָרְבָּנָם לְכָל־מִנְחָתָם
40   וּלְכָל־חַטָּאתָם וּלְכָל־אֲשָׁמָם אֲשֶׁר יָשִׁיבוּ לִי
41   קֹדֶשׁ קָדָשִׁים לְךָ הוּא וּלְבָנֶיךָ׃ בְּקֹדֶשׁ הַקֳּדָשִׁים
42   תֹּאכְלֶנּוּ כָּל־זָכָר יֹאכַל אֹתוֹ קֹדֶשׁ יִהְיֶה־לָּךְ׃
