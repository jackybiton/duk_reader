 1  וּפְקֻדָיו שִׁבְעָה וַחֲמִשִּׁים אֶלֶף וְאַרְבַּע מֵאוֹת׃ כָּל־
 2  הַפְּקֻדִים לְמַחֲנֵה יְהוּדָה מְאַת אֶלֶף וּשְׁמֹנִים אֶלֶף
 3   וְשֵׁשֶׁת־אֲלָפִים וְאַרְבַּע־מֵאוֹת לְצִבְאֹתָם רִאשֹׁנָה
 4   יִסָּעוּ׃ [ס]             דֶּגֶל מַחֲנֵה רְאוּבֵן
 5   תֵּימָנָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי רְאוּבֵן אֱלִיצוּר בֶּן־
 6  שְׁדֵיאוּר׃ וּצְבָאוֹ וּפְקֻדָיו שִׁשָּׁה וְאַרְבָּעִים אֶלֶף
 7   וַחֲמֵשׁ מֵאוֹת׃ וְהַחוֹנִם עָלָיו מַטֵּה שִׁמְעוֹן וְנָשִׂיא לִבְנֵי
 8   שִׁמְעוֹן שְׁלֻמִיאֵל בֶּן־צוּרִישַׁדָּי׃ וּצְבָאוֹ וּפְקֻדֵיהֶם
 9   תִּשְׁעָה וַחֲמִשִּׁים אֶלֶף וּשְׁלֹשׁ מֵאוֹת׃ וּמַטֵּה גָּד
10   וְנָשִׂיא לִבְנֵי גָד אֶלְיָסָף בֶּן־רְעוּאֵל׃ וּצְבָאוֹ וּפְקֻדֵיהֶם
11   חֲמִשָּׁה וְאַרְבָּעִים אֶלֶף וְשֵׁשׁ מֵאוֹת וַחֲמִשִּׁים׃ כָּל־
12  הַפְּקֻדִים לְמַחֲנֵה רְאוּבֵן מְאַת אֶלֶף וְאֶחָד וַחֲמִשִּׁים
13   אֶלֶף וְאַרְבַּע־מֵאוֹת וַחֲמִשִּׁים לְצִבְאֹתָם וּשְׁנִיִּם
14   יִסָּעוּ׃ [ס]             וְנָסַע אֹהֶל־מוֹעֵד
15   מַחֲנֵה הַלְוִיִּם בְּתוֹךְ הַמַּחֲנֹת כַּאֲשֶׁר יַחֲנוּ כֵּן יִסָּעוּ
16   אִישׁ עַל־יָדוֹ לְדִגְלֵיהֶם׃ [ס]             דֶּגֶל
17   מַחֲנֵה אֶפְרַיִם לְצִבְאֹתָם יָמָּה וְנָשִׂיא לִבְנֵי אֶפְרַיִם
18   אֱלִישָׁמָע בֶּן־עַמִּיהוּד׃ וּצְבָאוֹ וּפְקֻדֵיהֶם אַרְבָּעִים
19   אֶלֶף וַחֲמֵשׁ מֵאוֹת׃ וְעָלָיו מַטֵּה מְנַשֶּׁה וְנָשִׂיא לִבְנֵי
20   מְנַשֶּׁה גַּמְלִיאֵל בֶּן־פְּדָהצוּר׃ וּצְבָאוֹ וּפְקֻדֵיהֶם שְׁנַיִם
21   וּשְׁלֹשִׁים אֶלֶף וּמָאתָיִם׃ וּמַטֵּה בִּנְיָמִן וְנָשִׂיא לִבְנֵי
22   בִנְיָמִן אֲבִידָן בֶּן־גִּדְעֹנִי׃ וּצְבָאוֹ וּפְקֻדֵיהֶם חֲמִשָּׁה
23   וּשְׁלֹשִׁים אֶלֶף וְאַרְבַּע מֵאוֹת׃ כָּל־הַפְּקֻדִים לְמַחֲנֵה
24   אֶפְרַיִם מְאַת אֶלֶף וּשְׁמֹנַת־אֲלָפִים וּמֵאָה לְצִבְאֹתָם
25   וּשְׁלִשִׁים יִסָּעוּ׃ [ס]             דֶּגֶל
26   מַחֲנֵה דָן צָפֹנָה לְצִבְאֹתָם וְנָשִׂיא לִבְנֵי דָן אֲחִיעֶזֶר
27   בֶּן־עַמִּישַׁדָּי׃ וּצְבָאוֹ וּפְקֻדֵיהֶם שְׁנַיִם וְשִׁשִּׁים
28   אֶלֶף וּשְׁבַע מֵאוֹת׃ וְהַחֹנִים עָלָיו מַטֵּה אָשֵׁר
29   וְנָשִׂיא לִבְנֵי אָשֵׁר פַּגְעִיאֵל בֶּן־עָכְרָן׃ וּצְבָאוֹ
30   וּפְקֻדֵיהֶם אֶחָד וְאַרְבָּעִים אֶלֶף וַחֲמֵשׁ
31   מֵאוֹת׃ וּמַטֵּה נַפְתָּלִי וְנָשִׂיא לִבְנֵי נַפְתָּלִי אֲחִירַע בֶּן־
32  עֵינָן׃ וּצְבָאוֹ וּפְקֻדֵיהֶם שְׁלֹשָׁה וַחֲמִשִּׁים אֶלֶף
33   וְאַרְבַּע מֵאוֹת׃ כָּל־הַפְּקֻדִים לְמַחֲנֵה דָן מְאַת אֶלֶף
34   וְשִׁבְעָה וַחֲמִשִּׁים אֶלֶף וְשֵׁשׁ מֵאוֹת לָאַחֲרֹנָה
35   יִסְעוּ לְדִגְלֵיהֶם׃ [פ]
36   אֵלֶּה פְּקוּדֵי בְנֵי־יִשְׂרָאֵל לְבֵית אֲבֹתָם כָּל־פְּקוּדֵי
37   הַמַּחֲנֹת לְצִבְאֹתָם שֵׁשׁ־מֵאוֹת אֶלֶף וּשְׁלֹשֶׁת
38   אֲלָפִים וַחֲמֵשׁ מֵאוֹת וַחֲמִשִּׁים׃ וְהַלְוִיִּם לֹא הָתְפָּקְדוּ
39   בְּתוֹךְ בְּנֵי יִשְׂרָאֵל כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃
40   וַיַּעֲשׂוּ בְּנֵי יִשְׂרָאֵל כְּכֹל אֲשֶׁר־צִוָּה יְהוָה אֶת־
41  מֹשֶׁה כֵּן־חָנוּ לְדִגְלֵיהֶם וְכֵן נָסָעוּ אִישׁ לְמִשְׁפְּחֹתָיו
42   עַל־בֵּית אֲבֹתָיו׃ [פ]
