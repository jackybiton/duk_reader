 1  וְנִסְלַח לְכָל־עֲדַת בְּנֵי יִשְׂרָאֵל וְלַגֵּר הַגָּר בְּתוֹכָם
 2   כִּי לְכָל־הָעָם בִּשְׁגָגָה׃ [ס]              וְאִם־
 3  נֶפֶשׁ אַחַת תֶּחֱטָא בִשְׁגָגָה וְהִקְרִיבָה עֵז בַּת־שְׁנָתָהּ
 4   לְחַטָּאת׃ וְכִפֶּר הַכֹּהֵן עַל־הַנֶּפֶשׁ הַשֹּׁגֶגֶת בְּחֶטְאָה
 5   בִשְׁגָגָה לִפְנֵי יְהוָה לְכַפֵּר עָלָיו וְנִסְלַח לוֹ׃ הָאֶזְרָח
 6   בִּבְנֵי יִשְׂרָאֵל וְלַגֵּר הַגָּר בְּתוֹכָם תּוֹרָה אַחַת יִהְיֶה
 7   לָכֶם לָעֹשֶׂה בִּשְׁגָגָה׃ וְהַנֶּפֶשׁ אֲשֶׁר־תַּעֲשֶׂה  בְּיָד
 8   רָמָה מִן־הָאֶזְרָח וּמִן־הַגֵּר אֶת־יְהוָה הוּא מְגַדֵּף
 9   וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִקֶּרֶב עַמָּהּ׃ כִּי דְבַר־יְהוָה
10   בָּזָה וְאֶת־מִצְוָתוֹ הֵפַר הִכָּרֵת  תִּכָּרֵת הַנֶּפֶשׁ הַהִוא
11   עֲו‍ֹנָה בָהּ׃ [פ]
12   וַיִּהְיוּ בְנֵי־יִשְׂרָאֵל בַּמִּדְבָּר וַיִּמְצְאוּ אִישׁ מְקֹשֵׁשׁ
13   עֵצִים בְּיוֹם הַשַּׁבָּת׃ וַיַּקְרִיבוּ אֹתוֹ הַמֹּצְאִים אֹתוֹ
14   מְקֹשֵׁשׁ עֵצִים אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן וְאֶל כָּל־
15  הָעֵדָה׃ וַיַּנִּיחוּ אֹתוֹ בַּמִּשְׁמָר כִּי לֹא פֹרַשׁ מַה־
16  יֵּעָשֶׂה לוֹ׃ [ס]             וַיֹּאמֶר יְהוָה
17   אֶל־מֹשֶׁה מוֹת יוּמַת הָאִישׁ רָגוֹם אֹתוֹ בָאֲבָנִים כָּל־
18  הָעֵדָה מִחוּץ לַמַּחֲנֶה׃ וַיֹּצִיאוּ אֹתוֹ כָּל־הָעֵדָה אֶל־
19  מִחוּץ לַמַּחֲנֶה וַיִּרְגְּמוּ אֹתוֹ בָּאֲבָנִים וַיָּמֹת כַּאֲשֶׁר
20   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [פ]
21    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
22   וְאָמַרְתָּ אֲלֵהֶם וְעָשׂוּ לָהֶם צִיצִת עַל־כַּנְפֵי בִגְדֵיהֶם
23   לְדֹרֹתָם וְנָתְנוּ עַל־צִיצִת הַכָּנָף פְּתִיל תְּכֵלֶת׃ וְהָיָה
24   לָכֶם לְצִיצִת וּרְאִיתֶם אֹתוֹ וּזְכַרְתֶּם אֶת־כָּל־מִצְו‍ֹת
25   יְהוָה וַעֲשִׂיתֶם אֹתָם וְלֹא־תָתוּרוּ אַחֲרֵי לְבַבְכֶם
26   וְאַחֲרֵי עֵינֵיכֶם אֲשֶׁר־אַתֶּם זֹנִים אַחֲרֵיהֶם׃ לְמַעַן
27   תִּזְכְּרוּ וַעֲשִׂיתֶם אֶת־כָּל־מִצְו‍ֹתָי וִהְיִיתֶם קְדֹשִׁים
28   לֵאלֹהֵיכֶם׃ אֲנִי יְהוָה אֱלֹהֵיכֶם אֲשֶׁר הוֹצֵאתִי
29   אֶתְכֶם מֵאֶרֶץ מִצְרַיִם לִהְיוֹת לָכֶם לֵאלֹהִים אֲנִי
30   יְהוָה אֱלֹהֵיכֶם׃ [פ]
31  וַיִּקַּח קֹרַח בֶּן־יִצְהָר בֶּן־קְהָת בֶּן־לֵוִי וְדָתָן וַאֲבִירָם
32   בְּנֵי אֱלִיאָב וְאוֹן בֶּן־פֶּלֶת בְּנֵי רְאוּבֵן׃ וַיָּקֻמוּ לִפְנֵי
33   מֹשֶׁה וַאֲנָשִׁים מִבְּנֵי־יִשְׂרָאֵל חֲמִשִּׁים וּמָאתָיִם
34   נְשִׂיאֵי עֵדָה קְרִאֵי מוֹעֵד אַנְשֵׁי־שֵׁם׃ וַיִּקָּהֲלוּ עַל־
35  מֹשֶׁה וְעַל־אַהֲרֹן וַיֹּאמְרוּ אֲלֵהֶם רַב־לָכֶם כִּי
36   כָל־הָעֵדָה כֻּלָּם קְדֹשִׁים וּבְתוֹכָם יְהוָה וּמַדּוּעַ
37   תִּתְנַשְּׂאוּ עַל־קְהַל יְהוָה׃♢ וַיִּשְׁמַע מֹשֶׁה וַיִּפֹּל עַל־
38  פָּנָיו׃ וַיְדַבֵּר אֶל־קֹרַח וְאֶל־כָּל־עֲדָתוֹ לֵאמֹר בֹּקֶר
39   וְיֹדַע יְהוָה אֶת־אֲשֶׁר־לוֹ וְאֶת־הַקָּדוֹשׁ וְהִקְרִיב
40   אֵלָיו וְאֵת אֲשֶׁר יִבְחַר־בּוֹ יַקְרִיב אֵלָיו׃ זֹאת
41   עֲשׂוּ קְחוּ־לָכֶם מַחְתּוֹת קֹרַח וְכָל־עֲדָתוֹ׃ וּתְנוּ
42   בָהֵן  אֵשׁ וְשִׂימוּ עֲלֵיהֶן  קְטֹרֶת לִפְנֵי יְהוָה מָחָר
