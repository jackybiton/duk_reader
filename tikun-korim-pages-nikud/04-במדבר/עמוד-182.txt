 1  וַיַּעֲמֹד בְּמָקוֹם צָר אֲשֶׁר אֵין־דֶּרֶךְ לִנְטוֹת יָמִין
 2   וּשְׂמֹאול׃ וַתֵּרֶא הָאָתוֹן אֶת־מַלְאַךְ יְהוָה וַתִּרְבַּץ
 3   תַּחַת בִּלְעָם וַיִּחַר־אַף בִּלְעָם וַיַּךְ אֶת־הָאָתוֹן בַּמַּקֵּל׃
 4   וַיִּפְתַּח יְהוָה אֶת־פִּי הָאָתוֹן וַתֹּאמֶר לְבִלְעָם מֶה־
 5  עָשִׂיתִי לְךָ כִּי הִכִּיתַנִי זֶה שָׁלֹשׁ רְגָלִים׃ וַיֹּאמֶר
 6   בִּלְעָם לָאָתוֹן כִּי הִתְעַלַּלְתְּ בִּי לוּ יֶשׁ־חֶרֶב בְּיָדִי
 7   כִּי עַתָּה הֲרַגְתִּיךְ׃ וַתֹּאמֶר הָאָתוֹן אֶל־בִּלְעָם הֲלוֹא
 8   אָנֹכִי אֲתֹנְךָ אֲשֶׁר־רָכַבְתָּ עָלַי מֵעוֹדְךָ עַד־הַיּוֹם
 9   הַזֶּה הַהַסְכֵּן הִסְכַּנְתִּי לַעֲשׂוֹת לְךָ כֹּה וַיֹּאמֶר לֹא׃
10   וַיְגַל יְהוָה אֶת־עֵינֵי בִלְעָם וַיַּרְא אֶת־מַלְאַךְ יְהוָה
11   נִצָּב בַּדֶּרֶךְ וְחַרְבּוֹ שְׁלֻפָה בְּיָדוֹ וַיִּקֹּד וַיִּשְׁתַּחוּ
12   לְאַפָּיו׃ וַיֹּאמֶר אֵלָיו מַלְאַךְ יְהוָה עַל־מָה הִכִּיתָ אֶת־
13  אֲתֹנְךָ זֶה שָׁלוֹשׁ רְגָלִים הִנֵּה אָנֹכִי יָצָאתִי לְשָׂטָן
14   כִּי־יָרַט הַדֶּרֶךְ לְנֶגְדִּי׃ וַתִּרְאַנִי הָאָתוֹן וַתֵּט לְפָנַי זֶה
15   שָׁלֹשׁ רְגָלִים אוּלַי נָטְתָה מִפָּנַי כִּי עַתָּה גַּם־אֹתְכָה
16   הָרַגְתִּי וְאוֹתָהּ הֶחֱיֵיתִי׃ וַיֹּאמֶר בִּלְעָם אֶל־מַלְאַךְ
17   יְהוָה חָטָאתִי כִּי לֹא יָדַעְתִּי כִּי אַתָּה נִצָּב לִקְרָאתִי
18   בַּדָּרֶךְ וְעַתָּה אִם־רַע בְּעֵינֶיךָ אָשׁוּבָה לִּי׃ וַיֹּאמֶר
19   מַלְאַךְ יְהוָה אֶל־בִּלְעָם לֵךְ עִם־הָאֲנָשִׁים וְאֶפֶס אֶת־
20  הַדָּבָר אֲשֶׁר־אֲדַבֵּר אֵלֶיךָ אֹתוֹ תְדַבֵּר וַיֵּלֶךְ בִּלְעָם
21   עִם־שָׂרֵי בָלָק׃ וַיִּשְׁמַע בָּלָק כִּי בָא בִלְעָם וַיֵּצֵא
22   לִקְרָאתוֹ אֶל־עִיר מוֹאָב אֲשֶׁר עַל־גְּבוּל אַרְנֹן אֲשֶׁר
23   בִּקְצֵה הַגְּבוּל׃ וַיֹּאמֶר בָּלָק אֶל־בִּלְעָם הֲלֹא שָׁלֹחַ
24   שָׁלַחְתִּי אֵלֶיךָ לִקְרֹא־לָךְ לָמָּה לֹא־הָלַכְתָּ אֵלָי
25   הַאֻמְנָם לֹא אוּכַל כַּבְּדֶךָ׃ וַיֹּאמֶר בִּלְעָם אֶל־בָּלָק
26   הִנֵּה־בָאתִי אֵלֶיךָ עַתָּה הֲיָכֹל אוּכַל דַּבֵּר מְאוּמָה
27   הַדָּבָר אֲשֶׁר יָשִׂים אֱלֹהִים בְּפִי אֹתוֹ אֲדַבֵּר׃  וַיֵּלֶךְ
28   בִּלְעָם עִם־בָּלָק וַיָּבֹאוּ קִרְיַת חֻצוֹת׃ וַיִּזְבַּח בָּלָק
29   בָּקָר וָצֹאן וַיְשַׁלַּח לְבִלְעָם וְלַשָּׂרִים אֲשֶׁר אִתּוֹ׃
30   וַיְהִי בַבֹּקֶר וַיִּקַּח בָּלָק אֶת־בִּלְעָם וַיַּעֲלֵהוּ בָּמוֹת
31   בָּעַל וַיַּרְא מִשָּׁם קְצֵה הָעָם׃ וַיֹּאמֶר בִּלְעָם אֶל־בָּלָק
32   בְּנֵה־לִי בָזֶה שִׁבְעָה מִזְבְּחֹת וְהָכֵן לִי בָּזֶה שִׁבְעָה
33   פָרִים וְשִׁבְעָה אֵילִים׃ וַיַּעַשׂ בָּלָק כַּאֲשֶׁר דִּבֶּר
34   בִּלְעָם וַיַּעַל בָּלָק וּבִלְעָם פָּר וָאַיִל בַּמִּזְבֵּחַ׃ וַיֹּאמֶר
35   בִּלְעָם לְבָלָק הִתְיַצֵּב עַל־עֹלָתֶךָ וְאֵלְכָה אוּלַי
36   יִקָּרֵה יְהוָה לִקְרָאתִי וּדְבַר מַה־יַּרְאֵנִי וְהִגַּדְתִּי לָךְ
37   וַיֵּלֶךְ שֶׁפִי׃ וַיִּקָּר אֱלֹהִים אֶל־בִּלְעָם וַיֹּאמֶר אֵלָיו אֶת־
38  שִׁבְעַת הַמִּזְבְּחֹת עָרַכְתִּי וָאַעַל פָּר וָאַיִל בַּמִּזְבֵּחַ׃
39   וַיָּשֶׂם יְהוָה דָּבָר בְּפִי בִלְעָם וַיֹּאמֶר שׁוּב אֶל־בָּלָק
40   וְכֹה תְדַבֵּר׃ וַיָּשָׁב אֵלָיו וְהִנֵּה נִצָּב עַל־עֹלָתוֹ הוּא
41   וְכָל־שָׂרֵי מוֹאָב׃ וַיִּשָּׂא מְשָׁלוֹ וַיֹּאמַר מִן־אֲרָם יַנְחֵנִי
42   בָלָק מֶלֶךְ־מוֹאָב מֵהַרְרֵי־קֶדֶם לְכָה אָרָה־לִּי יַעֲקֹב
