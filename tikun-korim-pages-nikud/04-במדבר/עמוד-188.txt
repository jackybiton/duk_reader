 1  וִיהוֹשֻׁעַ בִּן־נוּן׃ [ס]             וַתִּקְרַבְנָה בְּנוֹת
 2   צְלָפְחָד בֶּן־חֵפֶר בֶּן־גִּלְעָד בֶּן־מָכִיר בֶּן־מְנַשֶּׁה
 3   לְמִשְׁפְּחֹת מְנַשֶּׁה בֶן־יוֹסֵף וְאֵלֶּה שְׁמוֹת בְּנֹתָיו
 4   מַחְלָה נֹעָה וְחָגְלָה וּמִלְכָּה וְתִרְצָה׃ וַתַּעֲמֹדְנָה לִפְנֵי
 5   מֹשֶׁה וְלִפְנֵי אֶלְעָזָר הַכֹּהֵן וְלִפְנֵי הַנְּשִׂיאִם וְכָל־
 6  הָעֵדָה פֶּתַח אֹהֶל־מוֹעֵד לֵאמֹר׃ אָבִינוּ מֵת בַּמִּדְבָּר
 7   וְהוּא לֹא־הָיָה בְּתוֹךְ הָעֵדָה הַנּוֹעָדִים עַל־יְהוָה בַּעֲדַת־
 8  קֹרַח כִּי־בְחֶטְאוֹ מֵת וּבָנִים לֹא־הָיוּ לוֹ׃ לָמָּה יִגָּרַע
 9   שֵׁם־אָבִינוּ מִתּוֹךְ מִשְׁפַּחְתּוֹ כִּי אֵין לוֹ בֵּן תְּנָה־לָּנוּ
10   אֲחֻזָּה בְּתוֹךְ אֲחֵי אָבִינוּ׃ וַיַּקְרֵב מֹשֶׁה אֶת־מִשְׁפָּטָן
11   לִפְנֵי יְהוָה׃ [פ]
12    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ כֵּן בְּנוֹת צְלָפְחָד
13   דֹּבְרֹת נָתֹן תִּתֵּן לָהֶם אֲחֻזַּת נַחֲלָה בְּתוֹךְ אֲחֵי
14   אֲבִיהֶם וְהַעֲבַרְתָּ אֶת־נַחֲלַת אֲבִיהֶן לָהֶן׃ וְאֶל־בְּנֵי
15   יִשְׂרָאֵל תְּדַבֵּר לֵאמֹר אִישׁ כִּי־יָמוּת וּבֵן אֵין לוֹ
16   וְהַעֲבַרְתֶּם אֶת־נַחֲלָתוֹ לְבִתּוֹ׃ וְאִם־אֵין לוֹ בַּת וּנְתַתֶּם
17   אֶת־נַחֲלָתוֹ לְאֶחָיו׃ וְאִם־אֵין לוֹ אַחִים וּנְתַתֶּם אֶת־
18  נַחֲלָתוֹ לַאֲחֵי אָבִיו׃ וְאִם־אֵין אַחִים לְאָבִיו וּנְתַתֶּם
19   אֶת־נַחֲלָתוֹ לִשְׁאֵרוֹ הַקָּרֹב אֵלָיו מִמִּשְׁפַּחְתּוֹ וְיָרַשׁ
20   אֹתָהּ וְהָיְתָה לִבְנֵי יִשְׂרָאֵל לְחֻקַּת מִשְׁפָּט כַּאֲשֶׁר
21   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [פ]
22   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה עֲלֵה אֶל־הַר הָעֲבָרִים הַזֶּה
23   וּרְאֵה אֶת־הָאָרֶץ אֲשֶׁר נָתַתִּי לִבְנֵי יִשְׂרָאֵל׃ וְרָאִיתָה
24   אֹתָהּ וְנֶאֱסַפְתָּ אֶל־עַמֶּיךָ גַּם־אָתָּה כַּאֲשֶׁר נֶאֱסַף
25   אַהֲרֹן אָחִיךָ׃ כַּאֲשֶׁר מְרִיתֶם פִּי בְּמִדְבַּר־צִן בִּמְרִיבַת
26   הָעֵדָה לְהַקְדִּישֵׁנִי בַמַּיִם לְעֵינֵיהֶם הֵם מֵי־מְרִיבַת
27   קָדֵשׁ מִדְבַּר־צִן׃ [ס]             וַיְדַבֵּר
28   מֹשֶׁה אֶל־יְהוָה לֵאמֹר׃ יִפְקֹד יְהוָה אֱלֹהֵי הָרוּחֹת
29   לְכָל־בָּשָׂר אִישׁ עַל־הָעֵדָה׃ אֲשֶׁר־יֵצֵא לִפְנֵיהֶם
30   וַאֲשֶׁר יָבֹא לִפְנֵיהֶם וַאֲשֶׁר יוֹצִיאֵם וַאֲשֶׁר יְבִיאֵם
31   וְלֹא תִהְיֶה עֲדַת יְהוָה כַּצֹּאן אֲשֶׁר אֵין־לָהֶם רֹעֶה׃
32   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה קַח־לְךָ אֶת־יְהוֹשֻׁעַ בִּן־נוּן
33   אִישׁ אֲשֶׁר־רוּחַ בּוֹ וְסָמַכְתָּ אֶת־יָדְךָ עָלָיו׃ וְהַעֲמַדְתָּ
34   אֹתוֹ לִפְנֵי אֶלְעָזָר הַכֹּהֵן וְלִפְנֵי כָּל־הָעֵדָה וְצִוִּיתָה
35   אֹתוֹ לְעֵינֵיהֶם׃ וְנָתַתָּה מֵהוֹדְךָ עָלָיו לְמַעַן יִשְׁמְעוּ
36   כָּל־עֲדַת בְּנֵי יִשְׂרָאֵל׃ וְלִפְנֵי אֶלְעָזָר הַכֹּהֵן יַעֲמֹד
37   וְשָׁאַל לוֹ בְּמִשְׁפַּט הָאוּרִים לִפְנֵי יְהוָה עַל־פִּיו
38   יֵצְאוּ וְעַל־פִּיו יָבֹאוּ הוּא וְכָל־בְּנֵי־יִשְׂרָאֵל אִתּוֹ וְכָל־
39  הָעֵדָה׃ וַיַּעַשׂ מֹשֶׁה כַּאֲשֶׁר צִוָּה יְהוָה אֹתוֹ וַיִּקַּח
40   אֶת־יְהוֹשֻׁעַ וַיַּעֲמִדֵהוּ לִפְנֵי אֶלְעָזָר הַכֹּהֵן וְלִפְנֵי
41   כָּל־הָעֵדָה׃ וַיִּסְמֹךְ אֶת־יָדָיו עָלָיו וַיְצַוֵּהוּ כַּאֲשֶׁר
42   דִּבֶּר יְהוָה בְּיַד־מֹשֶׁה׃ [פ]
