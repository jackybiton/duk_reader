 1  וּבַקָּטֹן כִּלָּה וַיִּמָּצֵא הַגָּבִיעַ בְּאַמְתַּחַת בִּנְיָמִן׃ וַיִּקְרְעוּ
 2   שִׂמְלֹתָם וַיַּעֲמֹס אִישׁ עַל־חֲמֹרוֹ וַיָּשֻׁבוּ הָעִירָה׃  וַיָּבֹא
 3   יְהוּדָה וְאֶחָיו בֵּיתָה יוֹסֵף וְהוּא עוֹדֶנּוּ שָׁם וַיִּפְּלוּ
 4   לְפָנָיו אָרְצָה׃ וַיֹּאמֶר לָהֶם יוֹסֵף מָה־הַמַּעֲשֶׂה הַזֶּה
 5   אֲשֶׁר עֲשִׂיתֶם הֲלוֹא יְדַעְתֶּם כִּי־נַחֵשׁ יְנַחֵשׁ אִישׁ
 6   אֲשֶׁר כָּמֹנִי׃ וַיֹּאמֶר יְהוּדָה מַה־נֹּאמַר לַאדֹנִי מַה־
 7  נְּדַבֵּר וּמַה־נִּצְטַדָּק הָאֱלֹהִים מָצָא אֶת־עֲו‍ֹן עֲבָדֶיךָ
 8   הִנֶּנּוּ עֲבָדִים לַאדֹנִי גַּם־אֲנַחְנוּ גַּם אֲשֶׁר־נִמְצָא הַגָּבִיעַ
 9   בְּיָדוֹ׃ וַיֹּאמֶר חָלִילָה לִּי מֵעֲשׂוֹת זֹאת הָאִישׁ אֲשֶׁר
10   נִמְצָא הַגָּבִיעַ בְּיָדוֹ הוּא יִהְיֶה־לִּי עָבֶד וְאַתֶּם עֲלוּ
11   לְשָׁלוֹם אֶל־אֲבִיכֶם׃ [ס]            וַיִּגַּשׁ אֵלָיו
12   יְהוּדָה וַיֹּאמֶר בִּי אֲדֹנִי יְדַבֶּר־נָא עַבְדְּךָ דָבָר בְּאָזְנֵי
13   אֲדֹנִי וְאַל־יִחַר אַפְּךָ בְּעַבְדֶּךָ כִּי כָמוֹךָ כְּפַרְעֹה׃
14   אֲדֹנִי שָׁאַל אֶת־עֲבָדָיו לֵאמֹר הֲיֵשׁ־לָכֶם אָב אוֹ־
15  אָח׃ וַנֹּאמֶר אֶל־אֲדֹנִי יֶשׁ־לָנוּ אָב זָקֵן וְיֶלֶד זְקֻנִים
16   קָטָן וְאָחִיו מֵת וַיִּוָּתֵר הוּא לְבַדּוֹ לְאִמּוֹ וְאָבִיו אֲהֵבוֹ׃♢
17   וַתֹּאמֶר אֶל־עֲבָדֶיךָ הוֹרִדֻהוּ אֵלָי וְאָשִׂימָה עֵינִי
18   עָלָיו׃ וַנֹּאמֶר אֶל־אֲדֹנִי לֹא־יוּכַל הַנַּעַר לַעֲזֹב אֶת־
19  אָבִיו וְעָזַב אֶת־אָבִיו וָמֵת׃ וַתֹּאמֶר אֶל־עֲבָדֶיךָ אִם־
20  לֹא יֵרֵד אֲחִיכֶם הַקָּטֹן אִתְּכֶם לֹא תֹסִפוּן לִרְאוֹת
21   פָּנָי׃ וַיְהִי כִּי עָלִינוּ אֶל־עַבְדְּךָ אָבִי וַנַּגֶּד־לוֹ אֵת דִּבְרֵי
22   אֲדֹנִי׃♢ וַיֹּאמֶר אָבִינוּ שֻׁבוּ שִׁבְרוּ־לָנוּ מְעַט־אֹכֶל׃
23   וַנֹּאמֶר לֹא נוּכַל לָרֶדֶת אִם־יֵשׁ אָחִינוּ הַקָּטֹן אִתָּנוּ
24   וְיָרַדְנוּ כִּי־לֹא נוּכַל לִרְאוֹת פְּנֵי הָאִישׁ וְאָחִינוּ הַקָּטֹן
25   אֵינֶנּוּ אִתָּנוּ׃ וַיֹּאמֶר עַבְדְּךָ אָבִי אֵלֵינוּ אַתֶּם יְדַעְתֶּם
26   כִּי שְׁנַיִם יָלְדָה־לִּי אִשְׁתִּי׃ וַיֵּצֵא הָאֶחָד מֵאִתִּי וָאֹמַר
27   אַךְ טָרֹף טֹרָף וְלֹא רְאִיתִיו עַד־הֵנָּה׃ וּלְקַחְתֶּם גַּם־
28  אֶת־זֶה מֵעִם פָּנַי וְקָרָהוּ אָסוֹן וְהוֹרַדְתֶּם אֶת־שֵׂיבָתִי
29   בְּרָעָה שְׁאֹלָה׃ וְעַתָּה כְּבֹאִי אֶל־עַבְדְּךָ אָבִי וְהַנַּעַר
30   אֵינֶנּוּ אִתָּנוּ וְנַפְשׁוֹ קְשׁוּרָה בְנַפְשׁוֹ׃♢  וְהָיָה כִּרְאוֹתוֹ כִּי־
31  אֵין הַנַּעַר וָמֵת וְהוֹרִידוּ עֲבָדֶיךָ אֶת־שֵׂיבַת עַבְדְּךָ
32   אָבִינוּ בְּיָגוֹן שְׁאֹלָה׃ כִּי עַבְדְּךָ עָרַב אֶת־הַנַּעַר מֵעִם
33   אָבִי לֵאמֹר אִם־לֹא אֲבִיאֶנּוּ אֵלֶיךָ וְחָטָאתִי לְאָבִי
34   כָּל־הַיָּמִים׃ וְעַתָּה יֵשֶׁב־נָא עַבְדְּךָ תַּחַת הַנַּעַר עֶבֶד
35   לַאדֹנִי וְהַנַּעַר יַעַל עִם־אֶחָיו׃ כִּי־אֵיךְ אֶעֱלֶה אֶל־אָבִי
36   וְהַנַּעַר אֵינֶנּוּ אִתִּי פֶּן אֶרְאֶה בָרָע אֲשֶׁר יִמְצָא אֶת־
37  אָבִי׃ וְלֹא־יָכֹל יוֹסֵף לְהִתְאַפֵּק לְכֹל הַנִּצָּבִים עָלָיו
38   וַיִּקְרָא הוֹצִיאוּ כָל־אִישׁ מֵעָלָי וְלֹא־עָמַד אִישׁ אִתּוֹ
39   בְּהִתְוַדַּע יוֹסֵף אֶל־אֶחָיו׃ וַיִּתֵּן אֶת־קֹלוֹ בִּבְכִי וַיִּשְׁמְעוּ
40   מִצְרַיִם וַיִּשְׁמַע בֵּית פַּרְעֹה׃ וַיֹּאמֶר יוֹסֵף אֶל־אֶחָיו
41   אֲנִי יוֹסֵף הַעוֹד אָבִי חָי וְלֹא־יָכְלוּ אֶחָיו לַעֲנוֹת אֹתוֹ
42   כִּי נִבְהֲלוּ מִפָּנָיו׃ וַיֹּאמֶר יוֹסֵף אֶל־אֶחָיו גְּשׁוּ־נָא אֵלַי
