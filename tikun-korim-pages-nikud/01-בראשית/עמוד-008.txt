 1  וַתֵּלֶךְ הַתֵּבָה עַל־פְּנֵי הַמָּיִם׃ וְהַמַּיִם גָּבְרוּ מְאֹד
 2   מְאֹד עַל־הָאָרֶץ וַיְכֻסּוּ כָּל־הֶהָרִים הַגְּבֹהִים אֲשֶׁר־תַּחַת
 3   כָּל־הַשָּׁמָיִם׃ חֲמֵשׁ עֶשְׂרֵה אַמָּה מִלְמַעְלָה גָּבְרוּ הַמָּיִם
 4   וַיְכֻסּוּ הֶהָרִים׃ וַיִּגְוַע כָּל־בָּשָׂר  הָרֹמֵשׂ עַל־הָאָרֶץ
 5   בָּעוֹף וּבַבְּהֵמָה וּבַחַיָּה וּבְכָל־הַשֶּׁרֶץ הַשֹּׁרֵץ עַל־
 6  הָאָרֶץ וְכֹל הָאָדָם׃ כֹּל אֲשֶׁר נִשְׁמַת־רוּחַ חַיִּים בְּאַפָּיו
 7   מִכֹּל אֲשֶׁר בֶּחָרָבָה מֵתוּ׃ וַיִּמַח אֶת־כָּל־הַיְקוּם  אֲשֶׁר
 8   עַל־פְּנֵי הָאֲדָמָה מֵאָדָם עַד־בְּהֵמָה עַד־רֶמֶשׂ וְעַד־
 9  עוֹף הַשָּׁמַיִם וַיִּמָּחוּ מִן־הָאָרֶץ וַיִּשָּׁאֶר אַךְ־נֹחַ וַאֲשֶׁר
10   אִתּוֹ בַּתֵּבָה׃ וַיִּגְבְּרוּ הַמַּיִם עַל־הָאָרֶץ חֲמִשִּׁים וּמְאַת
11   יוֹם׃ וַיִּזְכֹּר אֱלֹהִים אֶת־נֹחַ וְאֵת כָּל־הַחַיָּה וְאֶת־כָּל־
12  הַבְּהֵמָה אֲשֶׁר אִתּוֹ בַּתֵּבָה וַיַּעֲבֵר אֱלֹהִים רוּחַ עַל־
13  הָאָרֶץ וַיָּשֹׁכּוּ הַמָּיִם׃ וַיִּסָּכְרוּ מַעְיְנֹת תְּהוֹם וַאֲרֻבֹּת
14   הַשָּׁמָיִם וַיִּכָּלֵא הַגֶּשֶׁם מִן־הַשָּׁמָיִם׃ וַיָּשֻׁבוּ הַמַּיִם מֵעַל
15   הָאָרֶץ הָלוֹךְ וָשׁוֹב וַיַּחְסְרוּ הַמַּיִם מִקְצֵה חֲמִשִּׁים
16   וּמְאַת יוֹם׃ וַתָּנַח הַתֵּבָה בַּחֹדֶשׁ הַשְּׁבִיעִי בְּשִׁבְעָה־
17  עָשָׂר יוֹם לַחֹדֶשׁ עַל הָרֵי אֲרָרָט׃ וְהַמַּיִם הָיוּ הָלוֹךְ
18   וְחָסוֹר עַד הַחֹדֶשׁ הָעֲשִׂירִי בָּעֲשִׂירִי בְּאֶחָד לַחֹדֶשׁ
19   נִרְאוּ רָאשֵׁי הֶהָרִים׃ וַיְהִי מִקֵּץ אַרְבָּעִים יוֹם וַיִּפְתַּח
20   נֹחַ אֶת־חַלּוֹן הַתֵּבָה אֲשֶׁר עָשָׂה׃ וַיְשַׁלַּח אֶת־הָעֹרֵב
21   וַיֵּצֵא יָצוֹא וָשׁוֹב עַד־יְבֹשֶׁת הַמַּיִם מֵעַל הָאָרֶץ׃
22   וַיְשַׁלַּח אֶת־הַיּוֹנָה מֵאִתּוֹ לִרְאוֹת הֲקַלּוּ הַמַּיִם מֵעַל
23   פְּנֵי הָאֲדָמָה׃ וְלֹא־מָצְאָה הַיּוֹנָה מָנוֹחַ לְכַף־רַגְלָהּ
24   וַתָּשָׁב אֵלָיו אֶל־הַתֵּבָה כִּי־מַיִם עַל־פְּנֵי כָל־הָאָרֶץ
25   וַיִּשְׁלַח יָדוֹ וַיִּקָּחֶהָ וַיָּבֵא אֹתָהּ אֵלָיו אֶל־הַתֵּבָה׃ וַיָּחֶל
26   עוֹד שִׁבְעַת יָמִים אֲחֵרִים וַיֹּסֶף שַׁלַּח אֶת־הַיּוֹנָה מִן־
27  הַתֵּבָה׃ וַתָּבֹא אֵלָיו הַיּוֹנָה לְעֵת עֶרֶב וְהִנֵּה עֲלֵה־זַיִת
28   טָרָף בְּפִיהָ וַיֵּדַע נֹחַ כִּי־קַלּוּ הַמַּיִם מֵעַל הָאָרֶץ׃ וַיִּיָּחֶל
29   עוֹד שִׁבְעַת יָמִים אֲחֵרִים וַיְשַׁלַּח אֶת־הַיּוֹנָה וְלֹא־
30  יָסְפָה שׁוּב־אֵלָיו עוֹד׃ וַיְהִי בְּאַחַת וְשֵׁשׁ־מֵאוֹת שָׁנָה
31   בָּרִאשׁוֹן בְּאֶחָד לַחֹדֶשׁ חָרְבוּ הַמַּיִם מֵעַל הָאָרֶץ
32   וַיָּסַר נֹחַ אֶת־מִכְסֵה הַתֵּבָה וַיַּרְא וְהִנֵּה חָרְבוּ פְּנֵי
33   הָאֲדָמָה׃ וּבַחֹדֶשׁ הַשֵּׁנִי בְּשִׁבְעָה וְעֶשְׂרִים יוֹם לַחֹדֶשׁ
34   יָבְשָׁה הָאָרֶץ׃ [ס]              וַיְדַבֵּר אֱלֹהִים אֶל־
35  נֹחַ לֵאמֹר׃ צֵא מִן־הַתֵּבָה אַתָּה וְאִשְׁתְּךָ וּבָנֶיךָ וּנְשֵׁי־
36  בָנֶיךָ אִתָּךְ׃ כָּל־הַחַיָּה אֲשֶׁר־אִתְּךָ מִכָּל־בָּשָׂר בָּעוֹף
37   וּבַבְּהֵמָה וּבְכָל־הָרֶמֶשׂ הָרֹמֵשׂ עַל־הָאָרֶץ הַיְצֵא
38   אִתָּךְ וְשָׁרְצוּ בָאָרֶץ וּפָרוּ וְרָבוּ עַל־הָאָרֶץ׃ וַיֵּצֵא־נֹחַ
39   וּבָנָיו וְאִשְׁתּוֹ וּנְשֵׁי־בָנָיו אִתּוֹ׃ כָּל־הַחַיָּה כָּל־הָרֶמֶשׂ
40   וְכָל־הָעוֹף כֹּל רוֹמֵשׂ עַל־הָאָרֶץ לְמִשְׁפְּחֹתֵיהֶם יָצְאוּ
41   מִן־הַתֵּבָה׃ וַיִּבֶן נֹחַ מִזְבֵּחַ לַיהוָה וַיִּקַּח מִכֹּל  הַבְּהֵמָה
42   הַטְּהֹרָה וּמִכֹּל הָעוֹף הַטָּהוֹר וַיַּעַל עֹלֹת בַּמִּזְבֵּחַ׃
