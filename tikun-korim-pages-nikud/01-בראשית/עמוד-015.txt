 1  וַיֹּאמֶר אֵלָיו קְחָה לִי עֶגְלָה מְשֻׁלֶּשֶׁת וְעֵז מְשֻׁלֶּשֶׁת
 2   וְאַיִל מְשֻׁלָּשׁ וְתֹר וְגוֹזָל׃ וַיִּקַּח־לוֹ אֶת־כָּל־אֵלֶּה
 3   וַיְבַתֵּר אֹתָם בַּתָּוֶךְ וַיִּתֵּן אִישׁ־בִּתְרוֹ לִקְרַאת רֵעֵהוּ
 4   וְאֶת־הַצִּפֹּר לֹא בָתָר׃ וַיֵּרֶד הָעַיִט עַל־הַפְּגָרִים וַיַּשֵּׁב
 5   אֹתָם אַבְרָם׃ וַיְהִי הַשֶּׁמֶשׁ לָבוֹא וְתַרְדֵּמָה נָפְלָה עַל־
 6  אַבְרָם וְהִנֵּה אֵימָה חֲשֵׁכָה גְדֹלָה נֹפֶלֶת עָלָיו׃ וַיֹּאמֶר
 7   לְאַבְרָם יָדֹעַ תֵּדַע כִּי־גֵר  יִהְיֶה זַרְעֲךָ בְּאֶרֶץ לֹא
 8   לָהֶם וַעֲבָדוּם וְעִנּוּ אֹתָם אַרְבַּע מֵאוֹת שָׁנָה׃ וְגַם אֶת־
 9  הַגּוֹי אֲשֶׁר יַעֲבֹדוּ דָּן אָנֹכִי וְאַחֲרֵי־כֵן יֵצְאוּ בִּרְכֻשׁ
10   גָּדוֹל׃ וְאַתָּה תָּבוֹא אֶל־אֲבֹתֶיךָ בְּשָׁלוֹם תִּקָּבֵר
11   בְּשֵׂיבָה טוֹבָה׃ וְדוֹר רְבִיעִי יָשׁוּבוּ הֵנָּה כִּי לֹא־שָׁלֵם
12   עֲו‍ֹן הָאֱמֹרִי עַד־הֵנָּה׃ וַיְהִי הַשֶּׁמֶשׁ בָּאָה וַעֲלָטָה
13   הָיָה וְהִנֵּה תַנּוּר עָשָׁן וְלַפִּיד אֵשׁ אֲשֶׁר עָבַר בֵּין
14   הַגְּזָרִים הָאֵלֶּה׃ בַּיּוֹם הַהוּא כָּרַת יְהוָה אֶת־אַבְרָם
15   בְּרִית לֵאמֹר לְזַרְעֲךָ נָתַתִּי אֶת־הָאָרֶץ הַזֹּאת מִנְּהַר
16   מִצְרַיִם עַד־הַנָּהָר הַגָּדֹל נְהַר־פְּרָת׃ אֶת־הַקֵּינִי וְאֶת־
17  הַקְּנִזִּי וְאֵת הַקַּדְמֹנִי׃ וְאֶת־הַחִתִּי וְאֶת־הַפְּרִזִּי וְאֶת־
18  הָרְפָאִים׃ וְאֶת־הָאֱמֹרִי וְאֶת־הַכְּנַעֲנִי וְאֶת־הַגִּרְגָּשִׁי וְאֶת־
19  הַיְבוּסִי׃ [ס]             וְשָׂרַי אֵשֶׁת אַבְרָם לֹא
20   יָלְדָה לוֹ וְלָהּ שִׁפְחָה מִצְרִית וּשְׁמָהּ הָגָר׃ וַתֹּאמֶר
21   שָׂרַי אֶל־אַבְרָם הִנֵּה־נָא עֲצָרַנִי יְהוָה מִלֶּדֶת בֹּא־נָא
22   אֶל־שִׁפְחָתִי אוּלַי אִבָּנֶה מִמֶּנָּה וַיִּשְׁמַע אַבְרָם לְקוֹל
23   שָׂרָי׃ וַתִּקַּח שָׂרַי אֵשֶׁת־אַבְרָם אֶת־הָגָר הַמִּצְרִית
24   שִׁפְחָתָהּ מִקֵּץ עֶשֶׂר שָׁנִים לְשֶׁבֶת אַבְרָם בְּאֶרֶץ
25   כְּנָעַן וַתִּתֵּן אֹתָהּ לְאַבְרָם אִישָׁהּ לוֹ לְאִשָּׁה׃ וַיָּבֹא אֶל־
26  הָגָר וַתַּהַר וַתֵּרֶא כִּי הָרָתָה וַתֵּקַל גְּבִרְתָּהּ בְּעֵינֶיהָ׃
27   וַתֹּאמֶר שָׂרַי אֶל־אַבְרָם חֲמָסִי עָלֶיךָ אָנֹכִי נָתַתִּי
28   שִׁפְחָתִי בְּחֵיקֶךָ וַתֵּרֶא כִּי הָרָתָה וָאֵקַל בְּעֵינֶיהָ
29   יִשְׁפֹּט יְהוָה בֵּינִי וּבֵינֶיךָ׃ וַיֹּאמֶר אַבְרָם אֶל־שָׂרַי
30   הִנֵּה שִׁפְחָתֵךְ בְּיָדֵךְ עֲשִׂי־לָהּ הַטּוֹב בְּעֵינָיִךְ וַתְּעַנֶּהָ
31   שָׂרַי וַתִּבְרַח מִפָּנֶיהָ׃ וַיִּמְצָאָהּ מַלְאַךְ יְהוָה עַל־עֵין
32   הַמַּיִם בַּמִּדְבָּר עַל־הָעַיִן בְּדֶרֶךְ שׁוּר׃ וַיֹּאמַר הָגָר
33   שִׁפְחַת שָׂרַי אֵי־מִזֶּה בָאת וְאָנָה תֵלֵכִי וַתֹּאמֶר מִפְּנֵי
34   שָׂרַי גְּבִרְתִּי אָנֹכִי בֹּרַחַת׃ וַיֹּאמֶר לָהּ מַלְאַךְ יְהוָה
35   שׁוּבִי אֶל־גְּבִרְתֵּךְ וְהִתְעַנִּי תַּחַת יָדֶיהָ׃ וַיֹּאמֶר לָהּ
36   מַלְאַךְ יְהוָה הַרְבָּה אַרְבֶּה אֶת־זַרְעֵךְ וְלֹא יִסָּפֵר
37   מֵרֹב׃ וַיֹּאמֶר לָהּ מַלְאַךְ יְהוָה הִנָּךְ הָרָה וְיֹלַדְתְּ בֵּן
38   וְקָרָאת שְׁמוֹ יִשְׁמָעֵאל כִּי־שָׁמַע יְהוָה אֶל־עָנְיֵךְ׃
39   וְהוּא יִהְיֶה פֶּרֶא אָדָם יָדוֹ בַכֹּל וְיַד כֹּל בּוֹ וְעַל־פְּנֵי
40   כָל־אֶחָיו יִשְׁכֹּן׃ וַתִּקְרָא שֵׁם־יְהוָה הַדֹּבֵר אֵלֶיהָ
41   אַתָּה אֵל רֳאִי כִּי אָמְרָה הֲגַם הֲלֹם רָאִיתִי אַחֲרֵי רֹאִי׃
42   עַל־כֵּן קָרָא לַבְּאֵר בְּאֵר לַחַי רֹאִי הִנֵּה בֵין־קָדֵשׁ
