 1  וַיַּעֲבֹד עִמּוֹ עוֹד שֶׁבַע־שָׁנִים אֲחֵרוֹת׃ וַיַּרְא יְהוָה כִּי־
 2  שְׂנוּאָה לֵאָה וַיִּפְתַּח אֶת־רַחְמָהּ וְרָחֵל עֲקָרָה׃ וַתַּהַר
 3   לֵאָה וַתֵּלֶד בֵּן וַתִּקְרָא שְׁמוֹ רְאוּבֵן כִּי אָמְרָה כִּי־
 4  רָאָה יְהוָה בְּעָנְיִי כִּי עַתָּה יֶאֱהָבַנִי אִישִׁי׃ וַתַּהַר עוֹד
 5   וַתֵּלֶד בֵּן וַתֹּאמֶר כִּי־שָׁמַע יְהוָה כִּי־שְׂנוּאָה אָנֹכִי
 6   וַיִּתֶּן־לִי גַּם־אֶת־זֶה וַתִּקְרָא שְׁמוֹ שִׁמְעוֹן׃ וַתַּהַר עוֹד
 7   וַתֵּלֶד בֵּן וַתֹּאמֶר עַתָּה הַפַּעַם יִלָּוֶה אִישִׁי אֵלַי כִּי־
 8  יָלַדְתִּי לוֹ שְׁלֹשָׁה בָנִים עַל־כֵּן קָרָא־שְׁמוֹ לֵוִי׃ וַתַּהַר
 9   עוֹד וַתֵּלֶד בֵּן וַתֹּאמֶר הַפַּעַם אוֹדֶה אֶת־יְהוָה עַל־כֵּן
10   קָרְאָה שְׁמוֹ יְהוּדָה וַתַּעֲמֹד מִלֶּדֶת׃ וַתֵּרֶא רָחֵל כִּי
11   לֹא יָלְדָה לְיַעֲקֹב וַתְּקַנֵּא רָחֵל בַּאֲחֹתָהּ וַתֹּאמֶר אֶל־
12  יַעֲקֹב הָבָה־לִּי בָנִים וְאִם־אַיִן מֵתָה אָנֹכִי׃ וַיִּחַר־אַף
13   יַעֲקֹב בְּרָחֵל וַיֹּאמֶר הֲתַחַת אֱלֹהִים אָנֹכִי אֲשֶׁר־מָנַע
14   מִמֵּךְ פְּרִי־בָטֶן׃ וַתֹּאמֶר הִנֵּה אֲמָתִי בִלְהָה בֹּא אֵלֶיהָ
15   וְתֵלֵד עַל־בִּרְכַּי וְאִבָּנֶה גַם־אָנֹכִי מִמֶּנָּה׃ וַתִּתֶּן־לוֹ אֶת־
16  בִּלְהָה שִׁפְחָתָהּ לְאִשָּׁה וַיָּבֹא אֵלֶיהָ יַעֲקֹב׃ וַתַּהַר
17   בִּלְהָה וַתֵּלֶד לְיַעֲקֹב בֵּן׃ וַתֹּאמֶר רָחֵל דָּנַנִּי אֱלֹהִים
18   וְגַם שָׁמַע בְּקֹלִי וַיִּתֶּן־לִי בֵּן עַל־כֵּן קָרְאָה שְׁמוֹ דָּן׃
19   וַתַּהַר עוֹד וַתֵּלֶד בִּלְהָה שִׁפְחַת רָחֵל בֵּן שֵׁנִי לְיַעֲקֹב׃
20   וַתֹּאמֶר רָחֵל נַפְתּוּלֵי אֱלֹהִים  נִפְתַּלְתִּי עִם־אֲחֹתִי גַּם־
21  יָכֹלְתִּי וַתִּקְרָא שְׁמוֹ נַפְתָּלִי׃ וַתֵּרֶא לֵאָה כִּי עָמְדָה
22   מִלֶּדֶת וַתִּקַּח אֶת־זִלְפָּה שִׁפְחָתָהּ וַתִּתֵּן אֹתָהּ לְיַעֲקֹב
23   לְאִשָּׁה׃ וַתֵּלֶד זִלְפָּה שִׁפְחַת לֵאָה לְיַעֲקֹב בֵּן׃ וַתֹּאמֶר
24   לֵאָה בָּא גָד וַתִּקְרָא אֶת־שְׁמוֹ גָּד׃ וַתֵּלֶד זִלְפָּה שִׁפְחַת
25   לֵאָה בֵּן שֵׁנִי לְיַעֲקֹב׃ וַתֹּאמֶר לֵאָה בְּאָשְׁרִי כִּי
26   אִשְּׁרוּנִי בָּנוֹת וַתִּקְרָא אֶת־שְׁמוֹ אָשֵׁר׃  וַיֵּלֶךְ רְאוּבֵן
27   בִּימֵי קְצִיר־חִטִּים וַיִּמְצָא דוּדָאִים בַּשָּׂדֶה וַיָּבֵא אֹתָם
28   אֶל־לֵאָה אִמּוֹ וַתֹּאמֶר רָחֵל אֶל־לֵאָה תְּנִי־נָא לִי
29   מִדּוּדָאֵי בְּנֵךְ׃ וַתֹּאמֶר לָהּ הַמְעַט קַחְתֵּךְ אֶת־אִישִׁי
30   וְלָקַחַת גַּם אֶת־דּוּדָאֵי בְּנִי וַתֹּאמֶר רָחֵל לָכֵן יִשְׁכַּב
31   עִמָּךְ הַלַּיְלָה תַּחַת דּוּדָאֵי בְנֵךְ׃ וַיָּבֹא יַעֲקֹב מִן־הַשָּׂדֶה
32   בָּעֶרֶב וַתֵּצֵא לֵאָה לִקְרָאתוֹ וַתֹּאמֶר אֵלַי תָּבוֹא כִּי
33   שָׂכֹר שְׂכַרְתִּיךָ בְּדוּדָאֵי בְּנִי וַיִּשְׁכַּב עִמָּהּ בַּלַּיְלָה
34   הוּא׃ וַיִּשְׁמַע אֱלֹהִים אֶל־לֵאָה וַתַּהַר וַתֵּלֶד לְיַעֲקֹב
35   בֵּן חֲמִישִׁי׃  וַתֹּאמֶר לֵאָה נָתַן אֱלֹהִים שְׂכָרִי אֲשֶׁר־
36  נָתַתִּי שִׁפְחָתִי לְאִישִׁי וַתִּקְרָא שְׁמוֹ יִשָּׂשכָר׃ וַתַּהַר
37   עוֹד לֵאָה וַתֵּלֶד בֵּן־שִׁשִּׁי לְיַעֲקֹב׃ וַתֹּאמֶר לֵאָה
38   זְבָדַנִי אֱלֹהִים  אֹתִי זֵבֶד טוֹב הַפַּעַם יִזְבְּלֵנִי אִישִׁי כִּי־
39  יָלַדְתִּי לוֹ שִׁשָּׁה בָנִים וַתִּקְרָא אֶת־שְׁמוֹ זְבֻלוּן׃ וְאַחַר
40   יָלְדָה בַּת וַתִּקְרָא אֶת־שְׁמָהּ דִּינָה׃ וַיִּזְכֹּר אֱלֹהִים
41   אֶת־רָחֵל וַיִּשְׁמַע אֵלֶיהָ אֱלֹהִים וַיִּפְתַּח אֶת־רַחְמָהּ׃
42   וַתַּהַר וַתֵּלֶד בֵּן וַתֹּאמֶר אָסַף אֱלֹהִים אֶת־חֶרְפָּתִי׃
