 1  וַתָּבֹאנָה אֶל־קִרְבֶּנָה וְלֹא נוֹדַע כִּי־בָאוּ אֶל־קִרְבֶּנָה
 2   וּמַרְאֵיהֶן רַע כַּאֲשֶׁר בַּתְּחִלָּה וָאִיקָץ׃ וָאֵרֶא בַּחֲלֹמִי
 3   וְהִנֵּה  שֶׁבַע שִׁבֳּלִים עֹלֹת בְּקָנֶה אֶחָד מְלֵאֹת וְטֹבוֹת׃
 4   וְהִנֵּה שֶׁבַע שִׁבֳּלִים צְנֻמוֹת דַּקּוֹת שְׁדֻפוֹת קָדִים
 5   צֹמְחוֹת אַחֲרֵיהֶם׃ וַתִּבְלַעְןָ הַשִּׁבֳּלִים הַדַּקֹּת אֵת
 6   שֶׁבַע הַשִּׁבֳּלִים הַטֹּבוֹת וָאֹמַר אֶל־הַחַרְטֻמִּים וְאֵין
 7   מַגִּיד לִי׃ וַיֹּאמֶר יוֹסֵף אֶל־פַּרְעֹה חֲלוֹם פַּרְעֹה אֶחָד
 8   הוּא אֵת אֲשֶׁר הָאֱלֹהִים עֹשֶׂה הִגִּיד לְפַרְעֹה׃ שֶׁבַע
 9   פָּרֹת הַטֹּבֹת שֶׁבַע שָׁנִים הֵנָּה וְשֶׁבַע הַשִּׁבֳּלִים
10   הַטֹּבֹת שֶׁבַע שָׁנִים הֵנָּה חֲלוֹם אֶחָד הוּא׃ וְשֶׁבַע
11   הַפָּרוֹת הָרַקּוֹת וְהָרָעֹת הָעֹלֹת אַחֲרֵיהֶן שֶׁבַע שָׁנִים
12   הֵנָּה וְשֶׁבַע הַשִּׁבֳּלִים הָרֵקוֹת שְׁדֻפוֹת הַקָּדִים
13   יִהְיוּ שֶׁבַע שְׁנֵי רָעָב׃ הוּא הַדָּבָר אֲשֶׁר דִּבַּרְתִּי
14   אֶל־פַּרְעֹה אֲשֶׁר הָאֱלֹהִים עֹשֶׂה הֶרְאָה אֶת־פַּרְעֹה׃
15   הִנֵּה שֶׁבַע שָׁנִים בָּאוֹת שָׂבָע גָּדוֹל בְּכָל־אֶרֶץ
16   מִצְרָיִם׃ וְקָמוּ שֶׁבַע שְׁנֵי רָעָב אַחֲרֵיהֶן וְנִשְׁכַּח כָּל־
17  הַשָּׂבָע בְּאֶרֶץ מִצְרָיִם וְכִלָּה הָרָעָב אֶת־הָאָרֶץ׃ וְלֹא־
18  יִוָּדַע הַשָּׂבָע בָּאָרֶץ מִפְּנֵי הָרָעָב הַהוּא אַחֲרֵי־כֵן
19   כִּי־כָבֵד הוּא מְאֹד׃ וְעַל הִשָּׁנוֹת הַחֲלוֹם אֶל־פַּרְעֹה
20   פַּעֲמָיִם כִּי־נָכוֹן הַדָּבָר מֵעִם הָאֱלֹהִים וּמְמַהֵר
21   הָאֱלֹהִים לַעֲשֹׂתוֹ׃ וְעַתָּה יֵרֶא פַרְעֹה אִישׁ נָבוֹן
22   וְחָכָם וִישִׁיתֵהוּ עַל־אֶרֶץ מִצְרָיִם׃ יַעֲשֶׂה פַרְעֹה
23   וְיַפְקֵד פְּקִדִים עַל־הָאָרֶץ וְחִמֵּשׁ אֶת־אֶרֶץ מִצְרַיִם
24   בְּשֶׁבַע שְׁנֵי הַשָּׂבָע׃ וְיִקְבְּצוּ אֶת־כָּל־אֹכֶל הַשָּׁנִים
25   הַטֹּבוֹת הַבָּאֹת הָאֵלֶּה וְיִצְבְּרוּ־בָר תַּחַת יַד־פַּרְעֹה
26   אֹכֶל בֶּעָרִים וְשָׁמָרוּ׃ וְהָיָה הָאֹכֶל לְפִקָּדוֹן לָאָרֶץ
27   לְשֶׁבַע שְׁנֵי הָרָעָב אֲשֶׁר תִּהְיֶיןָ בְּאֶרֶץ מִצְרָיִם וְלֹא־
28  תִכָּרֵת הָאָרֶץ בָּרָעָב׃ וַיִּיטַב הַדָּבָר בְּעֵינֵי פַרְעֹה
29   וּבְעֵינֵי כָּל־עֲבָדָיו׃ וַיֹּאמֶר פַּרְעֹה אֶל־עֲבָדָיו הֲנִמְצָא
30   כָזֶה אִישׁ אֲשֶׁר רוּחַ אֱלֹהִים בּוֹ׃  וַיֹּאמֶר פַּרְעֹה
31   אֶל־יוֹסֵף אַחֲרֵי הוֹדִיעַ אֱלֹהִים אוֹתְךָ אֶת־כָּל־זֹאת
32   אֵין־נָבוֹן וְחָכָם כָּמוֹךָ׃ אַתָּה תִּהְיֶה עַל־בֵּיתִי וְעַל־
33  פִּיךָ יִשַּׁק כָּל־עַמִּי רַק הַכִּסֵּא אֶגְדַּל מִמֶּךָּ׃ וַיֹּאמֶר
34   פַּרְעֹה אֶל־יוֹסֵף רְאֵה נָתַתִּי אֹתְךָ עַל כָּל־אֶרֶץ
35   מִצְרָיִם׃ וַיָּסַר פַּרְעֹה אֶת־טַבַּעְתּוֹ מֵעַל יָדוֹ וַיִּתֵּן אֹתָהּ
36   עַל־יַד יוֹסֵף וַיַּלְבֵּשׁ אֹתוֹ בִּגְדֵי־שֵׁשׁ וַיָּשֶׂם רְבִד
37   הַזָּהָב עַל־צַוָּארוֹ׃ וַיַּרְכֵּב אֹתוֹ בְּמִרְכֶּבֶת הַמִּשְׁנֶה
38   אֲשֶׁר־לוֹ וַיִּקְרְאוּ לְפָנָיו אַבְרֵךְ וְנָתוֹן אֹתוֹ עַל כָּל־
39  אֶרֶץ מִצְרָיִם׃ וַיֹּאמֶר פַּרְעֹה אֶל־יוֹסֵף אֲנִי פַרְעֹה
40   וּבִלְעָדֶיךָ לֹא־יָרִים אִישׁ אֶת־יָדוֹ וְאֶת־רַגְלוֹ בְּכָל־
41  אֶרֶץ מִצְרָיִם׃ וַיִּקְרָא פַרְעֹה שֵׁם־יוֹסֵף צָפְנַת פַּעְנֵחַ
42   וַיִּתֶּן־לוֹ אֶת־אָסְנַת בַּת־פּוֹטִי פֶרַע כֹּהֵן אֹן לְאִשָּׁה
