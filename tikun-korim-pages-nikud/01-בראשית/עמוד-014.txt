 1  וַיֵּצֵא מֶלֶךְ־סְדֹם וּמֶלֶךְ עֲמֹרָה וּמֶלֶךְ אַדְמָה וּמֶלֶךְ
 2   צְבֹיִים וּמֶלֶךְ בֶּלַע הִוא־צֹעַר וַיַּעַרְכוּ אִתָּם מִלְחָמָה
 3   בְּעֵמֶק הַשִּׂדִּים׃ אֵת כְּדָרְלָעֹמֶר מֶלֶךְ עֵילָם וְתִדְעָל
 4   מֶלֶךְ גּוֹיִם וְאַמְרָפֶל מֶלֶךְ שִׁנְעָר וְאַרְיוֹךְ מֶלֶךְ אֶלָּסָר
 5   אַרְבָּעָה מְלָכִים אֶת־הַחֲמִשָּׁה׃ וְעֵמֶק הַשִּׂדִּים בֶּאֱרֹת
 6   בֶּאֱרֹת חֵמָר וַיָּנֻסוּ מֶלֶךְ־סְדֹם וַעֲמֹרָה וַיִּפְּלוּ־שָׁמָּה
 7   וְהַנִּשְׁאָרִים הֶרָה נָּסוּ׃ וַיִּקְחוּ אֶת־כָּל־רְכֻשׁ סְדֹם
 8   וַעֲמֹרָה וְאֶת־כָּל־אָכְלָם וַיֵּלֵכוּ׃ וַיִּקְחוּ אֶת־לוֹט וְאֶת־
 9  רְכֻשׁוֹ בֶּן־אֲחִי אַבְרָם וַיֵּלֵכוּ וְהוּא יֹשֵׁב בִּסְדֹם׃ וַיָּבֹא
10   הַפָּלִיט וַיַּגֵּד לְאַבְרָם הָעִבְרִי וְהוּא שֹׁכֵן בְּאֵלֹנֵי
11   מַמְרֵא הָאֱמֹרִי אֲחִי אֶשְׁכֹּל וַאֲחִי עָנֵר וְהֵם בַּעֲלֵי
12   בְרִית־אַבְרָם׃ וַיִּשְׁמַע אַבְרָם כִּי נִשְׁבָּה אָחִיו וַיָּרֶק
13   אֶת־חֲנִיכָיו יְלִידֵי בֵיתוֹ שְׁמֹנָה עָשָׂר וּשְׁלֹשׁ מֵאוֹת
14   וַיִּרְדֹּף עַד־דָּן׃ וַיֵּחָלֵק עֲלֵיהֶם  לַיְלָה הוּא וַעֲבָדָיו
15   וַיַּכֵּם וַיִּרְדְּפֵם עַד־חוֹבָה אֲשֶׁר מִשְּׂמֹאל לְדַמָּשֶׂק׃
16   וַיָּשֶׁב אֵת כָּל־הָרְכֻשׁ וְגַם אֶת־לוֹט אָחִיו וּרְכֻשׁוֹ
17   הֵשִׁיב וְגַם אֶת־הַנָּשִׁים וְאֶת־הָעָם׃ וַיֵּצֵא מֶלֶךְ־סְדֹם
18   לִקְרָאתוֹ אַחֲרֵי שׁוּבוֹ מֵהַכּוֹת אֶת־כְּדָרְלָעֹמֶר וְאֶת־
19  הַמְּלָכִים אֲשֶׁר אִתּוֹ אֶל־עֵמֶק שָׁוֵה הוּא עֵמֶק הַמֶּלֶךְ׃
20   וּמַלְכִּי־צֶדֶק מֶלֶךְ שָׁלֵם הוֹצִיא לֶחֶם וָיָיִן וְהוּא כֹהֵן
21   לְאֵל עֶלְיוֹן׃ וַיְבָרְכֵהוּ וַיֹּאמַר בָּרוּךְ אַבְרָם לְאֵל עֶלְיוֹן
22   קֹנֵה שָׁמַיִם וָאָרֶץ׃ וּבָרוּךְ אֵל עֶלְיוֹן אֲשֶׁר־מִגֵּן צָרֶיךָ
23   בְּיָדֶךָ וַיִּתֶּן־לוֹ מַעֲשֵׂר מִכֹּל׃  וַיֹּאמֶר מֶלֶךְ־סְדֹם אֶל־
24  אַבְרָם תֶּן־לִי הַנֶּפֶשׁ וְהָרְכֻשׁ קַח־לָךְ׃ וַיֹּאמֶר אַבְרָם
25   אֶל־מֶלֶךְ סְדֹם הֲרִמֹתִי יָדִי אֶל־יְהוָה אֵל עֶלְיוֹן קֹנֵה
26   שָׁמַיִם וָאָרֶץ׃ אִם־מִחוּט וְעַד שְׂרוֹךְ־נַעַל וְאִם־אֶקַּח
27   מִכָּל־אֲשֶׁר־לָךְ וְלֹא תֹאמַר אֲנִי הֶעֱשַׁרְתִּי אֶת־אַבְרָם׃
28   בִּלְעָדַי רַק אֲשֶׁר אָכְלוּ הַנְּעָרִים וְחֵלֶק הָאֲנָשִׁים
29   אֲשֶׁר הָלְכוּ אִתִּי עָנֵר אֶשְׁכֹּל וּמַמְרֵא הֵם יִקְחוּ
30   חֶלְקָם׃ [ס]             אַחַר  הַדְּבָרִים הָאֵלֶּה
31   הָיָה דְבַר־יְהוָה אֶל־אַבְרָם בַּמַּחֲזֶה לֵאמֹר אַל־תִּירָא
32   אַבְרָם אָנֹכִי מָגֵן לָךְ שְׂכָרְךָ הַרְבֵּה מְאֹד׃ וַיֹּאמֶר
33   אַבְרָם אֲדֹנָי יְהוִה מַה־תִּתֶּן־לִי וְאָנֹכִי הוֹלֵךְ עֲרִירִי
34   וּבֶן־מֶשֶׁק בֵּיתִי הוּא דַּמֶּשֶׂק אֱלִיעֶזֶר׃ וַיֹּאמֶר אַבְרָם
35   הֵן לִי לֹא נָתַתָּה זָרַע וְהִנֵּה בֶן־בֵּיתִי יוֹרֵשׁ אֹתִי׃ וְהִנֵּה
36   דְבַר־יְהוָה אֵלָיו לֵאמֹר לֹא יִירָשְׁךָ זֶה כִּי־אִם אֲשֶׁר
37   יֵצֵא מִמֵּעֶיךָ הוּא יִירָשֶׁךָ׃ וַיּוֹצֵא אֹתוֹ הַחוּצָה וַיֹּאמֶר
38   הַבֶּט־נָא הַשָּׁמַיְמָה וּסְפֹר הַכּוֹכָבִים אִם־תּוּכַל לִסְפֹּר
39   אֹתָם וַיֹּאמֶר לוֹ כֹּה יִהְיֶה זַרְעֶךָ׃ וְהֶאֱמִן בַּיהוָה
40   וַיַּחְשְׁבֶהָ לּוֹ צְדָקָה׃  וַיֹּאמֶר אֵלָיו אֲנִי יְהוָה אֲשֶׁר
41   הוֹצֵאתִיךָ מֵאוּר כַּשְׂדִּים לָתֶת לְךָ אֶת־הָאָרֶץ הַזֹּאת
42   לְרִשְׁתָּהּ׃ וַיֹּאמַר אֲדֹנָי יְהוִה בַּמָּה אֵדַע כִּי אִירָשֶׁנָּה׃
