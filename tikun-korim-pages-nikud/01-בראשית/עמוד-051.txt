 1  וַיֹּאמֶר לֹא־יֵרֵד בְּנִי עִמָּכֶם כִּי־אָחִיו מֵת וְהוּא לְבַדּוֹ
 2   נִשְׁאָר וּקְרָאָהוּ אָסוֹן בַּדֶּרֶךְ אֲשֶׁר תֵּלְכוּ־בָהּ
 3   וְהוֹרַדְתֶּם אֶת־שֵׂיבָתִי בְּיָגוֹן שְׁאוֹלָה׃ וְהָרָעָב כָּבֵד
 4   בָּאָרֶץ׃ וַיְהִי כַּאֲשֶׁר כִּלּוּ לֶאֱכֹל אֶת־הַשֶּׁבֶר אֲשֶׁר
 5   הֵבִיאוּ מִמִּצְרָיִם וַיֹּאמֶר אֲלֵיהֶם אֲבִיהֶם שֻׁבוּ שִׁבְרוּ־
 6  לָנוּ מְעַט־אֹכֶל׃ וַיֹּאמֶר אֵלָיו יְהוּדָה לֵאמֹר הָעֵד
 7   הֵעִד בָּנוּ הָאִישׁ לֵאמֹר לֹא־תִרְאוּ פָנַי בִּלְתִּי אֲחִיכֶם
 8   אִתְּכֶם׃ אִם־יֶשְׁךָ מְשַׁלֵּחַ אֶת־אָחִינוּ אִתָּנוּ נֵרְדָה
 9   וְנִשְׁבְּרָה לְךָ אֹכֶל׃ וְאִם־אֵינְךָ מְשַׁלֵּחַ לֹא נֵרֵד כִּי־
10  הָאִישׁ אָמַר אֵלֵינוּ לֹא־תִרְאוּ פָנַי בִּלְתִּי אֲחִיכֶם
11   אִתְּכֶם׃ וַיֹּאמֶר יִשְׂרָאֵל לָמָה הֲרֵעֹתֶם לִי לְהַגִּיד
12   לָאִישׁ הַעוֹד לָכֶם אָח׃ וַיֹּאמְרוּ שָׁאוֹל שָׁאַל־הָאִישׁ
13   לָנוּ וּלְמוֹלַדְתֵּנוּ לֵאמֹר הַעוֹד אֲבִיכֶם חַי הֲיֵשׁ לָכֶם
14   אָח וַנַּגֶּד־לוֹ עַל־פִּי הַדְּבָרִים הָאֵלֶּה הֲיָדוֹעַ נֵדַע כִּי
15   יֹאמַר הוֹרִידוּ אֶת־אֲחִיכֶם׃ וַיֹּאמֶר יְהוּדָה אֶל־יִשְׂרָאֵל
16   אָבִיו שִׁלְחָה הַנַּעַר אִתִּי וְנָקוּמָה וְנֵלֵכָה וְנִחְיֶה וְלֹא
17   נָמוּת גַּם־אֲנַחְנוּ גַם־אַתָּה גַּם־טַפֵּנוּ׃ אָנֹכִי אֶעֶרְבֶנּוּ
18   מִיָּדִי תְּבַקְשֶׁנּוּ אִם־לֹא הֲבִיאֹתִיו אֵלֶיךָ וְהִצַּגְתִּיו
19   לְפָנֶיךָ וְחָטָאתִי לְךָ כָּל־הַיָּמִים׃ כִּי לוּלֵא הִתְמַהְמָהְנוּ
20   כִּי־עַתָּה שַׁבְנוּ זֶה פַעֲמָיִם׃ וַיֹּאמֶר אֲלֵהֶם יִשְׂרָאֵל
21   אֲבִיהֶם אִם־כֵּן  אֵפוֹא זֹאת עֲשׂוּ קְחוּ מִזִּמְרַת הָאָרֶץ
22   בִּכְלֵיכֶם וְהוֹרִידוּ לָאִישׁ מִנְחָה מְעַט צֳרִי וּמְעַט
23   דְּבַשׁ נְכֹאת וָלֹט בָּטְנִים וּשְׁקֵדִים׃ וְכֶסֶף מִשְׁנֶה
24   קְחוּ בְיֶדְכֶם וְאֶת־הַכֶּסֶף הַמּוּשָׁב בְּפִי אַמְתְּחֹתֵיכֶם
25   תָּשִׁיבוּ בְיֶדְכֶם אוּלַי מִשְׁגֶּה הוּא׃ וְאֶת־אֲחִיכֶם
26   קָחוּ וְקוּמוּ שׁוּבוּ אֶל־הָאִישׁ׃ וְאֵל שַׁדַּי יִתֵּן לָכֶם
27   רַחֲמִים לִפְנֵי הָאִישׁ וְשִׁלַּח לָכֶם אֶת־אֲחִיכֶם אַחֵר
28   וְאֶת־בִּנְיָמִין וַאֲנִי כַּאֲשֶׁר שָׁכֹלְתִּי שָׁכָלְתִּי׃ וַיִּקְחוּ
29   הָאֲנָשִׁים אֶת־הַמִּנְחָה הַזֹּאת וּמִשְׁנֶה־כֶּסֶף לָקְחוּ
30   בְיָדָם וְאֶת־בִּנְיָמִן וַיָּקֻמוּ וַיֵּרְדוּ מִצְרַיִם וַיַּעַמְדוּ לִפְנֵי
31   יוֹסֵף׃  וַיַּרְא יוֹסֵף אִתָּם אֶת־בִּנְיָמִין וַיֹּאמֶר לַאֲשֶׁר עַל־
32  בֵּיתוֹ הָבֵא אֶת־הָאֲנָשִׁים הַבָּיְתָה וּטְבֹחַ טֶבַח וְהָכֵן
33   כִּי אִתִּי יֹאכְלוּ הָאֲנָשִׁים בַּצָּהֳרָיִם׃ וַיַּעַשׂ הָאִישׁ
34   כַּאֲשֶׁר אָמַר יוֹסֵף וַיָּבֵא הָאִישׁ אֶת־הָאֲנָשִׁים בֵּיתָה
35   יוֹסֵף׃ וַיִּירְאוּ הָאֲנָשִׁים כִּי הוּבְאוּ בֵּית יוֹסֵף וַיֹּאמְרוּ
36   עַל־דְּבַר הַכֶּסֶף הַשָּׁב בְּאַמְתְּחֹתֵינוּ בַּתְּחִלָּה אֲנַחְנוּ
37   מוּבָאִים לְהִתְגֹּלֵל עָלֵינוּ וּלְהִתְנַפֵּל עָלֵינוּ וְלָקַחַת
38   אֹתָנוּ לַעֲבָדִים וְאֶת־חֲמֹרֵינוּ׃ וַיִּגְּשׁוּ אֶל־הָאִישׁ
39   אֲשֶׁר עַל־בֵּית יוֹסֵף וַיְדַבְּרוּ אֵלָיו פֶּתַח הַבָּיִת׃ וַיֹּאמְרוּ בִּי
40   אֲדֹנִי יָרֹד יָרַדְנוּ בַּתְּחִלָּה לִשְׁבָּר־אֹכֶל׃ וַיְהִי כִּי־בָאנוּ
41   אֶל־הַמָּלוֹן וַנִּפְתְּחָה אֶת־אַמְתְּחֹתֵינוּ וְהִנֵּה כֶסֶף־אִישׁ
42   בְּפִי אַמְתַּחְתּוֹ כַּסְפֵּנוּ בְּמִשְׁקָלוֹ וַנָּשֶׁב אֹתוֹ בְּיָדֵנוּ׃
