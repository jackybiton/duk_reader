 1  וַיֹּאמֶר אֲלֵהֶם יוֹסֵף הוּא אֲשֶׁר דִּבַּרְתִּי אֲלֵכֶם לֵאמֹר
 2   מְרַגְּלִים אַתֶּם׃ בְּזֹאת תִּבָּחֵנוּ חֵי פַרְעֹה אִם־תֵּצְאוּ מִזֶּה
 3   כִּי אִם־בְּבוֹא אֲחִיכֶם הַקָּטֹן הֵנָּה׃ שִׁלְחוּ מִכֶּם אֶחָד
 4   וְיִקַּח אֶת־אֲחִיכֶם וְאַתֶּם הֵאָסְרוּ וְיִבָּחֲנוּ דִּבְרֵיכֶם
 5   הַאֱמֶת אִתְּכֶם וְאִם־לֹא חֵי פַרְעֹה כִּי מְרַגְּלִים אַתֶּם׃
 6   וַיֶּאֱסֹף אֹתָם אֶל־מִשְׁמָר שְׁלֹשֶׁת יָמִים׃ וַיֹּאמֶר אֲלֵהֶם
 7   יוֹסֵף בַּיּוֹם הַשְּׁלִישִׁי זֹאת עֲשׂוּ וִחְיוּ אֶת־הָאֱלֹהִים
 8   אֲנִי יָרֵא׃  אִם־כֵּנִים אַתֶּם אֲחִיכֶם אֶחָד יֵאָסֵר בְּבֵית
 9   מִשְׁמַרְכֶם וְאַתֶּם לְכוּ הָבִיאוּ שֶׁבֶר רַעֲבוֹן בָּתֵּיכֶם׃
10   וְאֶת־אֲחִיכֶם הַקָּטֹן תָּבִיאוּ אֵלַי וְיֵאָמְנוּ דִבְרֵיכֶם
11   וְלֹא תָמוּתוּ וַיַּעֲשׂוּ־כֵן׃ וַיֹּאמְרוּ אִישׁ אֶל־אָחִיו אֲבָל
12   אֲשֵׁמִים  אֲנַחְנוּ עַל־אָחִינוּ אֲשֶׁר רָאִינוּ צָרַת נַפְשׁוֹ
13   בְּהִתְחַנְנוֹ אֵלֵינוּ וְלֹא שָׁמָעְנוּ עַל־כֵּן בָּאָה אֵלֵינוּ
14   הַצָּרָה הַזֹּאת׃ וַיַּעַן רְאוּבֵן אֹתָם לֵאמֹר הֲלוֹא אָמַרְתִּי
15   אֲלֵיכֶם  לֵאמֹר אַל־תֶּחֶטְאוּ בַיֶּלֶד וְלֹא שְׁמַעְתֶּם וְגַם־
16  דָּמוֹ הִנֵּה נִדְרָשׁ׃ וְהֵם לֹא יָדְעוּ כִּי שֹׁמֵעַ יוֹסֵף כִּי
17   הַמֵּלִיץ בֵּינֹתָם׃ וַיִּסֹּב מֵעֲלֵיהֶם וַיֵּבְךְּ וַיָּשָׁב אֲלֵהֶם
18   וַיְדַבֵּר אֲלֵהֶם וַיִּקַּח מֵאִתָּם אֶת־שִׁמְעוֹן וַיֶּאֱסֹר אֹתוֹ
19   לְעֵינֵיהֶם׃ וַיְצַו יוֹסֵף וַיְמַלְאוּ אֶת־כְּלֵיהֶם בָּר וּלְהָשִׁיב
20   כַּסְפֵּיהֶם אִישׁ אֶל־שַׂקּוֹ וְלָתֵת לָהֶם צֵדָה לַדָּרֶךְ
21   וַיַּעַשׂ לָהֶם כֵּן׃ וַיִּשְׂאוּ אֶת־שִׁבְרָם עַל־חֲמֹרֵיהֶם
22   וַיֵּלְכוּ מִשָּׁם׃ וַיִּפְתַּח הָאֶחָד אֶת־שַׂקּוֹ לָתֵת מִסְפּוֹא
23   לַחֲמֹרוֹ בַּמָּלוֹן וַיַּרְא אֶת־כַּסְפּוֹ וְהִנֵּה־הוּא בְּפִי
24   אַמְתַּחְתּוֹ׃ וַיֹּאמֶר אֶל־אֶחָיו הוּשַׁב כַּסְפִּי וְגַם הִנֵּה
25   בְאַמְתַּחְתִּי וַיֵּצֵא לִבָּם וַיֶּחֶרְדוּ אִישׁ אֶל־אָחִיו לֵאמֹר
26   מַה־זֹּאת עָשָׂה אֱלֹהִים לָנוּ׃ וַיָּבֹאוּ אֶל־יַעֲקֹב אֲבִיהֶם
27   אַרְצָה כְּנָעַן וַיַּגִּידוּ לוֹ אֵת כָּל־הַקֹּרֹת אֹתָם לֵאמֹר׃
28   דִּבֶּר הָאִישׁ אֲדֹנֵי הָאָרֶץ אִתָּנוּ קָשׁוֹת וַיִּתֵּן אֹתָנוּ
29   כִּמְרַגְּלִים אֶת־הָאָרֶץ׃ וַנֹּאמֶר אֵלָיו כֵּנִים אֲנָחְנוּ לֹא
30   הָיִינוּ מְרַגְּלִים׃ שְׁנֵים־עָשָׂר אֲנַחְנוּ אַחִים בְּנֵי אָבִינוּ
31   הָאֶחָד אֵינֶנּוּ וְהַקָּטֹן הַיּוֹם אֶת־אָבִינוּ בְּאֶרֶץ כְּנָעַן׃
32   וַיֹּאמֶר אֵלֵינוּ הָאִישׁ אֲדֹנֵי הָאָרֶץ בְּזֹאת אֵדַע כִּי כֵנִים
33   אַתֶּם אֲחִיכֶם הָאֶחָד הַנִּיחוּ אִתִּי וְאֶת־רַעֲבוֹן בָּתֵּיכֶם
34   קְחוּ וָלֵכוּ׃ וְהָבִיאוּ אֶת־אֲחִיכֶם הַקָּטֹן אֵלַי וְאֵדְעָה
35   כִּי לֹא מְרַגְּלִים אַתֶּם כִּי כֵנִים אַתֶּם אֶת־אֲחִיכֶם אֶתֵּן
36   לָכֶם וְאֶת־הָאָרֶץ תִּסְחָרוּ׃ וַיְהִי הֵם מְרִיקִים שַׂקֵּיהֶם
37   וְהִנֵּה־אִישׁ צְרוֹר־כַּסְפּוֹ בְּשַׂקּוֹ וַיִּרְאוּ אֶת־צְרֹרוֹת
38   כַּסְפֵּיהֶם הֵמָּה וַאֲבִיהֶם וַיִּירָאוּ׃ וַיֹּאמֶר אֲלֵהֶם
39   יַעֲקֹב אֲבִיהֶם אֹתִי שִׁכַּלְתֶּם יוֹסֵף אֵינֶנּוּ וְשִׁמְעוֹן
40   אֵינֶנּוּ וְאֶת־בִּנְיָמִן תִּקָּחוּ עָלַי הָיוּ כֻלָּנָה׃ וַיֹּאמֶר
41   רְאוּבֵן אֶל־אָבִיו לֵאמֹר אֶת־שְׁנֵי בָנַי תָּמִית אִם־לֹא
42   אֲבִיאֶנּוּ אֵלֶיךָ תְּנָה אֹתוֹ עַל־יָדִי וַאֲנִי אֲשִׁיבֶנּוּ אֵלֶיךָ׃
