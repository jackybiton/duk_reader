1וַיִּקְרָאשֵׁם־הַמָּקוֹםהַהוּאמַחֲנָיִם׃
2וַיִּשְׁלַחיַעֲקֹבמַלְאָכִיםלְפָנָיואֶל־עֵשָׂואָחִיואַרְצָה
3שֵׂעִירשְׂדֵהאֱדוֹם׃וַיְצַואֹתָםלֵאמֹרכֹּהתֹאמְרוּן
4לַאדֹנִילְעֵשָׂוכֹּהאָמַרעַבְדְּךָיַעֲקֹבעִם־לָבָןגַּרְתִּי
5וָאֵחַרעַד־עָתָּה׃וַיְהִי־לִישׁוֹרוַחֲמוֹרצֹאןוְעֶבֶד
6וְשִׁפְחָהוָאֶשְׁלְחָהלְהַגִּידלַאדֹנִילִמְצֹא־חֵןבְּעֵינֶיךָ׃♢
7וַיָּשֻׁבוּהַמַּלְאָכִיםאֶל־יַעֲקֹבלֵאמֹרבָּאנוּאֶל־אָחִיךָ
8אֶל־עֵשָׂווְגַםהֹלֵךְלִקְרָאתְךָוְאַרְבַּע־מֵאוֹתאִישׁעִמּוֹ׃
9וַיִּירָאיַעֲקֹבמְאֹדוַיֵּצֶרלוֹוַיַּחַץאֶת־הָעָםאֲשֶׁר־אִתּוֹ
10וְאֶת־הַצֹּאןוְאֶת־הַבָּקָרוְהַגְּמַלִּיםלִשְׁנֵימַחֲנוֹת׃וַיֹּאמֶר
11אִם־יָבוֹאעֵשָׂואֶל־הַמַּחֲנֶההָאַחַתוְהִכָּהוּוְהָיָההַמַּחֲנֶה
12הַנִּשְׁאָרלִפְלֵיטָה׃♢וַיֹּאמֶריַעֲקֹבאֱלֹהֵיאָבִיאַבְרָהָם
13וֵאלֹהֵיאָבִייִצְחָקיְהוָההָאֹמֵראֵלַישׁוּבלְאַרְצְךָ
14וּלְמוֹלַדְתְּךָוְאֵיטִיבָהעִמָּךְ׃קָטֹנְתִּימִכֹּלהַחֲסָדִים
15וּמִכָּל־הָאֱמֶתאֲשֶׁרעָשִׂיתָאֶת־עַבְדֶּךָכִּיבְמַקְלִי
16עָבַרְתִּיאֶת־הַיַּרְדֵּןהַזֶּהוְעַתָּההָיִיתִילִשְׁנֵימַחֲנוֹת׃
17הַצִּילֵנִינָאמִיַּדאָחִימִיַּדעֵשָׂוכִּי־יָרֵאאָנֹכִיאֹתוֹפֶּן־
18יָבוֹאוְהִכַּנִיאֵםעַל־בָּנִים׃וְאַתָּהאָמַרְתָּהֵיטֵבאֵיטִיב
19עִמָּךְוְשַׂמְתִּיאֶת־זַרְעֲךָכְּחוֹלהַיָּםאֲשֶׁרלֹא־יִסָּפֵר
20מֵרֹב׃♢וַיָּלֶןשָׁםבַּלַּיְלָההַהוּאוַיִּקַּחמִן־הַבָּאבְיָדוֹ
21מִנְחָהלְעֵשָׂואָחִיו׃עִזִּיםמָאתַיִםוּתְיָשִׁיםעֶשְׂרִים
22רְחֵלִיםמָאתַיִםוְאֵילִיםעֶשְׂרִים׃גְּמַלִּיםמֵינִיקוֹת
23וּבְנֵיהֶםשְׁלֹשִׁיםפָּרוֹתאַרְבָּעִיםוּפָרִיםעֲשָׂרָה
24אֲתֹנֹתעֶשְׂרִיםוַעְיָרִםעֲשָׂרָה׃וַיִּתֵּןבְּיַד־עֲבָדָיועֵדֶר
25עֵדֶרלְבַדּוֹוַיֹּאמֶראֶל־עֲבָדָיועִבְרוּלְפָנַיוְרֶוַח
26תָּשִׂימוּבֵּיןעֵדֶרוּבֵיןעֵדֶר׃וַיְצַואֶת־הָרִאשׁוֹןלֵאמֹר
27כִּייִפְגָשְׁךָעֵשָׂואָחִיוּשְׁאֵלְךָלֵאמֹרלְמִי־אַתָּהוְאָנָה
28תֵלֵךְוּלְמִיאֵלֶּהלְפָנֶיךָ׃וְאָמַרְתָּלְעַבְדְּךָלְיַעֲקֹבמִנְחָה
29הִואשְׁלוּחָהלַאדֹנִילְעֵשָׂווְהִנֵּהגַם־הוּאאַחֲרֵינוּ׃וַיְצַו
30גַּםאֶת־הַשֵּׁנִיגַּםאֶת־הַשְּׁלִישִׁיגַּםאֶת־כָּל־הַהֹלְכִים
31אַחֲרֵיהָעֲדָרִיםלֵאמֹרכַּדָּבָרהַזֶּהתְּדַבְּרוּןאֶל־עֵשָׂו
32בְּמֹצַאֲכֶםאֹתוֹ׃וַאֲמַרְתֶּםגַּםהִנֵּהעַבְדְּךָיַעֲקֹבאַחֲרֵינוּ
33כִּי־אָמַראֲכַפְּרָהפָנָיובַּמִּנְחָההַהֹלֶכֶתלְפָנָיוְאַחֲרֵי־
34כֵןאֶרְאֶהפָנָיואוּלַייִשָּׂאפָנָי׃וַתַּעֲבֹרהַמִּנְחָהעַל־
35פָּנָיווְהוּאלָןבַּלַּיְלָה־הַהוּאבַּמַּחֲנֶה׃וַיָּקָםבַּלַּיְלָההוּא
36וַיִּקַּחאֶת־שְׁתֵּינָשָׁיווְאֶת־שְׁתֵּישִׁפְחֹתָיווְאֶת־אַחַד
37עָשָׂריְלָדָיווַיַּעֲבֹראֵתמַעֲבַריַבֹּק׃וַיִּקָּחֵםוַיַּעֲבִרֵם
38אֶת־הַנָּחַלוַיַּעֲבֵראֶת־אֲשֶׁר־לוֹ׃וַיִּוָּתֵריַעֲקֹבלְבַדּוֹ
39וַיֵּאָבֵקאִישׁעִמּוֹעַדעֲלוֹתהַשָּׁחַר׃וַיַּרְאכִּילֹאיָכֹל
40לוֹוַיִּגַּעבְּכַף־יְרֵכוֹוַתֵּקַעכַּף־יֶרֶךְיַעֲקֹבבְּהֵאָבְקוֹעִמּוֹ׃
41וַיֹּאמֶרשַׁלְּחֵנִיכִּיעָלָההַשָּׁחַרוַיֹּאמֶרלֹאאֲשַׁלֵּחֲךָ
