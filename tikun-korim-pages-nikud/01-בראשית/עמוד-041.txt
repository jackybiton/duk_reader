 1  וַיִּשְׁמַע יִשְׂרָאֵל [פ]
 2   וַיִּהְיוּ בְנֵי־יַעֲקֹב שְׁנֵים עָשָׂר׃ בְּנֵי לֵאָה בְּכוֹר יַעֲקֹב
 3   רְאוּבֵן וְשִׁמְעוֹן וְלֵוִי וִיהוּדָה וְיִשָּׂשכָר וּזְבֻלוּן׃ בְּנֵי
 4   רָחֵל יוֹסֵף וּבִנְיָמִן׃ וּבְנֵי בִלְהָה שִׁפְחַת רָחֵל דָּן וְנַפְתָּלִי׃
 5   וּבְנֵי זִלְפָּה שִׁפְחַת לֵאָה גָּד וְאָשֵׁר אֵלֶּה בְּנֵי יַעֲקֹב
 6   אֲשֶׁר יֻלַּד־לוֹ בְּפַדַּן אֲרָם׃ וַיָּבֹא יַעֲקֹב אֶל־יִצְחָק אָבִיו
 7   מַמְרֵא קִרְיַת הָאַרְבַּע הִוא חֶבְרוֹן אֲשֶׁר־גָּר־שָׁם
 8   אַבְרָהָם וְיִצְחָק׃ וַיִּהְיוּ יְמֵי יִצְחָק מְאַת שָׁנָה וּשְׁמֹנִים
 9   שָׁנָה׃ וַיִּגְוַע יִצְחָק וַיָּמָת וַיֵּאָסֶף אֶל־עַמָּיו זָקֵן וּשְׂבַע
10   יָמִים וַיִּקְבְּרוּ אֹתוֹ עֵשָׂו וְיַעֲקֹב בָּנָיו׃ [פ]
11   וְאֵלֶּה תֹּלְדוֹת עֵשָׂו הוּא אֱדוֹם׃ עֵשָׂו לָקַח אֶת־
12  נָשָׁיו מִבְּנוֹת כְּנָעַן אֶת־עָדָה בַּת־אֵילוֹן הַחִתִּי וְאֶת־
13  אָהֳלִיבָמָה בַּת־עֲנָה בַּת־צִבְעוֹן הַחִוִּי׃ וְאֶת־בָּשְׂמַת בַּת־
14  יִשְׁמָעֵאל אֲחוֹת נְבָיוֹת׃ וַתֵּלֶד עָדָה לְעֵשָׂו אֶת־אֱלִיפָז
15   וּבָשְׂמַת יָלְדָה אֶת־רְעוּאֵל׃ וְאָהֳלִיבָמָה יָלְדָה אֶת־
16  יְעוּשׁ וְאֶת־יַעְלָם וְאֶת־קֹרַח אֵלֶּה בְּנֵי עֵשָׂו אֲשֶׁר
17   יֻלְּדוּ־לוֹ בְּאֶרֶץ כְּנָעַן׃ וַיִּקַּח עֵשָׂו אֶת־נָשָׁיו וְאֶת־בָּנָיו
18   וְאֶת־בְּנֹתָיו וְאֶת־כָּל־נַפְשׁוֹת בֵּיתוֹ וְאֶת־מִקְנֵהוּ וְאֶת־
19  כָּל־בְּהֶמְתּוֹ וְאֵת כָּל־קִנְיָנוֹ אֲשֶׁר רָכַשׁ בְּאֶרֶץ כְּנָעַן
20   וַיֵּלֶךְ אֶל־אֶרֶץ מִפְּנֵי יַעֲקֹב אָחִיו׃ כִּי־הָיָה רְכוּשָׁם רָב
21   מִשֶּׁבֶת יַחְדָּו וְלֹא יָכְלָה אֶרֶץ מְגוּרֵיהֶם לָשֵׂאת אֹתָם
22   מִפְּנֵי מִקְנֵיהֶם׃ וַיֵּשֶׁב עֵשָׂו בְּהַר שֵׂעִיר עֵשָׂו הוּא
23   אֱדוֹם׃ וְאֵלֶּה תֹּלְדוֹת עֵשָׂו אֲבִי אֱדוֹם בְּהַר שֵׂעִיר׃
24   אֵלֶּה שְׁמוֹת בְּנֵי־עֵשָׂו אֱלִיפַז בֶּן־עָדָה אֵשֶׁת עֵשָׂו
25   רְעוּאֵל בֶּן־בָּשְׂמַת אֵשֶׁת עֵשָׂו׃ וַיִּהְיוּ בְּנֵי אֱלִיפָז תֵּימָן
26   אוֹמָר צְפוֹ וְגַעְתָּם וּקְנַז׃ וְתִמְנַע  הָיְתָה פִילֶגֶשׁ לֶאֱלִיפַז
27   בֶּן־עֵשָׂו וַתֵּלֶד לֶאֱלִיפַז אֶת־עֲמָלֵק אֵלֶּה בְּנֵי עָדָה
28   אֵשֶׁת עֵשָׂו׃ וְאֵלֶּה בְּנֵי רְעוּאֵל נַחַת וָזֶרַח שַׁמָּה וּמִזָּה
29   אֵלֶּה הָיוּ בְּנֵי בָשְׂמַת אֵשֶׁת עֵשָׂו׃ וְאֵלֶּה הָיוּ בְּנֵי
30   אָהֳלִיבָמָה בַת־עֲנָה בַּת־צִבְעוֹן אֵשֶׁת עֵשָׂו וַתֵּלֶד
31   לְעֵשָׂו אֶת־יעיש יְעוּשׁ וְאֶת־יַעְלָם וְאֶת־קֹרַח׃ אֵלֶּה אַלּוּפֵי
32   בְנֵי־עֵשָׂו בְּנֵי אֱלִיפַז בְּכוֹר עֵשָׂו אַלּוּף תֵּימָן אַלּוּף
33   אוֹמָר אַלּוּף צְפוֹ אַלּוּף קְנַז׃ אַלּוּף־קֹרַח אַלּוּף גַּעְתָּם
34   אַלּוּף עֲמָלֵק אֵלֶּה אַלּוּפֵי אֱלִיפַז בְּאֶרֶץ אֱדוֹם אֵלֶּה
35   בְּנֵי עָדָה׃ וְאֵלֶּה בְּנֵי רְעוּאֵל בֶּן־עֵשָׂו אַלּוּף נַחַת
36   אַלּוּף זֶרַח אַלּוּף שַׁמָּה אַלּוּף מִזָּה אֵלֶּה אַלּוּפֵי
37   רְעוּאֵל בְּאֶרֶץ אֱדוֹם אֵלֶּה בְּנֵי בָשְׂמַת אֵשֶׁת עֵשָׂו׃
38   וְאֵלֶּה בְּנֵי אָהֳלִיבָמָה אֵשֶׁת עֵשָׂו אַלּוּף יְעוּשׁ אַלּוּף
39   יַעְלָם אַלּוּף קֹרַח אֵלֶּה אַלּוּפֵי אָהֳלִיבָמָה בַּת־עֲנָה
40   אֵשֶׁת עֵשָׂו׃ אֵלֶּה בְנֵי־עֵשָׂו וְאֵלֶּה אַלּוּפֵיהֶם הוּא
41   אֱדוֹם׃ [ס]              אֵלֶּה בְנֵי־שֵׂעִיר הַחֹרִי
42   יֹשְׁבֵי הָאָרֶץ לוֹטָן וְשׁוֹבָל וְצִבְעוֹן וַעֲנָה׃ וְדִשׁוֹן וְאֵצֶר
