 1  וְאֵת אֲשֶׁר־בָּעִיר וְאֶת־אֲשֶׁר בַּשָּׂדֶה לָקָחוּ׃ וְאֶת־כָּל־
 2  חֵילָם וְאֶת־כָּל־טַפָּם וְאֶת־נְשֵׁיהֶם שָׁבוּ וַיָּבֹזּוּ וְאֵת כָּל־
 3  אֲשֶׁר בַּבָּיִת׃ וַיֹּאמֶר יַעֲקֹב אֶל־שִׁמְעוֹן וְאֶל־לֵוִי עֲכַרְתֶּם
 4   אֹתִי לְהַבְאִישֵׁנִי בְּיֹשֵׁב הָאָרֶץ בַּכְּנַעֲנִי וּבַפְּרִזִּי וַאֲנִי
 5   מְתֵי מִסְפָּר וְנֶאֶסְפוּ עָלַי וְהִכּוּנִי וְנִשְׁמַדְתִּי אֲנִי וּבֵיתִי׃
 6   וַיֹּאמְרוּ הַכְזוֹנָה יַעֲשֶׂה אֶת־אֲחוֹתֵנוּ׃ [פ]
 7   וַיֹּאמֶר אֱלֹהִים אֶל־יַעֲקֹב קוּם עֲלֵה בֵית־אֵל וְשֶׁב־
 8  שָׁם וַעֲשֵׂה־שָׁם מִזְבֵּחַ לָאֵל הַנִּרְאֶה אֵלֶיךָ בְּבָרְחֲךָ
 9   מִפְּנֵי עֵשָׂו אָחִיךָ׃ וַיֹּאמֶר יַעֲקֹב אֶל־בֵּיתוֹ וְאֶל כָּל־
10  אֲשֶׁר עִמּוֹ הָסִרוּ אֶת־אֱלֹהֵי הַנֵּכָר אֲשֶׁר בְּתֹכְכֶם
11   וְהִטַּהֲרוּ וְהַחֲלִיפוּ שִׂמְלֹתֵיכֶם׃ וְנָקוּמָה וְנַעֲלֶה בֵּית־אֵל
12   וְאֶעֱשֶׂה־שָּׁם מִזְבֵּחַ לָאֵל הָעֹנֶה אֹתִי בְּיוֹם צָרָתִי וַיְהִי
13   עִמָּדִי בַּדֶּרֶךְ אֲשֶׁר הָלָכְתִּי׃ וַיִּתְּנוּ אֶל־יַעֲקֹב אֵת כָּל־
14  אֱלֹהֵי הַנֵּכָר אֲשֶׁר בְּיָדָם וְאֶת־הַנְּזָמִים אֲשֶׁר בְּאָזְנֵיהֶם
15   וַיִּטְמֹן אֹתָם יַעֲקֹב תַּחַת הָאֵלָה אֲשֶׁר עִם־שְׁכֶם׃ וַיִּסָּעוּ
16   וַיְהִי  חִתַּת אֱלֹהִים עַל־הֶעָרִים אֲשֶׁר סְבִיבוֹתֵיהֶם
17   וְלֹא רָדְפוּ אַחֲרֵי בְּנֵי יַעֲקֹב׃ וַיָּבֹא יַעֲקֹב לוּזָה אֲשֶׁר
18   בְּאֶרֶץ כְּנַעַן הִוא בֵּית־אֵל הוּא וְכָל־הָעָם אֲשֶׁר־עִמּוֹ׃
19   וַיִּבֶן שָׁם מִזְבֵּחַ וַיִּקְרָא לַמָּקוֹם אֵל בֵּית־אֵל כִּי שָׁם
20   נִגְלוּ אֵלָיו הָאֱלֹהִים בְּבָרְחוֹ מִפְּנֵי אָחִיו׃ וַתָּמָת דְּבֹרָה
21   מֵינֶקֶת רִבְקָה וַתִּקָּבֵר מִתַּחַת לְבֵית־אֵל תַּחַת הָאַלּוֹן
22   וַיִּקְרָא שְׁמוֹ אַלּוֹן בָּכוּת׃ [פ]
23   וַיֵּרָא אֱלֹהִים אֶל־יַעֲקֹב עוֹד בְּבֹאוֹ מִפַּדַּן אֲרָם וַיְבָרֶךְ
24   אֹתוֹ׃ וַיֹּאמֶר־לוֹ אֱלֹהִים שִׁמְךָ יַעֲקֹב לֹא־יִקָּרֵא שִׁמְךָ
25   עוֹד יַעֲקֹב כִּי אִם־יִשְׂרָאֵל יִהְיֶה שְׁמֶךָ וַיִּקְרָא אֶת־
26  שְׁמוֹ יִשְׂרָאֵל׃ וַיֹּאמֶר לוֹ אֱלֹהִים אֲנִי אֵל שַׁדַּי פְּרֵה
27   וּרְבֵה גּוֹי וּקְהַל גּוֹיִם יִהְיֶה מִמֶּךָּ וּמְלָכִים מֵחֲלָצֶיךָ
28   יֵצֵאוּ׃  וְאֶת־הָאָרֶץ אֲשֶׁר נָתַתִּי לְאַבְרָהָם וּלְיִצְחָק לְךָ
29   אֶתְּנֶנָּה וּלְזַרְעֲךָ אַחֲרֶיךָ אֶתֵּן אֶת־הָאָרֶץ׃ וַיַּעַל מֵעָלָיו
30   אֱלֹהִים בַּמָּקוֹם אֲשֶׁר־דִּבֶּר אִתּוֹ׃ וַיַּצֵּב יַעֲקֹב מַצֵּבָה
31   בַּמָּקוֹם אֲשֶׁר־דִּבֶּר אִתּוֹ מַצֶּבֶת אָבֶן וַיַּסֵּךְ עָלֶיהָ נֶסֶךְ
32   וַיִּצֹק עָלֶיהָ שָׁמֶן׃ וַיִּקְרָא יַעֲקֹב אֶת־שֵׁם הַמָּקוֹם אֲשֶׁר
33   דִּבֶּר אִתּוֹ שָׁם אֱלֹהִים בֵּית־אֵל׃ וַיִּסְעוּ מִבֵּית אֵל
34   וַיְהִי־עוֹד כִּבְרַת־הָאָרֶץ לָבוֹא אֶפְרָתָה וַתֵּלֶד רָחֵל
35   וַתְּקַשׁ בְּלִדְתָּהּ׃ וַיְהִי בְהַקְשֹׁתָהּ בְּלִדְתָּהּ וַתֹּאמֶר לָהּ
36   הַמְיַלֶּדֶת אַל־תִּירְאִי כִּי־גַם־זֶה לָךְ בֵּן׃ וַיְהִי בְּצֵאת
37   נַפְשָׁהּ כִּי מֵתָה וַתִּקְרָא שְׁמוֹ בֶּן־אוֹנִי וְאָבִיו קָרָא־
38  לוֹ בִנְיָמִין׃ וַתָּמָת רָחֵל וַתִּקָּבֵר בְּדֶרֶךְ אֶפְרָתָה הִוא
39   בֵּית לָחֶם׃ וַיַּצֵּב יַעֲקֹב מַצֵּבָה עַל־קְבֻרָתָהּ הִוא מַצֶּבֶת
40   קְבֻרַת־רָחֵל עַד־הַיּוֹם׃ וַיִּסַּע יִשְׂרָאֵל וַיֵּט אָהֳלֹה
41   מֵהָלְאָה לְמִגְדַּל־עֵדֶר׃ וַיְהִי בִּשְׁכֹּן יִשְׂרָאֵל בָּאָרֶץ
42   הַהִוא וַיֵּלֶךְ רְאוּבֵן וַיִּשְׁכַּב אֶת־בִּלְהָה פִּילֶגֶשׁ אָבִיו
