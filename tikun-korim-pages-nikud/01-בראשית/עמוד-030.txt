 1  וַאֲמֻשְׁךָ בְּנִי הַאַתָּה זֶה בְּנִי עֵשָׂו אִם־לֹא׃ וַיִּגַּשׁ יַעֲקֹב
 2   אֶל־יִצְחָק אָבִיו וַיְמֻשֵּׁהוּ וַיֹּאמֶר הַקֹּל קוֹל יַעֲקֹב
 3   וְהַיָּדַיִם יְדֵי עֵשָׂו׃ וְלֹא הִכִּירוֹ כִּי־הָיוּ יָדָיו כִּידֵי עֵשָׂו
 4   אָחִיו שְׂעִרֹת וַיְבָרְכֵהוּ׃ וַיֹּאמֶר אַתָּה זֶה בְּנִי עֵשָׂו
 5   וַיֹּאמֶר אָנִי׃ וַיֹּאמֶר הַגִּשָׁה לִּי וְאֹכְלָה מִצֵּיד בְּנִי לְמַעַן
 6   תְּבָרֶכְךָ נַפְשִׁי וַיַּגֶּשׁ־לוֹ וַיֹּאכַל וַיָּבֵא לוֹ יַיִן וַיֵּשְׁתְּ׃
 7   וַיֹּאמֶר אֵלָיו יִצְחָק אָבִיו גְּשָׁה־נָּא וּשְׁקָה־לִּי בְּנִי׃
 8   וַיִּגַּשׁ וַיִּשַּׁק־לוֹ וַיָּרַח אֶת־רֵיחַ בְּגָדָיו וַיְבָרְכֵהוּ וַיֹּאמֶר
 9   רְאֵה רֵיחַ בְּנִי כְּרֵיחַ שָׂדֶה אֲשֶׁר בֵּרְכוֹ יְהוָה׃  וְיִתֶּן־
10  לְךָ הָאֱלֹהִים מִטַּל הַשָּׁמַיִם וּמִשְׁמַנֵּי הָאָרֶץ וְרֹב דָּגָן
11   וְתִירֹשׁ׃ יַעַבְדוּךָ עַמִּים וישתחו וְיִשְׁתַּחֲווּ לְךָ לְאֻמִּים הֱוֵה
12   גְבִיר לְאַחֶיךָ וְיִשְׁתַּחֲווּ לְךָ בְּנֵי אִמֶּךָ אֹרְרֶיךָ אָרוּר
13   וּמְבָרְכֶיךָ בָּרוּךְ׃ וַיְהִי כַּאֲשֶׁר כִּלָּה יִצְחָק לְבָרֵךְ אֶת־
14  יַעֲקֹב וַיְהִי אַךְ יָצֹא יָצָא יַעֲקֹב מֵאֵת פְּנֵי יִצְחָק אָבִיו
15   וְעֵשָׂו אָחִיו בָּא מִצֵּידוֹ׃ וַיַּעַשׂ גַּם־הוּא מַטְעַמִּים וַיָּבֵא
16   לְאָבִיו וַיֹּאמֶר לְאָבִיו יָקֻם אָבִי וְיֹאכַל מִצֵּיד בְּנוֹ
17   בַּעֲבֻר תְּבָרְכַנִּי נַפְשֶׁךָ׃ וַיֹּאמֶר לוֹ יִצְחָק אָבִיו מִי־אָתָּה
18   וַיֹּאמֶר אֲנִי בִּנְךָ בְכֹרְךָ עֵשָׂו׃ וַיֶּחֱרַד יִצְחָק חֲרָדָה
19   גְּדֹלָה עַד־מְאֹד וַיֹּאמֶר מִי־אֵפוֹא הוּא הַצָּד־צַיִד וַיָּבֵא
20   לִי וָאֹכַל מִכֹּל בְּטֶרֶם תָּבוֹא וָאֲבָרְכֵהוּ גַּם־בָּרוּךְ יִהְיֶה׃
21   כִּשְׁמֹעַ עֵשָׂו אֶת־דִּבְרֵי אָבִיו וַיִּצְעַק צְעָקָה גְּדֹלָה
22   וּמָרָה עַד־מְאֹד וַיֹּאמֶר לְאָבִיו בָּרְכֵנִי גַם־אָנִי אָבִי׃
23   וַיֹּאמֶר בָּא אָחִיךָ בְּמִרְמָה וַיִּקַּח בִּרְכָתֶךָ׃ וַיֹּאמֶר הֲכִי
24   קָרָא שְׁמוֹ יַעֲקֹב וַיַּעְקְבֵנִי זֶה פַעֲמַיִם אֶת־בְּכֹרָתִי
25   לָקָח וְהִנֵּה עַתָּה לָקַח בִּרְכָתִי וַיֹּאמַר הֲלֹא־אָצַלְתָּ לִּי
26   בְּרָכָה׃ וַיַּעַן יִצְחָק וַיֹּאמֶר לְעֵשָׂו הֵן גְּבִיר שַׂמְתִּיו לָךְ
27   וְאֶת־כָּל־אֶחָיו נָתַתִּי לוֹ לַעֲבָדִים וְדָגָן וְתִירֹשׁ סְמַכְתִּיו
28   וּלְכָה אֵפוֹא מָה אֶעֱשֶׂה בְּנִי׃ וַיֹּאמֶר עֵשָׂו אֶל־אָבִיו
29   הַבְרָכָה אַחַת הִוא־לְךָ אָבִי בָּרְכֵנִי גַם־אָנִי אָבִי וַיִּשָּׂא
30   עֵשָׂו קֹלוֹ וַיֵּבְךְּ׃ וַיַּעַן יִצְחָק אָבִיו וַיֹּאמֶר אֵלָיו הִנֵּה
31   מִשְׁמַנֵּי הָאָרֶץ יִהְיֶה מוֹשָׁבֶךָ וּמִטַּל הַשָּׁמַיִם מֵעָל׃
32   וְעַל־חַרְבְּךָ תִחְיֶה וְאֶת־אָחִיךָ תַּעֲבֹד וְהָיָה כַּאֲשֶׁר
33   תָּרִיד וּפָרַקְתָּ עֻלּוֹ מֵעַל צַוָּארֶךָ׃ וַיִּשְׂטֹם עֵשָׂו אֶת־
34  יַעֲקֹב עַל־הַבְּרָכָה אֲשֶׁר בֵּרְכוֹ אָבִיו וַיֹּאמֶר עֵשָׂו
35   בְּלִבּוֹ יִקְרְבוּ יְמֵי אֵבֶל אָבִי וְאַהַרְגָה אֶת־יַעֲקֹב אָחִי׃
36   וַיֻּגַּד לְרִבְקָה אֶת־דִּבְרֵי עֵשָׂו בְּנָהּ הַגָּדֹל וַתִּשְׁלַח
37   וַתִּקְרָא לְיַעֲקֹב בְּנָהּ הַקָּטָן וַתֹּאמֶר אֵלָיו הִנֵּה עֵשָׂו
38   אָחִיךָ מִתְנַחֵם לְךָ לְהָרְגֶךָ׃ וְעַתָּה בְנִי שְׁמַע בְּקֹלִי
39   וְקוּם בְּרַח־לְךָ אֶל־לָבָן אָחִי חָרָנָה׃ וְיָשַׁבְתָּ עִמּוֹ יָמִים
40   אֲחָדִים עַד אֲשֶׁר־תָּשׁוּב חֲמַת אָחִיךָ׃ עַד־שׁוּב אַף־
41  אָחִיךָ מִמְּךָ וְשָׁכַח אֵת אֲשֶׁר־עָשִׂיתָ לּוֹ וְשָׁלַחְתִּי
42   וּלְקַחְתִּיךָ מִשָּׁם לָמָה אֶשְׁכַּל גַּם־שְׁנֵיכֶם יוֹם אֶחָד׃
