 1  וַיֹּאמְרוּ לוֹ לֹא־נְכַחֵד מֵאֲדֹנִי כִּי אִם־תַּם הַכֶּסֶף וּמִקְנֵה
 2   הַבְּהֵמָה אֶל־אֲדֹנִי לֹא נִשְׁאַר לִפְנֵי אֲדֹנִי בִּלְתִּי אִם־
 3  גְּוִיָּתֵנוּ וְאַדְמָתֵנוּ׃ לָמָּה נָמוּת לְעֵינֶיךָ גַּם־אֲנַחְנוּ גַּם
 4   אַדְמָתֵנוּ קְנֵה־אֹתָנוּ וְאֶת־אַדְמָתֵנוּ בַּלָּחֶם וְנִהְיֶה אֲנַחְנוּ
 5   וְאַדְמָתֵנוּ עֲבָדִים לְפַרְעֹה וְתֶן־זֶרַע וְנִחְיֶה וְלֹא נָמוּת
 6   וְהָאֲדָמָה לֹא תֵשָׁם׃ וַיִּקֶן יוֹסֵף אֶת־כָּל־אַדְמַת מִצְרַיִם
 7   לְפַרְעֹה כִּי־מָכְרוּ מִצְרַיִם אִישׁ שָׂדֵהוּ כִּי־חָזַק עֲלֵהֶם
 8   הָרָעָב וַתְּהִי הָאָרֶץ לְפַרְעֹה׃ וְאֶת־הָעָם הֶעֱבִיר אֹתוֹ
 9   לֶעָרִים מִקְצֵה גְבוּל־מִצְרַיִם וְעַד־קָצֵהוּ׃ רַק אַדְמַת
10   הַכֹּהֲנִים לֹא קָנָה כִּי חֹק לַכֹּהֲנִים מֵאֵת פַּרְעֹה וְאָכְלוּ
11   אֶת־חֻקָּם אֲשֶׁר נָתַן לָהֶם פַּרְעֹה עַל־כֵּן לֹא מָכְרוּ
12   אֶת־אַדְמָתָם׃ וַיֹּאמֶר יוֹסֵף אֶל־הָעָם הֵן קָנִיתִי אֶתְכֶם
13   הַיּוֹם וְאֶת־אַדְמַתְכֶם לְפַרְעֹה הֵא־לָכֶם זֶרַע וּזְרַעְתֶּם
14   אֶת־הָאֲדָמָה׃ וְהָיָה בַּתְּבוּאֹת וּנְתַתֶּם חֲמִישִׁית לְפַרְעֹה
15   וְאַרְבַּע הַיָּדֹת יִהְיֶה לָכֶם לְזֶרַע הַשָּׂדֶה וּלְאָכְלְכֶם
16   וְלַאֲשֶׁר בְּבָתֵּיכֶם וְלֶאֱכֹל לְטַפְּכֶם׃  וַיֹּאמְרוּ הֶחֱיִתָנוּ
17   נִמְצָא־חֵן בְּעֵינֵי אֲדֹנִי וְהָיִינוּ עֲבָדִים לְפַרְעֹה׃ וַיָּשֶׂם
18   אֹתָהּ יוֹסֵף לְחֹק עַד־הַיּוֹם הַזֶּה עַל־אַדְמַת מִצְרַיִם
19   לְפַרְעֹה לַחֹמֶשׁ רַק אַדְמַת הַכֹּהֲנִים לְבַדָּם לֹא
20   הָיְתָה לְפַרְעֹה׃ וַיֵּשֶׁב יִשְׂרָאֵל בְּאֶרֶץ מִצְרַיִם בְּאֶרֶץ
21   גֹּשֶׁן וַיֵּאָחֲזוּ בָהּ וַיִּפְרוּ וַיִּרְבּוּ מְאֹד׃ וַיְחִי יַעֲקֹב בְּאֶרֶץ
22   מִצְרַיִם שְׁבַע עֶשְׂרֵה שָׁנָה וַיְהִי יְמֵי־יַעֲקֹב שְׁנֵי חַיָּיו
23   שֶׁבַע שָׁנִים וְאַרְבָּעִים וּמְאַת שָׁנָה׃ וַיִּקְרְבוּ יְמֵי־יִשְׂרָאֵל
24   לָמוּת וַיִּקְרָא  לִבְנוֹ לְיוֹסֵף וַיֹּאמֶר לוֹ אִם־נָא מָצָאתִי
25   חֵן בְּעֵינֶיךָ שִׂים־נָא יָדְךָ תַּחַת יְרֵכִי וְעָשִׂיתָ עִמָּדִי
26   חֶסֶד וֶאֱמֶת אַל־נָא תִקְבְּרֵנִי בְּמִצְרָיִם׃ וְשָׁכַבְתִּי עִם־
27  אֲבֹתַי וּנְשָׂאתַנִי מִמִּצְרַיִם וּקְבַרְתַּנִי בִּקְבֻרָתָם וַיֹּאמַר
28   אָנֹכִי אֶעֱשֶׂה כִדְבָרֶךָ׃ וַיֹּאמֶר הִשָּׁבְעָה לִי וַיִּשָּׁבַע לוֹ
29   וַיִּשְׁתַּחוּ יִשְׂרָאֵל עַל־רֹאשׁ הַמִּטָּה׃♢ [פ]
30   וַיְהִי אַחֲרֵי הַדְּבָרִים הָאֵלֶּה וַיֹּאמֶר לְיוֹסֵף הִנֵּה
31   אָבִיךָ חֹלֶה וַיִּקַּח אֶת־שְׁנֵי בָנָיו עִמּוֹ אֶת־מְנַשֶּׁה וְאֶת־
32  אֶפְרָיִם׃ וַיַּגֵּד לְיַעֲקֹב וַיֹּאמֶר הִנֵּה בִּנְךָ יוֹסֵף בָּא אֵלֶיךָ
33   וַיִּתְחַזֵּק יִשְׂרָאֵל וַיֵּשֶׁב עַל־הַמִּטָּה׃ וַיֹּאמֶר יַעֲקֹב אֶל־
34  יוֹסֵף אֵל שַׁדַּי נִרְאָה־אֵלַי בְּלוּז בְּאֶרֶץ כְּנָעַן וַיְבָרֶךְ
35   אֹתִי׃♢ וַיֹּאמֶר אֵלַי הִנְנִי מַפְרְךָ וְהִרְבִּיתִךָ וּנְתַתִּיךָ
36   לִקְהַל עַמִּים וְנָתַתִּי אֶת־הָאָרֶץ הַזֹּאת לְזַרְעֲךָ אַחֲרֶיךָ
37   אֲחֻזַּת עוֹלָם׃ וְעַתָּה שְׁנֵי־בָנֶיךָ הַנּוֹלָדִים לְךָ בְּאֶרֶץ
38   מִצְרַיִם עַד־בֹּאִי אֵלֶיךָ מִצְרַיְמָה לִי־הֵם אֶפְרַיִם
39   וּמְנַשֶּׁה כִּרְאוּבֵן וְשִׁמְעוֹן יִהְיוּ־לִי׃ וּמוֹלַדְתְּךָ אֲשֶׁר־
40  הוֹלַדְתָּ אַחֲרֵיהֶם לְךָ יִהְיוּ עַל שֵׁם אֲחֵיהֶם יִקָּרְאוּ
41   בְּנַחֲלָתָם׃ וַאֲנִי  בְּבֹאִי מִפַּדָּן מֵתָה עָלַי רָחֵל בְּאֶרֶץ
42   כְּנַעַן בַּדֶּרֶךְ בְּעוֹד כִּבְרַת־אֶרֶץ לָבֹא אֶפְרָתָה
