 1  וַיַּעַל אַבְרָם מִמִּצְרַיִם הוּא וְאִשְׁתּוֹ וְכָל־אֲשֶׁר־לוֹ
 2   וְלוֹט עִמּוֹ הַנֶּגְבָּה׃ וְאַבְרָם כָּבֵד מְאֹד בַּמִּקְנֶה בַּכֶּסֶף
 3   וּבַזָּהָב׃ וַיֵּלֶךְ לְמַסָּעָיו מִנֶּגֶב וְעַד־בֵּית־אֵל עַד־הַמָּקוֹם
 4   אֲשֶׁר־הָיָה שָׁם אָהֳלֹה בַּתְּחִלָּה בֵּין בֵּית־אֵל וּבֵין הָעָי׃
 5   אֶל־מְקוֹם הַמִּזְבֵּחַ אֲשֶׁר־עָשָׂה שָׁם בָּרִאשֹׁנָה וַיִּקְרָא
 6   שָׁם אַבְרָם בְּשֵׁם יְהוָה׃  וְגַם־לְלוֹט הַהֹלֵךְ אֶת־אַבְרָם
 7   הָיָה צֹאן־וּבָקָר וְאֹהָלִים׃ וְלֹא־נָשָׂא אֹתָם הָאָרֶץ
 8   לָשֶׁבֶת יַחְדָּו כִּי־הָיָה רְכוּשָׁם רָב וְלֹא יָכְלוּ לָשֶׁבֶת
 9   יַחְדָּו׃ וַיְהִי־רִיב בֵּין רֹעֵי מִקְנֵה־אַבְרָם וּבֵין רֹעֵי
10   מִקְנֵה־לוֹט וְהַכְּנַעֲנִי וְהַפְּרִזִּי אָז יֹשֵׁב בָּאָרֶץ׃ וַיֹּאמֶר
11   אַבְרָם אֶל־לוֹט אַל־נָא תְהִי מְרִיבָה בֵּינִי וּבֵינֶךָ וּבֵין
12   רֹעַי וּבֵין רֹעֶיךָ כִּי־אֲנָשִׁים אַחִים אֲנָחְנוּ׃ הֲלֹא כָל־
13  הָאָרֶץ לְפָנֶיךָ הִפָּרֶד נָא מֵעָלָי אִם־הַשְּׂמֹאל וְאֵימִנָה
14   וְאִם־הַיָּמִין וְאַשְׂמְאִילָה׃ וַיִּשָּׂא־לוֹט אֶת־עֵינָיו וַיַּרְא
15   אֶת־כָּל־כִּכַּר הַיַּרְדֵּן כִּי כֻלָּהּ מַשְׁקֶה לִפְנֵי  שַׁחֵת
16   יְהוָה אֶת־סְדֹם וְאֶת־עֲמֹרָה כְּגַן־יְהוָה כְּאֶרֶץ מִצְרַיִם
17   בֹּאֲכָה צֹעַר׃ וַיִּבְחַר־לוֹ לוֹט אֵת כָּל־כִּכַּר הַיַּרְדֵּן
18   וַיִּסַּע לוֹט מִקֶּדֶם וַיִּפָּרְדוּ אִישׁ מֵעַל אָחִיו׃ אַבְרָם
19   יָשַׁב בְּאֶרֶץ־כְּנָעַן וְלוֹט יָשַׁב בְּעָרֵי הַכִּכָּר וַיֶּאֱהַל
20   עַד־סְדֹם׃ וְאַנְשֵׁי סְדֹם רָעִים וְחַטָּאִים לַיהוָה מְאֹד׃
21   וַיהוָה אָמַר אֶל־אַבְרָם אַחֲרֵי הִפָּרֶד־לוֹט מֵעִמּוֹ
22   שָׂא נָא עֵינֶיךָ וּרְאֵה מִן־הַמָּקוֹם אֲשֶׁר־אַתָּה שָׁם
23   צָפֹנָה וָנֶגְבָּה וָקֵדְמָה וָיָמָּה׃ כִּי אֶת־כָּל־הָאָרֶץ אֲשֶׁר־
24  אַתָּה רֹאֶה לְךָ אֶתְּנֶנָּה וּלְזַרְעֲךָ עַד־עוֹלָם׃ וְשַׂמְתִּי
25   אֶת־זַרְעֲךָ כַּעֲפַר הָאָרֶץ אֲשֶׁר  אִם־יוּכַל אִישׁ
26   לִמְנוֹת אֶת־עֲפַר הָאָרֶץ גַּם־זַרְעֲךָ יִמָּנֶה׃ קוּם הִתְהַלֵּךְ
27   בָּאָרֶץ לְאָרְכָּהּ וּלְרָחְבָּהּ כִּי לְךָ אֶתְּנֶנָּה׃ וַיֶּאֱהַל
28   אַבְרָם וַיָּבֹא וַיֵּשֶׁב בְּאֵלֹנֵי מַמְרֵא אֲשֶׁר בְּחֶבְרוֹן
29   וַיִּבֶן־שָׁם מִזְבֵּחַ לַיהוָה׃ [פ]
30    וַיְהִי בִּימֵי אַמְרָפֶל מֶלֶךְ־שִׁנְעָר אַרְיוֹךְ מֶלֶךְ אֶלָּסָר
31   כְּדָרְלָעֹמֶר מֶלֶךְ עֵילָם וְתִדְעָל מֶלֶךְ גּוֹיִם׃ עָשׂוּ
32   מִלְחָמָה אֶת־בֶּרַע מֶלֶךְ סְדֹם וְאֶת־בִּרְשַׁע מֶלֶךְ
33   עֲמֹרָה שִׁנְאָב  מֶלֶךְ אַדְמָה וְשֶׁמְאֵבֶר מֶלֶךְ צְבוֺיִם
34   וּמֶלֶךְ בֶּלַע הִיא־צֹעַר׃ כָּל־אֵלֶּה חָבְרוּ אֶל־עֵמֶק
35   הַשִּׂדִּים הוּא יָם הַמֶּלַח׃ שְׁתֵּים עֶשְׂרֵה שָׁנָה עָבְדוּ
36   אֶת־כְּדָרְלָעֹמֶר וּשְׁלֹשׁ־עֶשְׂרֵה שָׁנָה מָרָדוּ׃ וּבְאַרְבַּע
37   עֶשְׂרֵה שָׁנָה בָּא כְדָרְלָעֹמֶר וְהַמְּלָכִים אֲשֶׁר אִתּוֹ
38   וַיַּכּוּ אֶת־רְפָאִים בְּעַשְׁתְּרֹת קַרְנַיִם וְאֶת־הַזּוּזִים בְּהָם
39   וְאֵת הָאֵימִים בְּשָׁוֵה קִרְיָתָיִם׃ וְאֶת־הַחֹרִי בְּהַרְרָם
40   שֵׂעִיר עַד אֵיל פָּארָן אֲשֶׁר עַל־הַמִּדְבָּר׃ וַיָּשֻׁבוּ
41   וַיָּבֹאוּ אֶל־עֵין מִשְׁפָּט הִוא קָדֵשׁ וַיַּכּוּ אֶת־כָּל־שְׂדֵה
42   הָעֲמָלֵקִי וְגַם אֶת־הָאֱמֹרִי הַיֹּשֵׁב בְּחַצְצֹן תָּמָר׃
