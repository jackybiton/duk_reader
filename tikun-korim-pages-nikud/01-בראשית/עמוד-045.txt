 1  וְגַם הִנֵּה הָרָה לִזְנוּנִים וַיֹּאמֶר יְהוּדָה הוֹצִיאוּהָ וְתִשָּׂרֵף׃
 2   הִוא מוּצֵאת וְהִיא שָׁלְחָה אֶל־חָמִיהָ לֵאמֹר לְאִישׁ
 3   אֲשֶׁר־אֵלֶּה לּוֹ אָנֹכִי הָרָה וַתֹּאמֶר הַכֶּר־נָא לְמִי
 4   הַחֹתֶמֶת וְהַפְּתִילִים וְהַמַּטֶּה הָאֵלֶּה׃ וַיַּכֵּר יְהוּדָה
 5   וַיֹּאמֶר צָדְקָה מִמֶּנִּי כִּי־עַל־כֵּן לֹא־נְתַתִּיהָ לְשֵׁלָה
 6   בְנִי וְלֹא־יָסַף עוֹד לְדַעְתָּהּ׃ וַיְהִי בְּעֵת לִדְתָּהּ וְהִנֵּה
 7   תְאוֹמִים בְּבִטְנָהּ׃ וַיְהִי בְלִדְתָּהּ וַיִּתֶּן־יָד וַתִּקַּח הַמְיַלֶּדֶת
 8   וַתִּקְשֹׁר עַל־יָדוֹ שָׁנִי לֵאמֹר זֶה יָצָא רִאשֹׁנָה׃ וַיְהִי
 9   כְּמֵשִׁיב יָדוֹ וְהִנֵּה יָצָא אָחִיו וַתֹּאמֶר מַה־פָּרַצְתָּ עָלֶיךָ
10   פָּרֶץ וַיִּקְרָא שְׁמוֹ פָּרֶץ׃ וְאַחַר יָצָא אָחִיו אֲשֶׁר עַל־
11  יָדוֹ הַשָּׁנִי וַיִּקְרָא שְׁמוֹ זָרַח׃ [ס]              וְיוֹסֵף
12   הוּרַד מִצְרָיְמָה וַיִּקְנֵהוּ פּוֹטִיפַר סְרִיס פַּרְעֹה שַׂר
13   הַטַּבָּחִים אִישׁ מִצְרִי מִיַּד הַיִּשְׁמְעֵאלִים אֲשֶׁר
14   הוֹרִדֻהוּ שָׁמָּה׃ וַיְהִי יְהוָה אֶת־יוֹסֵף וַיְהִי אִישׁ מַצְלִיחַ
15   וַיְהִי בְּבֵית אֲדֹנָיו הַמִּצְרִי׃ וַיַּרְא אֲדֹנָיו כִּי יְהוָה אִתּוֹ
16   וְכֹל אֲשֶׁר־הוּא עֹשֶׂה יְהוָה מַצְלִיחַ בְּיָדוֹ׃ וַיִּמְצָא
17   יוֹסֵף חֵן בְּעֵינָיו וַיְשָׁרֶת אֹתוֹ וַיַּפְקִדֵהוּ עַל־בֵּיתוֹ וְכָל־
18  יֶשׁ־לוֹ נָתַן בְּיָדוֹ׃ וַיְהִי מֵאָז הִפְקִיד אֹתוֹ בְּבֵיתוֹ וְעַל
19   כָּל־אֲשֶׁר יֶשׁ־לוֹ וַיְבָרֶךְ יְהוָה אֶת־בֵּית הַמִּצְרִי בִּגְלַל
20   יוֹסֵף וַיְהִי בִּרְכַּת יְהוָה בְּכָל־אֲשֶׁר יֶשׁ־לוֹ בַּבַּיִת
21   וּבַשָּׂדֶה׃ וַיַּעֲזֹב כָּל־אֲשֶׁר־לוֹ בְּיַד־יוֹסֵף וְלֹא־יָדַע
22   אִתּוֹ מְאוּמָה כִּי אִם־הַלֶּחֶם אֲשֶׁר־הוּא אוֹכֵל וַיְהִי
23   יוֹסֵף יְפֵה־תֹאַר וִיפֵה מַרְאֶה׃  וַיְהִי אַחַר הַדְּבָרִים
24   הָאֵלֶּה וַתִּשָּׂא אֵשֶׁת־אֲדֹנָיו אֶת־עֵינֶיהָ אֶל־יוֹסֵף
25   וַתֹּאמֶר שִׁכְבָה עִמִּי׃ וַיְמָאֵן  וַיֹּאמֶר אֶל־אֵשֶׁת אֲדֹנָיו
26   הֵן אֲדֹנִי לֹא־יָדַע אִתִּי מַה־בַּבָּיִת וְכֹל אֲשֶׁר־יֶשׁ־לוֹ
27   נָתַן בְּיָדִי׃ אֵינֶנּוּ גָדוֹל בַּבַּיִת הַזֶּה מִמֶּנִּי וְלֹא־חָשַׂךְ
28   מִמֶּנִּי מְאוּמָה כִּי אִם־אוֹתָךְ בַּאֲשֶׁר אַתְּ־אִשְׁתּוֹ וְאֵיךְ
29   אֶעֱשֶׂה הָרָעָה הַגְּדֹלָה הַזֹּאת וְחָטָאתִי לֵאלֹהִים׃ וַיְהִי
30   כְּדַבְּרָהּ אֶל־יוֹסֵף יוֹם  יוֹם וְלֹא־שָׁמַע אֵלֶיהָ לִשְׁכַּב
31   אֶצְלָהּ לִהְיוֹת עִמָּהּ׃ וַיְהִי כְּהַיּוֹם הַזֶּה וַיָּבֹא הַבַּיְתָה
32   לַעֲשׂוֹת מְלַאכְתּוֹ וְאֵין אִישׁ מֵאַנְשֵׁי הַבַּיִת שָׁם בַּבָּיִת׃
33   וַתִּתְפְּשֵׂהוּ בְּבִגְדוֹ לֵאמֹר שִׁכְבָה עִמִּי וַיַּעֲזֹב בִּגְדוֹ
34   בְּיָדָהּ וַיָּנָס וַיֵּצֵא הַחוּצָה׃ וַיְהִי כִּרְאוֹתָהּ כִּי־עָזַב בִּגְדוֹ
35   בְּיָדָהּ וַיָּנָס הַחוּצָה׃ וַתִּקְרָא לְאַנְשֵׁי בֵיתָהּ וַתֹּאמֶר
36   לָהֶם לֵאמֹר רְאוּ הֵבִיא לָנוּ אִישׁ עִבְרִי לְצַחֶק
37   בָּנוּ בָּא אֵלַי לִשְׁכַּב עִמִּי וָאֶקְרָא בְּקוֹל גָּדוֹל׃ וַיְהִי
38   כְשָׁמְעוֹ כִּי־הֲרִימֹתִי קוֹלִי וָאֶקְרָא וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי
39   וַיָּנָס וַיֵּצֵא הַחוּצָה׃ וַתַּנַּח בִּגְדוֹ אֶצְלָהּ עַד־בּוֹא אֲדֹנָיו
40   אֶל־בֵּיתוֹ׃ וַתְּדַבֵּר אֵלָיו כַּדְּבָרִים הָאֵלֶּה לֵאמֹר בָּא־
41  אֵלַי הָעֶבֶד הָעִבְרִי אֲשֶׁר־הֵבֵאתָ לָּנוּ לְצַחֶק בִּי׃ וַיְהִי
42   כַּהֲרִימִי קוֹלִי וָאֶקְרָא וַיַּעֲזֹב בִּגְדוֹ אֶצְלִי וַיָּנָס הַחוּצָה׃
