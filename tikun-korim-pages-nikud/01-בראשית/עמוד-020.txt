 1  וַתִּשְׁכַּב עִמּוֹ וְלֹא־יָדַע בְּשִׁכְבָהּ וּבְקֻמָהּ׃ וַתַּהֲרֶיןָ שְׁתֵּי
 2   בְנוֹת־לוֹט מֵאֲבִיהֶן׃ וַתֵּלֶד הַבְּכִירָה בֵּן וַתִּקְרָא שְׁמוֹ
 3   מוֹאָב הוּא אֲבִי־מוֹאָב עַד־הַיּוֹם׃ וְהַצְּעִירָה גַם־הִוא
 4   יָלְדָה בֵּן וַתִּקְרָא שְׁמוֹ בֶּן־עַמִּי הוּא אֲבִי בְנֵי־עַמּוֹן
 5   עַד־הַיּוֹם׃ [ס]             וַיִּסַּע מִשָּׁם אַבְרָהָם
 6   אַרְצָה הַנֶּגֶב וַיֵּשֶׁב בֵּין־קָדֵשׁ וּבֵין שׁוּר וַיָּגָר בִּגְרָר׃
 7   וַיֹּאמֶר אַבְרָהָם אֶל־שָׂרָה אִשְׁתּוֹ אֲחֹתִי הִוא וַיִּשְׁלַח
 8   אֲבִימֶלֶךְ מֶלֶךְ גְּרָר וַיִּקַּח אֶת־שָׂרָה׃ וַיָּבֹא אֱלֹהִים אֶל־
 9  אֲבִימֶלֶךְ בַּחֲלוֹם הַלָּיְלָה וַיֹּאמֶר לוֹ הִנְּךָ מֵת עַל־
10  הָאִשָּׁה אֲשֶׁר־לָקַחְתָּ וְהִוא בְּעֻלַת בָּעַל׃ וַאֲבִימֶלֶךְ
11   לֹא קָרַב אֵלֶיהָ וַיֹּאמַר אֲדֹנָי הֲגוֹי גַּם־צַדִּיק תַּהֲרֹג׃
12   הֲלֹא הוּא אָמַר־לִי אֲחֹתִי הִוא וְהִיא־גַם־הִוא אָמְרָה
13   אָחִי הוּא בְּתָם־לְבָבִי וּבְנִקְיֹן כַּפַּי עָשִׂיתִי זֹאת׃ וַיֹּאמֶר
14   אֵלָיו הָאֱלֹהִים בַּחֲלֹם גַּם אָנֹכִי יָדַעְתִּי כִּי בְתָם־לְבָבְךָ
15   עָשִׂיתָ זֹּאת וָאֶחְשֹׂךְ גַּם־אָנֹכִי אוֹתְךָ מֵחֲטוֹ־לִי עַל־כֵּן
16   לֹא־נְתַתִּיךָ לִנְגֹּעַ אֵלֶיהָ׃ וְעַתָּה הָשֵׁב אֵשֶׁת־הָאִישׁ כִּי־
17  נָבִיא הוּא וְיִתְפַּלֵּל בַּעַדְךָ וֶחְיֵה וְאִם־אֵינְךָ מֵשִׁיב דַּע
18   כִּי־מוֹת תָּמוּת אַתָּה וְכָל־אֲשֶׁר־לָךְ׃ וַיַּשְׁכֵּם אֲבִימֶלֶךְ
19   בַּבֹּקֶר וַיִּקְרָא לְכָל־עֲבָדָיו וַיְדַבֵּר אֶת־כָּל־הַדְּבָרִים
20   הָאֵלֶּה בְּאָזְנֵיהֶם וַיִּירְאוּ הָאֲנָשִׁים מְאֹד׃ וַיִּקְרָא
21   אֲבִימֶלֶךְ לְאַבְרָהָם וַיֹּאמֶר לוֹ מֶה־עָשִׂיתָ לָּנוּ וּמֶה־
22  חָטָאתִי לָךְ כִּי־הֵבֵאתָ עָלַי וְעַל־מַמְלַכְתִּי חֲטָאָה
23   גְדֹלָה מַעֲשִׂים אֲשֶׁר לֹא־יֵעָשׂוּ עָשִׂיתָ עִמָּדִי׃ וַיֹּאמֶר
24   אֲבִימֶלֶךְ אֶל־אַבְרָהָם מָה רָאִיתָ כִּי עָשִׂיתָ אֶת־הַדָּבָר
25   הַזֶּה׃ וַיֹּאמֶר אַבְרָהָם כִּי אָמַרְתִּי רַק אֵין־יִרְאַת אֱלֹהִים
26   בַּמָּקוֹם הַזֶּה וַהֲרָגוּנִי עַל־דְּבַר אִשְׁתִּי׃ וְגַם־אָמְנָה אֲחֹתִי
27   בַת־אָבִי הִוא אַךְ לֹא בַת־אִמִּי וַתְּהִי־לִי לְאִשָּׁה׃ וַיְהִי
28   כַּאֲשֶׁר הִתְעוּ אֹתִי אֱלֹהִים מִבֵּית אָבִי וָאֹמַר לָהּ זֶה
29   חַסְדֵּךְ אֲשֶׁר תַּעֲשִׂי עִמָּדִי אֶל כָּל־הַמָּקוֹם אֲשֶׁר נָבוֹא
30   שָׁמָּה אִמְרִי־לִי אָחִי הוּא׃ וַיִּקַּח אֲבִימֶלֶךְ צֹאן וּבָקָר
31   וַעֲבָדִים וּשְׁפָחֹת וַיִּתֵּן לְאַבְרָהָם וַיָּשֶׁב לוֹ אֵת שָׂרָה
32   אִשְׁתּוֹ׃ וַיֹּאמֶר אֲבִימֶלֶךְ הִנֵּה אַרְצִי לְפָנֶיךָ בַּטּוֹב
33   בְּעֵינֶיךָ שֵׁב׃ וּלְשָׂרָה אָמַר הִנֵּה נָתַתִּי אֶלֶף כֶּסֶף
34   לְאָחִיךְ הִנֵּה הוּא־לָךְ כְּסוּת עֵינַיִם לְכֹל אֲשֶׁר אִתָּךְ
35   וְאֵת כֹּל וְנֹכָחַת׃ וַיִּתְפַּלֵּל אַבְרָהָם אֶל־הָאֱלֹהִים וַיִּרְפָּא
36   אֱלֹהִים אֶת־אֲבִימֶלֶךְ וְאֶת־אִשְׁתּוֹ וְאַמְהֹתָיו וַיֵּלֵדוּ׃ כִּי־
37  עָצֹר עָצַר יְהוָה בְּעַד כָּל־רֶחֶם לְבֵית אֲבִימֶלֶךְ עַל־
38  דְּבַר שָׂרָה אֵשֶׁת אַבְרָהָם׃ [ס]             וַיהוָה
39   פָּקַד אֶת־שָׂרָה כַּאֲשֶׁר אָמָר וַיַּעַשׂ יְהוָה לְשָׂרָה
40   כַּאֲשֶׁר דִּבֵּר׃ וַתַּהַר וַתֵּלֶד שָׂרָה לְאַבְרָהָם בֵּן לִזְקֻנָיו
41   לַמּוֹעֵד אֲשֶׁר־דִּבֶּר אֹתוֹ אֱלֹהִים׃ וַיִּקְרָא אַבְרָהָם
42   אֶת־שֶׁם־בְּנוֹ הַנּוֹלַד־לוֹ אֲשֶׁר־יָלְדָה־לּוֹ שָׂרָה יִצְחָק׃
