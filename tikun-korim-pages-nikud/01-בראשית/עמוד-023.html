1וַיָּקֻמוּוַיֵּלְכוּיַחְדָּואֶל־בְּאֵרשָׁבַעוַיֵּשֶׁבאַבְרָהָם

4הִנֵּהיָלְדָהמִלְכָּהגַם־הִואבָּנִיםלְנָחוֹראָחִיךָ׃אֶת־

6וְאֶת־כֶּשֶׂדוְאֶת־חֲזוֹוְאֶת־פִּלְדָּשׁוְאֶת־יִדְלָףוְאֵת
7בְּתוּאֵל׃וּבְתוּאֵליָלַדאֶת־רִבְקָהשְׁמֹנָהאֵלֶּהיָלְדָה
8מִלְכָּהלְנָחוֹראֲחִיאַבְרָהָם׃וּפִילַגְשׁוֹוּשְׁמָהּ
9רְאוּמָהוַתֵּלֶדגַּם־הִואאֶת־טֶבַחוְאֶת־גַּחַםוְאֶת־
10תַּחַשׁוְאֶת־מַעֲכָה׃
11וַיִּהְיוּחַיֵּישָׂרָהמֵאָהשָׁנָהוְעֶשְׂרִיםשָׁנָהוְשֶׁבַע
12שָׁנִיםשְׁנֵיחַיֵּישָׂרָה׃וַתָּמָתשָׂרָהבְּקִרְיַתאַרְבַּע
13הִואחֶבְרוֹןבְּאֶרֶץכְּנָעַןוַיָּבֹאאַבְרָהָםלִסְפֹּדלְשָׂרָה
14וְלִבְכֹּתָהּ׃וַיָּקָםאַבְרָהָםמֵעַלפְּנֵימֵתוֹוַיְדַבֵּראֶל־בְּנֵי־
15חֵתלֵאמֹר׃גֵּר־וְתוֹשָׁבאָנֹכִיעִמָּכֶםתְּנוּלִיאֲחֻזַּת־
16קֶבֶרעִמָּכֶםוְאֶקְבְּרָהמֵתִימִלְּפָנָי׃וַיַּעֲנוּבְנֵי־חֵתאֶת־
17אַבְרָהָםלֵאמֹרלוֹ׃שְׁמָעֵנוּאֲדֹנִינְשִׂיאאֱלֹהִיםאַתָּה
18בְּתוֹכֵנוּבְּמִבְחַרקְבָרֵינוּקְבֹראֶת־מֵתֶךָאִישׁמִמֶּנּוּ
19אֶת־קִבְרוֹלֹא־יִכְלֶהמִמְּךָמִקְּבֹרמֵתֶךָ׃וַיָּקָםאַבְרָהָם
20וַיִּשְׁתַּחוּלְעַם־הָאָרֶץלִבְנֵי־חֵת׃♢וַיְדַבֵּראִתָּםלֵאמֹר
21אִם־יֵשׁאֶת־נַפְשְׁכֶםלִקְבֹּראֶת־מֵתִימִלְּפָנַישְׁמָעוּנִי
22וּפִגְעוּ־לִיבְּעֶפְרוֹןבֶּן־צֹחַר׃וְיִתֶּן־לִיאֶת־מְעָרַת
23הַמַּכְפֵּלָהאֲשֶׁר־לוֹאֲשֶׁרבִּקְצֵהשָׂדֵהוּבְּכֶסֶף
24מָלֵאיִתְּנֶנָּהלִּיבְּתוֹכְכֶםלַאֲחֻזַּת־קָבֶר׃וְעֶפְרוֹןיֹשֵׁב
25בְּתוֹךְבְּנֵי־חֵתוַיַּעַןעֶפְרוֹןהַחִתִּיאֶת־אַבְרָהָםבְּאָזְנֵי
26בְנֵי־חֵתלְכֹלבָּאֵישַׁעַר־עִירוֹלֵאמֹר׃לֹא־אֲדֹנִי
27שְׁמָעֵנִיהַשָּׂדֶהנָתַתִּילָךְוְהַמְּעָרָהאֲשֶׁר־בּוֹלְךָ
28נְתַתִּיהָלְעֵינֵיבְנֵי־עַמִּינְתַתִּיהָלָּךְקְבֹרמֵתֶךָ׃♢וַיִּשְׁתַּחוּ
29אַבְרָהָםלִפְנֵיעַםהָאָרֶץ׃וַיְדַבֵּראֶל־עֶפְרוֹןבְּאָזְנֵי
30עַם־הָאָרֶץלֵאמֹראַךְאִם־אַתָּהלוּשְׁמָעֵנִינָתַתִּי
31כֶּסֶףהַשָּׂדֶהקַחמִמֶּנִּיוְאֶקְבְּרָהאֶת־מֵתִישָׁמָּה׃
32וַיַּעַןעֶפְרוֹןאֶת־אַבְרָהָםלֵאמֹרלוֹ׃אֲדֹנִישְׁמָעֵנִי
33אֶרֶץאַרְבַּעמֵאֹתשֶׁקֶל־כֶּסֶףבֵּינִיוּבֵינְךָמַה־הִוא
34וְאֶת־מֵתְךָקְבֹר׃וַיִּשְׁמַעאַבְרָהָםאֶל־עֶפְרוֹןוַיִּשְׁקֹל
35אַבְרָהָםלְעֶפְרֹןאֶת־הַכֶּסֶףאֲשֶׁרדִּבֶּרבְּאָזְנֵיבְנֵי־
36חֵתאַרְבַּעמֵאוֹתשֶׁקֶלכֶּסֶףעֹבֵרלַסֹּחֵר׃♢וַיָּקָם
37שְׂדֵהעֶפְרוֹןאֲשֶׁרבַּמַּכְפֵּלָהאֲשֶׁרלִפְנֵימַמְרֵא
38הַשָּׂדֶהוְהַמְּעָרָהאֲשֶׁר־בּוֹוְכָל־הָעֵץאֲשֶׁרבַּשָּׂדֶה
39אֲשֶׁרבְּכָל־גְּבֻלוֹסָבִיב׃לְאַבְרָהָםלְמִקְנָהלְעֵינֵיבְנֵי־
40חֵתבְּכֹלבָּאֵישַׁעַר־עִירוֹ׃וְאַחֲרֵי־כֵןקָבַראַבְרָהָם
