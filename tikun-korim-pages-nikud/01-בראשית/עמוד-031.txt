 1  וַתֹּאמֶר רִבְקָה אֶל־יִצְחָק קַצְתִּי בְחַיַּי מִפְּנֵי בְּנוֹת חֵת
 2   אִם־לֹקֵחַ יַעֲקֹב אִשָּׁה מִבְּנוֹת־חֵת כָּאֵלֶּה מִבְּנוֹת
 3   הָאָרֶץ לָמָּה לִּי חַיִּים׃ וַיִּקְרָא יִצְחָק אֶל־יַעֲקֹב וַיְבָרֶךְ
 4   אֹתוֹ וַיְצַוֵּהוּ וַיֹּאמֶר לוֹ לֹא־תִקַּח אִשָּׁה מִבְּנוֹת כְּנָעַן׃
 5   קוּם לֵךְ פַּדֶּנָה אֲרָם בֵּיתָה בְתוּאֵל אֲבִי אִמֶּךָ וְקַח־לְךָ
 6   מִשָּׁם אִשָּׁה מִבְּנוֹת לָבָן אֲחִי אִמֶּךָ׃ וְאֵל שַׁדַּי יְבָרֵךְ
 7   אֹתְךָ וְיַפְרְךָ וְיַרְבֶּךָ וְהָיִיתָ לִקְהַל עַמִּים׃ וְיִתֶּן־לְךָ אֶת־
 8  בִּרְכַּת אַבְרָהָם לְךָ וּלְזַרְעֲךָ אִתָּךְ לְרִשְׁתְּךָ אֶת־אֶרֶץ
 9   מְגֻרֶיךָ אֲשֶׁר־נָתַן אֱלֹהִים לְאַבְרָהָם׃  וַיִּשְׁלַח יִצְחָק אֶת־
10  יַעֲקֹב וַיֵּלֶךְ פַּדֶּנָה אֲרָם אֶל־לָבָן בֶּן־בְּתוּאֵל הָאֲרַמִּי
11   אֲחִי רִבְקָה אֵם יַעֲקֹב וְעֵשָׂו׃ וַיַּרְא עֵשָׂו כִּי־בֵרַךְ
12   יִצְחָק אֶת־יַעֲקֹב וְשִׁלַּח אֹתוֹ פַּדֶּנָה אֲרָם לָקַחַת־
13  לוֹ מִשָּׁם אִשָּׁה בְּבָרְכוֹ אֹתוֹ וַיְצַו עָלָיו לֵאמֹר לֹא־
14  תִקַּח אִשָּׁה מִבְּנוֹת כְּנָעַן׃  וַיִּשְׁמַע יַעֲקֹב אֶל־אָבִיו וְאֶל־
15  אִמּוֹ וַיֵּלֶךְ פַּדֶּנָה אֲרָם׃ וַיַּרְא עֵשָׂו כִּי רָעוֹת בְּנוֹת כְּנָעַן
16   בְּעֵינֵי יִצְחָק אָבִיו׃ וַיֵּלֶךְ עֵשָׂו אֶל־יִשְׁמָעֵאל וַיִּקַּח
17   אֶת־מָחֲלַת  בַּת־יִשְׁמָעֵאל בֶּן־אַבְרָהָם אֲחוֹת נְבָיוֹת
18   עַל־נָשָׁיו לוֹ לְאִשָּׁה׃ [ס]            וַיֵּצֵא יַעֲקֹב
19   מִבְּאֵר שָׁבַע וַיֵּלֶךְ חָרָנָה׃ וַיִּפְגַּע בַּמָּקוֹם וַיָּלֶן שָׁם כִּי־
20  בָא הַשֶּׁמֶשׁ וַיִּקַּח מֵאַבְנֵי הַמָּקוֹם וַיָּשֶׂם מְרַאֲשֹׁתָיו
21   וַיִּשְׁכַּב בַּמָּקוֹם הַהוּא׃ וַיַּחֲלֹם וְהִנֵּה סֻלָּם מֻצָּב אַרְצָה
22   וְרֹאשׁוֹ מַגִּיעַ הַשָּׁמָיְמָה וְהִנֵּה מַלְאֲכֵי אֱלֹהִים עֹלִים
23   וְיֹרְדִים בּוֹ׃♢ וְהִנֵּה יְהוָה נִצָּב עָלָיו וַיֹּאמַר אֲנִי יְהוָה
24   אֱלֹהֵי אַבְרָהָם אָבִיךָ וֵאלֹהֵי יִצְחָק הָאָרֶץ אֲשֶׁר
25   אַתָּה שֹׁכֵב עָלֶיהָ לְךָ אֶתְּנֶנָּה וּלְזַרְעֶךָ׃ וְהָיָה זַרְעֲךָ
26   כַּעֲפַר הָאָרֶץ וּפָרַצְתָּ יָמָּה וָקֵדְמָה וְצָפֹנָה וָנֶגְבָּה
27   וְנִבְרְכוּ בְךָ כָּל־מִשְׁפְּחֹת הָאֲדָמָה וּבְזַרְעֶךָ׃ וְהִנֵּה
28   אָנֹכִי עִמָּךְ וּשְׁמַרְתִּיךָ בְּכֹל אֲשֶׁר־תֵּלֵךְ וַהֲשִׁבֹתִיךָ
29   אֶל־הָאֲדָמָה הַזֹּאת כִּי לֹא אֶעֱזָבְךָ עַד אֲשֶׁר אִם־
30  עָשִׂיתִי אֵת אֲשֶׁר־דִּבַּרְתִּי לָךְ׃ וַיִּיקַץ יַעֲקֹב מִשְּׁנָתוֹ
31   וַיֹּאמֶר אָכֵן יֵשׁ יְהוָה בַּמָּקוֹם הַזֶּה וְאָנֹכִי לֹא יָדָעְתִּי׃
32   וַיִּירָא וַיֹּאמַר מַה־נּוֹרָא הַמָּקוֹם הַזֶּה אֵין זֶה כִּי אִם־
33  בֵּית אֱלֹהִים וְזֶה שַׁעַר הַשָּׁמָיִם׃♢ וַיַּשְׁכֵּם יַעֲקֹב בַּבֹּקֶר
34   וַיִּקַּח אֶת־הָאֶבֶן אֲשֶׁר־שָׂם מְרַאֲשֹׁתָיו וַיָּשֶׂם אֹתָהּ
35   מַצֵּבָה וַיִּצֹק שֶׁמֶן עַל־רֹאשָׁהּ׃ וַיִּקְרָא אֶת־שֵׁם־הַמָּקוֹם
36   הַהוּא בֵּית־אֵל וְאוּלָם לוּז שֵׁם־הָעִיר לָרִאשֹׁנָה׃ וַיִּדַּר
37   יַעֲקֹב נֶדֶר לֵאמֹר אִם־יִהְיֶה אֱלֹהִים עִמָּדִי וּשְׁמָרַנִי
38   בַּדֶּרֶךְ הַזֶּה אֲשֶׁר אָנֹכִי הוֹלֵךְ וְנָתַן־לִי לֶחֶם לֶאֱכֹל
39   וּבֶגֶד לִלְבֹּשׁ׃ וְשַׁבְתִּי בְשָׁלוֹם אֶל־בֵּית אָבִי וְהָיָה
40   יְהוָה לִי לֵאלֹהִים׃ וְהָאֶבֶן הַזֹּאת אֲשֶׁר־שַׂמְתִּי מַצֵּבָה
41   יִהְיֶה בֵּית אֱלֹהִים וְכֹל אֲשֶׁר תִּתֶּן־לִי עַשֵּׂר אֲעַשְּׂרֶנּוּ
42   לָךְ׃♢  וַיִּשָּׂא יַעֲקֹב רַגְלָיו וַיֵּלֶךְ אַרְצָה בְנֵי־קֶדֶם׃ וַיַּרְא
