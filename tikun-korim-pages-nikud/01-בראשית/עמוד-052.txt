 1  וְכֶסֶף אַחֵר הוֹרַדְנוּ בְיָדֵנוּ לִשְׁבָּר־אֹכֶל לֹא יָדַעְנוּ
 2   מִי־שָׂם כַּסְפֵּנוּ בְּאַמְתְּחֹתֵינוּ׃ וַיֹּאמֶר שָׁלוֹם לָכֶם אַל־
 3  תִּירָאוּ אֱלֹהֵיכֶם וֵאלֹהֵי אֲבִיכֶם נָתַן לָכֶם מַטְמוֹן
 4   בְּאַמְתְּחֹתֵיכֶם כַּסְפְּכֶם בָּא אֵלָי וַיּוֹצֵא אֲלֵהֶם אֶת־
 5  שִׁמְעוֹן׃ וַיָּבֵא הָאִישׁ אֶת־הָאֲנָשִׁים בֵּיתָה יוֹסֵף וַיִּתֶּן־
 6  מַיִם וַיִּרְחֲצוּ רַגְלֵיהֶם וַיִּתֵּן מִסְפּוֹא לַחֲמֹרֵיהֶם׃ וַיָּכִינוּ
 7   אֶת־הַמִּנְחָה עַד־בּוֹא יוֹסֵף בַּצָּהֳרָיִם כִּי שָׁמְעוּ כִּי־
 8  שָׁם יֹאכְלוּ לָחֶם׃ וַיָּבֹא יוֹסֵף הַבַּיְתָה וַיָּבִיאּוּ לוֹ אֶת־
 9  הַמִּנְחָה אֲשֶׁר־בְּיָדָם הַבָּיְתָה וַיִּשְׁתַּחֲווּ־לוֹ אָרְצָה׃
10   וַיִּשְׁאַל לָהֶם לְשָׁלוֹם וַיֹּאמֶר הֲשָׁלוֹם אֲבִיכֶם הַזָּקֵן
11   אֲשֶׁר אֲמַרְתֶּם הַעוֹדֶנּוּ חָי׃ וַיֹּאמְרוּ שָׁלוֹם לְעַבְדְּךָ
12   לְאָבִינוּ עוֹדֶנּוּ חָי וַיִּקְּדוּ וישתחו וַיִּשְׁתַּחֲווּ׃ וַיִּשָּׂא עֵינָיו וַיַּרְא
13   אֶת־בִּנְיָמִין אָחִיו בֶּן־אִמּוֹ וַיֹּאמֶר הֲזֶה אֲחִיכֶם הַקָּטֹן
14   אֲשֶׁר אֲמַרְתֶּם אֵלָי וַיֹּאמַר אֱלֹהִים יָחְנְךָ בְּנִי׃  וַיְמַהֵר
15   יוֹסֵף כִּי־נִכְמְרוּ רַחֲמָיו אֶל־אָחִיו וַיְבַקֵּשׁ לִבְכּוֹת
16   וַיָּבֹא הַחַדְרָה וַיֵּבְךְּ שָׁמָּה׃ וַיִּרְחַץ פָּנָיו וַיֵּצֵא וַיִּתְאַפַּק
17   וַיֹּאמֶר שִׂימוּ לָחֶם׃ וַיָּשִׂימוּ לוֹ לְבַדּוֹ וְלָהֶם לְבַדָּם
18   וְלַמִּצְרִים הָאֹכְלִים אִתּוֹ לְבַדָּם כִּי לֹא יוּכְלוּן
19   הַמִּצְרִים לֶאֱכֹל אֶת־הָעִבְרִים לֶחֶם כִּי־תוֹעֵבָה הִוא
20   לְמִצְרָיִם׃ וַיֵּשְׁבוּ לְפָנָיו הַבְּכֹר כִּבְכֹרָתוֹ וְהַצָּעִיר
21   כִּצְעִרָתוֹ וַיִּתְמְהוּ הָאֲנָשִׁים אִישׁ אֶל־רֵעֵהוּ׃ וַיִּשָּׂא
22   מַשְׂאֹת מֵאֵת פָּנָיו אֲלֵהֶם וַתֵּרֶב מַשְׂאַת בִּנְיָמִן
23   מִמַּשְׂאֹת כֻּלָּם חָמֵשׁ יָדוֹת וַיִּשְׁתּוּ וַיִּשְׁכְּרוּ עִמּוֹ׃
24   וַיְצַו אֶת־אֲשֶׁר עַל־בֵּיתוֹ לֵאמֹר מַלֵּא אֶת־אַמְתְּחֹת
25   הָאֲנָשִׁים אֹכֶל כַּאֲשֶׁר יוּכְלוּן שְׂאֵת וְשִׂים כֶּסֶף־
26  אִישׁ בְּפִי אַמְתַּחְתּוֹ׃ וְאֶת־גְּבִיעִי גְּבִיעַ הַכֶּסֶף תָּשִׂים
27   בְּפִי אַמְתַּחַת הַקָּטֹן וְאֵת כֶּסֶף שִׁבְרוֹ וַיַּעַשׂ כִּדְבַר
28   יוֹסֵף אֲשֶׁר דִּבֵּר׃ הַבֹּקֶר אוֹר וְהָאֲנָשִׁים שֻׁלְּחוּ
29   הֵמָּה וַחֲמֹרֵיהֶם׃ הֵם יָצְאוּ אֶת־הָעִיר לֹא הִרְחִיקוּ
30   וְיוֹסֵף אָמַר לַאֲשֶׁר עַל־בֵּיתוֹ קוּם רְדֹף אַחֲרֵי
31   הָאֲנָשִׁים וְהִשַּׂגְתָּם וְאָמַרְתָּ אֲלֵהֶם לָמָּה שִׁלַּמְתֶּם
32   רָעָה תַּחַת טוֹבָה׃ הֲלוֹא זֶה אֲשֶׁר יִשְׁתֶּה אֲדֹנִי בּוֹ
33   וְהוּא נַחֵשׁ יְנַחֵשׁ בּוֹ הֲרֵעֹתֶם אֲשֶׁר עֲשִׂיתֶם׃ וַיַּשִּׂגֵם
34   וַיְדַבֵּר אֲלֵהֶם אֶת־הַדְּבָרִים הָאֵלֶּה׃ וַיֹּאמְרוּ אֵלָיו
35   לָמָּה יְדַבֵּר אֲדֹנִי כַּדְּבָרִים הָאֵלֶּה חָלִילָה לַעֲבָדֶיךָ
36   מֵעֲשׂוֹת כַּדָּבָר הַזֶּה׃ הֵן כֶּסֶף אֲשֶׁר מָצָאנוּ בְּפִי
37   אַמְתְּחֹתֵינוּ הֱשִׁיבֹנוּ אֵלֶיךָ מֵאֶרֶץ כְּנָעַן וְאֵיךְ נִגְנֹב
38   מִבֵּית אֲדֹנֶיךָ כֶּסֶף אוֹ זָהָב׃ אֲשֶׁר יִמָּצֵא אִתּוֹ מֵעֲבָדֶיךָ
39   וָמֵת וְגַם־אֲנַחְנוּ נִהְיֶה לַאדֹנִי לַעֲבָדִים׃ וַיֹּאמֶר גַּם־עַתָּה
40   כְדִבְרֵיכֶם כֶּן־הוּא אֲשֶׁר יִמָּצֵא אִתּוֹ יִהְיֶה־לִּי עָבֶד
41   וְאַתֶּם תִּהְיוּ נְקִיִּם׃ וַיְמַהֲרוּ וַיּוֹרִדוּ אִישׁ אֶת־אַמְתַּחְתּוֹ
42   אָרְצָה וַיִּפְתְּחוּ אִישׁ אַמְתַּחְתּוֹ׃ וַיְחַפֵּשׂ בַּגָּדוֹל הֵחֵל
