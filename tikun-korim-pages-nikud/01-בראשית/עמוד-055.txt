 1  וַיִּסַּע יִשְׂרָאֵל וְכָל־אֲשֶׁר־לוֹ וַיָּבֹא בְּאֵרָה שָּׁבַע
 2   וַיִּזְבַּח זְבָחִים לֵאלֹהֵי אָבִיו יִצְחָק׃ וַיֹּאמֶר אֱלֹהִים
 3   לְיִשְׂרָאֵל בְּמַרְאֹת הַלַּיְלָה וַיֹּאמֶר יַעֲקֹב  יַעֲקֹב
 4   וַיֹּאמֶר הִנֵּנִי׃ וַיֹּאמֶר אָנֹכִי הָאֵל אֱלֹהֵי אָבִיךָ אַל־תִּירָא
 5   מֵרְדָה מִצְרַיְמָה כִּי־לְגוֹי גָּדוֹל אֲשִׂימְךָ שָׁם׃ אָנֹכִי
 6   אֵרֵד עִמְּךָ מִצְרַיְמָה וְאָנֹכִי אַעַלְךָ גַם־עָלֹה וְיוֹסֵף
 7   יָשִׁית יָדוֹ עַל־עֵינֶיךָ׃ וַיָּקָם יַעֲקֹב מִבְּאֵר שָׁבַע
 8   וַיִּשְׂאוּ בְנֵי־יִשְׂרָאֵל אֶת־יַעֲקֹב אֲבִיהֶם וְאֶת־טַפָּם
 9   וְאֶת־נְשֵׁיהֶם בָּעֲגָלוֹת אֲשֶׁר־שָׁלַח פַּרְעֹה לָשֵׂאת
10   אֹתוֹ׃ וַיִּקְחוּ אֶת־מִקְנֵיהֶם וְאֶת־רְכוּשָׁם אֲשֶׁר רָכְשׁוּ
11   בְּאֶרֶץ כְּנַעַן וַיָּבֹאוּ מִצְרָיְמָה יַעֲקֹב וְכָל־זַרְעוֹ אִתּוֹ׃
12   בָּנָיו וּבְנֵי בָנָיו אִתּוֹ בְּנֹתָיו וּבְנוֹת בָּנָיו וְכָל־זַרְעוֹ הֵבִיא
13   אִתּוֹ מִצְרָיְמָה׃ [ס]             וְאֵלֶּה שְׁמוֹת בְּנֵי־
14  יִשְׂרָאֵל הַבָּאִים מִצְרַיְמָה יַעֲקֹב וּבָנָיו בְּכֹר יַעֲקֹב
15   רְאוּבֵן׃ וּבְנֵי רְאוּבֵן חֲנוֹךְ וּפַלּוּא וְחֶצְרֹן וְכַרְמִי׃ וּבְנֵי
16   שִׁמְעוֹן יְמוּאֵל וְיָמִין וְאֹהַד וְיָכִין וְצֹחַר וְשָׁאוּל בֶּן־
17  הַכְּנַעֲנִית׃ וּבְנֵי לֵוִי גֵּרְשׁוֹן קְהָת וּמְרָרִי׃ וּבְנֵי יְהוּדָה
18   עֵר וְאוֹנָן וְשֵׁלָה וָפֶרֶץ וָזָרַח וַיָּמָת עֵר וְאוֹנָן בְּאֶרֶץ
19   כְּנַעַן וַיִּהְיוּ בְנֵי־פֶרֶץ חֶצְרֹן וְחָמוּל׃ וּבְנֵי יִשָּׂשכָר
20   תּוֹלָע וּפֻוָה וְיוֹב וְשִׁמְרֹן׃ וּבְנֵי זְבֻלוּן סֶרֶד וְאֵלוֹן
21   וְיַחְלְאֵל׃ אֵלֶּה  בְּנֵי לֵאָה אֲשֶׁר יָלְדָה לְיַעֲקֹב בְּפַדַּן
22   אֲרָם וְאֵת דִּינָה בִתּוֹ כָּל־נֶפֶשׁ בָּנָיו וּבְנוֹתָיו שְׁלֹשִׁים
23   וְשָׁלֹשׁ׃ וּבְנֵי גָד צִפְיוֹן וְחַגִּי שׁוּנִי וְאֶצְבֹּן עֵרִי וַאֲרוֹדִי
24   וְאַרְאֵלִי׃ וּבְנֵי אָשֵׁר יִמְנָה וְיִשְׁוָה וְיִשְׁוִי וּבְרִיעָה
25   וְשֶׂרַח אֲחֹתָם וּבְנֵי בְרִיעָה חֶבֶר וּמַלְכִּיאֵל׃ אֵלֶּה בְּנֵי
26   זִלְפָּה אֲשֶׁר־נָתַן לָבָן לְלֵאָה בִתּוֹ וַתֵּלֶד אֶת־אֵלֶּה
27   לְיַעֲקֹב שֵׁשׁ עֶשְׂרֵה נָפֶשׁ׃ בְּנֵי רָחֵל אֵשֶׁת יַעֲקֹב
28   יוֹסֵף וּבִנְיָמִן׃ וַיִּוָּלֵד לְיוֹסֵף בְּאֶרֶץ מִצְרַיִם אֲשֶׁר
29   יָלְדָה־לּוֹ אָסְנַת בַּת־פּוֹטִי פֶרַע כֹּהֵן אֹן אֶת־מְנַשֶּׁה
30   וְאֶת־אֶפְרָיִם׃ וּבְנֵי בִנְיָמִן בֶּלַע וָבֶכֶר וְאַשְׁבֵּל גֵּרָא
31   וְנַעֲמָן אֵחִי וָרֹאשׁ מֻפִּים וְחֻפִּים וָאָרְדְּ׃ אֵלֶּה בְּנֵי רָחֵל
32   אֲשֶׁר יֻלַּד לְיַעֲקֹב כָּל־נֶפֶשׁ אַרְבָּעָה עָשָׂר׃ וּבְנֵי־דָן
33   חֻשִׁים׃ וּבְנֵי נַפְתָּלִי יַחְצְאֵל וְגוּנִי וְיֵצֶר וְשִׁלֵּם׃ אֵלֶּה בְּנֵי
34   בִלְהָה אֲשֶׁר־נָתַן לָבָן לְרָחֵל בִּתּוֹ וַתֵּלֶד אֶת־אֵלֶּה
35   לְיַעֲקֹב כָּל־נֶפֶשׁ שִׁבְעָה׃ כָּל־הַנֶּפֶשׁ הַבָּאָה לְיַעֲקֹב
36   מִצְרַיְמָה יֹצְאֵי יְרֵכוֹ מִלְּבַד נְשֵׁי בְנֵי־יַעֲקֹב כָּל־נֶפֶשׁ
37   שִׁשִּׁים וָשֵׁשׁ׃ וּבְנֵי יוֹסֵף אֲשֶׁר־יֻלַּד־לוֹ בְמִצְרַיִם
38   נֶפֶשׁ שְׁנָיִם כָּל־הַנֶּפֶשׁ לְבֵית־יַעֲקֹב הַבָּאָה מִצְרַיְמָה
39   שִׁבְעִים׃ [ס]              וְאֶת־יְהוּדָה שָׁלַח לְפָנָיו
40   אֶל־יוֹסֵף לְהוֹרֹת לְפָנָיו גֹּשְׁנָה וַיָּבֹאוּ אַרְצָה גֹּשֶׁן׃ וַיֶּאְסֹר
41   יוֹסֵף מֶרְכַּבְתּוֹ וַיַּעַל לִקְרַאת־יִשְׂרָאֵל אָבִיו גֹּשְׁנָה
42   וַיֵּרָא אֵלָיו וַיִּפֹּל עַל־צַוָּארָיו וַיֵּבְךְּ עַל־צַוָּארָיו עוֹד׃
