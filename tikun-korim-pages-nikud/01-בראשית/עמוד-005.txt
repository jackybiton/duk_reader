 1  וּמְתוּשָׁאֵל יָלַד אֶת־לָמֶךְ׃  וַיִּקַּח־לוֹ לֶמֶךְ שְׁתֵּי נָשִׁים
 2   שֵׁם הָאַחַת עָדָה וְשֵׁם הַשֵּׁנִית צִלָּה׃ וַתֵּלֶד עָדָה אֶת־
 3  יָבָל הוּא הָיָה אֲבִי יֹשֵׁב אֹהֶל וּמִקְנֶה׃ וְשֵׁם אָחִיו יוּבָל
 4   הוּא הָיָה אֲבִי כָּל־תֹּפֵשׂ כִּנּוֹר וְעוּגָב׃ וְצִלָּה גַם־הִוא
 5   יָלְדָה אֶת־תּוּבַל קַיִן לֹטֵשׁ כָּל־חֹרֵשׁ נְחֹשֶׁת וּבַרְזֶל
 6   וַאֲחוֹת תּוּבַל־קַיִן נַעֲמָה׃  וַיֹּאמֶר לֶמֶךְ לְנָשָׁיו עָדָה
 7   וְצִלָּה שְׁמַעַן קוֹלִי נְשֵׁי לֶמֶךְ הַאְזֵנָּה אִמְרָתִי כִּי אִישׁ
 8   הָרַגְתִּי לְפִצְעִי וְיֶלֶד לְחַבֻּרָתִי׃ כִּי שִׁבְעָתַיִם יֻקַּם־קָיִן
 9   וְלֶמֶךְ שִׁבְעִים וְשִׁבְעָה׃ וַיֵּדַע אָדָם עוֹד אֶת־אִשְׁתּוֹ
10   וַתֵּלֶד בֵּן וַתִּקְרָא אֶת־שְׁמוֹ שֵׁת כִּי שָׁת־לִי אֱלֹהִים
11   זֶרַע אַחֵר תַּחַת הֶבֶל כִּי הֲרָגוֹ קָיִן׃ וּלְשֵׁת גַּם־הוּא
12   יֻלַּד־בֵּן וַיִּקְרָא אֶת־שְׁמוֹ אֱנוֹשׁ אָז הוּחַל לִקְרֹא
13   בְּשֵׁם יְהוָה׃ [ס]             זֶה סֵפֶר תּוֹלְדֹת אָדָם
14   בְּיוֹם בְּרֹא אֱלֹהִים אָדָם בִּדְמוּת אֱלֹהִים עָשָׂה אֹתוֹ׃
15   זָכָר וּנְקֵבָה בְּרָאָם וַיְבָרֶךְ אֹתָם וַיִּקְרָא אֶת־שְׁמָם
16   אָדָם בְּיוֹם הִבָּרְאָם׃ וַיְחִי אָדָם שְׁלֹשִׁים וּמְאַת שָׁנָה
17   וַיּוֹלֶד בִּדְמוּתוֹ כְּצַלְמוֹ וַיִּקְרָא אֶת־שְׁמוֹ שֵׁת׃ וַיִּהְיוּ יְמֵי־
18  אָדָם אַחֲרֵי הוֹלִידוֹ אֶת־שֵׁת שְׁמֹנֶה מֵאֹת שָׁנָה וַיּוֹלֶד
19   בָּנִים וּבָנוֹת׃ וַיִּהְיוּ כָּל־יְמֵי אָדָם אֲשֶׁר־חַי תְּשַׁע מֵאוֹת
20   שָׁנָה וּשְׁלֹשִׁים שָׁנָה וַיָּמֹת׃ [ס]             וַיְחִי־
21  שֵׁת חָמֵשׁ שָׁנִים וּמְאַת שָׁנָה וַיּוֹלֶד אֶת־אֱנוֹשׁ׃
22   וַיְחִי־שֵׁת אַחֲרֵי הוֹלִידוֹ אֶת־אֱנוֹשׁ שֶׁבַע שָׁנִים
23   וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת׃ וַיִּהְיוּ כָּל־
24  יְמֵי־שֵׁת שְׁתֵּים עֶשְׂרֵה שָׁנָה וּתְשַׁע מֵאוֹת שָׁנָה
25   וַיָּמֹת׃ [ס]             וַיְחִי אֱנוֹשׁ תִּשְׁעִים שָׁנָה
26   וַיּוֹלֶד אֶת־קֵינָן׃ וַיְחִי אֱנוֹשׁ אַחֲרֵי הוֹלִידוֹ אֶת־קֵינָן
27   חֲמֵשׁ עֶשְׂרֵה שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים
28   וּבָנוֹת׃ וַיִּהְיוּ כָּל־יְמֵי אֱנוֹשׁ חָמֵשׁ שָׁנִים וּתְשַׁע מֵאוֹת
29   שָׁנָה וַיָּמֹת׃ [ס]             וַיְחִי קֵינָן שִׁבְעִים שָׁנָה
30   וַיּוֹלֶד אֶת־מַהֲלַלְאֵל׃ וַיְחִי קֵינָן אַחֲרֵי הוֹלִידוֹ אֶת־
31  מַהֲלַלְאֵל אַרְבָּעִים שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד
32   בָּנִים וּבָנוֹת׃ וַיִּהְיוּ כָּל־יְמֵי קֵינָן עֶשֶׂר שָׁנִים וּתְשַׁע
33   מֵאוֹת שָׁנָה וַיָּמֹת׃ [ס]             וַיְחִי מַהֲלַלְאֵל
34   חָמֵשׁ שָׁנִים וְשִׁשִּׁים שָׁנָה וַיּוֹלֶד אֶת־יָרֶד׃ וַיְחִי
35   מַהֲלַלְאֵל אַחֲרֵי הוֹלִידוֹ אֶת־יֶרֶד שְׁלֹשִׁים שָׁנָה
36   וּשְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת׃ וַיִּהְיוּ כָּל־יְמֵי
37   מַהֲלַלְאֵל חָמֵשׁ וְתִשְׁעִים שָׁנָה וּשְׁמֹנֶה מֵאוֹת שָׁנָה
38   וַיָּמֹת׃ [ס]             וַיְחִי־יֶרֶד שְׁתַּיִם וְשִׁשִּׁים שָׁנָה
39   וּמְאַת שָׁנָה וַיּוֹלֶד אֶת־חֲנוֹךְ׃ וַיְחִי־יֶרֶד אַחֲרֵי הוֹלִידוֹ
40   אֶת־חֲנוֹךְ שְׁמֹנֶה מֵאוֹת שָׁנָה וַיּוֹלֶד בָּנִים וּבָנוֹת׃ וַיִּהְיוּ
41   כָּל־יְמֵי־יֶרֶד שְׁתַּיִם וְשִׁשִּׁים שָׁנָה וּתְשַׁע מֵאוֹת שָׁנָה
42   וַיָּמֹת׃ [ס]             וַיְחִי חֲנוֹךְ חָמֵשׁ וְשִׁשִּׁים שָׁנָה
