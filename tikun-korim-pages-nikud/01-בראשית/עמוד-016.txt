 1  וּבֵין בָּרֶד׃ וַתֵּלֶד הָגָר לְאַבְרָם בֵּן וַיִּקְרָא אַבְרָם
 2   שֶׁם־בְּנוֹ אֲשֶׁר־יָלְדָה הָגָר יִשְׁמָעֵאל׃ וְאַבְרָם בֶּן־
 3  שְׁמֹנִים שָׁנָה וְשֵׁשׁ שָׁנִים בְּלֶדֶת־הָגָר אֶת־יִשְׁמָעֵאל
 4   לְאַבְרָם׃ [ס]             וַיְהִי אַבְרָם בֶּן־תִּשְׁעִים
 5   שָׁנָה וְתֵשַׁע שָׁנִים וַיֵּרָא יְהוָה אֶל־אַבְרָם וַיֹּאמֶר
 6   אֵלָיו אֲנִי־אֵל שַׁדַּי הִתְהַלֵּךְ לְפָנַי וֶהְיֵה תָמִים׃ וְאֶתְּנָה
 7   בְרִיתִי בֵּינִי וּבֵינֶךָ וְאַרְבֶּה אוֹתְךָ בִּמְאֹד מְאֹד׃ וַיִּפֹּל
 8   אַבְרָם עַל־פָּנָיו וַיְדַבֵּר אִתּוֹ אֱלֹהִים לֵאמֹר׃ אֲנִי הִנֵּה
 9   בְרִיתִי אִתָּךְ וְהָיִיתָ לְאַב הֲמוֹן גּוֹיִם׃ וְלֹא־יִקָּרֵא עוֹד
10   אֶת־שִׁמְךָ אַבְרָם וְהָיָה שִׁמְךָ אַבְרָהָם כִּי אַב־הֲמוֹן
11   גּוֹיִם נְתַתִּיךָ׃ וְהִפְרֵתִי אֹתְךָ בִּמְאֹד מְאֹד וּנְתַתִּיךָ לְגוֹיִם
12   וּמְלָכִים מִמְּךָ יֵצֵאוּ׃  וַהֲקִמֹתִי אֶת־בְּרִיתִי בֵּינִי וּבֵינֶךָ
13   וּבֵין זַרְעֲךָ אַחֲרֶיךָ לְדֹרֹתָם לִבְרִית עוֹלָם לִהְיוֹת לְךָ
14   לֵאלֹהִים וּלְזַרְעֲךָ אַחֲרֶיךָ׃ וְנָתַתִּי לְךָ וּלְזַרְעֲךָ אַחֲרֶיךָ
15   אֵת  אֶרֶץ מְגֻרֶיךָ אֵת כָּל־אֶרֶץ כְּנַעַן לַאֲחֻזַּת עוֹלָם
16   וְהָיִיתִי לָהֶם לֵאלֹהִים׃ וַיֹּאמֶר אֱלֹהִים אֶל־אַבְרָהָם
17   וְאַתָּה אֶת־בְּרִיתִי תִשְׁמֹר אַתָּה וְזַרְעֲךָ אַחֲרֶיךָ
18   לְדֹרֹתָם׃ זֹאת בְּרִיתִי אֲשֶׁר תִּשְׁמְרוּ בֵּינִי וּבֵינֵיכֶם
19   וּבֵין זַרְעֲךָ אַחֲרֶיךָ הִמּוֹל לָכֶם כָּל־זָכָר׃ וּנְמַלְתֶּם
20   אֵת בְּשַׂר עָרְלַתְכֶם וְהָיָה לְאוֹת בְּרִית בֵּינִי וּבֵינֵיכֶם׃
21   וּבֶן־שְׁמֹנַת יָמִים יִמּוֹל לָכֶם כָּל־זָכָר לְדֹרֹתֵיכֶם יְלִיד
22   בָּיִת וּמִקְנַת־כֶּסֶף מִכֹּל בֶּן־נֵכָר אֲשֶׁר לֹא מִזַּרְעֲךָ
23   הוּא׃ הִמּוֹל  יִמּוֹל יְלִיד בֵּיתְךָ וּמִקְנַת כַּסְפֶּךָ וְהָיְתָה
24   בְרִיתִי בִּבְשַׂרְכֶם לִבְרִית עוֹלָם׃ וְעָרֵל  זָכָר אֲשֶׁר
25   לֹא־יִמּוֹל אֶת־בְּשַׂר עָרְלָתוֹ וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא
26   מֵעַמֶּיהָ אֶת־בְּרִיתִי הֵפַר׃ [ס]             וַיֹּאמֶר
27   אֱלֹהִים אֶל־אַבְרָהָם שָׂרַי אִשְׁתְּךָ לֹא־תִקְרָא אֶת־
28  שְׁמָהּ שָׂרָי כִּי שָׂרָה שְׁמָהּ׃ וּבֵרַכְתִּי אֹתָהּ וְגַם נָתַתִּי
29   מִמֶּנָּה לְךָ בֵּן וּבֵרַכְתִּיהָ וְהָיְתָה לְגוֹיִם מַלְכֵי עַמִּים
30   מִמֶּנָּה יִהְיוּ׃ וַיִּפֹּל אַבְרָהָם עַל־פָּנָיו וַיִּצְחָק
31   וַיֹּאמֶר בְּלִבּוֹ הַלְּבֶן מֵאָה־שָׁנָה יִוָּלֵד וְאִם־שָׂרָה הֲבַת־
32  תִּשְׁעִים שָׁנָה תֵּלֵד׃ וַיֹּאמֶר אַבְרָהָם אֶל־הָאֱלֹהִים
33   לוּ יִשְׁמָעֵאל יִחְיֶה לְפָנֶיךָ׃ וַיֹּאמֶר אֱלֹהִים אֲבָל
34   שָׂרָה אִשְׁתְּךָ יֹלֶדֶת לְךָ בֵּן וְקָרָאתָ אֶת־שְׁמוֹ יִצְחָק
35   וַהֲקִמֹתִי אֶת־בְּרִיתִי אִתּוֹ לִבְרִית עוֹלָם לְזַרְעוֹ אַחֲרָיו׃
36   וּלְיִשְׁמָעֵאל שְׁמַעְתִּיךָ הִנֵּה  בֵּרַכְתִּי אֹתוֹ וְהִפְרֵיתִי
37   אֹתוֹ וְהִרְבֵּיתִי אֹתוֹ בִּמְאֹד מְאֹד שְׁנֵים־עָשָׂר נְשִׂיאִם
38   יוֹלִיד וּנְתַתִּיו לְגוֹי גָּדוֹל׃ וְאֶת־בְּרִיתִי אָקִים אֶת־יִצְחָק
39   אֲשֶׁר תֵּלֵד לְךָ שָׂרָה לַמּוֹעֵד הַזֶּה בַּשָּׁנָה הָאַחֶרֶת׃
40   וַיְכַל לְדַבֵּר אִתּוֹ וַיַּעַל אֱלֹהִים מֵעַל אַבְרָהָם׃ וַיִּקַּח
41   אַבְרָהָם אֶת־יִשְׁמָעֵאל בְּנוֹ וְאֵת כָּל־יְלִידֵי בֵיתוֹ
42   וְאֵת כָּל־מִקְנַת כַּסְפּוֹ כָּל־זָכָר בְּאַנְשֵׁי בֵּית אַבְרָהָם
