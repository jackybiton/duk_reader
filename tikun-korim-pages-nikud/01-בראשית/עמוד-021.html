1וַיָּמָלאַבְרָהָםאֶת־יִצְחָקבְּנוֹבֶּן־שְׁמֹנַתיָמִיםכַּאֲשֶׁר
2צִוָּהאֹתוֹאֱלֹהִים׃וְאַבְרָהָםבֶּן־מְאַתשָׁנָהבְּהִוָּלֶדלוֹ

7יִצְחָק׃וַתֵּרֶאשָׂרָהאֶת־בֶּן־הָגָרהַמִּצְרִיתאֲשֶׁר־יָלְדָה
8לְאַבְרָהָםמְצַחֵק׃וַתֹּאמֶרלְאַבְרָהָםגָּרֵשׁהָאָמָה
9הַזֹּאתוְאֶת־בְּנָהּכִּילֹאיִירַשׁבֶּן־הָאָמָההַזֹּאתעִם־
10בְּנִיעִם־יִצְחָק׃וַיֵּרַעהַדָּבָרמְאֹדבְּעֵינֵיאַבְרָהָם
11עַלאוֹדֹתבְּנוֹ׃וַיֹּאמֶראֱלֹהִיםאֶל־אַבְרָהָםאַל־יֵרַע
12בְּעֵינֶיךָעַל־הַנַּעַרוְעַל־אֲמָתֶךָכֹּלאֲשֶׁרתֹּאמַר
13אֵלֶיךָשָׂרָהשְׁמַעבְּקֹלָהּכִּיבְיִצְחָקיִקָּרֵאלְךָ
14זָרַע׃♢וְגַםאֶת־בֶּן־הָאָמָהלְגוֹיאֲשִׂימֶנּוּכִּיזַרְעֲךָהוּא׃
15וַיַּשְׁכֵּםאַבְרָהָםבַּבֹּקֶרוַיִּקַּח־לֶחֶםוְחֵמַתמַיִםוַיִּתֵּן
16אֶל־הָגָרשָׂםעַל־שִׁכְמָהּוְאֶת־הַיֶּלֶדוַיְשַׁלְּחֶהָוַתֵּלֶךְ
17וַתֵּתַעבְּמִדְבַּרבְּאֵרשָׁבַע׃וַיִּכְלוּהַמַּיִםמִן־הַחֵמֶת
18וַתַּשְׁלֵךְאֶת־הַיֶּלֶדתַּחַתאַחַדהַשִּׂיחִם׃וַתֵּלֶךְוַתֵּשֶׁב
19לָהּמִנֶּגֶדהַרְחֵקכִּמְטַחֲוֵיקֶשֶׁתכִּיאָמְרָהאַל־אֶרְאֶה
20בְּמוֹתהַיָּלֶדוַתֵּשֶׁבמִנֶּגֶדוַתִּשָּׂאאֶת־קֹלָהּוַתֵּבְךְּ׃
21וַיִּשְׁמַעאֱלֹהִיםאֶת־קוֹלהַנַּעַרוַיִּקְרָאמַלְאַךְאֱלֹהִים
22אֶל־הָגָרמִן־הַשָּׁמַיִםוַיֹּאמֶרלָהּמַה־לָּךְהָגָראַל־
23תִּירְאִיכִּי־שָׁמַעאֱלֹהִיםאֶל־קוֹלהַנַּעַרבַּאֲשֶׁרהוּא־

25לְגוֹיגָּדוֹלאֲשִׂימֶנּוּ׃וַיִּפְקַחאֱלֹהִיםאֶת־עֵינֶיהָוַתֵּרֶא
26בְּאֵרמָיִםוַתֵּלֶךְוַתְּמַלֵּאאֶת־הַחֵמֶתמַיִםוַתַּשְׁקְאֶת־
27הַנָּעַר׃וַיְהִיאֱלֹהִיםאֶת־הַנַּעַרוַיִּגְדָּלוַיֵּשֶׁבבַּמִּדְבָּר

29אִמּוֹאִשָּׁהמֵאֶרֶץמִצְרָיִם׃
30וַיְהִיבָּעֵתהַהִואוַיֹּאמֶראֲבִימֶלֶךְוּפִיכֹלשַׂר־צְבָאוֹ
31אֶל־אַבְרָהָםלֵאמֹראֱלֹהִיםעִמְּךָבְּכֹלאֲשֶׁר־אַתָּה
32עֹשֶׂה׃וְעַתָּההִשָּׁבְעָהלִּיבֵאלֹהִיםהֵנָּהאִם־תִּשְׁקֹר
33לִיוּלְנִינִיוּלְנֶכְדִּיכַּחֶסֶדאֲשֶׁר־עָשִׂיתִיעִמְּךָתַּעֲשֶׂה
34עִמָּדִיוְעִם־הָאָרֶץאֲשֶׁר־גַּרְתָּהבָּהּ׃וַיֹּאמֶראַבְרָהָם
35אָנֹכִיאִשָּׁבֵעַ׃וְהוֹכִחַאַבְרָהָםאֶת־אֲבִימֶלֶךְעַל־
36אֹדוֹתבְּאֵרהַמַּיִםאֲשֶׁרגָּזְלוּעַבְדֵיאֲבִימֶלֶךְ׃וַיֹּאמֶר
