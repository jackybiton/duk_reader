 1  וַיְהִי כִשְׁמֹעַ אֲדֹנָיו אֶת־דִּבְרֵי אִשְׁתּוֹ אֲשֶׁר דִּבְּרָה
 2   אֵלָיו לֵאמֹר כַּדְּבָרִים הָאֵלֶּה עָשָׂה לִי עַבְדֶּךָ
 3   וַיִּחַר אַפּוֹ׃ וַיִּקַּח אֲדֹנֵי יוֹסֵף אֹתוֹ וַיִּתְּנֵהוּ אֶל־בֵּית
 4   הַסֹּהַר מְקוֹם אֲשֶׁר־אֲסִירֵי הַמֶּלֶךְ אֲסוּרִים וַיְהִי־שָׁם
 5   בְּבֵית הַסֹּהַר׃ וַיְהִי יְהוָה אֶת־יוֹסֵף וַיֵּט אֵלָיו חָסֶד וַיִּתֵּן
 6   חִנּוֹ בְּעֵינֵי שַׂר בֵּית־הַסֹּהַר׃ וַיִּתֵּן שַׂר בֵּית־הַסֹּהַר
 7   בְּיַד־יוֹסֵף אֵת כָּל־הָאֲסִירִם אֲשֶׁר בְּבֵית הַסֹּהַר וְאֵת
 8   כָּל־אֲשֶׁר עֹשִׂים שָׁם הוּא הָיָה עֹשֶׂה׃ אֵין  שַׂר בֵּית־
 9  הַסֹּהַר רֹאֶה אֶת־כָּל־מְאוּמָה בְּיָדוֹ בַּאֲשֶׁר יְהוָה אִתּוֹ
10   וַאֲשֶׁר־הוּא עֹשֶׂה יְהוָה מַצְלִיחַ׃ [פ]
11    וַיְהִי אַחַר הַדְּבָרִים הָאֵלֶּה חָטְאוּ מַשְׁקֵה מֶלֶךְ־
12  מִצְרַיִם וְהָאֹפֶה לַאֲדֹנֵיהֶם לְמֶלֶךְ מִצְרָיִם׃ וַיִּקְצֹף
13   פַּרְעֹה עַל שְׁנֵי סָרִיסָיו עַל שַׂר הַמַּשְׁקִים וְעַל שַׂר
14   הָאוֹפִים׃ וַיִּתֵּן אֹתָם בְּמִשְׁמַר בֵּית שַׂר הַטַּבָּחִים אֶל־
15  בֵּית הַסֹּהַר מְקוֹם אֲשֶׁר יוֹסֵף אָסוּר שָׁם׃ וַיִּפְקֹד שַׂר
16   הַטַּבָּחִים אֶת־יוֹסֵף אִתָּם וַיְשָׁרֶת אֹתָם וַיִּהְיוּ יָמִים
17   בְּמִשְׁמָר׃ וַיַּחַלְמוּ חֲלוֹם שְׁנֵיהֶם אִישׁ חֲלֹמוֹ בְּלַיְלָה
18   אֶחָד אִישׁ כְּפִתְרוֹן חֲלֹמוֹ הַמַּשְׁקֶה וְהָאֹפֶה אֲשֶׁר
19   לְמֶלֶךְ מִצְרַיִם אֲשֶׁר אֲסוּרִים בְּבֵית הַסֹּהַר׃ וַיָּבֹא
20   אֲלֵיהֶם יוֹסֵף בַּבֹּקֶר וַיַּרְא אֹתָם וְהִנָּם זֹעֲפִים׃ וַיִּשְׁאַל
21   אֶת־סְרִיסֵי פַרְעֹה אֲשֶׁר אִתּוֹ בְמִשְׁמַר בֵּית אֲדֹנָיו
22   לֵאמֹר מַדּוּעַ פְּנֵיכֶם רָעִים הַיּוֹם׃ וַיֹּאמְרוּ אֵלָיו חֲלוֹם
23   חָלַמְנוּ וּפֹתֵר אֵין אֹתוֹ וַיֹּאמֶר אֲלֵהֶם יוֹסֵף הֲלוֹא
24   לֵאלֹהִים פִּתְרֹנִים סַפְּרוּ־נָא לִי׃ וַיְסַפֵּר שַׂר־הַמַּשְׁקִים
25   אֶת־חֲלֹמוֹ לְיוֹסֵף וַיֹּאמֶר לוֹ בַּחֲלוֹמִי וְהִנֵּה־גֶפֶן לְפָנָי׃
26   וּבַגֶּפֶן שְׁלֹשָׁה שָׂרִיגִם וְהִוא כְפֹרַחַת עָלְתָה נִצָּהּ
27   הִבְשִׁילוּ אַשְׁכְּלֹתֶיהָ עֲנָבִים׃ וְכוֹס פַּרְעֹה בְּיָדִי וָאֶקַּח
28   אֶת־הָעֲנָבִים וָאֶשְׂחַט אֹתָם אֶל־כּוֹס פַּרְעֹה וָאֶתֵּן
29   אֶת־הַכּוֹס עַל־כַּף פַּרְעֹה׃ וַיֹּאמֶר לוֹ יוֹסֵף זֶה פִּתְרֹנוֹ
30   שְׁלֹשֶׁת הַשָּׂרִגִים שְׁלֹשֶׁת יָמִים הֵם׃ בְּעוֹד  שְׁלֹשֶׁת
31   יָמִים יִשָּׂא פַרְעֹה אֶת־רֹאשֶׁךָ וַהֲשִׁיבְךָ עַל־כַּנֶּךָ
32   וְנָתַתָּ כוֹס־פַּרְעֹה בְּיָדוֹ כַּמִּשְׁפָּט הָרִאשׁוֹן אֲשֶׁר
33   הָיִיתָ מַשְׁקֵהוּ׃ כִּי אִם־זְכַרְתַּנִי אִתְּךָ כַּאֲשֶׁר יִיטַב לָךְ
34   וְעָשִׂיתָ־נָּא עִמָּדִי חָסֶד וְהִזְכַּרְתַּנִי אֶל־פַּרְעֹה וְהוֹצֵאתַנִי
35   מִן־הַבַּיִת הַזֶּה׃ כִּי־גֻנֹּב גֻּנַּבְתִּי מֵאֶרֶץ הָעִבְרִים וְגַם־
36  פֹּה לֹא־עָשִׂיתִי מְאוּמָה כִּי־שָׂמוּ אֹתִי בַּבּוֹר׃ וַיַּרְא
37   שַׂר־הָאֹפִים כִּי טוֹב פָּתָר וַיֹּאמֶר אֶל־יוֹסֵף אַף־אֲנִי
38   בַּחֲלוֹמִי וְהִנֵּה שְׁלֹשָׁה סַלֵּי חֹרִי עַל־רֹאשִׁי׃ וּבַסַּל
39   הָעֶלְיוֹן מִכֹּל מַאֲכַל פַּרְעֹה מַעֲשֵׂה אֹפֶה וְהָעוֹף
40   אֹכֵל אֹתָם מִן־הַסַּל מֵעַל רֹאשִׁי׃ וַיַּעַן יוֹסֵף וַיֹּאמֶר
41   זֶה פִּתְרֹנוֹ שְׁלֹשֶׁת הַסַּלִּים שְׁלֹשֶׁת יָמִים הֵם׃ בְּעוֹד
42    שְׁלֹשֶׁת יָמִים יִשָּׂא פַרְעֹה אֶת־רֹאשְׁךָ מֵעָלֶיךָ
