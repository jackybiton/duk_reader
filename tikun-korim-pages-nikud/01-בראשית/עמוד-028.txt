 1  וּלְזַרְעֲךָ אֶתֵּן אֶת־כָּל־הָאֲרָצֹת הָאֵל וַהֲקִמֹתִי אֶת־
 2  הַשְּׁבֻעָה אֲשֶׁר נִשְׁבַּעְתִּי לְאַבְרָהָם אָבִיךָ׃ וְהִרְבֵּיתִי
 3   אֶת־זַרְעֲךָ כְּכוֹכְבֵי הַשָּׁמַיִם וְנָתַתִּי לְזַרְעֲךָ אֵת כָּל־
 4  הָאֲרָצֹת הָאֵל וְהִתְבָּרְכוּ בְזַרְעֲךָ כֹּל גּוֹיֵי הָאָרֶץ׃ עֵקֶב
 5   אֲשֶׁר־שָׁמַע אַבְרָהָם בְּקֹלִי וַיִּשְׁמֹר מִשְׁמַרְתִּי מִצְו͏ֹתַי
 6   חֻקּוֹתַי וְתוֹרֹתָי׃♢  וַיֵּשֶׁב יִצְחָק בִּגְרָר׃ וַיִּשְׁאֲלוּ אַנְשֵׁי
 7   הַמָּקוֹם לְאִשְׁתּוֹ וַיֹּאמֶר אֲחֹתִי הִוא כִּי יָרֵא לֵאמֹר
 8   אִשְׁתִּי פֶּן־יַהַרְגֻנִי אַנְשֵׁי הַמָּקוֹם עַל־רִבְקָה כִּי־טוֹבַת
 9   מַרְאֶה הִוא׃ וַיְהִי כִּי אָרְכוּ־לוֹ שָׁם הַיָּמִים וַיַּשְׁקֵף
10   אֲבִימֶלֶךְ מֶלֶךְ פְּלִשְׁתִּים בְּעַד הַחַלּוֹן וַיַּרְא וְהִנֵּה
11   יִצְחָק מְצַחֵק אֵת רִבְקָה אִשְׁתּוֹ׃ וַיִּקְרָא אֲבִימֶלֶךְ
12   לְיִצְחָק וַיֹּאמֶר אַךְ הִנֵּה אִשְׁתְּךָ הִוא וְאֵיךְ אָמַרְתָּ
13   אֲחֹתִי הִוא וַיֹּאמֶר אֵלָיו יִצְחָק כִּי אָמַרְתִּי פֶּן־אָמוּת
14   עָלֶיהָ׃ וַיֹּאמֶר אֲבִימֶלֶךְ מַה־זֹּאת עָשִׂיתָ לָּנוּ כִּמְעַט
15   שָׁכַב אַחַד הָעָם אֶת־אִשְׁתֶּךָ וְהֵבֵאתָ עָלֵינוּ אָשָׁם׃
16   וַיְצַו אֲבִימֶלֶךְ אֶת־כָּל־הָעָם לֵאמֹר הַנֹּגֵעַ בָּאִישׁ הַזֶּה
17   וּבְאִשְׁתּוֹ מוֹת יוּמָת׃ וַיִּזְרַע יִצְחָק בָּאָרֶץ הַהִוא וַיִּמְצָא
18   בַּשָּׁנָה הַהִוא מֵאָה שְׁעָרִים וַיְבָרְכֵהוּ יְהוָה׃  וַיִּגְדַּל
19   הָאִישׁ וַיֵּלֶךְ הָלוֹךְ וְגָדֵל עַד כִּי־גָדַל מְאֹד׃ וַיְהִי־לוֹ
20   מִקְנֵה־צֹאן וּמִקְנֵה בָקָר וַעֲבֻדָּה רַבָּה וַיְקַנְאוּ אֹתוֹ
21   פְּלִשְׁתִּים׃ וְכָל־הַבְּאֵרֹת אֲשֶׁר חָפְרוּ עַבְדֵי אָבִיו
22   בִּימֵי אַבְרָהָם אָבִיו סִתְּמוּם פְּלִשְׁתִּים וַיְמַלְאוּם עָפָר׃
23   וַיֹּאמֶר אֲבִימֶלֶךְ אֶל־יִצְחָק לֵךְ מֵעִמָּנוּ כִּי־עָצַמְתָּ מִמֶּנּוּ
24   מְאֹד׃ וַיֵּלֶךְ מִשָּׁם יִצְחָק וַיִּחַן בְּנַחַל־גְּרָר וַיֵּשֶׁב שָׁם׃
25   וַיָּשָׁב יִצְחָק וַיַּחְפֹּר  אֶת־בְּאֵרֹת הַמַּיִם אֲשֶׁר חָפְרוּ
26   בִּימֵי אַבְרָהָם אָבִיו וַיְסַתְּמוּם פְּלִשְׁתִּים אַחֲרֵי מוֹת
27   אַבְרָהָם וַיִּקְרָא לָהֶן שֵׁמוֹת כַּשֵּׁמֹת אֲשֶׁר־קָרָא לָהֶן
28   אָבִיו׃ וַיַּחְפְּרוּ עַבְדֵי־יִצְחָק בַּנָּחַל וַיִּמְצְאוּ־שָׁם בְּאֵר
29   מַיִם חַיִּים׃ וַיָּרִיבוּ רֹעֵי גְרָר עִם־רֹעֵי יִצְחָק לֵאמֹר לָנוּ
30   הַמָּיִם וַיִּקְרָא שֵׁם־הַבְּאֵר עֵשֶׂק כִּי הִתְעַשְּׂקוּ עִמּוֹ׃
31   וַיַּחְפְּרוּ בְּאֵר אַחֶרֶת וַיָּרִיבוּ גַּם־עָלֶיהָ וַיִּקְרָא שְׁמָהּ
32   שִׂטְנָה׃ וַיַּעְתֵּק מִשָּׁם וַיַּחְפֹּר בְּאֵר אַחֶרֶת וְלֹא רָבוּ
33   עָלֶיהָ וַיִּקְרָא שְׁמָהּ רְחֹבוֹת וַיֹּאמֶר כִּי־עַתָּה הִרְחִיב
34   יְהוָה לָנוּ וּפָרִינוּ בָאָרֶץ׃  וַיַּעַל מִשָּׁם בְּאֵר שָׁבַע׃
35   וַיֵּרָא אֵלָיו יְהוָה בַּלַּיְלָה הַהוּא וַיֹּאמֶר אָנֹכִי אֱלֹהֵי
36   אַבְרָהָם אָבִיךָ אַל־תִּירָא כִּי־אִתְּךָ אָנֹכִי וּבֵרַכְתִּיךָ
37   וְהִרְבֵּיתִי אֶת־זַרְעֲךָ בַּעֲבוּר אַבְרָהָם עַבְדִּי׃ וַיִּבֶן
38   שָׁם מִזְבֵּחַ וַיִּקְרָא בְּשֵׁם יְהוָה וַיֶּט־שָׁם אָהֳלוֹ וַיִּכְרוּ־
39  שָׁם עַבְדֵי־יִצְחָק בְּאֵר׃ וַאֲבִימֶלֶךְ הָלַךְ אֵלָיו מִגְּרָר
40   וַאֲחֻזַּת מֵרֵעֵהוּ וּפִיכֹל שַׂר־צְבָאוֹ׃ וַיֹּאמֶר אֲלֵהֶם יִצְחָק
41   מַדּוּעַ בָּאתֶם אֵלָי וְאַתֶּם שְׂנֵאתֶם אֹתִי וַתְּשַׁלְּחוּנִי
42   מֵאִתְּכֶם׃ וַיֹּאמְרוּ רָאוֹ רָאִינוּ כִּי־הָיָה יְהוָה  עִמָּךְ
