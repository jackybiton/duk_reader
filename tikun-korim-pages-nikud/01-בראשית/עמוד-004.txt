 1  וְהוּא יִמְשָׁל־בָּךְ׃ [ס]             וּלְאָדָם אָמַר כִּי־
 2  שָׁמַעְתָּ לְקוֹל אִשְׁתֶּךָ וַתֹּאכַל מִן־הָעֵץ אֲשֶׁר צִוִּיתִיךָ
 3   לֵאמֹר לֹא תֹאכַל מִמֶּנּוּ אֲרוּרָה הָאֲדָמָה בַּעֲבוּרֶךָ
 4   בְּעִצָּבוֹן תֹּאכְלֶנָּה כֹּל יְמֵי חַיֶּיךָ׃ וְקוֹץ וְדַרְדַּר תַּצְמִיחַ
 5   לָךְ וְאָכַלְתָּ אֶת־עֵשֶׂב הַשָּׂדֶה׃ בְּזֵעַת אַפֶּיךָ תֹּאכַל
 6   לֶחֶם עַד שׁוּבְךָ אֶל־הָאֲדָמָה כִּי מִמֶּנָּה לֻקָּחְתָּ כִּי־עָפָר
 7   אַתָּה וְאֶל־עָפָר תָּשׁוּב׃ וַיִּקְרָא הָאָדָם שֵׁם אִשְׁתּוֹ חַוָּה
 8   כִּי הִוא הָיְתָה אֵם כָּל־חָי׃ וַיַּעַשׂ יְהוָה אֱלֹהִים לְאָדָם
 9   וּלְאִשְׁתּוֹ כָּתְנוֹת עוֹר וַיַּלְבִּשֵׁם׃ [פ]
10    וַיֹּאמֶר  יְהוָה אֱלֹהִים הֵן הָאָדָם הָיָה כְּאַחַד מִמֶּנּוּ
11   לָדַעַת טוֹב וָרָע וְעַתָּה  פֶּן־יִשְׁלַח יָדוֹ וְלָקַח גַּם מֵעֵץ
12   הַחַיִּים וְאָכַל וָחַי לְעֹלָם׃ וַיְשַׁלְּחֵהוּ יְהוָה אֱלֹהִים מִגַּן־
13  עֵדֶן לַעֲבֹד אֶת־הָאֲדָמָה אֲשֶׁר לֻקַּח מִשָּׁם׃ וַיְגָרֶשׁ
14   אֶת־הָאָדָם וַיַּשְׁכֵּן מִקֶּדֶם לְגַן־עֵדֶן אֶת־הַכְּרֻבִים
15   וְאֵת לַהַט הַחֶרֶב הַמִּתְהַפֶּכֶת לִשְׁמֹר אֶת־דֶּרֶךְ עֵץ
16   הַחַיִּים׃ [ס]             וְהָאָדָם יָדַע אֶת־חַוָּה
17   אִשְׁתּוֹ וַתַּהַר וַתֵּלֶד אֶת־קַיִן וַתֹּאמֶר קָנִיתִי אִישׁ אֶת־
18  יְהוָה׃ וַתֹּסֶף לָלֶדֶת אֶת־אָחִיו אֶת־הָבֶל וַיְהִי־הֶבֶל
19   רֹעֵה צֹאן וְקַיִן הָיָה עֹבֵד אֲדָמָה׃ וַיְהִי מִקֵּץ יָמִים
20   וַיָּבֵא קַיִן מִפְּרִי הָאֲדָמָה מִנְחָה לַיהוָה׃ וְהֶבֶל הֵבִיא
21   גַם־הוּא מִבְּכֹרוֹת צֹאנוֹ וּמֵחֶלְבֵהֶן וַיִּשַׁע יְהוָה אֶל־
22  הֶבֶל וְאֶל־מִנְחָתוֹ׃ וְאֶל־קַיִן וְאֶל־מִנְחָתוֹ לֹא שָׁעָה
23   וַיִּחַר לְקַיִן מְאֹד וַיִּפְּלוּ פָּנָיו׃ וַיֹּאמֶר יְהוָה אֶל־קָיִן
24   לָמָּה חָרָה לָךְ וְלָמָּה נָפְלוּ פָנֶיךָ׃ הֲלוֹא אִם־תֵּיטִיב
25   שְׂאֵת וְאִם לֹא תֵיטִיב לַפֶּתַח חַטָּאת רֹבֵץ וְאֵלֶיךָ
26   תְּשׁוּקָתוֹ וְאַתָּה תִּמְשָׁל־בּוֹ׃ וַיֹּאמֶר קַיִן אֶל־הֶבֶל
27   אָחִיו וַיְהִי בִּהְיוֹתָם בַּשָּׂדֶה וַיָּקָם קַיִן אֶל־הֶבֶל אָחִיו
28   וַיַּהַרְגֵהוּ׃ וַיֹּאמֶר יְהוָה אֶל־קַיִן אֵי הֶבֶל אָחִיךָ וַיֹּאמֶר
29   לֹא יָדַעְתִּי הֲשֹׁמֵר אָחִי אָנֹכִי׃ וַיֹּאמֶר מֶה עָשִׂיתָ קוֹל
30   דְּמֵי אָחִיךָ צֹעֲקִים אֵלַי מִן־הָאֲדָמָה׃ וְעַתָּה אָרוּר
31   אָתָּה מִן־הָאֲדָמָה אֲשֶׁר פָּצְתָה אֶת־פִּיהָ לָקַחַת אֶת־
32  דְּמֵי אָחִיךָ מִיָּדֶךָ׃ כִּי תַעֲבֹד אֶת־הָאֲדָמָה לֹא־תֹסֵף
33   תֵּת־כֹּחָהּ לָךְ נָע וָנָד תִּהְיֶה בָאָרֶץ׃ וַיֹּאמֶר קַיִן אֶל־
34  יְהוָה גָּדוֹל עֲו‍ֹנִי מִנְּשֹׂא׃ הֵן גֵּרַשְׁתָּ אֹתִי הַיּוֹם מֵעַל
35   פְּנֵי הָאֲדָמָה וּמִפָּנֶיךָ אֶסָּתֵר וְהָיִיתִי נָע וָנָד בָּאָרֶץ
36   וְהָיָה כָל־מֹצְאִי יַהַרְגֵנִי׃ וַיֹּאמֶר לוֹ יְהוָה לָכֵן כָּל־הֹרֵג
37   קַיִן שִׁבְעָתַיִם יֻקָּם וַיָּשֶׂם יְהוָה לְקַיִן אוֹת לְבִלְתִּי
38   הַכּוֹת־אֹתוֹ כָּל־מֹצְאוֹ׃ וַיֵּצֵא קַיִן מִלִּפְנֵי יְהוָה וַיֵּשֶׁב
39   בְּאֶרֶץ־נוֹד קִדְמַת־עֵדֶן׃ וַיֵּדַע קַיִן אֶת־אִשְׁתּוֹ וַתַּהַר
40   וַתֵּלֶד אֶת־חֲנוֹךְ וַיְהִי בֹּנֶה עִיר וַיִּקְרָא שֵׁם הָעִיר
41   כְּשֵׁם בְּנוֹ חֲנוֹךְ׃ וַיִּוָּלֵד לַחֲנוֹךְ אֶת־עִירָד וְעִירָד
42   יָלַד אֶת־מְחוּיָאֵל וּמְחִיָּיאֵל יָלַד אֶת־מְתוּשָׁאֵל
