 1  וַנֹּאמֶר תְּהִי נָא אָלָה בֵּינוֹתֵינוּ בֵּינֵינוּ וּבֵינֶךָ וְנִכְרְתָה
 2   בְרִית עִמָּךְ׃ אִם־תַּעֲשֵׂה עִמָּנוּ רָעָה כַּאֲשֶׁר לֹא
 3   נְגַעֲנוּךָ וְכַאֲשֶׁר עָשִׂינוּ עִמְּךָ רַק־טוֹב וַנְּשַׁלֵּחֲךָ
 4   בְּשָׁלוֹם אַתָּה עַתָּה בְּרוּךְ יְהוָה׃  וַיַּעַשׂ לָהֶם מִשְׁתֶּה
 5   וַיֹּאכְלוּ וַיִּשְׁתּוּ׃ וַיַּשְׁכִּימוּ בַבֹּקֶר וַיִּשָּׁבְעוּ אִישׁ לְאָחִיו
 6   וַיְשַׁלְּחֵם יִצְחָק וַיֵּלְכוּ מֵאִתּוֹ בְּשָׁלוֹם׃ וַיְהִי  בַּיּוֹם
 7   הַהוּא וַיָּבֹאוּ עַבְדֵי יִצְחָק וַיַּגִּדוּ לוֹ עַל־אֹדוֹת הַבְּאֵר
 8   אֲשֶׁר חָפָרוּ וַיֹּאמְרוּ לוֹ מָצָאנוּ מָיִם׃ וַיִּקְרָא אֹתָהּ
 9   שִׁבְעָה עַל־כֵּן שֵׁם־הָעִיר בְּאֵר שֶׁבַע עַד הַיּוֹם
10   הַזֶּה׃ [ס]             וַיְהִי עֵשָׂו בֶּן־אַרְבָּעִים
11   שָׁנָה וַיִּקַּח אִשָּׁה אֶת־יְהוּדִית בַּת־בְּאֵרִי הַחִתִּי וְאֶת־
12  בָּשְׂמַת בַּת־אֵילֹן הַחִתִּי׃ וַתִּהְיֶיןָ מֹרַת רוּחַ לְיִצְחָק
13   וּלְרִבְקָה׃ [ס]             וַיְהִי כִּי־זָקֵן יִצְחָק וַתִּכְהֶיןָ
14   עֵינָיו מֵרְאֹת וַיִּקְרָא אֶת־עֵשָׂו  בְּנוֹ הַגָּדֹל וַיֹּאמֶר אֵלָיו
15   בְּנִי וַיֹּאמֶר אֵלָיו הִנֵּנִי׃ וַיֹּאמֶר הִנֵּה־נָא זָקַנְתִּי לֹא יָדַעְתִּי
16   יוֹם מוֹתִי׃ וְעַתָּה שָׂא־נָא כֵלֶיךָ תֶּלְיְךָ וְקַשְׁתֶּךָ וְצֵא
17   הַשָּׂדֶה וְצוּדָה לִּי צָיִד׃ וַעֲשֵׂה־לִי מַטְעַמִּים כַּאֲשֶׁר
18   אָהַבְתִּי וְהָבִיאָה לִּי וְאֹכֵלָה בַּעֲבוּר תְּבָרֶכְךָ נַפְשִׁי
19   בְּטֶרֶם אָמוּת׃ וְרִבְקָה שֹׁמַעַת בְּדַבֵּר יִצְחָק אֶל־עֵשָׂו
20   בְּנוֹ וַיֵּלֶךְ עֵשָׂו הַשָּׂדֶה לָצוּד צַיִד לְהָבִיא׃ וְרִבְקָה
21   אָמְרָה אֶל־יַעֲקֹב בְּנָהּ לֵאמֹר הִנֵּה שָׁמַעְתִּי אֶת־אָבִיךָ
22   מְדַבֵּר אֶל־עֵשָׂו אָחִיךָ לֵאמֹר׃ הָבִיאָה לִּי צַיִד וַעֲשֵׂה־
23  לִי מַטְעַמִּים וְאֹכֵלָה וַאֲבָרֶכְכָה לִפְנֵי יְהוָה לִפְנֵי
24   מוֹתִי׃ וְעַתָּה בְנִי שְׁמַע בְּקֹלִי לַאֲשֶׁר אֲנִי מְצַוָּה אֹתָךְ׃
25   לֶךְ־נָא אֶל־הַצֹּאן וְקַח־לִי מִשָּׁם שְׁנֵי גְּדָיֵי עִזִּים טֹבִים
26   וְאֶעֱשֶׂה אֹתָם מַטְעַמִּים לְאָבִיךָ כַּאֲשֶׁר אָהֵב׃ וְהֵבֵאתָ
27   לְאָבִיךָ וְאָכָל בַּעֲבֻר אֲשֶׁר יְבָרֶכְךָ לִפְנֵי מוֹתוֹ׃ וַיֹּאמֶר
28   יַעֲקֹב אֶל־רִבְקָה אִמּוֹ הֵן עֵשָׂו אָחִי אִישׁ שָׂעִר
29   וְאָנֹכִי אִישׁ חָלָק׃ אוּלַי יְמֻשֵּׁנִי אָבִי וְהָיִיתִי בְעֵינָיו
30   כִּמְתַעְתֵּעַ וְהֵבֵאתִי עָלַי קְלָלָה וְלֹא בְרָכָה׃ וַתֹּאמֶר
31   לוֹ אִמּוֹ עָלַי קִלְלָתְךָ בְּנִי אַךְ שְׁמַע בְּקֹלִי וְלֵךְ קַח־לִי׃
32   וַיֵּלֶךְ וַיִּקַּח וַיָּבֵא לְאִמּוֹ וַתַּעַשׂ אִמּוֹ מַטְעַמִּים כַּאֲשֶׁר
33   אָהֵב אָבִיו׃ וַתִּקַּח רִבְקָה אֶת־בִּגְדֵי עֵשָׂו בְּנָהּ הַגָּדֹל
34   הַחֲמֻדֹת אֲשֶׁר אִתָּהּ בַּבָּיִת וַתַּלְבֵּשׁ אֶת־יַעֲקֹב בְּנָהּ
35   הַקָּטָן׃ וְאֵת עֹרֹת גְּדָיֵי הָעִזִּים הִלְבִּישָׁה עַל־יָדָיו וְעַל
36   חֶלְקַת צַוָּארָיו׃ וַתִּתֵּן אֶת־הַמַּטְעַמִּים וְאֶת־הַלֶּחֶם אֲשֶׁר
37   עָשָׂתָה בְּיַד יַעֲקֹב בְּנָהּ׃ וַיָּבֹא אֶל־אָבִיו וַיֹּאמֶר אָבִי
38   וַיֹּאמֶר הִנֶּנִּי מִי אַתָּה בְּנִי׃ וַיֹּאמֶר יַעֲקֹב אֶל־אָבִיו אָנֹכִי
39   עֵשָׂו בְּכֹרֶךָ עָשִׂיתִי כַּאֲשֶׁר דִּבַּרְתָּ אֵלָי קוּם־נָא שְׁבָה
40   וְאָכְלָה מִצֵּידִי בַּעֲבוּר תְּבָרְכַנִּי נַפְשֶׁךָ׃ וַיֹּאמֶר יִצְחָק
41   אֶל־בְּנוֹ מַה־זֶּה מִהַרְתָּ לִמְצֹא בְּנִי וַיֹּאמֶר כִּי הִקְרָה
42   יְהוָה אֱלֹהֶיךָ לְפָנָי׃ וַיֹּאמֶר יִצְחָק אֶל־יַעֲקֹב גְּשָׁה־נָּא
