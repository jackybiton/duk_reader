 1  וְכֹל אֲשֶׁר־לְךָ בָּעִיר הוֹצֵא מִן־הַמָּקוֹם׃ כִּי־מַשְׁחִתִים
 2   אֲנַחְנוּ אֶת־הַמָּקוֹם הַזֶּה כִּי־גָדְלָה צַעֲקָתָם אֶת־פְּנֵי
 3   יְהוָה וַיְשַׁלְּחֵנוּ יְהוָה לְשַׁחֲתָהּ׃ וַיֵּצֵא לוֹט וַיְדַבֵּר  אֶל־
 4  חֲתָנָיו  לֹקְחֵי בְנֹתָיו וַיֹּאמֶר קוּמוּ צְּאוּ מִן־הַמָּקוֹם הַזֶּה
 5   כִּי־מַשְׁחִית יְהוָה אֶת־הָעִיר וַיְהִי כִמְצַחֵק בְּעֵינֵי חֲתָנָיו׃
 6   וּכְמוֹ הַשַּׁחַר עָלָה וַיָּאִיצוּ הַמַּלְאָכִים בְּלוֹט לֵאמֹר
 7   קוּם קַח אֶת־אִשְׁתְּךָ וְאֶת־שְׁתֵּי בְנֹתֶיךָ הַנִּמְצָאֹת
 8   פֶּן־תִּסָּפֶה בַּעֲו‍ֹן הָעִיר׃ וַיִּתְמַהְמָהּ  וַיַּחֲזִיקוּ הָאֲנָשִׁים
 9   בְּיָדוֹ וּבְיַד־אִשְׁתּוֹ וּבְיַד שְׁתֵּי בְנֹתָיו בְּחֶמְלַת יְהוָה
10   עָלָיו וַיֹּצִאֻהוּ וַיַּנִּחֻהוּ מִחוּץ לָעִיר׃ וַיְהִי כְהוֹצִיאָם
11   אֹתָם הַחוּצָה וַיֹּאמֶר הִמָּלֵט עַל־נַפְשֶׁךָ אַל־תַּבִּיט
12   אַחֲרֶיךָ וְאַל־תַּעֲמֹד בְּכָל־הַכִּכָּר הָהָרָה הִמָּלֵט פֶּן־
13  תִּסָּפֶה׃ וַיֹּאמֶר לוֹט אֲלֵהֶם אַל־נָא אֲדֹנָי׃ הִנֵּה־נָא מָצָא
14   עַבְדְּךָ חֵן בְּעֵינֶיךָ וַתַּגְדֵּל חַסְדְּךָ אֲשֶׁר עָשִׂיתָ עִמָּדִי
15   לְהַחֲיוֹת אֶת־נַפְשִׁי וְאָנֹכִי לֹא אוּכַל לְהִמָּלֵט הָהָרָה
16   פֶּן־תִּדְבָּקַנִי הָרָעָה וָמַתִּי׃ הִנֵּה־נָא הָעִיר הַזֹּאת קְרֹבָה
17   לָנוּס שָׁמָּה וְהִוא מִצְעָר אִמָּלְטָה נָּא שָׁמָּה הֲלֹא
18   מִצְעָר הִוא וּתְחִי נַפְשִׁי׃  וַיֹּאמֶר אֵלָיו הִנֵּה נָשָׂאתִי
19   פָנֶיךָ גַּם לַדָּבָר הַזֶּה לְבִלְתִּי הָפְכִּי אֶת־הָעִיר אֲשֶׁר
20   דִּבַּרְתָּ׃ מַהֵר הִמָּלֵט שָׁמָּה כִּי לֹא אוּכַל לַעֲשׂוֹת
21   דָּבָר עַד־בֹּאֲךָ שָׁמָּה עַל־כֵּן קָרָא שֵׁם־הָעִיר צוֹעַר׃
22   הַשֶּׁמֶשׁ יָצָא עַל־הָאָרֶץ וְלוֹט בָּא צֹעֲרָה׃ וַיהוָה
23   הִמְטִיר עַל־סְדֹם וְעַל־עֲמֹרָה גָּפְרִית וָאֵשׁ מֵאֵת יְהוָה
24   מִן־הַשָּׁמָיִם׃ וַיַּהֲפֹךְ אֶת־הֶעָרִים הָאֵל וְאֵת כָּל־הַכִּכָּר
25   וְאֵת כָּל־יֹשְׁבֵי הֶעָרִים וְצֶמַח הָאֲדָמָה׃ וַתַּבֵּט אִשְׁתּוֹ
26   מֵאַחֲרָיו וַתְּהִי נְצִיב מֶלַח׃ וַיַּשְׁכֵּם אַבְרָהָם בַּבֹּקֶר אֶל־
27  הַמָּקוֹם אֲשֶׁר־עָמַד שָׁם אֶת־פְּנֵי יְהוָה׃ וַיַּשְׁקֵף עַל־
28  פְּנֵי סְדֹם וַעֲמֹרָה וְעַל־כָּל־פְּנֵי אֶרֶץ הַכִּכָּר וַיַּרְא
29   וְהִנֵּה עָלָה קִיטֹר הָאָרֶץ כְּקִיטֹר הַכִּבְשָׁן׃ וַיְהִי בְּשַׁחֵת
30   אֱלֹהִים אֶת־עָרֵי הַכִּכָּר וַיִּזְכֹּר אֱלֹהִים אֶת־אַבְרָהָם
31   וַיְשַׁלַּח אֶת־לוֹט מִתּוֹךְ הַהֲפֵכָה בַּהֲפֹךְ אֶת־הֶעָרִים
32   אֲשֶׁר־יָשַׁב בָּהֵן לוֹט׃ וַיַּעַל לוֹט מִצּוֹעַר וַיֵּשֶׁב בָּהָר
33   וּשְׁתֵּי בְנֹתָיו עִמּוֹ כִּי יָרֵא לָשֶׁבֶת בְּצוֹעַר וַיֵּשֶׁב
34   בַּמְּעָרָה הוּא וּשְׁתֵּי בְנֹתָיו׃ וַתֹּאמֶר הַבְּכִירָה אֶל־
35  הַצְּעִירָה אָבִינוּ זָקֵן וְאִישׁ אֵין בָּאָרֶץ לָבוֹא עָלֵינוּ
36   כְּדֶרֶךְ כָּל־הָאָרֶץ׃ לְכָה נַשְׁקֶה אֶת־אָבִינוּ יַיִן וְנִשְׁכְּבָה
37   עִמּוֹ וּנְחַיֶּה מֵאָבִינוּ זָרַע׃ וַתַּשְׁקֶיןָ אֶת־אֲבִיהֶן יַיִן
38   בַּלַּיְלָה הוּא וַתָּבֹא הַבְּכִירָה וַתִּשְׁכַּב אֶת־אָבִיהָ וְלֹא־
39  יָדַע בְּשִׁכְבָהּ וּבְקוּמָהּ׃ וַיְהִי מִמָּחֳרָת וַתֹּאמֶר הַבְּכִירָה
40   אֶל־הַצְּעִירָה הֵן־שָׁכַבְתִּי אֶמֶשׁ אֶת־אָבִי נַשְׁקֶנּוּ יַיִן גַּם־
41  הַלַּיְלָה וּבֹאִי שִׁכְבִי עִמּוֹ וּנְחַיֶּה מֵאָבִינוּ זָרַע׃ וַתַּשְׁקֶיןָ
42   גַּם בַּלַּיְלָה הַהוּא אֶת־אֲבִיהֶן יָיִן וַתָּקָם הַצְּעִירָה
