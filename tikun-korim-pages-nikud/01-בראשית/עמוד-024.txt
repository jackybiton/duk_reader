 1  וְהַמְּעָרָה אֲשֶׁר־בּוֹ לְאַבְרָהָם לַאֲחֻזַּת־קָבֶר מֵאֵת בְּנֵי־
 2  חֵת׃ [ס]             וְאַבְרָהָם זָקֵן בָּא בַּיָּמִים
 3   וַיהוָה בֵּרַךְ אֶת־אַבְרָהָם בַּכֹּל׃ וַיֹּאמֶר אַבְרָהָם אֶל־
 4  עַבְדּוֹ זְקַן בֵּיתוֹ הַמֹּשֵׁל בְּכָל־אֲשֶׁר־לוֹ שִׂים־נָא יָדְךָ
 5   תַּחַת יְרֵכִי׃ וְאַשְׁבִּיעֲךָ בַּיהוָה אֱלֹהֵי הַשָּׁמַיִם וֵאלֹהֵי
 6   הָאָרֶץ אֲשֶׁר לֹא־תִקַּח אִשָּׁה לִבְנִי מִבְּנוֹת הַכְּנַעֲנִי
 7   אֲשֶׁר אָנֹכִי יוֹשֵׁב בְּקִרְבּוֹ׃ כִּי אֶל־אַרְצִי וְאֶל־מוֹלַדְתִּי
 8   תֵּלֵךְ וְלָקַחְתָּ אִשָּׁה לִבְנִי לְיִצְחָק׃ וַיֹּאמֶר אֵלָיו הָעֶבֶד
 9   אוּלַי לֹא־תֹאבֶה הָאִשָּׁה לָלֶכֶת אַחֲרַי אֶל־הָאָרֶץ
10   הַזֹּאת הֶהָשֵׁב אָשִׁיב אֶת־בִּנְךָ אֶל־הָאָרֶץ אֲשֶׁר־יָצָאתָ
11   מִשָּׁם׃ וַיֹּאמֶר אֵלָיו אַבְרָהָם הִשָּׁמֶר לְךָ פֶּן־תָּשִׁיב
12   אֶת־בְּנִי שָׁמָּה׃ יְהוָה  אֱלֹהֵי הַשָּׁמַיִם אֲשֶׁר לְקָחַנִי
13   מִבֵּית אָבִי וּמֵאֶרֶץ מוֹלַדְתִּי וַאֲשֶׁר דִּבֶּר־לִי וַאֲשֶׁר
14   נִשְׁבַּע־לִי לֵאמֹר לְזַרְעֲךָ אֶתֵּן אֶת־הָאָרֶץ הַזֹּאת הוּא
15   יִשְׁלַח מַלְאָכוֹ לְפָנֶיךָ וְלָקַחְתָּ אִשָּׁה לִבְנִי מִשָּׁם׃ וְאִם־
16  לֹא תֹאבֶה הָאִשָּׁה לָלֶכֶת אַחֲרֶיךָ וְנִקִּיתָ מִשְּׁבֻעָתִי
17   זֹאת רַק אֶת־בְּנִי לֹא תָשֵׁב שָׁמָּה׃ וַיָּשֶׂם הָעֶבֶד אֶת־
18  יָדוֹ תַּחַת יֶרֶךְ אַבְרָהָם אֲדֹנָיו וַיִּשָּׁבַע לוֹ עַל־הַדָּבָר
19   הַזֶּה׃  וַיִּקַּח הָעֶבֶד עֲשָׂרָה גְמַלִּים מִגְּמַלֵּי אֲדֹנָיו וַיֵּלֶךְ
20   וְכָל־טוּב אֲדֹנָיו בְּיָדוֹ וַיָּקָם וַיֵּלֶךְ אֶל־אֲרַם נַהֲרַיִם
21   אֶל־עִיר נָחוֹר׃ וַיַּבְרֵךְ הַגְּמַלִּים מִחוּץ לָעִיר אֶל־בְּאֵר
22   הַמָּיִם לְעֵת עֶרֶב לְעֵת צֵאת הַשֹּׁאֲבֹת׃ וַיֹּאמַר  יְהוָה
23   אֱלֹהֵי אֲדֹנִי אַבְרָהָם הַקְרֵה־נָא לְפָנַי הַיּוֹם וַעֲשֵׂה־
24  חֶסֶד עִם אֲדֹנִי אַבְרָהָם׃ הִנֵּה אָנֹכִי נִצָּב עַל־עֵין הַמָּיִם
25   וּבְנוֹת אַנְשֵׁי הָעִיר יֹצְאֹת לִשְׁאֹב מָיִם׃ וְהָיָה הַנַּעֲרָ
26   אֲשֶׁר אֹמַר אֵלֶיהָ הַטִּי־נָא כַדֵּךְ וְאֶשְׁתֶּה וְאָמְרָה
27   שְׁתֵה וְגַם־גְּמַלֶּיךָ אַשְׁקֶה אֹתָהּ הֹכַחְתָּ לְעַבְדְּךָ
28   לְיִצְחָק וּבָהּ אֵדַע כִּי־עָשִׂיתָ חֶסֶד עִם־אֲדֹנִי׃ וַיְהִי־
29  הוּא טֶרֶם כִּלָּה לְדַבֵּר וְהִנֵּה רִבְקָה יֹצֵאת אֲשֶׁר
30   יֻלְּדָה לִבְתוּאֵל בֶּן־מִלְכָּה אֵשֶׁת נָחוֹר אֲחִי אַבְרָהָם
31   וְכַדָּהּ עַל־שִׁכְמָהּ׃ וְהַנַּעֲרָ טֹבַת מַרְאֶה מְאֹד בְּתוּלָה
32   וְאִישׁ לֹא יְדָעָהּ וַתֵּרֶד הָעַיְנָה וַתְּמַלֵּא כַדָּהּ וַתָּעַל׃
33   וַיָּרָץ הָעֶבֶד לִקְרָאתָהּ וַיֹּאמֶר הַגְמִיאִינִי נָא מְעַט־מַיִם
34   מִכַּדֵּךְ׃ וַתֹּאמֶר שְׁתֵה אֲדֹנִי וַתְּמַהֵר וַתֹּרֶד כַּדָּהּ עַל־
35  יָדָהּ וַתַּשְׁקֵהוּ׃ וַתְּכַל לְהַשְׁקֹתוֹ וַתֹּאמֶר גַּם לִגְמַלֶּיךָ
36   אֶשְׁאָב עַד אִם־כִּלּוּ לִשְׁתֹּת׃ וַתְּמַהֵר וַתְּעַר כַּדָּהּ אֶל־
37  הַשֹּׁקֶת וַתָּרָץ עוֹד אֶל־הַבְּאֵר לִשְׁאֹב וַתִּשְׁאַב לְכָל־
38  גְּמַלָּיו׃ וְהָאִישׁ מִשְׁתָּאֵה לָהּ מַחֲרִישׁ לָדַעַת הַהִצְלִיחַ
39   יְהוָה דַּרְכּוֹ אִם־לֹא׃ וַיְהִי כַּאֲשֶׁר כִּלּוּ הַגְּמַלִּים
40   לִשְׁתּוֹת וַיִּקַּח הָאִישׁ נֶזֶם זָהָב בֶּקַע מִשְׁקָלוֹ וּשְׁנֵי
41   צְמִידִים עַל־יָדֶיהָ עֲשָׂרָה זָהָב מִשְׁקָלָם׃ וַיֹּאמֶר בַּת־
42  מִי אַתְּ הַגִּידִי נָא לִי הֲיֵשׁ בֵּית־אָבִיךְ מָקוֹם לָנוּ לָלִין׃
