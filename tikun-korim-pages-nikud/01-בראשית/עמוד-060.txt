 1  וַיִּמְלְאוּ־לוֹ אַרְבָּעִים יוֹם כִּי כֵּן יִמְלְאוּ יְמֵי הַחֲנֻטִים
 2   וַיִּבְכּוּ אֹתוֹ מִצְרַיִם שִׁבְעִים יוֹם׃ וַיַּעַבְרוּ יְמֵי בְכִיתוֹ
 3   וַיְדַבֵּר יוֹסֵף אֶל־בֵּית פַּרְעֹה לֵאמֹר אִם־נָא מָצָאתִי
 4   חֵן בְּעֵינֵיכֶם דַּבְּרוּ־נָא בְּאָזְנֵי פַרְעֹה לֵאמֹר׃ אָבִי
 5   הִשְׁבִּיעַנִי לֵאמֹר הִנֵּה אָנֹכִי מֵת בְּקִבְרִי אֲשֶׁר
 6   כָּרִיתִי לִי בְּאֶרֶץ כְּנַעַן שָׁמָּה תִּקְבְּרֵנִי וְעַתָּה אֶעֱלֶה־
 7  נָּא וְאֶקְבְּרָה אֶת־אָבִי וְאָשׁוּבָה׃ וַיֹּאמֶר פַּרְעֹה עֲלֵה
 8   וּקְבֹר אֶת־אָבִיךָ כַּאֲשֶׁר הִשְׁבִּיעֶךָ׃ וַיַּעַל יוֹסֵף
 9   לִקְבֹּר אֶת־אָבִיו וַיַּעֲלוּ אִתּוֹ כָּל־עַבְדֵי פַרְעֹה זִקְנֵי
10   בֵיתוֹ וְכֹל זִקְנֵי אֶרֶץ־מִצְרָיִם׃ וְכֹל בֵּית יוֹסֵף וְאֶחָיו
11   וּבֵית אָבִיו רַק טַפָּם וְצֹאנָם וּבְקָרָם עָזְבוּ בְּאֶרֶץ
12   גֹּשֶׁן׃ וַיַּעַל עִמּוֹ גַּם־רֶכֶב גַּם־פָּרָשִׁים וַיְהִי הַמַּחֲנֶה
13   כָּבֵד מְאֹד׃ וַיָּבֹאוּ עַד־גֹּרֶן הָאָטָד אֲשֶׁר בְּעֵבֶר
14   הַיַּרְדֵּן וַיִּסְפְּדוּ־שָׁם מִסְפֵּד גָּדוֹל וְכָבֵד מְאֹד וַיַּעַשׂ
15   לְאָבִיו אֵבֶל שִׁבְעַת יָמִים׃ וַיַּרְא יוֹשֵׁב הָאָרֶץ
16   הַכְּנַעֲנִי אֶת־הָאֵבֶל בְּגֹרֶן הָאָטָד וַיֹּאמְרוּ אֵבֶל־כָּבֵד
17   זֶה לְמִצְרָיִם עַל־כֵּן קָרָא שְׁמָהּ אָבֵל מִצְרַיִם אֲשֶׁר
18   בְּעֵבֶר הַיַּרְדֵּן׃ וַיַּעֲשׂוּ בָנָיו לוֹ כֵּן כַּאֲשֶׁר צִוָּם׃ וַיִּשְׂאוּ
19   אֹתוֹ בָנָיו אַרְצָה כְּנַעַן וַיִּקְבְּרוּ אֹתוֹ בִּמְעָרַת שְׂדֵה
20   הַמַּכְפֵּלָה אֲשֶׁר קָנָה אַבְרָהָם אֶת־הַשָּׂדֶה לַאֲחֻזַּת־
21  קֶבֶר מֵאֵת עֶפְרֹן הַחִתִּי עַל־פְּנֵי מַמְרֵא׃ וַיָּשָׁב יוֹסֵף
22   מִצְרַיְמָה הוּא וְאֶחָיו וְכָל־הָעֹלִים אִתּוֹ לִקְבֹּר אֶת־
23  אָבִיו אַחֲרֵי קָבְרוֹ אֶת־אָבִיו׃ וַיִּרְאוּ אֲחֵי־יוֹסֵף כִּי־מֵת
24   אֲבִיהֶם וַיֹּאמְרוּ לוּ יִשְׂטְמֵנוּ יוֹסֵף וְהָשֵׁב יָשִׁיב לָנוּ
25   אֵת כָּל־הָרָעָה אֲשֶׁר גָּמַלְנוּ אֹתוֹ׃ וַיְצַוּוּ אֶל־יוֹסֵף
26   לֵאמֹר אָבִיךָ צִוָּה לִפְנֵי מוֹתוֹ לֵאמֹר׃ כֹּה־תֹאמְרוּ
27   לְיוֹסֵף אָנָּא שָׂא נָא פֶּשַׁע אַחֶיךָ וְחַטָּאתָם כִּי־רָעָה
28   גְמָלוּךָ וְעַתָּה שָׂא נָא לְפֶשַׁע עַבְדֵי אֱלֹהֵי אָבִיךָ
29   וַיֵּבְךְּ יוֹסֵף בְּדַבְּרָם אֵלָיו׃ וַיֵּלְכוּ גַּם־אֶחָיו וַיִּפְּלוּ לְפָנָיו
30   וַיֹּאמְרוּ הִנֶּנּוּ לְךָ לַעֲבָדִים׃ וַיֹּאמֶר אֲלֵהֶם יוֹסֵף אַל־
31  תִּירָאוּ כִּי הֲתַחַת אֱלֹהִים אָנִי׃ וְאַתֶּם חֲשַׁבְתֶּם עָלַי
32   רָעָה אֱלֹהִים חֲשָׁבָהּ לְטֹבָה לְמַעַן עֲשֹׂה כַּיּוֹם הַזֶּה
33   לְהַחֲיֹת עַם־רָב׃  וְעַתָּה אַל־תִּירָאוּ אָנֹכִי אֲכַלְכֵּל
34   אֶתְכֶם וְאֶת־טַפְּכֶם וַיְנַחֵם אוֹתָם וַיְדַבֵּר עַל־לִבָּם׃
35   וַיֵּשֶׁב יוֹסֵף בְּמִצְרַיִם הוּא וּבֵית אָבִיו וַיְחִי יוֹסֵף מֵאָה
36   וָעֶשֶׂר שָׁנִים׃  וַיַּרְא יוֹסֵף לְאֶפְרַיִם בְּנֵי שִׁלֵּשִׁים גַּם
37   בְּנֵי מָכִיר בֶּן־מְנַשֶּׁה יֻלְּדוּ עַל־בִּרְכֵּי יוֹסֵף׃ וַיֹּאמֶר
38   יוֹסֵף אֶל־אֶחָיו אָנֹכִי מֵת וֵאלֹהִים פָּקֹד יִפְקֹד אֶתְכֶם
39   וְהֶעֱלָה אֶתְכֶם מִן־הָאָרֶץ הַזֹּאת אֶל־הָאָרֶץ אֲשֶׁר
40   נִשְׁבַּע לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב׃ וַיַּשְׁבַּע יוֹסֵף
41   אֶת־בְּנֵי יִשְׂרָאֵל לֵאמֹר פָּקֹד יִפְקֹד אֱלֹהִים אֶתְכֶם
42   וְהַעֲלִתֶם אֶת־עַצְמֹתַי מִזֶּה׃ וַיָּמָת יוֹסֵף בֶּן־מֵאָה
