 1  וַיֵּצֵא יוֹסֵף עַל־אֶרֶץ מִצְרָיִם׃ וְיוֹסֵף בֶּן־שְׁלֹשִׁים שָׁנָה
 2   בְּעָמְדוֹ לִפְנֵי פַּרְעֹה מֶלֶךְ־מִצְרָיִם וַיֵּצֵא יוֹסֵף מִלִּפְנֵי
 3   פַרְעֹה וַיַּעֲבֹר בְּכָל־אֶרֶץ מִצְרָיִם׃ וַתַּעַשׂ הָאָרֶץ
 4   בְּשֶׁבַע שְׁנֵי הַשָּׂבָע לִקְמָצִים׃ וַיִּקְבֹּץ אֶת־כָּל־אֹכֶל
 5   שֶׁבַע שָׁנִים אֲשֶׁר הָיוּ בְּאֶרֶץ מִצְרַיִם וַיִּתֶּן־אֹכֶל
 6   בֶּעָרִים אֹכֶל שְׂדֵה־הָעִיר אֲשֶׁר סְבִיבֹתֶיהָ נָתַן
 7   בְּתוֹכָהּ׃ וַיִּצְבֹּר יוֹסֵף בָּר כְּחוֹל הַיָּם הַרְבֵּה מְאֹד
 8   עַד כִּי־חָדַל לִסְפֹּר כִּי־אֵין מִסְפָּר׃ וּלְיוֹסֵף יֻלַּד שְׁנֵי
 9   בָנִים בְּטֶרֶם תָּבוֹא שְׁנַת הָרָעָב אֲשֶׁר יָלְדָה־לּוֹ
10   אָסְנַת בַּת־פּוֹטִי פֶרַע כֹּהֵן אוֹן׃ וַיִּקְרָא יוֹסֵף אֶת־שֵׁם
11   הַבְּכוֹר מְנַשֶּׁה כִּי־נַשַּׁנִי אֱלֹהִים אֶת־כָּל־עֲמָלִי וְאֵת
12   כָּל־בֵּית אָבִי׃ וְאֵת שֵׁם הַשֵּׁנִי קָרָא אֶפְרָיִם כִּי־הִפְרַנִי
13   אֱלֹהִים בְּאֶרֶץ עָנְיִי׃  וַתִּכְלֶינָה שֶׁבַע שְׁנֵי הַשָּׂבָע
14   אֲשֶׁר הָיָה בְּאֶרֶץ מִצְרָיִם׃ וַתְּחִלֶּינָה שֶׁבַע שְׁנֵי
15   הָרָעָב לָבוֹא כַּאֲשֶׁר אָמַר יוֹסֵף וַיְהִי רָעָב בְּכָל־
16  הָאֲרָצוֹת וּבְכָל־אֶרֶץ מִצְרַיִם הָיָה לָחֶם׃ וַתִּרְעַב כָּל־
17  אֶרֶץ מִצְרַיִם וַיִּצְעַק הָעָם אֶל־פַּרְעֹה לַלָּחֶם וַיֹּאמֶר
18   פַּרְעֹה לְכָל־מִצְרַיִם לְכוּ אֶל־יוֹסֵף אֲשֶׁר־יֹאמַר
19   לָכֶם תַּעֲשׂוּ׃ וְהָרָעָב הָיָה עַל כָּל־פְּנֵי הָאָרֶץ וַיִּפְתַּח
20   יוֹסֵף אֶת־כָּל־אֲשֶׁר בָּהֶם וַיִּשְׁבֹּר לְמִצְרַיִם וַיֶּחֱזַק
21   הָרָעָב בְּאֶרֶץ מִצְרָיִם׃ וְכָל־הָאָרֶץ בָּאוּ מִצְרַיְמָה
22   לִשְׁבֹּר אֶל־יוֹסֵף כִּי־חָזַק הָרָעָב בְּכָל־הָאָרֶץ׃ וַיַּרְא
23   יַעֲקֹב כִּי יֶשׁ־שֶׁבֶר בְּמִצְרָיִם וַיֹּאמֶר יַעֲקֹב לְבָנָיו
24   לָמָּה תִּתְרָאוּ׃ וַיֹּאמֶר הִנֵּה שָׁמַעְתִּי כִּי יֶשׁ־שֶׁבֶר
25   בְּמִצְרָיִם רְדוּ־שָׁמָּה וְשִׁבְרוּ־לָנוּ מִשָּׁם וְנִחְיֶה וְלֹא
26   נָמוּת׃ וַיֵּרְדוּ אֲחֵי־יוֹסֵף עֲשָׂרָה לִשְׁבֹּר בָּר מִמִּצְרָיִם׃
27   וְאֶת־בִּנְיָמִין אֲחִי יוֹסֵף לֹא־שָׁלַח יַעֲקֹב אֶת־אֶחָיו כִּי
28   אָמַר פֶּן־יִקְרָאֶנּוּ אָסוֹן׃ וַיָּבֹאוּ בְּנֵי יִשְׂרָאֵל לִשְׁבֹּר
29   בְּתוֹךְ הַבָּאִים כִּי־הָיָה הָרָעָב בְּאֶרֶץ כְּנָעַן׃ וְיוֹסֵף
30   הוּא הַשַּׁלִּיט עַל־הָאָרֶץ הוּא הַמַּשְׁבִּיר לְכָל־עַם
31   הָאָרֶץ וַיָּבֹאוּ אֲחֵי יוֹסֵף וַיִּשְׁתַּחֲווּ־לוֹ אַפַּיִם אָרְצָה׃
32   וַיַּרְא יוֹסֵף אֶת־אֶחָיו וַיַּכִּרֵם וַיִּתְנַכֵּר אֲלֵיהֶם וַיְדַבֵּר
33   אִתָּם קָשׁוֹת וַיֹּאמֶר אֲלֵהֶם מֵאַיִן בָּאתֶם וַיֹּאמְרוּ
34   מֵאֶרֶץ כְּנַעַן לִשְׁבָּר־אֹכֶל׃ וַיַּכֵּר יוֹסֵף אֶת־אֶחָיו
35   וְהֵם לֹא הִכִּרֻהוּ׃ וַיִּזְכֹּר יוֹסֵף אֵת הַחֲלֹמוֹת אֲשֶׁר
36   חָלַם לָהֶם וַיֹּאמֶר אֲלֵהֶם מְרַגְּלִים אַתֶּם לִרְאוֹת
37   אֶת־עֶרְוַת הָאָרֶץ בָּאתֶם׃ וַיֹּאמְרוּ אֵלָיו לֹא אֲדֹנִי
38   וַעֲבָדֶיךָ בָּאוּ לִשְׁבָּר־אֹכֶל׃ כֻּלָּנוּ בְּנֵי אִישׁ־אֶחָד נָחְנוּ
39   כֵּנִים אֲנַחְנוּ לֹא־הָיוּ עֲבָדֶיךָ מְרַגְּלִים׃ וַיֹּאמֶר אֲלֵהֶם
40   לֹא כִּי־עֶרְוַת הָאָרֶץ בָּאתֶם לִרְאוֹת׃ וַיֹּאמְרוּ שְׁנֵים
41   עָשָׂר עֲבָדֶיךָ אַחִים  אֲנַחְנוּ בְּנֵי אִישׁ־אֶחָד בְּאֶרֶץ
42   כְּנָעַן וְהִנֵּה הַקָּטֹן אֶת־אָבִינוּ הַיּוֹם וְהָאֶחָד אֵינֶנּוּ׃
