 1  וַיָּשֶׂם שַׂק בְּמָתְנָיו וַיִּתְאַבֵּל עַל־בְּנוֹ יָמִים רַבִּים׃
 2   וַיָּקֻמוּ כָל־בָּנָיו וְכָל־בְּנֹתָיו לְנַחֲמוֹ וַיְמָאֵן לְהִתְנַחֵם
 3   וַיֹּאמֶר כִּי־אֵרֵד אֶל־בְּנִי אָבֵל שְׁאֹלָה וַיֵּבְךְּ אֹתוֹ
 4   אָבִיו׃ וְהַמְּדָנִים מָכְרוּ אֹתוֹ אֶל־מִצְרָיִם לְפוֹטִיפַר
 5   סְרִיס פַּרְעֹה שַׂר הַטַּבָּחִים׃ [פ]
 6    וַיְהִי בָּעֵת הַהִוא וַיֵּרֶד יְהוּדָה מֵאֵת אֶחָיו וַיֵּט עַד־
 7  אִישׁ עֲדֻלָּמִי וּשְׁמוֹ חִירָה׃ וַיַּרְא־שָׁם יְהוּדָה בַּת־אִישׁ
 8   כְּנַעֲנִי וּשְׁמוֹ שׁוּעַ וַיִּקָּחֶהָ וַיָּבֹא אֵלֶיהָ׃ וַתַּהַר וַתֵּלֶד
 9   בֵּן וַיִּקְרָא אֶת־שְׁמוֹ עֵר׃ וַתַּהַר עוֹד וַתֵּלֶד בֵּן וַתִּקְרָא
10   אֶת־שְׁמוֹ אוֹנָן׃ וַתֹּסֶף עוֹד וַתֵּלֶד בֵּן וַתִּקְרָא אֶת־שְׁמוֹ
11   שֵׁלָה וְהָיָה בִכְזִיב בְּלִדְתָּהּ אֹתוֹ׃ וַיִּקַּח יְהוּדָה אִשָּׁה
12   לְעֵר בְּכוֹרוֹ וּשְׁמָהּ תָּמָר׃ וַיְהִי עֵר בְּכוֹר יְהוּדָה רַע
13   בְּעֵינֵי יְהוָה וַיְמִתֵהוּ יְהוָה׃ וַיֹּאמֶר יְהוּדָה לְאוֹנָן בֹּא
14   אֶל־אֵשֶׁת אָחִיךָ וְיַבֵּם אֹתָהּ וְהָקֵם זֶרַע לְאָחִיךָ׃
15   וַיֵּדַע אוֹנָן כִּי לֹּא לוֹ יִהְיֶה הַזָּרַע וְהָיָה אִם־בָּא אֶל־
16  אֵשֶׁת אָחִיו וְשִׁחֵת אַרְצָה לְבִלְתִּי נְתָן־זֶרַע לְאָחִיו׃
17   וַיֵּרַע בְּעֵינֵי יְהוָה אֲשֶׁר עָשָׂה וַיָּמֶת גַּם־אֹתוֹ׃ וַיֹּאמֶר
18   יְהוּדָה לְתָמָר כַּלָּתוֹ שְׁבִי אַלְמָנָה בֵית־אָבִיךְ עַד־
19  יִגְדַּל שֵׁלָה בְנִי כִּי אָמַר פֶּן־יָמוּת גַּם־הוּא כְּאֶחָיו
20   וַתֵּלֶךְ תָּמָר וַתֵּשֶׁב בֵּית אָבִיהָ׃ וַיִּרְבּוּ הַיָּמִים וַתָּמָת
21   בַּת־שׁוּעַ אֵשֶׁת־יְהוּדָה וַיִּנָּחֶם יְהוּדָה וַיַּעַל עַל־גֹּזְזֵי
22   צֹאנוֹ הוּא וְחִירָה רֵעֵהוּ הָעֲדֻלָּמִי תִּמְנָתָה׃ וַיֻּגַּד לְתָמָר
23   לֵאמֹר הִנֵּה חָמִיךְ עֹלֶה תִמְנָתָה לָגֹז צֹאנוֹ׃ וַתָּסַר בִּגְדֵי
24   אַלְמְנוּתָהּ מֵעָלֶיהָ וַתְּכַס בַּצָּעִיף וַתִּתְעַלָּף וַתֵּשֶׁב
25   בְּפֶתַח עֵינַיִם אֲשֶׁר עַל־דֶּרֶךְ תִּמְנָתָה כִּי רָאֲתָה כִּי־
26  גָדַל שֵׁלָה וְהִוא לֹא־נִתְּנָה לוֹ לְאִשָּׁה׃ וַיִּרְאֶהָ יְהוּדָה
27   וַיַּחְשְׁבֶהָ לְזוֹנָה כִּי כִסְּתָה פָּנֶיהָ׃ וַיֵּט אֵלֶיהָ אֶל־הַדֶּרֶךְ
28   וַיֹּאמֶר הָבָה־נָּא אָבוֹא אֵלַיִךְ כִּי לֹא יָדַע כִּי כַלָּתוֹ
29   הִוא וַתֹּאמֶר מַה־תִּתֶּן־לִי כִּי תָבוֹא אֵלָי׃ וַיֹּאמֶר אָנֹכִי
30   אֲשַׁלַּח גְּדִי־עִזִּים מִן־הַצֹּאן וַתֹּאמֶר אִם־תִּתֵּן עֵרָבוֹן
31   עַד שָׁלְחֶךָ׃ וַיֹּאמֶר מָה הָעֵרָבוֹן אֲשֶׁר אֶתֶּן־לָךְ
32   וַתֹּאמֶר חֹתָמְךָ וּפְתִילֶךָ וּמַטְּךָ אֲשֶׁר בְּיָדֶךָ וַיִּתֶּן־לָהּ
33   וַיָּבֹא אֵלֶיהָ וַתַּהַר לוֹ׃ וַתָּקָם וַתֵּלֶךְ וַתָּסַר צְעִיפָהּ
34   מֵעָלֶיהָ וַתִּלְבַּשׁ בִּגְדֵי אַלְמְנוּתָהּ׃ וַיִּשְׁלַח יְהוּדָה אֶת־
35  גְּדִי הָעִזִּים בְּיַד רֵעֵהוּ הָעֲדֻלָּמִי לָקַחַת הָעֵרָבוֹן מִיַּד
36   הָאִשָּׁה וְלֹא מְצָאָהּ׃ וַיִּשְׁאַל אֶת־אַנְשֵׁי מְקֹמָהּ לֵאמֹר
37   אַיֵּה הַקְּדֵשָׁה הִוא בָעֵינַיִם עַל־הַדָּרֶךְ וַיֹּאמְרוּ לֹא־
38  הָיְתָה בָזֶה קְדֵשָׁה׃ וַיָּשָׁב אֶל־יְהוּדָה וַיֹּאמֶר לֹא
39   מְצָאתִיהָ וְגַם אַנְשֵׁי הַמָּקוֹם אָמְרוּ לֹא־הָיְתָה בָזֶה
40   קְדֵשָׁה׃ וַיֹּאמֶר יְהוּדָה תִּקַּח־לָהּ פֶּן נִהְיֶה לָבוּז הִנֵּה
41   שָׁלַחְתִּי הַגְּדִי הַזֶּה וְאַתָּה לֹא מְצָאתָהּ׃ וַיְהִי  כְּמִשְׁלֹשׁ
42   חֳדָשִׁים וַיֻּגַּד לִיהוּדָה לֵאמֹר זָנְתָה תָּמָר כַּלָּתֶךָ
