 1  וַיַּרְא אֹתָהּ שְׁכֶם בֶּן־חֲמוֹר הַחִוִּי נְשִׂיא הָאָרֶץ וַיִּקַּח
 2   אֹתָהּ וַיִּשְׁכַּב אֹתָהּ וַיְעַנֶּהָ׃ וַתִּדְבַּק נַפְשׁוֹ בְּדִינָה בַּת־
 3  יַעֲקֹב וַיֶּאֱהַב אֶת־הַנַּעֲרָ וַיְדַבֵּר עַל־לֵב הַנַּעֲרָ׃ וַיֹּאמֶר
 4   שְׁכֶם אֶל־חֲמוֹר אָבִיו לֵאמֹר קַח־לִי אֶת־הַיַּלְדָּה
 5   הַזֹּאת לְאִשָּׁה׃ וְיַעֲקֹב שָׁמַע כִּי טִמֵּא אֶת־דִּינָה בִתּוֹ
 6   וּבָנָיו הָיוּ אֶת־מִקְנֵהוּ בַּשָּׂדֶה וְהֶחֱרִשׁ יַעֲקֹב עַד־בֹּאָם׃
 7   וַיֵּצֵא חֲמוֹר אֲבִי־שְׁכֶם אֶל־יַעֲקֹב לְדַבֵּר אִתּוֹ׃ וּבְנֵי
 8   יַעֲקֹב בָּאוּ מִן־הַשָּׂדֶה כְּשָׁמְעָם וַיִּתְעַצְּבוּ הָאֲנָשִׁים
 9   וַיִּחַר לָהֶם מְאֹד כִּי־נְבָלָה עָשָׂה בְיִשְׂרָאֵל לִשְׁכַּב אֶת־
10  בַּת־יַעֲקֹב וְכֵן לֹא יֵעָשֶׂה׃ וַיְדַבֵּר חֲמוֹר אִתָּם לֵאמֹר
11   שְׁכֶם בְּנִי חָשְׁקָה נַפְשׁוֹ בְּבִתְּכֶם תְּנוּ נָא אֹתָהּ לוֹ
12   לְאִשָּׁה׃ וְהִתְחַתְּנוּ אֹתָנוּ בְּנֹתֵיכֶם תִּתְּנוּ־לָנוּ וְאֶת־בְּנֹתֵינוּ
13   תִּקְחוּ לָכֶם׃ וְאִתָּנוּ תֵּשֵׁבוּ וְהָאָרֶץ תִּהְיֶה לִפְנֵיכֶם שְׁבוּ
14   וּסְחָרוּהָ וְהֵאָחֲזוּ בָּהּ׃ וַיֹּאמֶר שְׁכֶם אֶל־אָבִיהָ וְאֶל־
15  אַחֶיהָ אֶמְצָא־חֵן בְּעֵינֵיכֶם וַאֲשֶׁר תֹּאמְרוּ אֵלַי אֶתֵּן׃
16   הַרְבּוּ עָלַי מְאֹד מֹהַר וּמַתָּן וְאֶתְּנָה כַּאֲשֶׁר תֹּאמְרוּ
17   אֵלָי וּתְנוּ־לִי אֶת־הַנַּעֲרָ לְאִשָּׁה׃ וַיַּעֲנוּ בְנֵי־יַעֲקֹב אֶת־
18  שְׁכֶם וְאֶת־חֲמוֹר אָבִיו בְּמִרְמָה וַיְדַבֵּרוּ אֲשֶׁר טִמֵּא
19   אֵת דִּינָה אֲחֹתָם׃ וַיֹּאמְרוּ אֲלֵיהֶם לֹא נוּכַל לַעֲשׂוֹת
20   הַדָּבָר הַזֶּה לָתֵת אֶת־אֲחֹתֵנוּ לְאִישׁ אֲשֶׁר־לוֹ עָרְלָה
21   כִּי־חֶרְפָּה הִוא לָנוּ׃ אַךְ־בְּזֹאת נֵאוֹת לָכֶם אִם תִּהְיוּ
22   כָמֹנוּ לְהִמֹּל לָכֶם כָּל־זָכָר׃ וְנָתַנּוּ אֶת־בְּנֹתֵינוּ לָכֶם וְאֶת־
23  בְּנֹתֵיכֶם נִקַּח־לָנוּ וְיָשַׁבְנוּ אִתְּכֶם וְהָיִינוּ לְעַם אֶחָד׃
24   וְאִם־לֹא תִשְׁמְעוּ אֵלֵינוּ לְהִמּוֹל וְלָקַחְנוּ אֶת־בִּתֵּנוּ
25   וְהָלָכְנוּ׃ וַיִּיטְבוּ דִבְרֵיהֶם בְּעֵינֵי חֲמוֹר וּבְעֵינֵי שְׁכֶם בֶּן־
26  חֲמוֹר׃ וְלֹא־אֵחַר הַנַּעַר לַעֲשׂוֹת הַדָּבָר כִּי חָפֵץ בְּבַת־
27  יַעֲקֹב וְהוּא נִכְבָּד מִכֹּל בֵּית אָבִיו׃ וַיָּבֹא חֲמוֹר וּשְׁכֶם
28   בְּנוֹ אֶל־שַׁעַר עִירָם וַיְדַבְּרוּ אֶל־אַנְשֵׁי עִירָם לֵאמֹר׃
29   הָאֲנָשִׁים הָאֵלֶּה שְׁלֵמִים הֵם אִתָּנוּ וְיֵשְׁבוּ בָאָרֶץ
30   וְיִסְחֲרוּ אֹתָהּ וְהָאָרֶץ הִנֵּה רַחֲבַת־יָדַיִם לִפְנֵיהֶם אֶת־
31  בְּנֹתָם נִקַּח־לָנוּ לְנָשִׁים וְאֶת־בְּנֹתֵינוּ נִתֵּן לָהֶם׃ אַךְ־
32  בְּזֹאת יֵאֹתוּ לָנוּ הָאֲנָשִׁים לָשֶׁבֶת אִתָּנוּ לִהְיוֹת לְעַם
33   אֶחָד בְּהִמּוֹל לָנוּ כָּל־זָכָר כַּאֲשֶׁר הֵם נִמֹּלִים׃ מִקְנֵהֶם
34   וְקִנְיָנָם וְכָל־בְּהֶמְתָּם הֲלוֹא לָנוּ הֵם אַךְ נֵאוֹתָה לָהֶם
35   וְיֵשְׁבוּ אִתָּנוּ׃ וַיִּשְׁמְעוּ אֶל־חֲמוֹר וְאֶל־שְׁכֶם בְּנוֹ כָּל־
36  יֹצְאֵי שַׁעַר עִירוֹ וַיִּמֹּלוּ כָּל־זָכָר כָּל־יֹצְאֵי שַׁעַר עִירוֹ׃
37   וַיְהִי בַיּוֹם הַשְּׁלִישִׁי בִּהְיוֹתָם כֹּאֲבִים וַיִּקְחוּ שְׁנֵי־בְנֵי־
38  יַעֲקֹב שִׁמְעוֹן וְלֵוִי אֲחֵי דִינָה אִישׁ חַרְבּוֹ וַיָּבֹאוּ עַל־
39  הָעִיר בֶּטַח וַיַּהַרְגוּ כָּל־זָכָר׃ וְאֶת־חֲמוֹר וְאֶת־שְׁכֶם
40   בְּנוֹ הָרְגוּ לְפִי־חָרֶב וַיִּקְחוּ אֶת־דִּינָה מִבֵּית שְׁכֶם
41   וַיֵּצֵאוּ׃ בְּנֵי יַעֲקֹב בָּאוּ עַל־הַחֲלָלִים וַיָּבֹזּוּ הָעִיר אֲשֶׁר
42   טִמְּאוּ אֲחוֹתָם׃ אֶת־צֹאנָם וְאֶת־בְּקָרָם וְאֶת־חֲמֹרֵיהֶם
