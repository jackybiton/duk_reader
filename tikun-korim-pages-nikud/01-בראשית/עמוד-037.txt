 1  וַיִּקְרָא שֵׁם־הַמָּקוֹם הַהוּא מַחֲנָיִם׃ [פ]
 2  וַיִּשְׁלַח יַעֲקֹב מַלְאָכִים לְפָנָיו אֶל־עֵשָׂו אָחִיו אַרְצָה
 3   שֵׂעִיר שְׂדֵה אֱדוֹם׃ וַיְצַו אֹתָם לֵאמֹר כֹּה תֹאמְרוּן
 4   לַאדֹנִי לְעֵשָׂו כֹּה אָמַר עַבְדְּךָ יַעֲקֹב עִם־לָבָן גַּרְתִּי
 5   וָאֵחַר עַד־עָתָּה׃ וַיְהִי־לִי שׁוֹר וַחֲמוֹר צֹאן וְעֶבֶד
 6   וְשִׁפְחָה וָאֶשְׁלְחָה לְהַגִּיד לַאדֹנִי לִמְצֹא־חֵן בְּעֵינֶיךָ׃♢
 7   וַיָּשֻׁבוּ הַמַּלְאָכִים אֶל־יַעֲקֹב לֵאמֹר בָּאנוּ אֶל־אָחִיךָ
 8   אֶל־עֵשָׂו וְגַם הֹלֵךְ לִקְרָאתְךָ וְאַרְבַּע־מֵאוֹת אִישׁ עִמּוֹ׃
 9   וַיִּירָא יַעֲקֹב מְאֹד וַיֵּצֶר לוֹ וַיַּחַץ אֶת־הָעָם אֲשֶׁר־אִתּוֹ
10   וְאֶת־הַצֹּאן וְאֶת־הַבָּקָר וְהַגְּמַלִּים לִשְׁנֵי מַחֲנוֹת׃ וַיֹּאמֶר
11   אִם־יָבוֹא עֵשָׂו אֶל־הַמַּחֲנֶה הָאַחַת וְהִכָּהוּ וְהָיָה הַמַּחֲנֶה
12   הַנִּשְׁאָר לִפְלֵיטָה׃♢ וַיֹּאמֶר יַעֲקֹב אֱלֹהֵי אָבִי אַבְרָהָם
13   וֵאלֹהֵי אָבִי יִצְחָק יְהוָה הָאֹמֵר אֵלַי שׁוּב לְאַרְצְךָ
14   וּלְמוֹלַדְתְּךָ וְאֵיטִיבָה עִמָּךְ׃ קָטֹנְתִּי מִכֹּל הַחֲסָדִים
15   וּמִכָּל־הָאֱמֶת אֲשֶׁר עָשִׂיתָ אֶת־עַבְדֶּךָ כִּי בְמַקְלִי
16   עָבַרְתִּי אֶת־הַיַּרְדֵּן הַזֶּה וְעַתָּה הָיִיתִי לִשְׁנֵי מַחֲנוֹת׃
17   הַצִּילֵנִי נָא מִיַּד אָחִי מִיַּד עֵשָׂו כִּי־יָרֵא אָנֹכִי אֹתוֹ פֶּן־
18  יָבוֹא וְהִכַּנִי אֵם עַל־בָּנִים׃ וְאַתָּה אָמַרְתָּ הֵיטֵב אֵיטִיב
19   עִמָּךְ וְשַׂמְתִּי אֶת־זַרְעֲךָ כְּחוֹל הַיָּם אֲשֶׁר לֹא־יִסָּפֵר
20   מֵרֹב׃♢  וַיָּלֶן שָׁם בַּלַּיְלָה הַהוּא וַיִּקַּח מִן־הַבָּא בְיָדוֹ
21   מִנְחָה לְעֵשָׂו אָחִיו׃ עִזִּים מָאתַיִם וּתְיָשִׁים עֶשְׂרִים
22   רְחֵלִים מָאתַיִם וְאֵילִים עֶשְׂרִים׃ גְּמַלִּים מֵינִיקוֹת
23   וּבְנֵיהֶם שְׁלֹשִׁים פָּרוֹת אַרְבָּעִים וּפָרִים עֲשָׂרָה
24   אֲתֹנֹת עֶשְׂרִים וַעְיָרִם עֲשָׂרָה׃ וַיִּתֵּן בְּיַד־עֲבָדָיו עֵדֶר
25   עֵדֶר לְבַדּוֹ וַיֹּאמֶר אֶל־עֲבָדָיו עִבְרוּ לְפָנַי וְרֶוַח
26   תָּשִׂימוּ בֵּין עֵדֶר וּבֵין עֵדֶר׃ וַיְצַו אֶת־הָרִאשׁוֹן לֵאמֹר
27   כִּי יִפְגָשְׁךָ עֵשָׂו אָחִי וּשְׁאֵלְךָ לֵאמֹר לְמִי־אַתָּה וְאָנָה
28   תֵלֵךְ וּלְמִי אֵלֶּה לְפָנֶיךָ׃ וְאָמַרְתָּ לְעַבְדְּךָ לְיַעֲקֹב מִנְחָה
29   הִוא שְׁלוּחָה לַאדֹנִי לְעֵשָׂו וְהִנֵּה גַם־הוּא אַחֲרֵינוּ׃ וַיְצַו
30   גַּם אֶת־הַשֵּׁנִי גַּם אֶת־הַשְּׁלִישִׁי גַּם אֶת־כָּל־הַהֹלְכִים
31   אַחֲרֵי הָעֲדָרִים לֵאמֹר כַּדָּבָר הַזֶּה תְּדַבְּרוּן אֶל־עֵשָׂו
32   בְּמֹצַאֲכֶם אֹתוֹ׃ וַאֲמַרְתֶּם גַּם הִנֵּה עַבְדְּךָ יַעֲקֹב אַחֲרֵינוּ
33   כִּי־אָמַר אֲכַפְּרָה פָנָיו בַּמִּנְחָה הַהֹלֶכֶת לְפָנָי וְאַחֲרֵי־
34  כֵן אֶרְאֶה פָנָיו אוּלַי יִשָּׂא פָנָי׃ וַתַּעֲבֹר הַמִּנְחָה עַל־
35  פָּנָיו וְהוּא לָן בַּלַּיְלָה־הַהוּא בַּמַּחֲנֶה׃ וַיָּקָם  בַּלַּיְלָה הוּא
36   וַיִּקַּח אֶת־שְׁתֵּי נָשָׁיו וְאֶת־שְׁתֵּי שִׁפְחֹתָיו וְאֶת־אַחַד
37   עָשָׂר יְלָדָיו וַיַּעֲבֹר אֵת מַעֲבַר יַבֹּק׃ וַיִּקָּחֵם וַיַּעֲבִרֵם
38   אֶת־הַנָּחַל וַיַּעֲבֵר אֶת־אֲשֶׁר־לוֹ׃ וַיִּוָּתֵר יַעֲקֹב לְבַדּוֹ
39   וַיֵּאָבֵק אִישׁ עִמּוֹ עַד עֲלוֹת הַשָּׁחַר׃ וַיַּרְא כִּי לֹא יָכֹל
40   לוֹ וַיִּגַּע בְּכַף־יְרֵכוֹ וַתֵּקַע כַּף־יֶרֶךְ יַעֲקֹב בְּהֵאָבְקוֹ עִמּוֹ׃
41   וַיֹּאמֶר שַׁלְּחֵנִי כִּי עָלָה הַשָּׁחַר וַיֹּאמֶר לֹא אֲשַׁלֵּחֲךָ
42   כִּי אִם־בֵּרַכְתָּנִי׃ וַיֹּאמֶר אֵלָיו מַה־שְּׁמֶךָ וַיֹּאמֶר יַעֲקֹב׃
