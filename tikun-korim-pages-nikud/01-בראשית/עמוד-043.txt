 1  וַיַּחֲלֹם עוֹד חֲלוֹם אַחֵר וַיְסַפֵּר אֹתוֹ לְאֶחָיו וַיֹּאמֶר
 2   הִנֵּה חָלַמְתִּי חֲלוֹם עוֹד וְהִנֵּה הַשֶּׁמֶשׁ וְהַיָּרֵחַ וְאַחַד
 3   עָשָׂר כּוֹכָבִים מִשְׁתַּחֲוִים לִי׃ וַיְסַפֵּר אֶל־אָבִיו וְאֶל־
 4  אֶחָיו וַיִּגְעַר־בּוֹ אָבִיו וַיֹּאמֶר לוֹ מָה הַחֲלוֹם הַזֶּה אֲשֶׁר
 5   חָלָמְתָּ הֲבוֹא נָבוֹא אֲנִי וְאִמְּךָ וְאַחֶיךָ לְהִשְׁתַּחֲו͏ֹת
 6   לְךָ אָרְצָה׃ וַיְקַנְאוּ־בוֹ אֶחָיו וְאָבִיו שָׁמַר אֶת־הַדָּבָר׃♢
 7    וַיֵּלְכוּ אֶחָיו לִרְעוֹת אֶת־צֹאן אֲבִיהֶם בִּשְׁכֶם׃ וַיֹּאמֶר
 8   יִשְׂרָאֵל אֶל־יוֹסֵף הֲלוֹא אַחֶיךָ רֹעִים בִּשְׁכֶם לְכָה
 9   וְאֶשְׁלָחֲךָ אֲלֵיהֶם וַיֹּאמֶר לוֹ הִנֵּנִי׃ וַיֹּאמֶר לוֹ לֶךְ־נָא
10   רְאֵה אֶת־שְׁלוֹם אַחֶיךָ וְאֶת־שְׁלוֹם הַצֹּאן וַהֲשִׁבֵנִי
11   דָּבָר וַיִּשְׁלָחֵהוּ מֵעֵמֶק חֶבְרוֹן וַיָּבֹא שְׁכֶמָה׃ וַיִּמְצָאֵהוּ
12   אִישׁ וְהִנֵּה תֹעֶה בַּשָּׂדֶה וַיִּשְׁאָלֵהוּ הָאִישׁ לֵאמֹר
13   מַה־תְּבַקֵּשׁ׃ וַיֹּאמֶר אֶת־אַחַי אָנֹכִי מְבַקֵּשׁ הַגִּידָה־
14  נָּא לִי אֵיפֹה הֵם רֹעִים׃ וַיֹּאמֶר הָאִישׁ נָסְעוּ מִזֶּה כִּי
15   שָׁמַעְתִּי אֹמְרִים נֵלְכָה דֹּתָיְנָה וַיֵּלֶךְ יוֹסֵף אַחַר אֶחָיו
16   וַיִּמְצָאֵם בְּדֹתָן׃ וַיִּרְאוּ אֹתוֹ מֵרָחֹק וּבְטֶרֶם יִקְרַב
17   אֲלֵיהֶם וַיִּתְנַכְּלוּ אֹתוֹ לַהֲמִיתוֹ׃ וַיֹּאמְרוּ אִישׁ אֶל־אָחִיו
18   הִנֵּה בַּעַל הַחֲלֹמוֹת הַלָּזֶה בָּא׃ וְעַתָּה  לְכוּ וְנַהַרְגֵהוּ
19   וְנַשְׁלִכֵהוּ בְּאַחַד הַבֹּרוֹת וְאָמַרְנוּ חַיָּה רָעָה אֲכָלָתְהוּ
20   וְנִרְאֶה מַה־יִּהְיוּ חֲלֹמֹתָיו׃ וַיִּשְׁמַע רְאוּבֵן וַיַּצִּלֵהוּ
21   מִיָּדָם וַיֹּאמֶר לֹא נַכֶּנּוּ נָפֶשׁ׃ וַיֹּאמֶר אֲלֵהֶם  רְאוּבֵן
22   אַל־תִּשְׁפְּכוּ־דָם הַשְׁלִיכוּ אֹתוֹ אֶל־הַבּוֹר הַזֶּה אֲשֶׁר
23   בַּמִּדְבָּר וְיָד אַל־תִּשְׁלְחוּ־בוֹ לְמַעַן הַצִּיל אֹתוֹ מִיָּדָם
24   לַהֲשִׁיבוֹ אֶל־אָבִיו׃  וַיְהִי כַּאֲשֶׁר־בָּא יוֹסֵף אֶל־אֶחָיו
25   וַיַּפְשִׁיטוּ אֶת־יוֹסֵף אֶת־כֻּתָּנְתּוֹ אֶת־כְּתֹנֶת הַפַּסִּים
26   אֲשֶׁר עָלָיו׃ וַיִּקָּחֻהוּ וַיַּשְׁלִכוּ אֹתוֹ הַבֹּרָה וְהַבּוֹר רֵק
27   אֵין בּוֹ מָיִם׃ וַיֵּשְׁבוּ לֶאֱכָל־לֶחֶם וַיִּשְׂאוּ עֵינֵיהֶם וַיִּרְאוּ
28   וְהִנֵּה אֹרְחַת יִשְׁמְעֵאלִים בָּאָה מִגִּלְעָד וּגְמַלֵּיהֶם
29   נֹשְׂאִים נְכֹאת וּצְרִי וָלֹט הוֹלְכִים לְהוֹרִיד מִצְרָיְמָה׃
30   וַיֹּאמֶר יְהוּדָה אֶל־אֶחָיו מַה־בֶּצַע כִּי נַהֲרֹג אֶת־אָחִינוּ
31   וְכִסִּינוּ אֶת־דָּמוֹ׃ לְכוּ וְנִמְכְּרֶנּוּ לַיִּשְׁמְעֵאלִים וְיָדֵנוּ אַל־
32  תְּהִי־בוֹ כִּי־אָחִינוּ בְשָׂרֵנוּ הוּא וַיִּשְׁמְעוּ אֶחָיו׃ וַיַּעַבְרוּ
33   אֲנָשִׁים מִדְיָנִים סֹחֲרִים וַיִּמְשְׁכוּ וַיַּעֲלוּ אֶת־יוֹסֵף מִן־
34  הַבּוֹר וַיִּמְכְּרוּ אֶת־יוֹסֵף לַיִּשְׁמְעֵאלִים בְּעֶשְׂרִים
35   כָּסֶף וַיָּבִיאוּ אֶת־יוֹסֵף מִצְרָיְמָה׃ וַיָּשָׁב רְאוּבֵן אֶל־
36  הַבּוֹר וְהִנֵּה אֵין־יוֹסֵף בַּבּוֹר וַיִּקְרַע אֶת־בְּגָדָיו׃ וַיָּשָׁב
37   אֶל־אֶחָיו וַיֹּאמַר הַיֶּלֶד אֵינֶנּוּ וַאֲנִי אָנָה אֲנִי־בָא׃ וַיִּקְחוּ
38   אֶת־כְּתֹנֶת יוֹסֵף וַיִּשְׁחֲטוּ שְׂעִיר עִזִּים וַיִּטְבְּלוּ אֶת־
39  הַכֻּתֹּנֶת בַּדָּם׃ וַיְשַׁלְּחוּ אֶת־כְּתֹנֶת הַפַּסִּים וַיָּבִיאוּ אֶל־
40  אֲבִיהֶם וַיֹּאמְרוּ זֹאת מָצָאנוּ הַכֶּר־נָא הַכְּתֹנֶת בִּנְךָ
41   הִוא אִם־לֹא׃ וַיַּכִּירָהּ וַיֹּאמֶר כְּתֹנֶת בְּנִי חַיָּה רָעָה
42   אֲכָלָתְהוּ טָרֹף טֹרַף יוֹסֵף׃ וַיִּקְרַע יַעֲקֹב שִׂמְלֹתָיו
