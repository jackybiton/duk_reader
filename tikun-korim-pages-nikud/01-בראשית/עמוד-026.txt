 1  וְאִם־לֹא הַגִּידוּ לִי וְאֶפְנֶה עַל־יָמִין אוֹ עַל־שְׂמֹאל׃
 2   וַיַּעַן לָבָן וּבְתוּאֵל וַיֹּאמְרוּ מֵיְהוָה יָצָא הַדָּבָר לֹא
 3   נוּכַל דַּבֵּר אֵלֶיךָ רַע אוֹ־טוֹב׃ הִנֵּה־רִבְקָה לְפָנֶיךָ קַח
 4   וָלֵךְ וּתְהִי אִשָּׁה לְבֶן־אֲדֹנֶיךָ כַּאֲשֶׁר דִּבֶּר יְהוָה׃ וַיְהִי
 5   כַּאֲשֶׁר שָׁמַע עֶבֶד אַבְרָהָם אֶת־דִּבְרֵיהֶם וַיִּשְׁתַּחוּ
 6   אַרְצָה לַיהוָה׃  וַיּוֹצֵא הָעֶבֶד כְּלֵי־כֶסֶף וּכְלֵי זָהָב
 7   וּבְגָדִים וַיִּתֵּן לְרִבְקָה וּמִגְדָּנֹת נָתַן לְאָחִיהָ וּלְאִמָּהּ׃
 8   וַיֹּאכְלוּ וַיִּשְׁתּוּ הוּא וְהָאֲנָשִׁים אֲשֶׁר־עִמּוֹ וַיָּלִינוּ
 9   וַיָּקוּמוּ בַבֹּקֶר וַיֹּאמֶר שַׁלְּחֻנִי לַאדֹנִי׃ וַיֹּאמֶר אָחִיהָ
10   וְאִמָּהּ תֵּשֵׁב הַנַּעֲרָ אִתָּנוּ יָמִים אוֹ עָשׂוֹר אַחַר תֵּלֵךְ׃
11   וַיֹּאמֶר אֲלֵהֶם אַל־תְּאַחֲרוּ אֹתִי וַיהוָה הִצְלִיחַ דַּרְכִּי
12   שַׁלְּחוּנִי וְאֵלְכָה לַאדֹנִי׃ וַיֹּאמְרוּ נִקְרָא לַנַּעֲרָ וְנִשְׁאֲלָה
13   אֶת־פִּיהָ׃ וַיִּקְרְאוּ לְרִבְקָה וַיֹּאמְרוּ אֵלֶיהָ הֲתֵלְכִי עִם־
14  הָאִישׁ הַזֶּה וַתֹּאמֶר אֵלֵךְ׃ וַיְשַׁלְּחוּ אֶת־רִבְקָה אֲחֹתָם
15   וְאֶת־מֵנִקְתָּהּ וְאֶת־עֶבֶד אַבְרָהָם וְאֶת־אֲנָשָׁיו׃ וַיְבָרְכוּ
16   אֶת־רִבְקָה וַיֹּאמְרוּ לָהּ אֲחֹתֵנוּ אַתְּ הֲיִי לְאַלְפֵי
17   רְבָבָה וְיִירַשׁ זַרְעֵךְ אֵת שַׁעַר שֹׂנְאָיו׃ וַתָּקָם רִבְקָה
18   וְנַעֲרֹתֶיהָ וַתִּרְכַּבְנָה עַל־הַגְּמַלִּים וַתֵּלַכְנָה אַחֲרֵי
19   הָאִישׁ וַיִּקַּח הָעֶבֶד אֶת־רִבְקָה וַיֵּלַךְ׃ וְיִצְחָק בָּא
20   מִבּוֹא בְּאֵר לַחַי רֹאִי וְהוּא יוֹשֵׁב בְּאֶרֶץ הַנֶּגֶב׃ וַיֵּצֵא
21   יִצְחָק לָשׂוּחַ בַּשָּׂדֶה לִפְנוֹת עָרֶב וַיִּשָּׂא עֵינָיו וַיַּרְא
22   וְהִנֵּה גְמַלִּים בָּאִים׃ וַתִּשָּׂא רִבְקָה אֶת־עֵינֶיהָ וַתֵּרֶא אֶת־
23  יִצְחָק וַתִּפֹּל מֵעַל הַגָּמָל׃ וַתֹּאמֶר אֶל־הָעֶבֶד מִי־הָאִישׁ
24   הַלָּזֶה הַהֹלֵךְ בַּשָּׂדֶה לִקְרָאתֵנוּ וַיֹּאמֶר הָעֶבֶד הוּא
25   אֲדֹנִי וַתִּקַּח הַצָּעִיף וַתִּתְכָּס׃ וַיְסַפֵּר הָעֶבֶד לְיִצְחָק
26   אֵת כָּל־הַדְּבָרִים אֲשֶׁר עָשָׂה׃ וַיְבִאֶהָ יִצְחָק הָאֹהֱלָה
27   שָׂרָה אִמּוֹ וַיִּקַּח אֶת־רִבְקָה וַתְּהִי־לוֹ לְאִשָּׁה וַיֶּאֱהָבֶהָ
28   וַיִּנָּחֵם יִצְחָק אַחֲרֵי אִמּוֹ׃ [פ]
29    וַיֹּסֶף אַבְרָהָם וַיִּקַּח אִשָּׁה וּשְׁמָהּ קְטוּרָה׃ וַתֵּלֶד לוֹ
30   אֶת־זִמְרָן וְאֶת־יָקְשָׁן וְאֶת־מְדָן וְאֶת־מִדְיָן וְאֶת־יִשְׁבָּק
31   וְאֶת־שׁוּחַ׃ וְיָקְשָׁן יָלַד אֶת־שְׁבָא וְאֶת־דְּדָן וּבְנֵי דְדָן
32   הָיוּ אַשּׁוּרִם וּלְטוּשִׁם וּלְאֻמִּים׃ וּבְנֵי מִדְיָן עֵיפָה וָעֵפֶר
33   וַחֲנֹךְ וַאֲבִידָע וְאֶלְדָּעָה כָּל־אֵלֶּה בְּנֵי קְטוּרָה׃ וַיִּתֵּן
34   אַבְרָהָם אֶת־כָּל־אֲשֶׁר־לוֹ לְיִצְחָק׃ וְלִבְנֵי הַפִּילַגְשִׁים
35   אֲשֶׁר לְאַבְרָהָם נָתַן אַבְרָהָם מַתָּנֹת וַיְשַׁלְּחֵם מֵעַל
36   יִצְחָק בְּנוֹ בְּעוֹדֶנּוּ חַי קֵדְמָה אֶל־אֶרֶץ קֶדֶם׃ וְאֵלֶּה יְמֵי
37   שְׁנֵי־חַיֵּי אַבְרָהָם אֲשֶׁר־חָי מְאַת שָׁנָה וְשִׁבְעִים שָׁנָה
38   וְחָמֵשׁ שָׁנִים׃ וַיִּגְוַע וַיָּמָת אַבְרָהָם בְּשֵׂיבָה טוֹבָה זָקֵן
39   וְשָׂבֵעַ וַיֵּאָסֶף אֶל־עַמָּיו׃ וַיִּקְבְּרוּ אֹתוֹ יִצְחָק וְיִשְׁמָעֵאל
40   בָּנָיו אֶל־מְעָרַת הַמַּכְפֵּלָה אֶל־שְׂדֵה עֶפְרֹן בֶּן־צֹחַר
41   הַחִתִּי אֲשֶׁר עַל־פְּנֵי מַמְרֵא׃ הַשָּׂדֶה אֲשֶׁר־קָנָה אַבְרָהָם
42   מֵאֵת בְּנֵי־חֵת שָׁמָּה קֻבַּר אַבְרָהָם וְשָׂרָה אִשְׁתּוֹ׃
