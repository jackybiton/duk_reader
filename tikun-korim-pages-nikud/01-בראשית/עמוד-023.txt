 1  וַיָּקֻמוּ וַיֵּלְכוּ יַחְדָּו אֶל־בְּאֵר שָׁבַע וַיֵּשֶׁב אַבְרָהָם
 2   בִּבְאֵר שָׁבַע׃ [פ]
 3    וַיְהִי אַחֲרֵי הַדְּבָרִים הָאֵלֶּה וַיֻּגַּד לְאַבְרָהָם לֵאמֹר
 4   הִנֵּה יָלְדָה מִלְכָּה גַם־הִוא בָּנִים לְנָחוֹר אָחִיךָ׃ אֶת־
 5  עוּץ בְּכֹרוֹ וְאֶת־בּוּז אָחִיו וְאֶת־קְמוּאֵל אֲבִי אֲרָם׃
 6   וְאֶת־כֶּשֶׂד וְאֶת־חֲזוֹ וְאֶת־פִּלְדָּשׁ וְאֶת־יִדְלָף וְאֵת
 7   בְּתוּאֵל׃ וּבְתוּאֵל יָלַד אֶת־רִבְקָה שְׁמֹנָה אֵלֶּה יָלְדָה
 8   מִלְכָּה לְנָחוֹר אֲחִי אַבְרָהָם׃ וּפִילַגְשׁוֹ וּשְׁמָהּ
 9   רְאוּמָה וַתֵּלֶד גַּם־הִוא אֶת־טֶבַח וְאֶת־גַּחַם וְאֶת־
10  תַּחַשׁ וְאֶת־מַעֲכָה׃ [פ]
11  וַיִּהְיוּ חַיֵּי שָׂרָה מֵאָה שָׁנָה וְעֶשְׂרִים שָׁנָה וְשֶׁבַע
12   שָׁנִים שְׁנֵי חַיֵּי שָׂרָה׃ וַתָּמָת שָׂרָה בְּקִרְיַת אַרְבַּע
13   הִוא חֶבְרוֹן בְּאֶרֶץ כְּנָעַן וַיָּבֹא אַבְרָהָם לִסְפֹּד לְשָׂרָה
14   וְלִבְכֹּתָהּ׃ וַיָּקָם אַבְרָהָם מֵעַל פְּנֵי מֵתוֹ וַיְדַבֵּר אֶל־בְּנֵי־
15  חֵת לֵאמֹר׃ גֵּר־וְתוֹשָׁב אָנֹכִי עִמָּכֶם תְּנוּ לִי אֲחֻזַּת־
16  קֶבֶר עִמָּכֶם וְאֶקְבְּרָה מֵתִי מִלְּפָנָי׃ וַיַּעֲנוּ בְנֵי־חֵת אֶת־
17  אַבְרָהָם לֵאמֹר לוֹ׃ שְׁמָעֵנוּ  אֲדֹנִי נְשִׂיא אֱלֹהִים אַתָּה
18   בְּתוֹכֵנוּ בְּמִבְחַר קְבָרֵינוּ קְבֹר אֶת־מֵתֶךָ אִישׁ מִמֶּנּוּ
19   אֶת־קִבְרוֹ לֹא־יִכְלֶה מִמְּךָ מִקְּבֹר מֵתֶךָ׃ וַיָּקָם אַבְרָהָם
20   וַיִּשְׁתַּחוּ לְעַם־הָאָרֶץ לִבְנֵי־חֵת׃♢ וַיְדַבֵּר אִתָּם לֵאמֹר
21   אִם־יֵשׁ אֶת־נַפְשְׁכֶם לִקְבֹּר אֶת־מֵתִי מִלְּפָנַי שְׁמָעוּנִי
22   וּפִגְעוּ־לִי בְּעֶפְרוֹן בֶּן־צֹחַר׃ וְיִתֶּן־לִי אֶת־מְעָרַת
23   הַמַּכְפֵּלָה אֲשֶׁר־לוֹ אֲשֶׁר בִּקְצֵה שָׂדֵהוּ בְּכֶסֶף
24   מָלֵא יִתְּנֶנָּה לִּי בְּתוֹכְכֶם לַאֲחֻזַּת־קָבֶר׃ וְעֶפְרוֹן יֹשֵׁב
25   בְּתוֹךְ בְּנֵי־חֵת וַיַּעַן עֶפְרוֹן הַחִתִּי אֶת־אַבְרָהָם בְּאָזְנֵי
26   בְנֵי־חֵת לְכֹל בָּאֵי שַׁעַר־עִירוֹ לֵאמֹר׃ לֹא־אֲדֹנִי
27   שְׁמָעֵנִי הַשָּׂדֶה נָתַתִּי לָךְ וְהַמְּעָרָה אֲשֶׁר־בּוֹ לְךָ
28   נְתַתִּיהָ לְעֵינֵי בְנֵי־עַמִּי נְתַתִּיהָ לָּךְ קְבֹר מֵתֶךָ׃♢ וַיִּשְׁתַּחוּ
29   אַבְרָהָם לִפְנֵי עַם הָאָרֶץ׃ וַיְדַבֵּר אֶל־עֶפְרוֹן בְּאָזְנֵי
30   עַם־הָאָרֶץ לֵאמֹר אַךְ אִם־אַתָּה לוּ שְׁמָעֵנִי נָתַתִּי
31   כֶּסֶף הַשָּׂדֶה קַח מִמֶּנִּי וְאֶקְבְּרָה אֶת־מֵתִי שָׁמָּה׃
32   וַיַּעַן עֶפְרוֹן אֶת־אַבְרָהָם לֵאמֹר לוֹ׃ אֲדֹנִי שְׁמָעֵנִי
33   אֶרֶץ אַרְבַּע מֵאֹת שֶׁקֶל־כֶּסֶף בֵּינִי וּבֵינְךָ מַה־הִוא
34   וְאֶת־מֵתְךָ קְבֹר׃ וַיִּשְׁמַע אַבְרָהָם אֶל־עֶפְרוֹן וַיִּשְׁקֹל
35   אַבְרָהָם לְעֶפְרֹן אֶת־הַכֶּסֶף אֲשֶׁר דִּבֶּר בְּאָזְנֵי בְנֵי־
36  חֵת אַרְבַּע מֵאוֹת שֶׁקֶל כֶּסֶף עֹבֵר לַסֹּחֵר׃♢  וַיָּקָם
37   שְׂדֵה עֶפְרוֹן אֲשֶׁר בַּמַּכְפֵּלָה אֲשֶׁר לִפְנֵי מַמְרֵא
38   הַשָּׂדֶה וְהַמְּעָרָה אֲשֶׁר־בּוֹ וְכָל־הָעֵץ אֲשֶׁר בַּשָּׂדֶה
39   אֲשֶׁר בְּכָל־גְּבֻלוֹ סָבִיב׃ לְאַבְרָהָם לְמִקְנָה לְעֵינֵי בְנֵי־
40  חֵת בְּכֹל בָּאֵי שַׁעַר־עִירוֹ׃ וְאַחֲרֵי־כֵן קָבַר אַבְרָהָם
41   אֶת־שָׂרָה אִשְׁתּוֹ אֶל־מְעָרַת שְׂדֵה הַמַּכְפֵּלָה עַל־
42  פְּנֵי מַמְרֵא הִוא חֶבְרוֹן בְּאֶרֶץ כְּנָעַן׃ וַיָּקָם הַשָּׂדֶה
