 1  וַיֹּאמֶר לֹא יַעֲקֹב יֵאָמֵר עוֹד שִׁמְךָ כִּי אִם־יִשְׂרָאֵל
 2   כִּי־שָׂרִיתָ עִם־אֱלֹהִים וְעִם־אֲנָשִׁים וַתּוּכָל׃ וַיִּשְׁאַל
 3   יַעֲקֹב וַיֹּאמֶר הַגִּידָה־נָּא שְׁמֶךָ וַיֹּאמֶר לָמָּה זֶּה תִּשְׁאַל
 4   לִשְׁמִי וַיְבָרֶךְ אֹתוֹ שָׁם׃  וַיִּקְרָא יַעֲקֹב שֵׁם הַמָּקוֹם
 5   פְּנִיאֵל כִּי־רָאִיתִי אֱלֹהִים פָּנִים אֶל־פָּנִים וַתִּנָּצֵל נַפְשִׁי׃
 6   וַיִּזְרַח־לוֹ הַשֶּׁמֶשׁ כַּאֲשֶׁר עָבַר אֶת־פְּנוּאֵל וְהוּא
 7   צֹלֵעַ עַל־יְרֵכוֹ׃ עַל־כֵּן לֹא־יֹאכְלוּ בְנֵי־יִשְׂרָאֵל אֶת־
 8  גִּיד הַנָּשֶׁה אֲשֶׁר עַל־כַּף הַיָּרֵךְ עַד הַיּוֹם הַזֶּה כִּי
 9   נָגַע בְּכַף־יֶרֶךְ יַעֲקֹב בְּגִיד הַנָּשֶׁה׃ וַיִּשָּׂא יַעֲקֹב עֵינָיו
10   וַיַּרְא וְהִנֵּה עֵשָׂו בָּא וְעִמּוֹ אַרְבַּע מֵאוֹת אִישׁ וַיַּחַץ
11   אֶת־הַיְלָדִים עַל־לֵאָה וְעַל־רָחֵל וְעַל שְׁתֵּי הַשְּׁפָחוֹת׃
12   וַיָּשֶׂם אֶת־הַשְּׁפָחוֹת וְאֶת־יַלְדֵיהֶן רִאשֹׁנָה וְאֶת־
13  לֵאָה וִילָדֶיהָ אַחֲרֹנִים וְאֶת־רָחֵל וְאֶת־יוֹסֵף אַחֲרֹנִים׃
14   וְהוּא עָבַר לִפְנֵיהֶם וַיִּשְׁתַּחוּ אַרְצָה שֶׁבַע פְּעָמִים
15   עַד־גִּשְׁתּוֹ עַד־אָחִיו׃ וַיָּרָץ עֵשָׂו לִקְרָאתוֹ וַיְחַבְּקֵהוּ
16   וַיִּפֹּל עַל־צַוָּארָו וַיִּשָּׁקֵהוּ וַיִּבְכּוּ׃ וַיִּשָּׂא אֶת־עֵינָיו
17   וַיַּרְא אֶת־הַנָּשִׁים וְאֶת־הַיְלָדִים וַיֹּאמֶר מִי־אֵלֶּה לָּךְ
18   וַיֹּאמַר הַיְלָדִים אֲשֶׁר־חָנַן אֱלֹהִים אֶת־עַבְדֶּךָ׃  וַתִּגַּשְׁןָ
19   הַשְּׁפָחוֹת הֵנָּה וְיַלְדֵיהֶן וַתִּשְׁתַּחֲוֶיןָ׃ וַתִּגַּשׁ גַּם־לֵאָה
20   וִילָדֶיהָ וַיִּשְׁתַּחֲווּ וְאַחַר נִגַּשׁ יוֹסֵף וְרָחֵל וַיִּשְׁתַּחֲווּ׃
21   וַיֹּאמֶר מִי לְךָ כָּל־הַמַּחֲנֶה הַזֶּה אֲשֶׁר פָּגָשְׁתִּי וַיֹּאמֶר
22   לִמְצֹא־חֵן בְּעֵינֵי אֲדֹנִי׃ וַיֹּאמֶר עֵשָׂו יֶשׁ־לִי רָב אָחִי
23   יְהִי לְךָ אֲשֶׁר־לָךְ׃ וַיֹּאמֶר יַעֲקֹב אַל־נָא אִם־נָא מָצָאתִי
24   חֵן בְּעֵינֶיךָ וְלָקַחְתָּ מִנְחָתִי מִיָּדִי כִּי עַל־כֵּן רָאִיתִי
25   פָנֶיךָ כִּרְאֹת פְּנֵי אֱלֹהִים וַתִּרְצֵנִי׃ קַח־נָא אֶת־בִּרְכָתִי
26   אֲשֶׁר הֻבָאת לָךְ כִּי־חַנַּנִי אֱלֹהִים וְכִי יֶשׁ־לִי־כֹל
27   וַיִּפְצַר־בּוֹ וַיִּקָּח׃ וַיֹּאמֶר נִסְעָה וְנֵלֵכָה וְאֵלְכָה לְנֶגְדֶּךָ׃
28   וַיֹּאמֶר אֵלָיו אֲדֹנִי יֹדֵעַ כִּי־הַיְלָדִים רַכִּים וְהַצֹּאן
29   וְהַבָּקָר עָלוֹת עָלָי וּדְפָקוּם יוֹם אֶחָד וָמֵתוּ כָּל־הַצֹּאן׃
30   יַעֲבָר־נָא אֲדֹנִי לִפְנֵי עַבְדּוֹ וַאֲנִי אֶתְנָהֲלָה לְאִטִּי לְרֶגֶל
31   הַמְּלָאכָה אֲשֶׁר־לְפָנַי וּלְרֶגֶל הַיְלָדִים עַד אֲשֶׁר־אָבֹא
32   אֶל־אֲדֹנִי שֵׂעִירָה׃ וַיֹּאמֶר עֵשָׂו אַצִּיגָה־נָּא עִמְּךָ מִן־
33  הָעָם אֲשֶׁר אִתִּי וַיֹּאמֶר לָמָּה זֶּה אֶמְצָא־חֵן בְּעֵינֵי
34   אֲדֹנִי׃ וַיָּשָׁב בַּיּוֹם הַהוּא עֵשָׂו לְדַרְכּוֹ שֵׂעִירָה׃ וְיַעֲקֹב
35   נָסַע סֻכֹּתָה וַיִּבֶן לוֹ בָּיִת וּלְמִקְנֵהוּ עָשָׂה סֻכֹּת עַל־כֵּן
36   קָרָא שֵׁם־הַמָּקוֹם סֻכּוֹת׃ [ס]             וַיָּבֹא
37   יַעֲקֹב שָׁלֵם עִיר שְׁכֶם אֲשֶׁר בְּאֶרֶץ כְּנַעַן בְּבֹאוֹ
38   מִפַּדַּן אֲרָם וַיִּחַן אֶת־פְּנֵי הָעִיר׃ וַיִּקֶן אֶת־חֶלְקַת
39   הַשָּׂדֶה אֲשֶׁר נָטָה־שָׁם אָהֳלוֹ מִיַּד בְּנֵי־חֲמוֹר אֲבִי
40   שְׁכֶם בְּמֵאָה קְשִׂיטָה׃ וַיַּצֶּב־שָׁם מִזְבֵּחַ וַיִּקְרָא־לוֹ
41   אֵל אֱלֹהֵי יִשְׂרָאֵל׃ [ס]              וַתֵּצֵא דִינָה
42   בַּת־לֵאָה אֲשֶׁר יָלְדָה לְיַעֲקֹב לִרְאוֹת בִּבְנוֹת הָאָרֶץ׃
