 1  יְהוּדָה אַתָּה יוֹדוּךָ אַחֶיךָ יָדְךָ בְּעֹרֶף אֹיְבֶיךָ יִשְׁתַּחֲווּ
 2   לְךָ בְּנֵי אָבִיךָ׃ גּוּר אַרְיֵה יְהוּדָה מִטֶּרֶף בְּנִי עָלִיתָ
 3   כָּרַע רָבַץ כְּאַרְיֵה וּכְלָבִיא מִי יְקִימֶנּוּ׃ לֹא־יָסוּר שֵׁבֶט
 4   מִיהוּדָה וּמְחֹקֵק מִבֵּין רַגְלָיו עַד כִּי־יָבֹא שִׁילֹה וְלוֹ
 5   יִקְּהַת עַמִּים׃ אֹסְרִי לַגֶּפֶן עִירֹה וְלַשֹּׂרֵקָה בְּנִי אֲתֹנוֹ
 6   כִּבֵּס בַּיַּיִן לְבֻשׁוֹ וּבְדַם־עֲנָבִים סוּתֹה׃ חַכְלִילִי עֵינַיִם
 7   מִיָּיִן וּלְבֶן־שִׁנַּיִם מֵחָלָב׃ [פ]
 8   זְבוּלֻן לְחוֹף יַמִּים יִשְׁכֹּן וְהוּא לְחוֹף אֳנִיֹּת וְיַרְכָתוֹ
 9   עַל־צִידֹן׃ [פ]
10   יִשָּׂשכָר חֲמֹר גָּרֶם רֹבֵץ בֵּין הַמִּשְׁפְּתָיִם׃ וַיַּרְא מְנֻחָה
11   כִּי טוֹב וְאֶת־הָאָרֶץ כִּי נָעֵמָה וַיֵּט שִׁכְמוֹ לִסְבֹּל וַיְהִי
12   לְמַס־עֹבֵד׃ [ס]             דָּן יָדִין עַמּוֹ כְּאַחַד
13   שִׁבְטֵי יִשְׂרָאֵל׃ יְהִי־דָן נָחָשׁ עֲלֵי־דֶרֶךְ שְׁפִיפֹן עֲלֵי־
14  אֹרַח הַנֹּשֵׁךְ עִקְּבֵי־סוּס וַיִּפֹּל רֹכְבוֹ אָחוֹר׃ לִישׁוּעָתְךָ
15   קִוִּיתִי יְהוָה׃ [ס]              גָּד גְּדוּד יְגוּדֶנּוּ וְהוּא
16   יָגֻד עָקֵב׃ [ס]             מֵאָשֵׁר שְׁמֵנָה לַחְמוֹ
17   וְהוּא יִתֵּן מַעֲדַנֵּי־מֶלֶךְ׃ [ס]             נַפְתָּלִי
18   אַיָּלָה שְׁלֻחָה הַנֹּתֵן אִמְרֵי־שָׁפֶר׃ [ס]             בֵּן
19   פֹּרָת יוֹסֵף בֵּן פֹּרָת עֲלֵי־עָיִן בָּנוֹת צָעֲדָה עֲלֵי־שׁוּר׃
20   וַיְמָרְרֻהוּ וָרֹבּוּ וַיִּשְׂטְמֻהוּ בַּעֲלֵי חִצִּים׃ וַתֵּשֶׁב בְּאֵיתָן
21   קַשְׁתּוֹ וַיָּפֹזּוּ זְרֹעֵי יָדָיו מִידֵי אֲבִיר יַעֲקֹב מִשָּׁם
22   רֹעֶה אֶבֶן יִשְׂרָאֵל׃ מֵאֵל אָבִיךָ וְיַעְזְרֶךָּ וְאֵת שַׁדַּי
23   וִיבָרְכֶךָּ בִּרְכֹת שָׁמַיִם מֵעָל בִּרְכֹת תְּהוֹם רֹבֶצֶת
24   תָּחַת בִּרְכֹת שָׁדַיִם וָרָחַם׃ בִּרְכֹת אָבִיךָ גָּבְרוּ עַל־
25  בִּרְכֹת הוֹרַי עַד־תַּאֲוַת גִּבְעֹת עוֹלָם תִּהְיֶיןָ לְרֹאשׁ
26   יוֹסֵף וּלְקָדְקֹד נְזִיר אֶחָיו׃ [פ]
27    בִּנְיָמִין זְאֵב יִטְרָף בַּבֹּקֶר יֹאכַל עַד וְלָעֶרֶב יְחַלֵּק
28   שָׁלָל׃ כָּל־אֵלֶּה שִׁבְטֵי יִשְׂרָאֵל שְׁנֵים עָשָׂר וְזֹאת
29   אֲשֶׁר־דִּבֶּר לָהֶם אֲבִיהֶם וַיְבָרֶךְ אוֹתָם אִישׁ
30   אֲשֶׁר כְּבִרְכָתוֹ בֵּרַךְ אֹתָם׃ וַיְצַו אוֹתָם וַיֹּאמֶר
31   אֲלֵהֶם אֲנִי נֶאֱסָף אֶל־עַמִּי קִבְרוּ אֹתִי אֶל־אֲבֹתָי
32   אֶל־הַמְּעָרָה אֲשֶׁר בִּשְׂדֵה עֶפְרוֹן הַחִתִּי׃ בַּמְּעָרָה
33   אֲשֶׁר בִּשְׂדֵה הַמַּכְפֵּלָה אֲשֶׁר עַל־פְּנֵי־מַמְרֵא
34   בְּאֶרֶץ כְּנָעַן אֲשֶׁר קָנָה אַבְרָהָם אֶת־הַשָּׂדֶה מֵאֵת
35   עֶפְרֹן הַחִתִּי לַאֲחֻזַּת־קָבֶר׃ שָׁמָּה קָבְרוּ אֶת־אַבְרָהָם
36   וְאֵת שָׂרָה אִשְׁתּוֹ שָׁמָּה קָבְרוּ אֶת־יִצְחָק וְאֵת
37   רִבְקָה אִשְׁתּוֹ וְשָׁמָּה קָבַרְתִּי אֶת־לֵאָה׃ מִקְנֵה
38   הַשָּׂדֶה וְהַמְּעָרָה אֲשֶׁר־בּוֹ מֵאֵת בְּנֵי־חֵת׃ וַיְכַל
39   יַעֲקֹב לְצַוּ͏ֹת אֶת־בָּנָיו וַיֶּאֱסֹף רַגְלָיו אֶל־הַמִּטָּה וַיִּגְוַע
40   וַיֵּאָסֶף אֶל־עַמָּיו׃ וַיִּפֹּל יוֹסֵף עַל־פְּנֵי אָבִיו וַיֵּבְךְּ
41   עָלָיו וַיִּשַּׁק־לוֹ׃ וַיְצַו יוֹסֵף אֶת־עֲבָדָיו אֶת־הָרֹפְאִים
42   לַחֲנֹט אֶת־אָבִיו וַיַּחַנְטוּ הָרֹפְאִים אֶת־יִשְׂרָאֵל׃
