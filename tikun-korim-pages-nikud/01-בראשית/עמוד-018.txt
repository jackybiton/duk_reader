 1  וַיֹּאמַר הַאַף תִּסְפֶּה צַדִּיק עִם־רָשָׁע׃ אוּלַי יֵשׁ
 2   חֲמִשִּׁים צַדִּיקִם בְּתוֹךְ הָעִיר הַאַף תִּסְפֶּה וְלֹא־תִשָּׂא
 3   לַמָּקוֹם לְמַעַן חֲמִשִּׁים הַצַּדִּיקִם אֲשֶׁר בְּקִרְבָּהּ׃
 4   חָלִלָה לְּךָ מֵעֲשֹׂת  כַּדָּבָר הַזֶּה לְהָמִית צַדִּיק עִם־
 5  רָשָׁע וְהָיָה כַצַּדִּיק כָּרָשָׁע חָלִלָה לָּךְ הֲשֹׁפֵט כָּל־
 6  הָאָרֶץ לֹא יַעֲשֶׂה מִשְׁפָּט׃ וַיֹּאמֶר יְהוָה אִם־אֶמְצָא
 7   בִסְדֹם חֲמִשִּׁים צַדִּיקִם בְּתוֹךְ הָעִיר וְנָשָׂאתִי לְכָל־
 8  הַמָּקוֹם בַּעֲבוּרָם׃ וַיַּעַן אַבְרָהָם וַיֹּאמַר הִנֵּה־נָא
 9   הוֹאַלְתִּי לְדַבֵּר אֶל־אֲדֹנָי וְאָנֹכִי עָפָר וָאֵפֶר׃ אוּלַי
10   יַחְסְרוּן חֲמִשִּׁים הַצַּדִּיקִם חֲמִשָּׁה הֲתַשְׁחִית בַּחֲמִשָּׁה
11   אֶת־כָּל־הָעִיר וַיֹּאמֶר לֹא אַשְׁחִית אִם־אֶמְצָא שָׁם
12   אַרְבָּעִים וַחֲמִשָּׁה׃ וַיֹּסֶף עוֹד לְדַבֵּר אֵלָיו וַיֹּאמַר
13   אוּלַי יִמָּצְאוּן שָׁם אַרְבָּעִים וַיֹּאמֶר לֹא אֶעֱשֶׂה בַּעֲבוּר
14   הָאַרְבָּעִים׃ וַיֹּאמֶר אַל־נָא יִחַר לַאדֹנָי וַאֲדַבֵּרָה אוּלַי
15   יִמָּצְאוּן שָׁם שְׁלֹשִׁים וַיֹּאמֶר לֹא אֶעֱשֶׂה אִם־אֶמְצָא
16   שָׁם שְׁלֹשִׁים׃ וַיֹּאמֶר הִנֵּה־נָא הוֹאַלְתִּי לְדַבֵּר אֶל־
17  אֲדֹנָי אוּלַי יִמָּצְאוּן שָׁם עֶשְׂרִים וַיֹּאמֶר לֹא אַשְׁחִית
18   בַּעֲבוּר הָעֶשְׂרִים׃ וַיֹּאמֶר אַל־נָא יִחַר לַאדֹנָי וַאֲדַבְּרָה
19   אַךְ־הַפַּעַם אוּלַי יִמָּצְאוּן שָׁם עֲשָׂרָה וַיֹּאמֶר לֹא
20   אַשְׁחִית בַּעֲבוּר הָעֲשָׂרָה׃ וַיֵּלֶךְ יְהוָה כַּאֲשֶׁר כִּלָּה
21   לְדַבֵּר אֶל־אַבְרָהָם וְאַבְרָהָם שָׁב לִמְקֹמוֹ׃  וַיָּבֹאוּ
22   שְׁנֵי הַמַּלְאָכִים סְדֹמָה בָּעֶרֶב וְלוֹט יֹשֵׁב בְּשַׁעַר־
23  סְדֹם וַיַּרְא־לוֹט וַיָּקָם לִקְרָאתָם וַיִּשְׁתַּחוּ אַפַּיִם אָרְצָה׃
24   וַיֹּאמֶר הִנֶּה נָּא־אֲדֹנַי סוּרוּ נָא אֶל־בֵּית עַבְדְּכֶם וְלִינוּ
25   וְרַחֲצוּ רַגְלֵיכֶם וְהִשְׁכַּמְתֶּם וַהֲלַכְתֶּם לְדַרְכְּכֶם
26   וַיֹּאמְרוּ לֹּא כִּי בָרְחוֹב נָלִין׃ וַיִּפְצַר־בָּם מְאֹד וַיָּסֻרוּ
27   אֵלָיו וַיָּבֹאוּ אֶל־בֵּיתוֹ וַיַּעַשׂ לָהֶם מִשְׁתֶּה וּמַצּוֹת אָפָה
28   וַיֹּאכֵלוּ׃ טֶרֶם יִשְׁכָּבוּ וְאַנְשֵׁי הָעִיר אַנְשֵׁי סְדֹם נָסַבּוּ
29   עַל־הַבַּיִת מִנַּעַר וְעַד־זָקֵן כָּל־הָעָם מִקָּצֶה׃ וַיִּקְרְאוּ
30   אֶל־לוֹט וַיֹּאמְרוּ לוֹ אַיֵּה הָאֲנָשִׁים אֲשֶׁר־בָּאוּ אֵלֶיךָ
31   הַלָּיְלָה הוֹצִיאֵם אֵלֵינוּ וְנֵדְעָה אֹתָם׃ וַיֵּצֵא אֲלֵהֶם לוֹט
32   הַפֶּתְחָה וְהַדֶּלֶת סָגַר אַחֲרָיו׃ וַיֹּאמַר אַל־נָא אַחַי תָּרֵעוּ׃
33   הִנֵּה־נָא לִי שְׁתֵּי בָנוֹת אֲשֶׁר לֹא־יָדְעוּ אִישׁ אוֹצִיאָה־
34  נָּא אֶתְהֶן אֲלֵיכֶם וַעֲשׂוּ לָהֶן כַּטּוֹב בְּעֵינֵיכֶם רַק
35   לָאֲנָשִׁים הָאֵל אַל־תַּעֲשׂוּ דָבָר כִּי־עַל־כֵּן בָּאוּ בְּצֵל
36   קֹרָתִי׃ וַיֹּאמְרוּ  גֶּשׁ־הָלְאָה וַיֹּאמְרוּ הָאֶחָד בָּא־לָגוּר
37   וַיִּשְׁפֹּט שָׁפוֹט עַתָּה נָרַע לְךָ מֵהֶם וַיִּפְצְרוּ בָאִישׁ
38   בְּלוֹט מְאֹד וַיִּגְּשׁוּ לִשְׁבֹּר הַדָּלֶת׃ וַיִּשְׁלְחוּ הָאֲנָשִׁים
39   אֶת־יָדָם וַיָּבִיאוּ אֶת־לוֹט אֲלֵיהֶם הַבָּיְתָה וְאֶת־הַדֶּלֶת
40   סָגָרוּ׃ וְאֶת־הָאֲנָשִׁים אֲשֶׁר־פֶּתַח הַבַּיִת הִכּוּ בַּסַּנְוֵרִים
41   מִקָּטֹן וְעַד־גָּדוֹל וַיִּלְאוּ לִמְצֹא הַפָּתַח׃ וַיֹּאמְרוּ
42   הָאֲנָשִׁים אֶל־לוֹט עֹד מִי־לְךָ פֹה חָתָן וּבָנֶיךָ וּבְנֹתֶיךָ
