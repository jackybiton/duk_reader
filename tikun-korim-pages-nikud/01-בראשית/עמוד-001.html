1בְּרֵאשִׁיתבָּרָאאֱלֹהִיםאֵתהַשָּׁמַיִםוְאֵתהָאָרֶץ׃
2וְהָאָרֶץהָיְתָהתֹהוּוָבֹהוּוְחֹשֶׁךְעַל־פְּנֵיתְהוֹםוְרוּחַ
3אֱלֹהִיםמְרַחֶפֶתעַל־פְּנֵיהַמָּיִם׃וַיֹּאמֶראֱלֹהִיםיְהִי
4אוֹרוַיְהִי־אוֹר׃וַיַּרְאאֱלֹהִיםאֶת־הָאוֹרכִּי־טוֹב
5וַיַּבְדֵּלאֱלֹהִיםבֵּיןהָאוֹרוּבֵיןהַחֹשֶׁךְ׃וַיִּקְרָא
6אֱלֹהִיםלָאוֹריוֹםוְלַחֹשֶׁךְקָרָאלָיְלָהוַיְהִי־

8וַיֹּאמֶראֱלֹהִיםיְהִירָקִיעַבְּתוֹךְהַמָּיִםוִיהִימַבְדִּיל
9בֵּיןמַיִםלָמָיִם׃וַיַּעַשׂאֱלֹהִיםאֶת־הָרָקִיעַוַיַּבְדֵּל
10בֵּיןהַמַּיִםאֲשֶׁרמִתַּחַתלָרָקִיעַוּבֵיןהַמַּיִםאֲשֶׁר
11מֵעַללָרָקִיעַוַיְהִי־כֵן׃וַיִּקְרָאאֱלֹהִיםלָרָקִיעַשָׁמָיִם
12וַיְהִי־עֶרֶבוַיְהִי־בֹקֶריוֹםשֵׁנִי׃♢
13וַיֹּאמֶראֱלֹהִיםיִקָּווּהַמַּיִםמִתַּחַתהַשָּׁמַיִםאֶל־מָקוֹם
14אֶחָדוְתֵרָאֶההַיַּבָּשָׁהוַיְהִי־כֵן׃וַיִּקְרָאאֱלֹהִים
15לַיַּבָּשָׁהאֶרֶץוּלְמִקְוֵההַמַּיִםקָרָאיַמִּיםוַיַּרְא
16אֱלֹהִיםכִּי־טוֹב׃וַיֹּאמֶראֱלֹהִיםתַּדְשֵׁאהָאָרֶץ
17דֶּשֶׁאעֵשֶׂבמַזְרִיעַזֶרַעעֵץפְּרִיעֹשֶׂהפְּרִי
18לְמִינוֹאֲשֶׁרזַרְעוֹ־בוֹעַל־הָאָרֶץוַיְהִי־כֵן׃וַתּוֹצֵאהָאָרֶץ
19דֶּשֶׁאעֵשֶׂבמַזְרִיעַזֶרַעלְמִינֵהוּוְעֵץעֹשֶׂה־פְּרִי
20אֲשֶׁרזַרְעוֹ־בוֹלְמִינֵהוּוַיַּרְאאֱלֹהִיםכִּי־טוֹב׃וַיְהִי־
21עֶרֶבוַיְהִי־בֹקֶריוֹםשְׁלִישִׁי׃♢
22וַיֹּאמֶראֱלֹהִיםיְהִימְאֹרֹתבִּרְקִיעַהַשָּׁמַיִםלְהַבְדִּיל
23בֵּיןהַיּוֹםוּבֵיןהַלָּיְלָהוְהָיוּלְאֹתֹתוּלְמוֹעֲדִים
24וּלְיָמִיםוְשָׁנִים׃וְהָיוּלִמְאוֹרֹתבִּרְקִיעַהַשָּׁמַיִם
25לְהָאִירעַל־הָאָרֶץוַיְהִי־כֵן׃וַיַּעַשׂאֱלֹהִיםאֶת־שְׁנֵי
26הַמְּאֹרֹתהַגְּדֹלִיםאֶת־הַמָּאוֹרהַגָּדֹללְמֶמְשֶׁלֶת
27הַיּוֹםוְאֶת־הַמָּאוֹרהַקָּטֹןלְמֶמְשֶׁלֶתהַלַּיְלָהוְאֵת
28הַכּוֹכָבִים׃וַיִּתֵּןאֹתָםאֱלֹהִיםבִּרְקִיעַהַשָּׁמָיִם
29לְהָאִירעַל־הָאָרֶץ׃וְלִמְשֹׁלבַּיּוֹםוּבַלַּיְלָהוּלְהַבְדִּיל
30בֵּיןהָאוֹרוּבֵיןהַחֹשֶׁךְוַיַּרְאאֱלֹהִיםכִּי־טוֹב׃וַיְהִי־
31עֶרֶבוַיְהִי־בֹקֶריוֹםרְבִיעִי׃
32וַיֹּאמֶראֱלֹהִיםיִשְׁרְצוּהַמַּיִםשֶׁרֶץנֶפֶשׁחַיָּה
33וְעוֹףיְעוֹפֵףעַל־הָאָרֶץעַל־פְּנֵירְקִיעַהַשָּׁמָיִם׃
34וַיִּבְרָאאֱלֹהִיםאֶת־הַתַּנִּינִםהַגְּדֹלִיםוְאֵתכָּל־
35נֶפֶשׁהַחַיָּההָרֹמֶשֶׂתאֲשֶׁרשָׁרְצוּהַמַּיִםלְמִינֵהֶם
36וְאֵתכָּל־עוֹףכָּנָףלְמִינֵהוּוַיַּרְאאֱלֹהִיםכִּי־טוֹב׃
37וַיְבָרֶךְאֹתָםאֱלֹהִיםלֵאמֹרפְּרוּוּרְבוּוּמִלְאוּאֶת־
38הַמַּיִםבַּיַּמִּיםוְהָעוֹףיִרֶבבָּאָרֶץ׃וַיְהִי־עֶרֶבוַיְהִי־
39בֹקֶריוֹםחֲמִישִׁי׃
40וַיֹּאמֶראֱלֹהִיםתּוֹצֵאהָאָרֶץנֶפֶשׁחַיָּהלְמִינָהּ
