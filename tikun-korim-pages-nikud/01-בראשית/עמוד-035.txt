 1  וְאַתֵּנָה יְדַעְתֶּן כִּי בְּכָל־כֹּחִי עָבַדְתִּי אֶת־אֲבִיכֶן׃
 2   וַאֲבִיכֶן הֵתֶל בִּי וְהֶחֱלִף אֶת־מַשְׂכֻּרְתִּי עֲשֶׂרֶת מֹנִים
 3   וְלֹא־נְתָנוֹ אֱלֹהִים לְהָרַע עִמָּדִי׃ אִם־כֹּה יֹאמַר נְקֻדִּים
 4   יִהְיֶה שְׂכָרֶךָ וְיָלְדוּ כָל־הַצֹּאן נְקֻדִּים וְאִם־כֹּה יֹאמַר
 5   עֲקֻדִּים יִהְיֶה שְׂכָרֶךָ וְיָלְדוּ כָל־הַצֹּאן עֲקֻדִּים׃ וַיַּצֵּל
 6   אֱלֹהִים אֶת־מִקְנֵה אֲבִיכֶם וַיִּתֶּן־לִי׃ וַיְהִי בְּעֵת יַחֵם
 7   הַצֹּאן וָאֶשָּׂא עֵינַי וָאֵרֶא בַּחֲלוֹם וְהִנֵּה הָעַתֻּדִים
 8   הָעֹלִים עַל־הַצֹּאן עֲקֻדִּים נְקֻדִּים וּבְרֻדִּים׃ וַיֹּאמֶר אֵלַי
 9   מַלְאַךְ הָאֱלֹהִים בַּחֲלוֹם יַעֲקֹב וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר
10   שָׂא־נָא עֵינֶיךָ וּרְאֵה כָּל־הָעַתֻּדִים הָעֹלִים עַל־הַצֹּאן
11   עֲקֻדִּים נְקֻדִּים וּבְרֻדִּים כִּי רָאִיתִי אֵת כָּל־אֲשֶׁר לָבָן
12   עֹשֶׂה לָּךְ׃ אָנֹכִי הָאֵל בֵּית־אֵל אֲשֶׁר מָשַׁחְתָּ שָּׁם
13   מַצֵּבָה אֲשֶׁר נָדַרְתָּ לִּי שָׁם נֶדֶר עַתָּה קוּם צֵא מִן־
14  הָאָרֶץ הַזֹּאת וְשׁוּב אֶל־אֶרֶץ מוֹלַדְתֶּךָ׃ וַתַּעַן רָחֵל
15   וְלֵאָה וַתֹּאמַרְנָה לוֹ הַעוֹד לָנוּ חֵלֶק וְנַחֲלָה בְּבֵית
16   אָבִינוּ׃ הֲלוֹא נָכְרִיּוֹת נֶחְשַׁבְנוּ לוֹ כִּי מְכָרָנוּ וַיֹּאכַל גַּם־
17  אָכוֹל אֶת־כַּסְפֵּנוּ׃ כִּי כָל־הָעֹשֶׁר אֲשֶׁר הִצִּיל אֱלֹהִים
18   מֵאָבִינוּ לָנוּ הוּא וּלְבָנֵינוּ וְעַתָּה כֹּל אֲשֶׁר אָמַר אֱלֹהִים
19   אֵלֶיךָ עֲשֵׂה׃  וַיָּקָם יַעֲקֹב וַיִּשָּׂא אֶת־בָּנָיו וְאֶת־נָשָׁיו עַל־
20  הַגְּמַלִּים׃ וַיִּנְהַג אֶת־כָּל־מִקְנֵהוּ וְאֶת־כָּל־רְכֻשׁוֹ אֲשֶׁר
21   רָכָשׁ מִקְנֵה קִנְיָנוֹ אֲשֶׁר רָכַשׁ בְּפַדַּן אֲרָם לָבוֹא אֶל־
22  יִצְחָק אָבִיו אַרְצָה כְּנָעַן׃ וְלָבָן הָלַךְ לִגְזֹז אֶת־צֹאנוֹ
23   וַתִּגְנֹב רָחֵל אֶת־הַתְּרָפִים אֲשֶׁר לְאָבִיהָ׃ וַיִּגְנֹב יַעֲקֹב
24   אֶת־לֵב לָבָן הָאֲרַמִּי עַל־בְּלִי הִגִּיד לוֹ כִּי בֹרֵחַ הוּא׃
25   וַיִּבְרַח הוּא וְכָל־אֲשֶׁר־לוֹ וַיָּקָם וַיַּעֲבֹר אֶת־הַנָּהָר
26   וַיָּשֶׂם אֶת־פָּנָיו הַר הַגִּלְעָד׃ וַיֻּגַּד לְלָבָן בַּיּוֹם הַשְּׁלִישִׁי
27   כִּי בָרַח יַעֲקֹב׃ וַיִּקַּח אֶת־אֶחָיו עִמּוֹ וַיִּרְדֹּף אַחֲרָיו
28   דֶּרֶךְ שִׁבְעַת יָמִים וַיַּדְבֵּק אֹתוֹ בְּהַר הַגִּלְעָד׃ וַיָּבֹא
29   אֱלֹהִים אֶל־לָבָן הָאֲרַמִּי בַּחֲלֹם הַלָּיְלָה וַיֹּאמֶר לוֹ
30   הִשָּׁמֶר לְךָ פֶּן־תְּדַבֵּר עִם־יַעֲקֹב מִטּוֹב עַד־רָע׃ וַיַּשֵּׂג
31   לָבָן אֶת־יַעֲקֹב וְיַעֲקֹב תָּקַע אֶת־אָהֳלוֹ בָּהָר וְלָבָן תָּקַע
32   אֶת־אֶחָיו בְּהַר הַגִּלְעָד׃ וַיֹּאמֶר לָבָן לְיַעֲקֹב מֶה עָשִׂיתָ
33   וַתִּגְנֹב אֶת־לְבָבִי וַתְּנַהֵג אֶת־בְּנֹתַי כִּשְׁבֻיוֹת חָרֶב׃ לָמָּה
34   נַחְבֵּאתָ לִבְרֹחַ וַתִּגְנֹב אֹתִי וְלֹא־הִגַּדְתָּ לִּי וָאֲשַׁלֵּחֲךָ
35   בְּשִׂמְחָה וּבְשִׁרִים בְּתֹף וּבְכִנּוֹר׃ וְלֹא נְטַשְׁתַּנִי לְנַשֵּׁק
36   לְבָנַי וְלִבְנֹתָי עַתָּה הִסְכַּלְתָּ עֲשׂוֹ׃ יֶשׁ־לְאֵל יָדִי לַעֲשׂוֹת
37   עִמָּכֶם רָע וֵאלֹהֵי אֲבִיכֶם אֶמֶשׁ  אָמַר אֵלַי לֵאמֹר
38   הִשָּׁמֶר לְךָ מִדַּבֵּר עִם־יַעֲקֹב מִטּוֹב עַד־רָע׃ וְעַתָּה
39   הָלֹךְ הָלַכְתָּ כִּי־נִכְסֹף נִכְסַפְתָּה לְבֵית אָבִיךָ לָמָּה
40   גָנַבְתָּ אֶת־אֱלֹהָי׃ וַיַּעַן יַעֲקֹב וַיֹּאמֶר לְלָבָן כִּי יָרֵאתִי כִּי
41   אָמַרְתִּי פֶּן־תִּגְזֹל אֶת־בְּנוֹתֶיךָ מֵעִמִּי׃ עִם אֲשֶׁר תִּמְצָא
42   אֶת־אֱלֹהֶיךָ לֹא יִחְיֶה נֶגֶד אַחֵינוּ הַכֶּר־לְךָ מָה עִמָּדִי
