 1  וְדִישָׁן אֵלֶּה אַלּוּפֵי הַחֹרִי בְּנֵי שֵׂעִיר בְּאֶרֶץ אֱדוֹם׃
 2   וַיִּהְיוּ בְנֵי־לוֹטָן חֹרִי וְהֵימָם וַאֲחוֹת לוֹטָן תִּמְנָע׃ וְאֵלֶּה
 3   בְּנֵי שׁוֹבָל עַלְוָן וּמָנַחַת וְעֵיבָל שְׁפוֹ וְאוֹנָם׃ וְאֵלֶּה בְנֵי־
 4  צִבְעוֹן וְאַיָּה וַעֲנָה הוּא עֲנָה אֲשֶׁר מָצָא אֶת־הַיֵּמִם
 5   בַּמִּדְבָּר בִּרְעֹתוֹ אֶת־הַחֲמֹרִים לְצִבְעוֹן אָבִיו׃ וְאֵלֶּה
 6   בְנֵי־עֲנָה דִּשֹׁן וְאָהֳלִיבָמָה בַּת־עֲנָה׃ וְאֵלֶּה בְּנֵי דִישָׁן
 7   חֶמְדָּן וְאֶשְׁבָּן וְיִתְרָן וּכְרָן׃ אֵלֶּה בְּנֵי־אֵצֶר בִּלְהָן וְזַעֲוָן
 8   וַעֲקָן׃ אֵלֶּה בְנֵי־דִישָׁן עוּץ וַאֲרָן׃ אֵלֶּה אַלּוּפֵי הַחֹרִי
 9   אַלּוּף לוֹטָן אַלּוּף שׁוֹבָל אַלּוּף צִבְעוֹן אַלּוּף עֲנָה׃
10   אַלּוּף דִּשֹׁן אַלּוּף אֵצֶר אַלּוּף דִּישָׁן אֵלֶּה אַלּוּפֵי
11   הַחֹרִי לְאַלֻּפֵיהֶם בְּאֶרֶץ שֵׂעִיר׃ [פ]
12   וְאֵלֶּה הַמְּלָכִים אֲשֶׁר מָלְכוּ בְּאֶרֶץ אֱדוֹם לִפְנֵי מְלָךְ־
13  מֶלֶךְ לִבְנֵי יִשְׂרָאֵל׃ וַיִּמְלֹךְ בֶּאֱדוֹם בֶּלַע בֶּן־בְּעוֹר
14   וְשֵׁם עִירוֹ דִּנְהָבָה׃ וַיָּמָת בָּלַע וַיִּמְלֹךְ תַּחְתָּיו יוֹבָב
15   בֶּן־זֶרַח מִבָּצְרָה׃ וַיָּמָת יוֹבָב וַיִּמְלֹךְ תַּחְתָּיו חֻשָׁם
16   מֵאֶרֶץ הַתֵּימָנִי׃ וַיָּמָת חֻשָׁם וַיִּמְלֹךְ תַּחְתָּיו הֲדַד בֶּן־
17  בְּדַד הַמַּכֶּה אֶת־מִדְיָן בִּשְׂדֵה מוֹאָב וְשֵׁם עִירוֹ עֲוִית׃
18   וַיָּמָת הֲדָד וַיִּמְלֹךְ תַּחְתָּיו שַׂמְלָה מִמַּשְׂרֵקָה׃ וַיָּמָת
19   שַׂמְלָה וַיִּמְלֹךְ תַּחְתָּיו שָׁאוּל מֵרְחֹבוֹת הַנָּהָר׃ וַיָּמָת
20   שָׁאוּל וַיִּמְלֹךְ תַּחְתָּיו בַּעַל חָנָן בֶּן־עַכְבּוֹר׃ וַיָּמָת
21   בַּעַל חָנָן בֶּן־עַכְבּוֹר וַיִּמְלֹךְ תַּחְתָּיו הֲדַר וְשֵׁם
22   עִירוֹ פָּעוּ וְשֵׁם אִשְׁתּוֹ מְהֵיטַבְאֵל בַּת־מַטְרֵד בַּת
23   מֵי זָהָב׃  וְאֵלֶּה שְׁמוֹת אַלּוּפֵי עֵשָׂו לְמִשְׁפְּחֹתָם
24   לִמְקֹמֹתָם בִּשְׁמֹתָם אַלּוּף תִּמְנָע אַלּוּף עַלְוָה אַלּוּף
25   יְתֵת׃ אַלּוּף אָהֳלִיבָמָה אַלּוּף אֵלָה אַלּוּף פִּינֹן׃ אַלּוּף
26   קְנַז אַלּוּף תֵּימָן אַלּוּף מִבְצָר׃ אַלּוּף מַגְדִּיאֵל אַלּוּף
27   עִירָם אֵלֶּה  אַלּוּפֵי אֱדוֹם לְמֹשְׁבֹתָם בְּאֶרֶץ אֲחֻזָּתָם
28   הוּא עֵשָׂו אֲבִי אֱדוֹם׃ [פ]
29  וַיֵּשֶׁב יַעֲקֹב בְּאֶרֶץ מְגוּרֵי אָבִיו בְּאֶרֶץ כְּנָעַן׃ אֵלֶּה
30   תֹּלְדוֹת יַעֲקֹב יוֹסֵף בֶּן־שְׁבַע־עֶשְׂרֵה שָׁנָה הָיָה רֹעֶה
31   אֶת־אֶחָיו בַּצֹּאן וְהוּא נַעַר אֶת־בְּנֵי בִלְהָה וְאֶת־בְּנֵי
32   זִלְפָּה נְשֵׁי אָבִיו וַיָּבֵא יוֹסֵף אֶת־דִּבָּתָם רָעָה אֶל־
33  אֲבִיהֶם׃ וְיִשְׂרָאֵל אָהַב אֶת־יוֹסֵף מִכָּל־בָּנָיו כִּי־בֶן־
34  זְקֻנִים הוּא לוֹ וְעָשָׂה לוֹ כְּתֹנֶת פַּסִּים׃♢ וַיִּרְאוּ אֶחָיו כִּי־
35  אֹתוֹ אָהַב אֲבִיהֶם מִכָּל־אֶחָיו וַיִּשְׂנְאוּ אֹתוֹ וְלֹא יָכְלוּ
36   דַּבְּרוֹ לְשָׁלֹם׃ וַיַּחֲלֹם יוֹסֵף חֲלוֹם וַיַּגֵּד לְאֶחָיו וַיּוֹסִפוּ
37   עוֹד שְׂנֹא אֹתוֹ׃ וַיֹּאמֶר אֲלֵיהֶם שִׁמְעוּ־נָא הַחֲלוֹם
38   הַזֶּה אֲשֶׁר חָלָמְתִּי׃ וְהִנֵּה אֲנַחְנוּ מְאַלְּמִים אֲלֻמִּים
39   בְּתוֹךְ הַשָּׂדֶה וְהִנֵּה קָמָה אֲלֻמָּתִי וְגַם־נִצָּבָה וְהִנֵּה
40   תְסֻבֶּינָה אֲלֻמֹּתֵיכֶם וַתִּשְׁתַּחֲוֶיןָ לַאֲלֻמָּתִי׃♢ וַיֹּאמְרוּ
41   לוֹ אֶחָיו הֲמָלֹךְ תִּמְלֹךְ עָלֵינוּ אִם־מָשׁוֹל תִּמְשֹׁל
42   בָּנוּ וַיּוֹסִפוּ עוֹד שְׂנֹא אֹתוֹ עַל־חֲלֹמֹתָיו וְעַל־דְּבָרָיו׃
