 1  וַיֹּאמֶר יְהוָה אֱלֹהִים לֹא־טוֹב הֱיוֹת הָאָדָם לְבַדּוֹ
 2   אֶעֱשֶׂה־לּוֹ עֵזֶר כְּנֶגְדּוֹ׃ וַיִּצֶר יְהוָה אֱלֹהִים מִן־הָאֲדָמָה
 3   כָּל־חַיַּת הַשָּׂדֶה וְאֵת כָּל־עוֹף הַשָּׁמַיִם וַיָּבֵא אֶל־
 4  הָאָדָם לִרְאוֹת מַה־יִּקְרָא־לוֹ וְכֹל אֲשֶׁר יִקְרָא־לוֹ
 5   הָאָדָם נֶפֶשׁ חַיָּה הוּא שְׁמוֹ׃  וַיִּקְרָא הָאָדָם שֵׁמוֹת
 6   לְכָל־הַבְּהֵמָה וּלְעוֹף הַשָּׁמַיִם וּלְכֹל חַיַּת הַשָּׂדֶה
 7   וּלְאָדָם לֹא־מָצָא עֵזֶר כְּנֶגְדּוֹ׃ וַיַּפֵּל יְהוָה אֱלֹהִים
 8    תַּרְדֵּמָה עַל־הָאָדָם וַיִּישָׁן וַיִּקַּח אַחַת מִצַּלְעֹתָיו וַיִּסְגֹּר
 9   בָּשָׂר תַּחְתֶּנָּה׃ וַיִּבֶן יְהוָה אֱלֹהִים  אֶת־הַצֵּלָע אֲשֶׁר־
10  לָקַח מִן־הָאָדָם לְאִשָּׁה וַיְבִאֶהָ אֶל־הָאָדָם׃ וַיֹּאמֶר
11   הָאָדָם זֹאת הַפַּעַם עֶצֶם מֵעֲצָמַי וּבָשָׂר מִבְּשָׂרִי
12   לְזֹאת יִקָּרֵא אִשָּׁה כִּי מֵאִישׁ לֻקְחָה־זֹּאת׃ עַל־כֵּן יַעֲזָב־
13  אִישׁ אֶת־אָבִיו וְאֶת־אִמּוֹ וְדָבַק בְּאִשְׁתּוֹ וְהָיוּ לְבָשָׂר
14   אֶחָד׃ וַיִּהְיוּ שְׁנֵיהֶם עֲרוּמִּים הָאָדָם וְאִשְׁתּוֹ וְלֹא
15   יִתְבֹּשָׁשׁוּ׃ וְהַנָּחָשׁ הָיָה עָרוּם מִכֹּל חַיַּת הַשָּׂדֶה אֲשֶׁר
16   עָשָׂה יְהוָה אֱלֹהִים וַיֹּאמֶר אֶל־הָאִשָּׁה אַף כִּי־אָמַר
17   אֱלֹהִים לֹא תֹאכְלוּ מִכֹּל עֵץ הַגָּן׃ וַתֹּאמֶר הָאִשָּׁה אֶל־
18  הַנָּחָשׁ מִפְּרִי עֵץ־הַגָּן נֹאכֵל׃ וּמִפְּרִי הָעֵץ אֲשֶׁר בְּתוֹךְ־
19  הַגָּן אָמַר אֱלֹהִים לֹא תֹאכְלוּ מִמֶּנּוּ וְלֹא תִגְּעוּ בּוֹ פֶּן־
20  תְּמֻתוּן׃ וַיֹּאמֶר הַנָּחָשׁ אֶל־הָאִשָּׁה לֹא־מוֹת תְּמֻתוּן׃ כִּי
21   יֹדֵעַ אֱלֹהִים כִּי בְּיוֹם אֲכָלְכֶם מִמֶּנּוּ וְנִפְקְחוּ עֵינֵיכֶם
22   וִהְיִיתֶם כֵּאלֹהִים יֹדְעֵי טוֹב וָרָע׃ וַתֵּרֶא הָאִשָּׁה כִּי
23   טוֹב הָעֵץ לְמַאֲכָל וְכִי תַאֲוָה־הוּא לָעֵינַיִם וְנֶחְמָד
24   הָעֵץ לְהַשְׂכִּיל וַתִּקַּח מִפִּרְיוֹ וַתֹּאכַל וַתִּתֵּן גַּם־
25  לְאִישָׁהּ עִמָּהּ וַיֹּאכַל׃ וַתִּפָּקַחְנָה עֵינֵי שְׁנֵיהֶם וַיֵּדְעוּ
26   כִּי עֵירֻמִּם הֵם וַיִּתְפְּרוּ עֲלֵה תְאֵנָה וַיַּעֲשׂוּ לָהֶם חֲגֹרֹת׃
27   וַיִּשְׁמְעוּ אֶת־קוֹל יְהוָה אֱלֹהִים מִתְהַלֵּךְ בַּגָּן לְרוּחַ
28   הַיּוֹם וַיִּתְחַבֵּא הָאָדָם וְאִשְׁתּוֹ מִפְּנֵי יְהוָה אֱלֹהִים
29   בְּתוֹךְ עֵץ הַגָּן׃ וַיִּקְרָא יְהוָה אֱלֹהִים אֶל־הָאָדָם
30   וַיֹּאמֶר לוֹ אַיֶּכָּה׃ וַיֹּאמֶר אֶת־קֹלְךָ שָׁמַעְתִּי בַּגָּן וָאִירָא
31   כִּי־עֵירֹם אָנֹכִי וָאֵחָבֵא׃ וַיֹּאמֶר מִי הִגִּיד לְךָ כִּי עֵירֹם
32   אָתָּה הֲמִן־הָעֵץ אֲשֶׁר צִוִּיתִיךָ לְבִלְתִּי אֲכָל־מִמֶּנּוּ
33   אָכָלְתָּ׃ וַיֹּאמֶר הָאָדָם הָאִשָּׁה אֲשֶׁר נָתַתָּה עִמָּדִי הִוא
34   נָתְנָה־לִּי מִן־הָעֵץ וָאֹכֵל׃ וַיֹּאמֶר יְהוָה אֱלֹהִים לָאִשָּׁה
35   מַה־זֹּאת עָשִׂית וַתֹּאמֶר הָאִשָּׁה הַנָּחָשׁ הִשִּׁיאַנִי
36   וָאֹכֵל׃ וַיֹּאמֶר יְהוָה אֱלֹהִים  אֶל־הַנָּחָשׁ כִּי עָשִׂיתָ זֹּאת
37   אָרוּר אַתָּה מִכָּל־הַבְּהֵמָה וּמִכֹּל חַיַּת הַשָּׂדֶה עַל־
38  גְּחֹנְךָ תֵלֵךְ וְעָפָר תֹּאכַל כָּל־יְמֵי חַיֶּיךָ׃ וְאֵיבָה  אָשִׁית
39   בֵּינְךָ וּבֵין הָאִשָּׁה וּבֵין זַרְעֲךָ וּבֵין זַרְעָהּ הוּא יְשׁוּפְךָ
40   רֹאשׁ וְאַתָּה תְּשׁוּפֶנּוּ עָקֵב׃ [ס]             אֶל־
41  הָאִשָּׁה אָמַר הַרְבָּה אַרְבֶּה עִצְּבוֹנֵךְ וְהֵרֹנֵךְ
42   בְּעֶצֶב תֵּלְדִי בָנִים וְאֶל־אִישֵׁךְ תְּשׁוּקָתֵךְ
