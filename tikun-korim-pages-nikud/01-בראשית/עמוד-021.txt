 1  וַיָּמָל אַבְרָהָם אֶת־יִצְחָק בְּנוֹ בֶּן־שְׁמֹנַת יָמִים כַּאֲשֶׁר
 2   צִוָּה אֹתוֹ אֱלֹהִים׃  וְאַבְרָהָם בֶּן־מְאַת שָׁנָה בְּהִוָּלֶד לוֹ
 3   אֵת יִצְחָק בְּנוֹ׃ וַתֹּאמֶר שָׂרָה צְחֹק עָשָׂה לִי אֱלֹהִים
 4   כָּל־הַשֹּׁמֵעַ יִצְחַק־לִי׃ וַתֹּאמֶר מִי מִלֵּל לְאַבְרָהָם
 5   הֵינִיקָה בָנִים שָׂרָה כִּי־יָלַדְתִּי בֵן לִזְקֻנָיו׃ וַיִּגְדַּל הַיֶּלֶד
 6   וַיִּגָּמַל וַיַּעַשׂ אַבְרָהָם מִשְׁתֶּה גָדוֹל בְּיוֹם הִגָּמֵל אֶת־
 7  יִצְחָק׃ וַתֵּרֶא שָׂרָה אֶת־בֶּן־הָגָר הַמִּצְרִית אֲשֶׁר־יָלְדָה
 8   לְאַבְרָהָם מְצַחֵק׃ וַתֹּאמֶר לְאַבְרָהָם גָּרֵשׁ הָאָמָה
 9   הַזֹּאת וְאֶת־בְּנָהּ כִּי לֹא יִירַשׁ בֶּן־הָאָמָה הַזֹּאת עִם־
10  בְּנִי עִם־יִצְחָק׃ וַיֵּרַע הַדָּבָר מְאֹד בְּעֵינֵי אַבְרָהָם
11   עַל אוֹדֹת בְּנוֹ׃ וַיֹּאמֶר אֱלֹהִים אֶל־אַבְרָהָם אַל־יֵרַע
12   בְּעֵינֶיךָ עַל־הַנַּעַר וְעַל־אֲמָתֶךָ כֹּל אֲשֶׁר תֹּאמַר
13   אֵלֶיךָ שָׂרָה שְׁמַע בְּקֹלָהּ כִּי בְיִצְחָק יִקָּרֵא לְךָ
14   זָרַע׃ ♢  וְגַם אֶת־בֶּן־הָאָמָה לְגוֹי אֲשִׂימֶנּוּ כִּי זַרְעֲךָ הוּא׃
15   וַיַּשְׁכֵּם אַבְרָהָם  בַּבֹּקֶר וַיִּקַּח־לֶחֶם וְחֵמַת מַיִם וַיִּתֵּן
16   אֶל־הָגָר שָׂם עַל־שִׁכְמָהּ וְאֶת־הַיֶּלֶד וַיְשַׁלְּחֶהָ וַתֵּלֶךְ
17   וַתֵּתַע בְּמִדְבַּר בְּאֵר שָׁבַע׃ וַיִּכְלוּ הַמַּיִם מִן־הַחֵמֶת
18   וַתַּשְׁלֵךְ אֶת־הַיֶּלֶד תַּחַת אַחַד הַשִּׂיחִם׃ וַתֵּלֶךְ וַתֵּשֶׁב
19   לָהּ מִנֶּגֶד הַרְחֵק כִּמְטַחֲוֵי קֶשֶׁת כִּי אָמְרָה אַל־אֶרְאֶה
20   בְּמוֹת הַיָּלֶד וַתֵּשֶׁב מִנֶּגֶד וַתִּשָּׂא אֶת־קֹלָהּ וַתֵּבְךְּ׃
21   וַיִּשְׁמַע אֱלֹהִים אֶת־קוֹל הַנַּעַר וַיִּקְרָא מַלְאַךְ אֱלֹהִים
22    אֶל־הָגָר מִן־הַשָּׁמַיִם וַיֹּאמֶר לָהּ מַה־לָּךְ הָגָר אַל־
23  תִּירְאִי כִּי־שָׁמַע אֱלֹהִים אֶל־קוֹל הַנַּעַר בַּאֲשֶׁר הוּא־
24  שָׁם׃ קוּמִי שְׂאִי אֶת־הַנַּעַר וְהַחֲזִיקִי אֶת־יָדֵךְ בּוֹ כִּי־
25  לְגוֹי גָּדוֹל אֲשִׂימֶנּוּ׃ וַיִּפְקַח אֱלֹהִים אֶת־עֵינֶיהָ וַתֵּרֶא
26   בְּאֵר מָיִם וַתֵּלֶךְ וַתְּמַלֵּא אֶת־הַחֵמֶת מַיִם וַתַּשְׁקְ אֶת־
27  הַנָּעַר׃ וַיְהִי אֱלֹהִים אֶת־הַנַּעַר וַיִּגְדָּל וַיֵּשֶׁב בַּמִּדְבָּר
28   וַיְהִי רֹבֶה קַשָּׁת׃ וַיֵּשֶׁב בְּמִדְבַּר פָּארָן וַתִּקַּח־לוֹ
29   אִמּוֹ אִשָּׁה מֵאֶרֶץ מִצְרָיִם׃[פ]
30   וַיְהִי בָּעֵת הַהִוא וַיֹּאמֶר אֲבִימֶלֶךְ וּפִיכֹל שַׂר־צְבָאוֹ
31   אֶל־אַבְרָהָם לֵאמֹר אֱלֹהִים עִמְּךָ בְּכֹל אֲשֶׁר־אַתָּה
32   עֹשֶׂה׃ וְעַתָּה הִשָּׁבְעָה לִּי בֵאלֹהִים הֵנָּה אִם־תִּשְׁקֹר
33   לִי וּלְנִינִי וּלְנֶכְדִּי כַּחֶסֶד אֲשֶׁר־עָשִׂיתִי עִמְּךָ תַּעֲשֶׂה
34   עִמָּדִי וְעִם־הָאָרֶץ אֲשֶׁר־גַּרְתָּה בָּהּ׃ וַיֹּאמֶר אַבְרָהָם
35   אָנֹכִי אִשָּׁבֵעַ׃ וְהוֹכִחַ אַבְרָהָם אֶת־אֲבִימֶלֶךְ עַל־
36  אֹדוֹת בְּאֵר הַמַּיִם אֲשֶׁר גָּזְלוּ עַבְדֵי אֲבִימֶלֶךְ׃ וַיֹּאמֶר
37   אֲבִימֶלֶךְ לֹא יָדַעְתִּי מִי עָשָׂה אֶת־הַדָּבָר הַזֶּה וְגַם־
38  אַתָּה לֹא־הִגַּדְתָּ לִּי וְגַם אָנֹכִי לֹא שָׁמַעְתִּי בִּלְתִּי
39   הַיּוֹם׃ וַיִּקַּח אַבְרָהָם צֹאן וּבָקָר וַיִּתֵּן לַאֲבִימֶלֶךְ
40   וַיִּכְרְתוּ שְׁנֵיהֶם בְּרִית׃ וַיַּצֵּב אַבְרָהָם אֶת־שֶׁבַע
41   כִּבְשֹׂת הַצֹּאן לְבַדְּהֶן׃ וַיֹּאמֶר אֲבִימֶלֶךְ אֶל־אַבְרָהָם
42   מָה הֵנָּה שֶׁבַע כְּבָשֹׂת הָאֵלֶּה אֲשֶׁר הִצַּבְתָּ לְבַדָּנָה׃
