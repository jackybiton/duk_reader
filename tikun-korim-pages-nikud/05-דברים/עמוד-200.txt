 1  וְלֹא־תִקְחוּ כֹפֶר לָנוּס אֶל־עִיר מִקְלָטוֹ לָשׁוּב
 2   לָשֶׁבֶת בָּאָרֶץ עַד־מוֹת הַכֹּהֵן׃ וְלֹא־תַחֲנִיפוּ אֶת־
 3  הָאָרֶץ אֲשֶׁר אַתֶּם בָּהּ כִּי הַדָּם הוּא יַחֲנִיף אֶת־
 4  הָאָרֶץ וְלָאָרֶץ לֹא־יְכֻפַּר לַדָּם אֲשֶׁר שֻׁפַּךְ־בָּהּ
 5   כִּי־אִם בְּדַם שֹׁפְכוֹ׃ וְלֹא תְטַמֵּא אֶת־הָאָרֶץ אֲשֶׁר
 6   אַתֶּם יֹשְׁבִים בָּהּ אֲשֶׁר אֲנִי שֹׁכֵן בְּתוֹכָהּ כִּי אֲנִי
 7   יְהוָה שֹׁכֵן בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׃ [פ]
 8    וַיִּקְרְבוּ רָאשֵׁי הָאָבוֹת לְמִשְׁפַּחַת בְּנֵי־גִלְעָד בֶּן־
 9  מָכִיר בֶּן־מְנַשֶּׁה מִמִּשְׁפְּחֹת בְּנֵי יוֹסֵף וַיְדַבְּרוּ לִפְנֵי
10   מֹשֶׁה וְלִפְנֵי הַנְּשִׂאִים רָאשֵׁי אָבוֹת לִבְנֵי יִשְׂרָאֵל׃
11   וַיֹּאמְרוּ אֶת־אֲדֹנִי צִוָּה יְהוָה לָתֵת אֶת־הָאָרֶץ בְּנַחֲלָה
12   בְּגוֹרָל לִבְנֵי יִשְׂרָאֵל וַאדֹנִי צֻוָּה בַיהוָה לָתֵת אֶת־
13  נַחֲלַת צְלָפְחָד אָחִינוּ לִבְנֹתָיו׃ וְהָיוּ לְאֶחָד מִבְּנֵי
14   שִׁבְטֵי בְנֵי־יִשְׂרָאֵל לְנָשִׁים וְנִגְרְעָה נַחֲלָתָן מִנַּחֲלַת
15   אֲבֹתֵינוּ וְנוֹסַף עַל נַחֲלַת הַמַּטֶּה אֲשֶׁר תִּהְיֶינָה לָהֶם
16   וּמִגֹּרַל נַחֲלָתֵנוּ יִגָּרֵעַ׃ וְאִם־יִהְיֶה הַיֹּבֵל לִבְנֵי יִשְׂרָאֵל
17   וְנוֹסְפָה נַחֲלָתָן עַל נַחֲלַת הַמַּטֶּה אֲשֶׁר תִּהְיֶינָה לָהֶם
18   וּמִנַּחֲלַת מַטֵּה אֲבֹתֵינוּ יִגָּרַע נַחֲלָתָן׃ וַיְצַו מֹשֶׁה אֶת־
19  בְּנֵי יִשְׂרָאֵל עַל־פִּי יְהוָה לֵאמֹר כֵּן מַטֵּה בְנֵי־יוֹסֵף
20   דֹּבְרִים׃ זֶה הַדָּבָר אֲשֶׁר־צִוָּה יְהוָה לִבְנוֹת צְלָפְחָד
21   לֵאמֹר לַטּוֹב בְּעֵינֵיהֶם תִּהְיֶינָה לְנָשִׁים אַךְ לְמִשְׁפַּחַת
22   מַטֵּה אֲבִיהֶם תִּהְיֶינָה לְנָשִׁים׃ וְלֹא־תִסֹּב נַחֲלָה לִבְנֵי
23   יִשְׂרָאֵל מִמַּטֶּה אֶל־מַטֶּה כִּי אִישׁ בְּנַחֲלַת מַטֵּה
24   אֲבֹתָיו יִדְבְּקוּ בְּנֵי יִשְׂרָאֵל׃ וְכָל־בַּת יֹרֶשֶׁת נַחֲלָה
25   מִמַּטּוֹת בְּנֵי יִשְׂרָאֵל לְאֶחָד מִמִּשְׁפַּחַת מַטֵּה אָבִיהָ
26   תִּהְיֶה לְאִשָּׁה לְמַעַן יִירְשׁוּ בְּנֵי יִשְׂרָאֵל אִישׁ נַחֲלַת
27   אֲבֹתָיו׃ וְלֹא־תִסֹּב נַחֲלָה מִמַּטֶּה לְמַטֶּה אַחֵר כִּי־אִישׁ
28   בְּנַחֲלָתוֹ יִדְבְּקוּ מַטּוֹת בְּנֵי יִשְׂרָאֵל׃ כַּאֲשֶׁר צִוָּה
29   יְהוָה אֶת־מֹשֶׁה כֵּן עָשׂוּ בְּנוֹת צְלָפְחָד׃  וַתִּהְיֶינָה
30   מַחְלָה תִרְצָה וְחָגְלָה וּמִלְכָּה וְנֹעָה בְּנוֹת צְלָפְחָד
31   לִבְנֵי דֹדֵיהֶן לְנָשִׁים׃ מִמִּשְׁפְּחֹת בְּנֵי־מְנַשֶּׁה בֶן־יוֹסֵף
32   הָיוּ לְנָשִׁים וַתְּהִי נַחֲלָתָן עַל־מַטֵּה מִשְׁפַּחַת אֲבִיהֶן׃
33   אֵלֶּה הַמִּצְו͏ֹת וְהַמִּשְׁפָּטִים אֲשֶׁר צִוָּה יְהוָה בְּיַד־
34  מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל בְּעַרְבֹת מוֹאָב עַל יַרְדֵּן יְרֵחוֹ׃

39  אֵלֶּה הַדְּבָרִים אֲשֶׁר דִּבֶּר מֹשֶׁה אֶל־כָּל־יִשְׂרָאֵל
40   בְּעֵבֶר הַיַּרְדֵּן בַּמִּדְבָּר בָּעֲרָבָה מוֹל סוּף בֵּין־
41  פָּארָן וּבֵין־תֹּפֶל וְלָבָן וַחֲצֵרֹת וְדִי זָהָב׃ אַחַד עָשָׂר
42   יוֹם מֵחֹרֵב דֶּרֶךְ הַר־שֵׂעִיר עַד קָדֵשׁ בַּרְנֵעַ׃
