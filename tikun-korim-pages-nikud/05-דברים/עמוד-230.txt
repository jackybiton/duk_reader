 1  וַאֲשֶׁר שָׂכַר עָלֶיךָ אֶת־בִּלְעָם בֶּן־בְּעוֹר מִפְּתוֹר
 2   אֲרַם נַהֲרַיִם לְקַלְלֶךָּ׃ וְלֹא־אָבָה יְהוָה אֱלֹהֶיךָ לִשְׁמֹעַ
 3   אֶל־בִּלְעָם וַיַּהֲפֹךְ יְהוָה אֱלֹהֶיךָ לְּךָ אֶת־הַקְּלָלָה
 4   לִבְרָכָה כִּי אֲהֵבְךָ יְהוָה אֱלֹהֶיךָ׃ לֹא־תִדְרֹשׁ שְׁלֹמָם
 5   וְטֹבָתָם כָּל־יָמֶיךָ לְעוֹלָם׃ [ס]              לֹא־
 6  תְתַעֵב אֲדֹמִי כִּי אָחִיךָ הוּא לֹא־תְתַעֵב מִצְרִי כִּי־גֵר
 7   הָיִיתָ בְאַרְצוֹ׃ בָּנִים אֲשֶׁר־יִוָּלְדוּ לָהֶם דּוֹר שְׁלִישִׁי
 8   יָבֹא לָהֶם בִּקְהַל יְהוָה׃ [ס]             כִּי־תֵצֵא
 9   מַחֲנֶה עַל־אֹיְבֶיךָ וְנִשְׁמַרְתָּ מִכֹּל דָּבָר רָע׃ כִּי־יִהְיֶה
10   בְךָ אִישׁ אֲשֶׁר לֹא־יִהְיֶה טָהוֹר מִקְּרֵה־לָיְלָה וְיָצָא
11   אֶל־מִחוּץ לַמַּחֲנֶה לֹא יָבֹא אֶל־תּוֹךְ הַמַּחֲנֶה׃ וְהָיָה
12   לִפְנוֹת־עֶרֶב יִרְחַץ בַּמָּיִם וּכְבֹא הַשֶּׁמֶשׁ יָבֹא אֶל־
13  תּוֹךְ הַמַּחֲנֶה׃ וְיָד תִּהְיֶה לְךָ מִחוּץ לַמַּחֲנֶה וְיָצָאתָ
14   שָּׁמָּה חוּץ׃ וְיָתֵד תִּהְיֶה לְךָ עַל־אֲזֵנֶךָ וְהָיָה בְּשִׁבְתְּךָ
15   חוּץ וְחָפַרְתָּה בָהּ וְשַׁבְתָּ וְכִסִּיתָ אֶת־צֵאָתֶךָ׃ כִּי יְהוָה
16   אֱלֹהֶיךָ מִתְהַלֵּךְ  בְּקֶרֶב מַחֲנֶךָ לְהַצִּילְךָ וְלָתֵת אֹיְבֶיךָ
17   לְפָנֶיךָ וְהָיָה מַחֲנֶיךָ קָדוֹשׁ וְלֹא־יִרְאֶה בְךָ עֶרְוַת
18   דָּבָר וְשָׁב מֵאַחֲרֶיךָ׃ [ס]             לֹא־תַסְגִּיר
19   עֶבֶד אֶל־אֲדֹנָיו אֲשֶׁר־יִנָּצֵל אֵלֶיךָ מֵעִם אֲדֹנָיו׃ עִמְּךָ
20   יֵשֵׁב בְּקִרְבְּךָ בַּמָּקוֹם אֲשֶׁר־יִבְחַר בְּאַחַד שְׁעָרֶיךָ
21   בַּטּוֹב לוֹ לֹא תּוֹנֶנּוּ׃ [ס]             לֹא־תִהְיֶה
22   קְדֵשָׁה מִבְּנוֹת יִשְׂרָאֵל וְלֹא־יִהְיֶה קָדֵשׁ מִבְּנֵי
23   יִשְׂרָאֵל׃ לֹא־תָבִיא אֶתְנַן זוֹנָה וּמְחִיר כֶּלֶב בֵּית
24   יְהוָה אֱלֹהֶיךָ לְכָל־נֶדֶר כִּי תוֹעֲבַת יְהוָה אֱלֹהֶיךָ
25   גַּם־שְׁנֵיהֶם׃ [ס]             לֹא־תַשִּׁיךְ לְאָחִיךָ
26   נֶשֶׁךְ כֶּסֶף נֶשֶׁךְ אֹכֶל נֶשֶׁךְ כָּל־דָּבָר אֲשֶׁר יִשָּׁךְ׃
27   לַנָּכְרִי תַשִּׁיךְ וּלְאָחִיךָ לֹא תַשִּׁיךְ לְמַעַן יְבָרֶכְךָ
28   יְהוָה אֱלֹהֶיךָ בְּכֹל מִשְׁלַח יָדֶךָ עַל־הָאָרֶץ אֲשֶׁר־
29  אַתָּה בָא־שָׁמָּה לְרִשְׁתָּהּ׃ [ס]             כִּי־
30  תִדֹּר נֶדֶר לַיהוָה אֱלֹהֶיךָ לֹא תְאַחֵר לְשַׁלְּמוֹ כִּי־
31  דָרֹשׁ יִדְרְשֶׁנּוּ יְהוָה אֱלֹהֶיךָ מֵעִמָּךְ וְהָיָה בְךָ חֵטְא׃
32   וְכִי תֶחְדַּל לִנְדֹּר לֹא־יִהְיֶה בְךָ חֵטְא׃ מוֹצָא שְׂפָתֶיךָ
33   תִּשְׁמֹר וְעָשִׂיתָ כַּאֲשֶׁר נָדַרְתָּ לַיהוָה אֱלֹהֶיךָ נְדָבָה
34   אֲשֶׁר דִּבַּרְתָּ בְּפִיךָ׃ [ס]              כִּי
35   תָבֹא בְּכֶרֶם רֵעֶךָ וְאָכַלְתָּ עֲנָבִים כְּנַפְשְׁךָ שָׂבְעֶךָ
36   וְאֶל־כֶּלְיְךָ לֹא תִתֵּן׃ [ס]             כִּי
37   תָבֹא בְּקָמַת רֵעֶךָ וְקָטַפְתָּ מְלִילֹת בְּיָדֶךָ וְחֶרְמֵשׁ
38   לֹא תָנִיף עַל קָמַת רֵעֶךָ׃ [ס]             כִּי־
39  יִקַּח אִישׁ אִשָּׁה וּבְעָלָהּ וְהָיָה אִם־לֹא תִמְצָא־חֵן
40   בְּעֵינָיו כִּי־מָצָא בָהּ עֶרְוַת דָּבָר וְכָתַב לָהּ סֵפֶר
41   כְּרִיתֻת וְנָתַן בְּיָדָהּ וְשִׁלְּחָהּ מִבֵּיתוֹ׃ וְיָצְאָה מִבֵּיתוֹ
42   וְהָלְכָה וְהָיְתָה לְאִישׁ־אַחֵר׃ וּשְׂנֵאָהּ הָאִישׁ הָאַחֲרוֹן
