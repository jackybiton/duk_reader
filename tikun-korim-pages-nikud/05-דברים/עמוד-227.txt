 1  וְאָכַלְתָּ אֶת־שְׁלַל אֹיְבֶיךָ אֲשֶׁר נָתַן יְהוָה אֱלֹהֶיךָ
 2   לָךְ׃ כֵּן תַּעֲשֶׂה לְכָל־הֶעָרִים הָרְחֹקֹת מִמְּךָ מְאֹד
 3   אֲשֶׁר לֹא־מֵעָרֵי הַגּוֹיִם־הָאֵלֶּה הֵנָּה׃ רַק מֵעָרֵי הָעַמִּים
 4   הָאֵלֶּה אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה לֹא תְחַיֶּה
 5   כָּל־נְשָׁמָה׃ כִּי־הַחֲרֵם תַּחֲרִימֵם הַחִתִּי וְהָאֱמֹרִי
 6   הַכְּנַעֲנִי וְהַפְּרִזִּי הַחִוִּי וְהַיְבוּסִי כַּאֲשֶׁר צִוְּךָ יְהוָה
 7   אֱלֹהֶיךָ׃ לְמַעַן אֲשֶׁר לֹא־יְלַמְּדוּ אֶתְכֶם לַעֲשׂוֹת כְּכֹל
 8   תּוֹעֲבֹתָם אֲשֶׁר עָשׂוּ לֵאלֹהֵיהֶם וַחֲטָאתֶם לַיהוָה
 9   אֱלֹהֵיכֶם׃ [ס]             כִּי־תָצוּר אֶל־עִיר
10   יָמִים רַבִּים לְהִלָּחֵם עָלֶיהָ לְתָפְשָׂהּ לֹא־תַשְׁחִית
11   אֶת־עֵצָהּ לִנְדֹּחַ עָלָיו גַּרְזֶן כִּי מִמֶּנּוּ תֹאכֵל וְאֹתוֹ לֹא
12   תִכְרֹת כִּי הָאָדָם עֵץ הַשָּׂדֶה לָבֹא מִפָּנֶיךָ בַּמָּצוֹר׃
13   רַק עֵץ אֲשֶׁר־תֵּדַע כִּי־לֹא־עֵץ מַאֲכָל הוּא אֹתוֹ
14   תַשְׁחִית וְכָרָתָּ וּבָנִיתָ מָצוֹר עַל־הָעִיר אֲשֶׁר־הִוא
15   עֹשָׂה עִמְּךָ מִלְחָמָה עַד רִדְתָּהּ׃ [פ]
16   כִּי־יִמָּצֵא חָלָל בָּאֲדָמָה אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ
17   לְרִשְׁתָּהּ נֹפֵל בַּשָּׂדֶה לֹא נוֹדַע מִי הִכָּהוּ׃ וְיָצְאוּ
18   זְקֵנֶיךָ וְשֹׁפְטֶיךָ וּמָדְדוּ אֶל־הֶעָרִים אֲשֶׁר סְבִיבֹת
19   הֶחָלָל׃ וְהָיָה הָעִיר הַקְּרֹבָה אֶל־הֶחָלָל וְלָקְחוּ
20   זִקְנֵי הָעִיר הַהִוא עֶגְלַת בָּקָר אֲשֶׁר לֹא־עֻבַּד בָּהּ
21   אֲשֶׁר לֹא־מָשְׁכָה בְּעֹל׃ וְהוֹרִדוּ זִקְנֵי הָעִיר הַהִוא
22   אֶת־הָעֶגְלָה אֶל־נַחַל אֵיתָן אֲשֶׁר לֹא־יֵעָבֵד בּוֹ וְלֹא
23   יִזָּרֵעַ וְעָרְפוּ־שָׁם אֶת־הָעֶגְלָה בַּנָּחַל׃ וְנִגְּשׁוּ הַכֹּהֲנִים
24   בְּנֵי לֵוִי כִּי בָם בָּחַר יְהוָה אֱלֹהֶיךָ לְשָׁרְתוֹ וּלְבָרֵךְ
25   בְּשֵׁם יְהוָה וְעַל־פִּיהֶם יִהְיֶה כָּל־רִיב וְכָל־נָגַע׃ וְכֹל
26   זִקְנֵי הָעִיר הַהִוא הַקְּרֹבִים אֶל־הֶחָלָל יִרְחֲצוּ אֶת־
27  יְדֵיהֶם עַל־הָעֶגְלָה הָעֲרוּפָה בַנָּחַל׃  וְעָנוּ וְאָמְרוּ יָדֵינוּ
28   לֹא שָׁפְכוּ אֶת־הַדָּם הַזֶּה וְעֵינֵינוּ לֹא רָאוּ׃ כַּפֵּר
29   לְעַמְּךָ יִשְׂרָאֵל אֲשֶׁר־פָּדִיתָ יְהוָה וְאַל־תִּתֵּן דָּם
30   נָקִי בְּקֶרֶב עַמְּךָ יִשְׂרָאֵל וְנִכַּפֵּר לָהֶם הַדָּם׃ וְאַתָּה
31   תְּבַעֵר הַדָּם הַנָּקִי מִקִּרְבֶּךָ כִּי־תַעֲשֶׂה הַיָּשָׁר בְּעֵינֵי
32   יְהוָה׃ [ס]            כִּי־תֵצֵא לַמִּלְחָמָה עַל־
33  אֹיְבֶיךָ וּנְתָנוֹ יְהוָה אֱלֹהֶיךָ בְּיָדֶךָ וְשָׁבִיתָ שִׁבְיוֹ׃ וְרָאִיתָ
34   בַּשִּׁבְיָה אֵשֶׁת יְפַת־תֹּאַר וְחָשַׁקְתָּ בָהּ וְלָקַחְתָּ לְךָ
35   לְאִשָּׁה׃ וַהֲבֵאתָהּ אֶל־תּוֹךְ בֵּיתֶךָ וְגִלְּחָה אֶת־רֹאשָׁהּ
36   וְעָשְׂתָה אֶת־צִפָּרְנֶיהָ׃ וְהֵסִירָה אֶת־שִׂמְלַת שִׁבְיָהּ
37   מֵעָלֶיהָ וְיָשְׁבָה בְּבֵיתֶךָ וּבָכְתָה אֶת־אָבִיהָ וְאֶת־אִמָּהּ
38   יֶרַח יָמִים וְאַחַר כֵּן תָּבוֹא אֵלֶיהָ וּבְעַלְתָּהּ וְהָיְתָה
39   לְךָ לְאִשָּׁה׃ וְהָיָה אִם־לֹא חָפַצְתָּ בָּהּ וְשִׁלַּחְתָּהּ
40   לְנַפְשָׁהּ וּמָכֹר לֹא־תִמְכְּרֶנָּה בַּכָּסֶף לֹא־תִתְעַמֵּר
41   בָּהּ תַּחַת אֲשֶׁר עִנִּיתָהּ׃♢ [ס]             כִּי־תִהְיֶיןָ
42   לְאִישׁ שְׁתֵּי נָשִׁים הָאַחַת אֲהוּבָה וְהָאַחַת שְׂנוּאָה
