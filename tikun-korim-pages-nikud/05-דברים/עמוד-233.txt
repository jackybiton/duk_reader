 1  וְלָקַח הַכֹּהֵן הַטֶּנֶא מִיָּדֶךָ וְהִנִּיחוֹ לִפְנֵי מִזְבַּח יְהוָה
 2   אֱלֹהֶיךָ׃ וְעָנִיתָ וְאָמַרְתָּ לִפְנֵי  יְהוָה אֱלֹהֶיךָ אֲרַמִּי אֹבֵד
 3   אָבִי וַיֵּרֶד מִצְרַיְמָה וַיָּגָר שָׁם בִּמְתֵי מְעָט וַיְהִי־שָׁם
 4   לְגוֹי גָּדוֹל עָצוּם וָרָב׃ וַיָּרֵעוּ אֹתָנוּ הַמִּצְרִים וַיְעַנּוּנוּ
 5   וַיִּתְּנוּ עָלֵינוּ עֲבֹדָה קָשָׁה׃ וַנִּצְעַק אֶל־יְהוָה אֱלֹהֵי
 6   אֲבֹתֵינוּ וַיִּשְׁמַע יְהוָה אֶת־קֹלֵנוּ וַיַּרְא אֶת־עָנְיֵנוּ וְאֶת־
 7  עֲמָלֵנוּ וְאֶת־לַחֲצֵנוּ׃ וַיּוֹצִאֵנוּ יְהוָה מִמִּצְרַיִם בְּיָד חֲזָקָה
 8   וּבִזְרֹעַ נְטוּיָה וּבְמֹרָא גָּדֹל וּבְאֹתוֹת וּבְמֹפְתִים׃ וַיְבִאֵנוּ
 9   אֶל־הַמָּקוֹם הַזֶּה וַיִּתֶּן־לָנוּ אֶת־הָאָרֶץ הַזֹּאת אֶרֶץ זָבַת
10   חָלָב וּדְבָשׁ׃ וְעַתָּה הִנֵּה הֵבֵאתִי אֶת־רֵאשִׁית פְּרִי
11   הָאֲדָמָה אֲשֶׁר־נָתַתָּה לִּי יְהוָה וְהִנַּחְתּוֹ לִפְנֵי יְהוָה
12   אֱלֹהֶיךָ וְהִשְׁתַּחֲוִיתָ לִפְנֵי יְהוָה אֱלֹהֶיךָ׃ וְשָׂמַחְתָּ בְכָל־
13  הַטּוֹב אֲשֶׁר נָתַן־לְךָ יְהוָה אֱלֹהֶיךָ וּלְבֵיתֶךָ אַתָּה
14   וְהַלֵּוִי וְהַגֵּר אֲשֶׁר בְּקִרְבֶּךָ׃♢ [ס]              כִּי
15   תְכַלֶּה לַעְשֵׂר אֶת־כָּל־מַעְשַׂר תְּבוּאָתְךָ בַּשָּׁנָה
16   הַשְּׁלִישִׁת שְׁנַת הַמַּעֲשֵׂר וְנָתַתָּה לַלֵּוִי לַגֵּר לַיָּתוֹם
17   וְלָאַלְמָנָה וְאָכְלוּ בִשְׁעָרֶיךָ וְשָׂבֵעוּ׃ וְאָמַרְתָּ לִפְנֵי
18   יְהוָה אֱלֹהֶיךָ בִּעַרְתִּי הַקֹּדֶשׁ מִן־הַבַּיִת וְגַם נְתַתִּיו
19   לַלֵּוִי וְלַגֵּר לַיָּתוֹם וְלָאַלְמָנָה כְּכָל־מִצְוָתְךָ אֲשֶׁר
20   צִוִּיתָנִי לֹא־עָבַרְתִּי מִמִּצְו͏ֹתֶיךָ וְלֹא שָׁכָחְתִּי׃ לֹא־
21  אָכַלְתִּי בְאֹנִי מִמֶּנּוּ וְלֹא־בִעַרְתִּי מִמֶּנּוּ בְּטָמֵא וְלֹא־
22  נָתַתִּי מִמֶּנּוּ לְמֵת שָׁמַעְתִּי בְּקוֹל יְהוָה אֱלֹהָי עָשִׂיתִי
23   כְּכֹל אֲשֶׁר צִוִּיתָנִי׃ הַשְׁקִיפָה מִמְּעוֹן קָדְשְׁךָ מִן־
24  הַשָּׁמַיִם וּבָרֵךְ אֶת־עַמְּךָ אֶת־יִשְׂרָאֵל וְאֵת הָאֲדָמָה
25   אֲשֶׁר נָתַתָּה לָנוּ כַּאֲשֶׁר נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ אֶרֶץ
26   זָבַת חָלָב וּדְבָשׁ׃♢ [ס]              הַיּוֹם הַזֶּה
27   יְהוָה אֱלֹהֶיךָ מְצַוְּךָ לַעֲשׂוֹת אֶת־הַחֻקִּים הָאֵלֶּה
28   וְאֶת־הַמִּשְׁפָּטִים וְשָׁמַרְתָּ וְעָשִׂיתָ אוֹתָם בְּכָל־לְבָבְךָ
29   וּבְכָל־נַפְשֶׁךָ׃ אֶת־יְהוָה הֶאֱמַרְתָּ הַיּוֹם לִהְיוֹת לְךָ
30   לֵאלֹהִים וְלָלֶכֶת בִּדְרָכָיו וְלִשְׁמֹר חֻקָּיו וּמִצְו͏ֹתָיו
31   וּמִשְׁפָּטָיו וְלִשְׁמֹעַ בְּקֹלוֹ׃ וַיהוָה הֶאֱמִירְךָ הַיּוֹם
32   לִהְיוֹת לוֹ לְעַם סְגֻלָּה כַּאֲשֶׁר דִּבֶּר־לָךְ וְלִשְׁמֹר כָּל־
33  מִצְו͏ֹתָיו׃ וּלְתִתְּךָ עֶלְיוֹן עַל כָּל־הַגּוֹיִם אֲשֶׁר עָשָׂה
34   לִתְהִלָּה וּלְשֵׁם וּלְתִפְאָרֶת וְלִהְיֹתְךָ עַם־קָדֹשׁ
35   לַיהוָה אֱלֹהֶיךָ כַּאֲשֶׁר דִּבֵּר׃ [פ]
36    וַיְצַו מֹשֶׁה וְזִקְנֵי יִשְׂרָאֵל אֶת־הָעָם לֵאמֹר שָׁמֹר
37   אֶת־כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם׃
38   וְהָיָה בַּיּוֹם אֲשֶׁר תַּעַבְרוּ אֶת־הַיַּרְדֵּן אֶל־הָאָרֶץ
39   אֲשֶׁר־יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ וַהֲקֵמֹתָ לְךָ אֲבָנִים
40   גְּדֹלוֹת וְשַׂדְתָּ אֹתָם בַּשִּׂיד׃ וְכָתַבְתָּ עֲלֵיהֶן אֶת־כָּל־
41  דִּבְרֵי הַתּוֹרָה הַזֹּאת בְּעָבְרֶךָ לְמַעַן אֲשֶׁר תָּבֹא אֶל־
42  הָאָרֶץ אֲשֶׁר־יְהוָה אֱלֹהֶיךָ  נֹתֵן לְךָ אֶרֶץ זָבַת חָלָב
