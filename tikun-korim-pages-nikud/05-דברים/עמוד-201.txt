 1  וַיְהִי בְּאַרְבָּעִים שָׁנָה בְּעַשְׁתֵּי־עָשָׂר חֹדֶשׁ בְּאֶחָד
 2   לַחֹדֶשׁ דִּבֶּר מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל כְּכֹל אֲשֶׁר
 3   צִוָּה יְהוָה אֹתוֹ אֲלֵהֶם׃♢ אַחֲרֵי הַכֹּתוֹ אֵת סִיחֹן מֶלֶךְ
 4   הָאֱמֹרִי אֲשֶׁר יוֹשֵׁב בְּחֶשְׁבּוֹן וְאֵת עוֹג מֶלֶךְ הַבָּשָׁן
 5   אֲשֶׁר־יוֹשֵׁב בְּעַשְׁתָּרֹת בְּאֶדְרֶעִי׃ בְּעֵבֶר הַיַּרְדֵּן
 6   בְּאֶרֶץ מוֹאָב הוֹאִיל מֹשֶׁה בֵּאֵר אֶת־הַתּוֹרָה הַזֹּאת
 7   לֵאמֹר׃ יְהוָה אֱלֹהֵינוּ דִּבֶּר אֵלֵינוּ בְּחֹרֵב לֵאמֹר רַב־
 8  לָכֶם שֶׁבֶת בָּהָר הַזֶּה׃ פְּנוּ  וּסְעוּ לָכֶם וּבֹאוּ הַר
 9   הָאֱמֹרִי וְאֶל־כָּל־שְׁכֵנָיו בָּעֲרָבָה בָהָר וּבַשְּׁפֵלָה
10   וּבַנֶּגֶב וּבְחוֹף הַיָּם אֶרֶץ הַכְּנַעֲנִי וְהַלְּבָנוֹן עַד־הַנָּהָר
11   הַגָּדֹל נְהַר־פְּרָת׃♢ רְאֵה נָתַתִּי לִפְנֵיכֶם אֶת־הָאָרֶץ בֹּאוּ
12   וּרְשׁוּ אֶת־הָאָרֶץ אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֵיכֶם
13   לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב לָתֵת לָהֶם וּלְזַרְעָם
14   אַחֲרֵיהֶם׃ וָאֹמַר אֲלֵכֶם בָּעֵת הַהִוא לֵאמֹר לֹא־אוּכַל
15   לְבַדִּי שְׂאֵת אֶתְכֶם׃ יְהוָה אֱלֹהֵיכֶם הִרְבָּה אֶתְכֶם
16   וְהִנְּכֶם הַיּוֹם כְּכוֹכְבֵי הַשָּׁמַיִם לָרֹב׃ יְהוָה אֱלֹהֵי
17   אֲבוֹתֵכֶם יֹסֵף עֲלֵיכֶם כָּכֶם אֶלֶף פְּעָמִים וִיבָרֵךְ
18   אֶתְכֶם כַּאֲשֶׁר דִּבֶּר לָכֶם׃♢  אֵיכָה אֶשָּׂא לְבַדִּי טָרְחֲכֶם
19   וּמַשַּׂאֲכֶם וְרִיבְכֶם׃ הָבוּ לָכֶם אֲנָשִׁים חֲכָמִים וּנְבֹנִים
20   וִידֻעִים לְשִׁבְטֵיכֶם וַאֲשִׂימֵם בְּרָאשֵׁיכֶם׃ וַתַּעֲנוּ
21   אֹתִי וַתֹּאמְרוּ טוֹב־הַדָּבָר אֲשֶׁר־דִּבַּרְתָּ לַעֲשׂוֹת׃
22   וָאֶקַּח אֶת־רָאשֵׁי שִׁבְטֵיכֶם אֲנָשִׁים חֲכָמִים וִידֻעִים
23   וָאֶתֵּן אוֹתָם רָאשִׁים עֲלֵיכֶם שָׂרֵי אֲלָפִים וְשָׂרֵי
24   מֵאוֹת וְשָׂרֵי חֲמִשִּׁים וְשָׂרֵי עֲשָׂרֹת וְשֹׁטְרִים
25   לְשִׁבְטֵיכֶם׃ וָאֲצַוֶּה אֶת־שֹׁפְטֵיכֶם בָּעֵת הַהִוא לֵאמֹר
26   שָׁמֹעַ בֵּין־אֲחֵיכֶם וּשְׁפַטְתֶּם צֶדֶק בֵּין־אִישׁ וּבֵין־
27  אָחִיו וּבֵין גֵּרוֹ׃ לֹא־תַכִּירוּ פָנִים בַּמִּשְׁפָּט כַּקָּטֹן
28   כַּגָּדֹל תִּשְׁמָעוּן לֹא תָגוּרוּ מִפְּנֵי־אִישׁ כִּי הַמִּשְׁפָּט
29   לֵאלֹהִים הוּא וְהַדָּבָר אֲשֶׁר יִקְשֶׁה מִכֶּם תַּקְרִבוּן
30   אֵלַי וּשְׁמַעְתִּיו׃ וָאֲצַוֶּה אֶתְכֶם בָּעֵת הַהִוא אֵת כָּל־
31  הַדְּבָרִים אֲשֶׁר תַּעֲשׂוּן׃ וַנִּסַּע מֵחֹרֵב וַנֵּלֶךְ אֵת כָּל־
32  הַמִּדְבָּר הַגָּדוֹל וְהַנּוֹרָא הַהוּא אֲשֶׁר רְאִיתֶם דֶּרֶךְ
33   הַר הָאֱמֹרִי כַּאֲשֶׁר צִוָּה יְהוָה אֱלֹהֵינוּ אֹתָנוּ וַנָּבֹא עַד
34   קָדֵשׁ בַּרְנֵעַ׃ וָאֹמַר אֲלֵכֶם בָּאתֶם עַד־הַר הָאֱמֹרִי
35   אֲשֶׁר־יְהוָה אֱלֹהֵינוּ נֹתֵן לָנוּ׃ רְאֵה נָתַן יְהוָה אֱלֹהֶיךָ
36   לְפָנֶיךָ אֶת־הָאָרֶץ עֲלֵה רֵשׁ כַּאֲשֶׁר דִּבֶּר יְהוָה
37   אֱלֹהֵי אֲבֹתֶיךָ לָךְ אַל־תִּירָא וְאַל־תֵּחָת׃  וַתִּקְרְבוּן אֵלַי
38   כֻּלְּכֶם וַתֹּאמְרוּ נִשְׁלְחָה אֲנָשִׁים לְפָנֵינוּ וְיַחְפְּרוּ־לָנוּ
39   אֶת־הָאָרֶץ וְיָשִׁבוּ אֹתָנוּ דָּבָר אֶת־הַדֶּרֶךְ אֲשֶׁר נַעֲלֶה־
40  בָּהּ וְאֵת הֶעָרִים אֲשֶׁר נָבֹא אֲלֵיהֶן׃ וַיִּיטַב בְּעֵינַי
41   הַדָּבָר וָאֶקַּח מִכֶּם שְׁנֵים עָשָׂר אֲנָשִׁים אִישׁ אֶחָד
42   לַשָּׁבֶט׃ וַיִּפְנוּ וַיַּעֲלוּ הָהָרָה וַיָּבֹאוּ עַד־נַחַל אֶשְׁכֹּל
