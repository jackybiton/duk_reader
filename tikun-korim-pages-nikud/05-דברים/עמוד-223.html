1וְהָאַלְמָנָהאֲשֶׁרבִּשְׁעָרֶיךָ׃שִׁבְעַתיָמִיםתָּחֹגלַיהוָה
2אֱלֹהֶיךָבַּמָּקוֹםאֲשֶׁר־יִבְחַריְהוָהכִּייְבָרֶכְךָיְהוָה
3אֱלֹהֶיךָבְּכֹלתְּבוּאָתְךָוּבְכֹלמַעֲשֵׂהיָדֶיךָוְהָיִיתָאַךְ

5פְּנֵייְהוָהאֱלֹהֶיךָבַּמָּקוֹםאֲשֶׁריִבְחָרבְּחַגהַמַּצּוֹת
6וּבְחַגהַשָּׁבֻעוֹתוּבְחַגהַסֻּכּוֹתוְלֹאיֵרָאֶהאֶת־פְּנֵי
7יְהוָהרֵיקָם׃אִישׁכְּמַתְּנַתיָדוֹכְּבִרְכַּתיְהוָהאֱלֹהֶיךָ

9תִּתֶּן־לְךָבְּכָל־שְׁעָרֶיךָאֲשֶׁריְהוָהאֱלֹהֶיךָנֹתֵןלְךָ
10לִשְׁבָטֶיךָוְשָׁפְטוּאֶת־הָעָםמִשְׁפַּט־צֶדֶק׃לֹא־תַטֶּה
11מִשְׁפָּטלֹאתַכִּירפָּנִיםוְלֹא־תִקַּחשֹׁחַדכִּיהַשֹּׁחַד
12יְעַוֵּרעֵינֵיחֲכָמִיםוִיסַלֵּףדִּבְרֵיצַדִּיקִם׃צֶדֶקצֶדֶק
13תִּרְדֹּףלְמַעַןתִּחְיֶהוְיָרַשְׁתָּאֶת־הָאָרֶץאֲשֶׁר־יְהוָה

16תַּעֲשֶׂה־לָּךְ׃וְלֹא־תָקִיםלְךָמַצֵּבָהאֲשֶׁרשָׂנֵאיְהוָה

21נֹתֵןלָךְאִישׁאוֹ־אִשָּׁהאֲשֶׁריַעֲשֶׂהאֶת־הָרַעבְּעֵינֵי
22יְהוָה־אֱלֹהֶיךָלַעֲבֹרבְּרִיתוֹ׃וַיֵּלֶךְוַיַּעֲבֹדאֱלֹהִים

25וְדָרַשְׁתָּהֵיטֵבוְהִנֵּהאֱמֶתנָכוֹןהַדָּבָרנֶעֶשְׂתָה
26הַתּוֹעֵבָההַזֹּאתבְּיִשְׂרָאֵל׃וְהוֹצֵאתָאֶת־הָאִישׁהַהוּא
27אוֹאֶת־הָאִשָּׁההַהִואאֲשֶׁרעָשׂוּאֶת־הַדָּבָרהָרָע
28הַזֶּהאֶל־שְׁעָרֶיךָאֶת־הָאִישׁאוֹאֶת־הָאִשָּׁהוּסְקַלְתָּם
