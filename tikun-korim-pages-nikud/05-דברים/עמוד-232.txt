 1  וְנִגְּשׁוּ אֶל־הַמִּשְׁפָּט וּשְׁפָטוּם וְהִצְדִּיקוּ אֶת־הַצַּדִּיק
 2   וְהִרְשִׁיעוּ אֶת־הָרָשָׁע׃ וְהָיָה אִם־בִּן הַכּוֹת הָרָשָׁע
 3   וְהִפִּילוֹ הַשֹּׁפֵט וְהִכָּהוּ לְפָנָיו כְּדֵי רִשְׁעָתוֹ בְּמִסְפָּר׃
 4   אַרְבָּעִים יַכֶּנּוּ לֹא יֹסִיף פֶּן־יֹסִיף לְהַכֹּתוֹ עַל־אֵלֶּה
 5   מַכָּה רַבָּה וְנִקְלָה אָחִיךָ לְעֵינֶיךָ׃ לֹא־תַחְסֹם שׁוֹר
 6   בְּדִישׁוֹ׃ [ס]             כִּי־יֵשְׁבוּ אַחִים
 7   יַחְדָּו וּמֵת אַחַד מֵהֶם וּבֵן אֵין־לוֹ לֹא־תִהְיֶה אֵשֶׁת־
 8  הַמֵּת הַחוּצָה לְאִישׁ זָר יְבָמָהּ יָבֹא עָלֶיהָ וּלְקָחָהּ
 9   לוֹ לְאִשָּׁה וְיִבְּמָהּ׃ וְהָיָה הַבְּכוֹר אֲשֶׁר תֵּלֵד יָקוּם
10   עַל־שֵׁם אָחִיו הַמֵּת וְלֹא־יִמָּחֶה שְׁמוֹ מִיִּשְׂרָאֵל׃
11   וְאִם־לֹא יַחְפֹּץ הָאִישׁ לָקַחַת אֶת־יְבִמְתּוֹ וְעָלְתָה
12   יְבִמְתּוֹ הַשַּׁעְרָה אֶל־הַזְּקֵנִים וְאָמְרָה מֵאֵן יְבָמִי
13   לְהָקִים לְאָחִיו שֵׁם בְּיִשְׂרָאֵל לֹא אָבָה יַבְּמִי׃
14   וְקָרְאוּ־לוֹ זִקְנֵי־עִירוֹ וְדִבְּרוּ אֵלָיו וְעָמַד וְאָמַר
15   לֹא חָפַצְתִּי לְקַחְתָּהּ׃ וְנִגְּשָׁה יְבִמְתּוֹ אֵלָיו לְעֵינֵי
16   הַזְּקֵנִים וְחָלְצָה נַעֲלוֹ מֵעַל רַגְלוֹ וְיָרְקָה בְּפָנָיו
17   וְעָנְתָה וְאָמְרָה כָּכָה יֵעָשֶׂה לָאִישׁ אֲשֶׁר לֹא־יִבְנֶה
18   אֶת־בֵּית אָחִיו׃ וְנִקְרָא שְׁמוֹ בְּיִשְׂרָאֵל בֵּית חֲלוּץ
19   הַנָּעַל׃ [ס]             כִּי־יִנָּצוּ אֲנָשִׁים יַחְדָּו
20   אִישׁ וְאָחִיו וְקָרְבָה אֵשֶׁת הָאֶחָד לְהַצִּיל אֶת־אִישָׁהּ
21   מִיַּד מַכֵּהוּ וְשָׁלְחָה יָדָהּ וְהֶחֱזִיקָה בִּמְבֻשָׁיו׃ וְקַצֹּתָה
22   אֶת־כַּפָּהּ לֹא תָחוֹס עֵינֶךָ׃ [ס]             לֹא־
23  יִהְיֶה לְךָ בְּכִיסְךָ אֶבֶן וָאָבֶן גְּדוֹלָה וּקְטַנָּה׃ לֹא־
24  יִהְיֶה לְךָ בְּבֵיתְךָ אֵיפָה וְאֵיפָה גְּדוֹלָה וּקְטַנָּה׃ אֶבֶן
25   שְׁלֵמָה וָצֶדֶק יִהְיֶה־לָּךְ אֵיפָה שְׁלֵמָה וָצֶדֶק יִהְיֶה־
26  לָּךְ לְמַעַן יַאֲרִיכוּ יָמֶיךָ עַל הָאֲדָמָה אֲשֶׁר־יְהוָה
27   אֱלֹהֶיךָ נֹתֵן לָךְ׃ כִּי תוֹעֲבַת יְהוָה אֱלֹהֶיךָ כָּל־עֹשֵׂה
28   אֵלֶּה כֹּל עֹשֵׂה עָוֶל׃ [פ]
29    זָכוֹר אֵת אֲשֶׁר־עָשָׂה לְךָ עֲמָלֵק בַּדֶּרֶךְ בְּצֵאתְכֶם
30   מִמִּצְרָיִם׃ אֲשֶׁר קָרְךָ בַּדֶּרֶךְ וַיְזַנֵּב בְּךָ כָּל־הַנֶּחֱשָׁלִים
31   אַחֲרֶיךָ וְאַתָּה עָיֵף וְיָגֵעַ וְלֹא יָרֵא אֱלֹהִים׃ וְהָיָה בְּהָנִיחַ
32   יְהוָה אֱלֹהֶיךָ  לְךָ מִכָּל־אֹיְבֶיךָ מִסָּבִיב בָּאָרֶץ אֲשֶׁר
33   יְהוָה־אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה לְרִשְׁתָּהּ תִּמְחֶה אֶת־זֵכֶר
34   עֲמָלֵק מִתַּחַת הַשָּׁמָיִם לֹא תִּשְׁכָּח׃ [פ]
35  וְהָיָה כִּי־תָבוֹא אֶל־הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן
36   לְךָ נַחֲלָה וִירִשְׁתָּהּ וְיָשַׁבְתָּ בָּהּ׃ וְלָקַחְתָּ מֵרֵאשִׁית
37   כָּל־פְּרִי הָאֲדָמָה אֲשֶׁר תָּבִיא מֵאַרְצְךָ אֲשֶׁר
38   יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ וְשַׂמְתָּ בַטֶּנֶא וְהָלַכְתָּ אֶל־
39  הַמָּקוֹם אֲשֶׁר יִבְחַר יְהוָה אֱלֹהֶיךָ לְשַׁכֵּן שְׁמוֹ
40   שָׁם׃ וּבָאתָ אֶל־הַכֹּהֵן אֲשֶׁר יִהְיֶה בַּיָּמִים הָהֵם
41   וְאָמַרְתָּ אֵלָיו הִגַּדְתִּי הַיּוֹם לַיהוָה אֱלֹהֶיךָ כִּי־בָאתִי
42   אֶל־הָאָרֶץ אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֵינוּ לָתֶת לָנוּ׃♢
