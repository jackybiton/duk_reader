 1  וְהָאַלְמָנָה אֲשֶׁר בִּשְׁעָרֶיךָ׃ שִׁבְעַת יָמִים תָּחֹג לַיהוָה
 2   אֱלֹהֶיךָ בַּמָּקוֹם אֲשֶׁר־יִבְחַר יְהוָה כִּי יְבָרֶכְךָ יְהוָה
 3   אֱלֹהֶיךָ בְּכֹל תְּבוּאָתְךָ וּבְכֹל מַעֲשֵׂה יָדֶיךָ וְהָיִיתָ אַךְ
 4   שָׂמֵחַ׃ שָׁלוֹשׁ פְּעָמִים  בַּשָּׁנָה יֵרָאֶה כָל־זְכוּרְךָ אֶת־
 5  פְּנֵי  יְהוָה אֱלֹהֶיךָ בַּמָּקוֹם אֲשֶׁר יִבְחָר בְּחַג הַמַּצּוֹת
 6   וּבְחַג הַשָּׁבֻעוֹת וּבְחַג הַסֻּכּוֹת וְלֹא יֵרָאֶה אֶת־פְּנֵי
 7   יְהוָה רֵיקָם׃ אִישׁ כְּמַתְּנַת יָדוֹ כְּבִרְכַּת יְהוָה אֱלֹהֶיךָ
 8   אֲשֶׁר נָתַן־לָךְ׃ [ס]            שֹׁפְטִים וְשֹׁטְרִים
 9   תִּתֶּן־לְךָ בְּכָל־שְׁעָרֶיךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ
10   לִשְׁבָטֶיךָ וְשָׁפְטוּ אֶת־הָעָם מִשְׁפַּט־צֶדֶק׃ לֹא־תַטֶּה
11   מִשְׁפָּט לֹא תַכִּיר פָּנִים וְלֹא־תִקַּח שֹׁחַד כִּי הַשֹּׁחַד
12   יְעַוֵּר עֵינֵי חֲכָמִים וִיסַלֵּף דִּבְרֵי צַדִּיקִם׃ צֶדֶק צֶדֶק
13   תִּרְדֹּף לְמַעַן תִּחְיֶה וְיָרַשְׁתָּ אֶת־הָאָרֶץ אֲשֶׁר־יְהוָה
14   אֱלֹהֶיךָ נֹתֵן לָךְ׃♢ [ס]             לֹא־תִטַּע
15   לְךָ אֲשֵׁרָה כָּל־עֵץ אֵצֶל מִזְבַּח יְהוָה אֱלֹהֶיךָ אֲשֶׁר
16   תַּעֲשֶׂה־לָּךְ׃ וְלֹא־תָקִים לְךָ מַצֵּבָה אֲשֶׁר שָׂנֵא יְהוָה
17   אֱלֹהֶיךָ׃ [ס]             לֹא־תִזְבַּח לַיהוָה
18   אֱלֹהֶיךָ שׁוֹר וָשֶׂה אֲשֶׁר יִהְיֶה בוֹ מוּם כֹּל דָּבָר רָע
19   כִּי תוֹעֲבַת יְהוָה אֱלֹהֶיךָ הוּא׃ [ס]             כִּי־
20  יִמָּצֵא בְקִרְבְּךָ בְּאַחַד שְׁעָרֶיךָ אֲשֶׁר־יְהוָה אֱלֹהֶיךָ
21   נֹתֵן לָךְ אִישׁ אוֹ־אִשָּׁה אֲשֶׁר יַעֲשֶׂה אֶת־הָרַע בְּעֵינֵי
22   יְהוָה־אֱלֹהֶיךָ לַעֲבֹר בְּרִיתוֹ׃ וַיֵּלֶךְ וַיַּעֲבֹד אֱלֹהִים
23   אֲחֵרִים וַיִּשְׁתַּחוּ לָהֶם וְלַשֶּׁמֶשׁ  אוֹ לַיָּרֵחַ אוֹ לְכָל־
24  צְבָא הַשָּׁמַיִם אֲשֶׁר לֹא־צִוִּיתִי׃ וְהֻגַּד־לְךָ וְשָׁמָעְתָּ
25   וְדָרַשְׁתָּ הֵיטֵב וְהִנֵּה אֱמֶת נָכוֹן הַדָּבָר נֶעֶשְׂתָה
26   הַתּוֹעֵבָה הַזֹּאת בְּיִשְׂרָאֵל׃ וְהוֹצֵאתָ אֶת־הָאִישׁ הַהוּא
27   אוֹ אֶת־הָאִשָּׁה הַהִוא אֲשֶׁר עָשׂוּ אֶת־הַדָּבָר הָרָע
28   הַזֶּה אֶל־שְׁעָרֶיךָ אֶת־הָאִישׁ אוֹ אֶת־הָאִשָּׁה וּסְקַלְתָּם
29   בָּאֲבָנִים וָמֵתוּ׃ עַל־פִּי  שְׁנַיִם עֵדִים אוֹ שְׁלֹשָׁה עֵדִים
30   יוּמַת הַמֵּת לֹא יוּמַת עַל־פִּי עֵד אֶחָד׃ יַד הָעֵדִים
31   תִּהְיֶה־בּוֹ בָרִאשֹׁנָה לַהֲמִיתוֹ וְיַד כָּל־הָעָם בָּאַחֲרֹנָה
32   וּבִעַרְתָּ הָרָע מִקִּרְבֶּךָ׃ [פ]
33   כִּי יִפָּלֵא מִמְּךָ דָבָר לַמִּשְׁפָּט בֵּין־דָּם  לְדָם בֵּין־
34  דִּין לְדִין וּבֵין נֶגַע לָנֶגַע דִּבְרֵי רִיבֹת בִּשְׁעָרֶיךָ
35   וְקַמְתָּ וְעָלִיתָ אֶל־הַמָּקוֹם אֲשֶׁר יִבְחַר יְהוָה אֱלֹהֶיךָ
36   בּוֹ׃ וּבָאתָ אֶל־הַכֹּהֲנִים הַלְוִיִּם וְאֶל־הַשֹּׁפֵט אֲשֶׁר
37   יִהְיֶה בַּיָּמִים הָהֵם וְדָרַשְׁתָּ וְהִגִּידוּ לְךָ אֵת דְּבַר
38   הַמִּשְׁפָּט׃ וְעָשִׂיתָ עַל־פִּי הַדָּבָר אֲשֶׁר יַגִּידוּ לְךָ
39   מִן־הַמָּקוֹם הַהוּא אֲשֶׁר יִבְחַר יְהוָה וְשָׁמַרְתָּ
40   לַעֲשׂוֹת כְּכֹל אֲשֶׁר יוֹרוּךָ׃♢ עַל־פִּי הַתּוֹרָה אֲשֶׁר
41   יוֹרוּךָ וְעַל־הַמִּשְׁפָּט אֲשֶׁר־יֹאמְרוּ לְךָ תַּעֲשֶׂה לֹא
42   תָסוּר מִן־הַדָּבָר אֲשֶׁר־יַגִּידוּ לְךָ יָמִין וּשְׂמֹאל׃
