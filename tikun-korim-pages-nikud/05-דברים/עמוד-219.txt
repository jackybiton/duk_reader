 1  וְאֶת־בְּנֹתֵיהֶם יִשְׂרְפוּ בָאֵשׁ לֵאלֹהֵיהֶם׃ אֵת כָּל־הַדָּבָר
 2   אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם אֹתוֹ תִשְׁמְרוּ לַעֲשׂוֹת לֹא־
 3  תֹסֵף עָלָיו וְלֹא תִגְרַע מִמֶּנּוּ׃ [פ]
 4   כִּי־יָקוּם בְּקִרְבְּךָ נָבִיא אוֹ חֹלֵם חֲלוֹם וְנָתַן אֵלֶיךָ
 5   אוֹת אוֹ מוֹפֵת׃ וּבָא הָאוֹת וְהַמּוֹפֵת אֲשֶׁר־דִּבֶּר
 6   אֵלֶיךָ לֵאמֹר נֵלְכָה אַחֲרֵי אֱלֹהִים אֲחֵרִים אֲשֶׁר
 7   לֹא־יְדַעְתָּם וְנָעָבְדֵם׃ לֹא תִשְׁמַע אֶל־דִּבְרֵי הַנָּבִיא
 8   הַהוּא אוֹ אֶל־חוֹלֵם הַחֲלוֹם הַהוּא כִּי מְנַסֶּה יְהוָה
 9   אֱלֹהֵיכֶם אֶתְכֶם לָדַעַת הֲיִשְׁכֶם אֹהֲבִים אֶת־יְהוָה
10   אֱלֹהֵיכֶם בְּכָל־לְבַבְכֶם וּבְכָל־נַפְשְׁכֶם׃ אַחֲרֵי
11   יְהוָה אֱלֹהֵיכֶם תֵּלֵכוּ וְאֹתוֹ תִירָאוּ וְאֶת־מִצְו͏ֹתָיו
12   תִּשְׁמֹרוּ וּבְקֹלוֹ תִשְׁמָעוּ וְאֹתוֹ תַעֲבֹדוּ וּבוֹ תִדְבָּקוּן׃
13   וְהַנָּבִיא הַהוּא אוֹ חֹלֵם הַחֲלוֹם הַהוּא יוּמָת כִּי
14   דִבֶּר־סָרָה עַל־יְהוָה אֱלֹהֵיכֶם הַמּוֹצִיא אֶתְכֶם
15   מֵאֶרֶץ מִצְרַיִם וְהַפֹּדְךָ מִבֵּית עֲבָדִים לְהַדִּיחֲךָ מִן־
16  הַדֶּרֶךְ אֲשֶׁר צִוְּךָ יְהוָה אֱלֹהֶיךָ לָלֶכֶת בָּהּ וּבִעַרְתָּ
17   הָרָע מִקִּרְבֶּךָ׃ [ס]             כִּי
18   יְסִיתְךָ אָחִיךָ בֶן־אִמֶּךָ אוֹ־בִנְךָ אוֹ־בִתְּךָ אוֹ  אֵשֶׁת
19   חֵיקֶךָ אוֹ רֵעֲךָ אֲשֶׁר כְּנַפְשְׁךָ בַּסֵּתֶר לֵאמֹר נֵלְכָה
20   וְנַעַבְדָה אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא יָדַעְתָּ אַתָּה
21   וַאֲבֹתֶיךָ׃ מֵאֱלֹהֵי הָעַמִּים אֲשֶׁר סְבִיבֹתֵיכֶם הַקְּרֹבִים
22   אֵלֶיךָ אוֹ הָרְחֹקִים מִמֶּךָּ מִקְצֵה הָאָרֶץ וְעַד־קְצֵה
23   הָאָרֶץ׃ לֹא־תֹאבֶה לוֹ וְלֹא תִשְׁמַע אֵלָיו וְלֹא־
24  תָחוֹס עֵינְךָ עָלָיו וְלֹא־תַחְמֹל וְלֹא־תְכַסֶּה עָלָיו׃
25   כִּי הָרֹג תַּהַרְגֶנּוּ יָדְךָ תִּהְיֶה־בּוֹ בָרִאשׁוֹנָה לַהֲמִיתוֹ
26   וְיַד כָּל־הָעָם בָּאַחֲרֹנָה׃ וּסְקַלְתּוֹ בָאֲבָנִים וָמֵת
27   כִּי בִקֵּשׁ לְהַדִּיחֲךָ מֵעַל יְהוָה אֱלֹהֶיךָ הַמּוֹצִיאֲךָ
28   מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים׃ וְכָל־יִשְׂרָאֵל יִשְׁמְעוּ
29   וְיִרָאוּן וְלֹא־יוֹסִפוּ לַעֲשׂוֹת כַּדָּבָר הָרָע הַזֶּה
30   בְּקִרְבֶּךָ׃ [ס]             כִּי־תִשְׁמַע בְּאַחַת
31   עָרֶיךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ לָשֶׁבֶת שָׁם
32   לֵאמֹר׃ יָצְאוּ אֲנָשִׁים בְּנֵי־בְלִיַּעַל מִקִּרְבֶּךָ וַיַּדִּיחוּ
33   אֶת־יֹשְׁבֵי עִירָם לֵאמֹר נֵלְכָה וְנַעַבְדָה אֱלֹהִים
34   אֲחֵרִים אֲשֶׁר לֹא־יְדַעְתֶּם׃ וְדָרַשְׁתָּ וְחָקַרְתָּ וְשָׁאַלְתָּ
35   הֵיטֵב וְהִנֵּה אֱמֶת נָכוֹן הַדָּבָר נֶעֶשְׂתָה הַתּוֹעֵבָה
36   הַזֹּאת בְּקִרְבֶּךָ׃ הַכֵּה תַכֶּה אֶת־יֹשְׁבֵי הָעִיר הַהִוא
37   לְפִי־חָרֶב הַחֲרֵם אֹתָהּ וְאֶת־כָּל־אֲשֶׁר־בָּהּ וְאֶת־
38  בְּהֶמְתָּהּ לְפִי־חָרֶב׃ וְאֶת־כָּל־שְׁלָלָהּ תִּקְבֹּץ אֶל־
39  תּוֹךְ רְחֹבָהּ וְשָׂרַפְתָּ בָאֵשׁ אֶת־הָעִיר וְאֶת־כָּל־
40  שְׁלָלָהּ כָּלִיל לַיהוָה אֱלֹהֶיךָ וְהָיְתָה תֵּל עוֹלָם
41   לֹא תִבָּנֶה עוֹד׃ וְלֹא־יִדְבַּק בְּיָדְךָ מְאוּמָה מִן־הַחֵרֶם
42   לְמַעַן יָשׁוּב יְהוָה מֵחֲרוֹן אַפּוֹ וְנָתַן־לְךָ רַחֲמִים
