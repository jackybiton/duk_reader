 1  וְהַנִּסְתָּרִים מִפָּנֶיךָ׃ לֹא תַעֲרֹץ מִפְּנֵיהֶם כִּי־יְהוָה
 2   אֱלֹהֶיךָ בְּקִרְבֶּךָ אֵל גָּדוֹל וְנוֹרָא׃♢ וְנָשַׁל יְהוָה אֱלֹהֶיךָ
 3   אֶת־הַגּוֹיִם הָאֵל מִפָּנֶיךָ מְעַט מְעָט לֹא תוּכַל כַּלֹּתָם
 4   מַהֵר פֶּן־תִּרְבֶּה עָלֶיךָ חַיַּת הַשָּׂדֶה׃ וּנְתָנָם יְהוָה
 5   אֱלֹהֶיךָ לְפָנֶיךָ וְהָמָם מְהוּמָה גְדֹלָה עַד הִשָּׁמְדָם׃
 6   וְנָתַן מַלְכֵיהֶם בְּיָדֶךָ וְהַאֲבַדְתָּ אֶת־שְׁמָם מִתַּחַת
 7   הַשָּׁמָיִם לֹא־יִתְיַצֵּב אִישׁ בְּפָנֶיךָ עַד הִשְׁמִדְךָ אֹתָם׃
 8   פְּסִילֵי אֱלֹהֵיהֶם תִּשְׂרְפוּן בָּאֵשׁ לֹא־תַחְמֹד כֶּסֶף
 9   וְזָהָב עֲלֵיהֶם וְלָקַחְתָּ לָךְ פֶּן תִּוָּקֵשׁ בּוֹ כִּי תוֹעֲבַת
10   יְהוָה אֱלֹהֶיךָ הוּא׃ וְלֹא־תָבִיא תוֹעֵבָה אֶל־בֵּיתֶךָ
11   וְהָיִיתָ חֵרֶם כָּמֹהוּ שַׁקֵּץ  תְּשַׁקְּצֶנּוּ וְתַעֵב  תְּתַעֲבֶנּוּ
12   כִּי־חֵרֶם הוּא׃ [פ]
13   כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם תִּשְׁמְרוּן
14   לַעֲשׂוֹת לְמַעַן תִּחְיוּן וּרְבִיתֶם וּבָאתֶם וִירִשְׁתֶּם אֶת־
15  הָאָרֶץ אֲשֶׁר־נִשְׁבַּע יְהוָה לַאֲבֹתֵיכֶם׃ וְזָכַרְתָּ אֶת־
16  כָּל־הַדֶּרֶךְ אֲשֶׁר הוֹלִיכְךָ יְהוָה אֱלֹהֶיךָ זֶה אַרְבָּעִים
17   שָׁנָה בַּמִּדְבָּר לְמַעַן עַנֹּתְךָ לְנַסֹּתְךָ לָדַעַת אֶת־אֲשֶׁר
18   בִּלְבָבְךָ הֲתִשְׁמֹר מִצְו͏ֹתָו אִם־לֹא׃ וַיְעַנְּךָ וַיַּרְעִבֶךָ
19   וַיַּאֲכִלְךָ אֶת־הַמָּן אֲשֶׁר לֹא־יָדַעְתָּ וְלֹא יָדְעוּן אֲבֹתֶיךָ
20   לְמַעַן הוֹדִיעֲךָ כִּי לֹא עַל־הַלֶּחֶם לְבַדּוֹ יִחְיֶה הָאָדָם
21   כִּי עַל־כָּל־מוֹצָא פִי־יְהוָה יִחְיֶה הָאָדָם׃ שִׂמְלָתְךָ
22   לֹא בָלְתָה מֵעָלֶיךָ וְרַגְלְךָ לֹא בָצֵקָה זֶה אַרְבָּעִים
23   שָׁנָה׃♢ וְיָדַעְתָּ עִם־לְבָבֶךָ כִּי כַּאֲשֶׁר יְיַסֵּר אִישׁ אֶת־
24  בְּנוֹ יְהוָה אֱלֹהֶיךָ מְיַסְּרֶךָּ׃ וְשָׁמַרְתָּ אֶת־מִצְו͏ֹת יְהוָה
25   אֱלֹהֶיךָ לָלֶכֶת בִּדְרָכָיו וּלְיִרְאָה אֹתוֹ׃ כִּי יְהוָה
26   אֱלֹהֶיךָ מְבִיאֲךָ אֶל־אֶרֶץ טוֹבָה אֶרֶץ נַחֲלֵי מָיִם
27   עֲיָנֹת וּתְהֹמֹת יֹצְאִים בַּבִּקְעָה וּבָהָר׃ אֶרֶץ חִטָּה
28   וּשְׂעֹרָה וְגֶפֶן וּתְאֵנָה וְרִמּוֹן אֶרֶץ־זֵית שֶׁמֶן וּדְבָשׁ׃
29   אֶרֶץ אֲשֶׁר לֹא בְמִסְכֵּנֻת תֹּאכַל־בָּהּ לֶחֶם לֹא־
30  תֶחְסַר כֹּל בָּהּ אֶרֶץ אֲשֶׁר אֲבָנֶיהָ בַרְזֶל וּמֵהֲרָרֶיהָ
31   תַּחְצֹב נְחֹשֶׁת׃ וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת־יְהוָה
32   אֱלֹהֶיךָ עַל־הָאָרֶץ הַטֹּבָה אֲשֶׁר נָתַן־לָךְ׃♢  הִשָּׁמֶר לְךָ
33   פֶּן־תִּשְׁכַּח אֶת־יְהוָה אֱלֹהֶיךָ לְבִלְתִּי שְׁמֹר מִצְו͏ֹתָיו
34   וּמִשְׁפָּטָיו וְחֻקֹּתָיו אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם׃ פֶּן־
35  תֹּאכַל וְשָׂבָעְתָּ וּבָתִּים טֹבִים תִּבְנֶה וְיָשָׁבְתָּ׃ וּבְקָרְךָ
36   וְצֹאנְךָ יִרְבְּיֻן וְכֶסֶף וְזָהָב יִרְבֶּה־לָּךְ וְכֹל אֲשֶׁר־
37  לְךָ יִרְבֶּה׃ וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת־יְהוָה אֱלֹהֶיךָ
38   הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים׃ הַמּוֹלִיכְךָ
39   בַּמִּדְבָּר  הַגָּדֹל וְהַנּוֹרָא נָחָשׁ  שָׂרָף וְעַקְרָב וְצִמָּאוֹן
40   אֲשֶׁר אֵין־מָיִם הַמּוֹצִיא לְךָ מַיִם מִצּוּר הַחַלָּמִישׁ׃
41   הַמַּאֲכִלְךָ מָן בַּמִּדְבָּר אֲשֶׁר לֹא־יָדְעוּן אֲבֹתֶיךָ
42   לְמַעַן עַנֹּתְךָ וּלְמַעַן נַסֹּתֶךָ לְהֵיטִבְךָ בְּאַחֲרִיתֶךָ׃
