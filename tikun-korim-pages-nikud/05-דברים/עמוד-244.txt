 1  וּבַדָּבָר הַזֶּה תַּאֲרִיכוּ יָמִים עַל־הָאֲדָמָה אֲשֶׁר אַתֶּם
 2   עֹבְרִים אֶת־הַיַּרְדֵּן שָׁמָּה לְרִשְׁתָּהּ׃ [פ]
 3    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה בְּעֶצֶם הַיּוֹם הַזֶּה
 4   לֵאמֹר׃ עֲלֵה אֶל־הַר הָעֲבָרִים הַזֶּה הַר־נְבוֹ
 5   אֲשֶׁר בְּאֶרֶץ מוֹאָב אֲשֶׁר עַל־פְּנֵי יְרֵחוֹ וּרְאֵה
 6   אֶת־אֶרֶץ כְּנַעַן אֲשֶׁר אֲנִי נֹתֵן לִבְנֵי יִשְׂרָאֵל
 7   לַאֲחֻזָּה׃ וּמֻת בָּהָר אֲשֶׁר אַתָּה עֹלֶה שָׁמָּה
 8   וְהֵאָסֵף אֶל־עַמֶּיךָ כַּאֲשֶׁר־מֵת אַהֲרֹן אָחִיךָ
 9   בְּהֹר הָהָר וַיֵּאָסֶף אֶל־עַמָּיו׃ עַל אֲשֶׁר מְעַלְתֶּם
10   בִּי בְּתוֹךְ בְּנֵי יִשְׂרָאֵל בְּמֵי־מְרִיבַת קָדֵשׁ
11   מִדְבַּר־צִן עַל אֲשֶׁר לֹא־קִדַּשְׁתֶּם אוֹתִי בְּתוֹךְ
12   בְּנֵי יִשְׂרָאֵל׃ כִּי מִנֶּגֶד תִּרְאֶה אֶת־הָאָרֶץ וְשָׁמָּה
13   לֹא תָבוֹא אֶל־הָאָרֶץ אֲשֶׁר־אֲנִי נֹתֵן לִבְנֵי
14   יִשְׂרָאֵל׃ [פ]
15  וְזֹאת הַבְּרָכָה אֲשֶׁר בֵּרַךְ מֹשֶׁה אִישׁ הָאֱלֹהִים
16   אֶת־בְּנֵי יִשְׂרָאֵל לִפְנֵי מוֹתוֹ׃ וַיֹּאמַר יְהוָה מִסִּינַי
17   בָּא וְזָרַח מִשֵּׂעִיר לָמוֹ הוֹפִיעַ מֵהַר פָּארָן וְאָתָה
18   מֵרִבְבֹת קֹדֶשׁ מִימִינוֹ אֵשׁ דָּת לָמוֹ׃ אַף חֹבֵב
19   עַמִּים כָּל־קְדֹשָׁיו בְּיָדֶךָ וְהֵם תֻּכּוּ לְרַגְלֶךָ יִשָּׂא
20   מִדַּבְּרֹתֶיךָ׃ תּוֹרָה צִוָּה־לָנוּ מֹשֶׁה מוֹרָשָׁה קְהִלַּת
21   יַעֲקֹב׃ וַיְהִי בִישֻׁרוּן מֶלֶךְ בְּהִתְאַסֵּף רָאשֵׁי עָם יַחַד
22   שִׁבְטֵי יִשְׂרָאֵל׃ יְחִי רְאוּבֵן וְאַל־יָמֹת וִיהִי מְתָיו
23   מִסְפָּר׃ [ס]             וְזֹאת לִיהוּדָה וַיֹּאמַר
24   שְׁמַע יְהוָה קוֹל יְהוּדָה וְאֶל־עַמּוֹ תְּבִיאֶנּוּ יָדָיו רָב
25   לוֹ וְעֵזֶר מִצָּרָיו תִּהְיֶה׃♢ [פ]
26    וּלְלֵוִי אָמַר תֻּמֶּיךָ וְאוּרֶיךָ לְאִישׁ חֲסִידֶךָ אֲשֶׁר
27   נִסִּיתוֹ בְּמַסָּה תְּרִיבֵהוּ עַל־מֵי מְרִיבָה׃ הָאֹמֵר
28   לְאָבִיו וּלְאִמּוֹ לֹא רְאִיתִיו וְאֶת־אֶחָיו לֹא הִכִּיר
29   וְאֶת־בָּנָו לֹא יָדָע כִּי שָׁמְרוּ אִמְרָתֶךָ וּבְרִיתְךָ יִנְצֹרוּ׃
30   יוֹרוּ מִשְׁפָּטֶיךָ לְיַעֲקֹב וְתוֹרָתְךָ לְיִשְׂרָאֵל יָשִׂימוּ
31   קְטוֹרָה בְּאַפֶּךָ וְכָלִיל עַל־מִזְבְּחֶךָ׃ בָּרֵךְ יְהוָה חֵילוֹ
32   וּפֹעַל יָדָיו תִּרְצֶה מְחַץ מָתְנַיִם קָמָיו וּמְשַׂנְאָיו מִן־
33  יְקוּמוּן׃ [ס]             לְבִנְיָמִן אָמַר
34   יְדִיד יְהוָה יִשְׁכֹּן לָבֶטַח עָלָיו חֹפֵף עָלָיו כָּל־הַיּוֹם
35   וּבֵין כְּתֵפָיו שָׁכֵן׃♢ [ס]             וּלְיוֹסֵף
36   אָמַר מְבֹרֶכֶת יְהוָה אַרְצוֹ מִמֶּגֶד שָׁמַיִם מִטָּל
37   וּמִתְּהוֹם רֹבֶצֶת תָּחַת׃ וּמִמֶּגֶד תְּבוּאֹת שָׁמֶשׁ
38   וּמִמֶּגֶד גֶּרֶשׁ יְרָחִים׃ וּמֵרֹאשׁ הַרְרֵי־קֶדֶם וּמִמֶּגֶד
39   גִּבְעוֹת עוֹלָם׃ וּמִמֶּגֶד אֶרֶץ וּמְלֹאָהּ וּרְצוֹן שֹׁכְנִי
40   סְנֶה תָּבוֹאתָה לְרֹאשׁ יוֹסֵף וּלְקָדְקֹד נְזִיר אֶחָיו׃
41   בְּכוֹר שׁוֹרוֹ הָדָר לוֹ וְקַרְנֵי רְאֵם קַרְנָיו בָּהֶם
42   עַמִּים יְנַגַּח יַחְדָּו אַפְסֵי־אָרֶץ וְהֵם רִבְבוֹת אֶפְרַיִם
