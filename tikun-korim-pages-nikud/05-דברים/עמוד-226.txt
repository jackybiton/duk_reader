 1  וְשָׁלְחוּ זִקְנֵי עִירוֹ וְלָקְחוּ אֹתוֹ מִשָּׁם וְנָתְנוּ אֹתוֹ בְּיַד
 2   גֹּאֵל הַדָּם וָמֵת׃ לֹא־תָחוֹס עֵינְךָ עָלָיו וּבִעַרְתָּ דַם־
 3  הַנָּקִי מִיִּשְׂרָאֵל וְטוֹב לָךְ׃ [ס]              לֹא
 4   תַסִּיג גְּבוּל רֵעֲךָ אֲשֶׁר גָּבְלוּ רִאשֹׁנִים בְּנַחֲלָתְךָ
 5   אֲשֶׁר תִּנְחַל בָּאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ
 6   לְרִשְׁתָּהּ׃ [ס]             לֹא־יָקוּם עֵד אֶחָד
 7   בְּאִישׁ לְכָל־עָו͏ֹן וּלְכָל־חַטָּאת בְּכָל־חֵטְא אֲשֶׁר
 8   יֶחֱטָא עַל־פִּי  שְׁנֵי עֵדִים אוֹ עַל־פִּי שְׁלֹשָׁה־עֵדִים
 9   יָקוּם דָּבָר׃ כִּי־יָקוּם עֵד־חָמָס בְּאִישׁ לַעֲנוֹת בּוֹ
10   סָרָה׃ וְעָמְדוּ שְׁנֵי־הָאֲנָשִׁים אֲשֶׁר־לָהֶם הָרִיב לִפְנֵי
11   יְהוָה לִפְנֵי הַכֹּהֲנִים וְהַשֹּׁפְטִים אֲשֶׁר יִהְיוּ בַּיָּמִים
12   הָהֵם׃ וְדָרְשׁוּ הַשֹּׁפְטִים הֵיטֵב וְהִנֵּה עֵד־שֶׁקֶר הָעֵד
13   שֶׁקֶר עָנָה בְאָחִיו׃ וַעֲשִׂיתֶם לוֹ כַּאֲשֶׁר זָמַם לַעֲשׂוֹת
14   לְאָחִיו וּבִעַרְתָּ הָרָע מִקִּרְבֶּךָ׃ וְהַנִּשְׁאָרִים יִשְׁמְעוּ
15   וְיִרָאוּ וְלֹא־יֹסִפוּ לַעֲשׂוֹת עוֹד כַּדָּבָר הָרָע הַזֶּה
16   בְּקִרְבֶּךָ׃ וְלֹא תָחוֹס עֵינֶךָ נֶפֶשׁ בְּנֶפֶשׁ עַיִן בְּעַיִן שֵׁן
17   בְּשֵׁן יָד בְּיָד רֶגֶל בְּרָגֶל׃ [ס]             כִּי־
18  תֵצֵא לַמִּלְחָמָה עַל־אֹיְבֶךָ וְרָאִיתָ סוּס וָרֶכֶב עַם רַב
19   מִמְּךָ לֹא תִירָא מֵהֶם כִּי־יְהוָה אֱלֹהֶיךָ עִמָּךְ הַמַּעַלְךָ
20   מֵאֶרֶץ מִצְרָיִם׃ וְהָיָה כְּקָרָבְכֶם אֶל־הַמִּלְחָמָה וְנִגַּשׁ
21   הַכֹּהֵן וְדִבֶּר אֶל־הָעָם׃ וְאָמַר אֲלֵהֶם שְׁמַע יִשְׂרָאֵל
22   אַתֶּם קְרֵבִים הַיּוֹם לַמִּלְחָמָה עַל־אֹיְבֵיכֶם אַל־יֵרַךְ
23   לְבַבְכֶם אַל־תִּירְאוּ וְאַל־תַּחְפְּזוּ וְאַל־תַּעַרְצוּ מִפְּנֵיהֶם׃
24   כִּי יְהוָה אֱלֹהֵיכֶם הַהֹלֵךְ עִמָּכֶם לְהִלָּחֵם לָכֶם עִם־
25  אֹיְבֵיכֶם לְהוֹשִׁיעַ אֶתְכֶם׃ וְדִבְּרוּ הַשֹּׁטְרִים אֶל־
26  הָעָם לֵאמֹר מִי־הָאִישׁ אֲשֶׁר בָּנָה בַיִת־חָדָשׁ וְלֹא
27   חֲנָכוֹ יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ פֶּן־יָמוּת בַּמִּלְחָמָה וְאִישׁ
28   אַחֵר יַחְנְכֶנּוּ׃ וּמִי־הָאִישׁ אֲשֶׁר־נָטַע כֶּרֶם וְלֹא חִלְּלוֹ
29   יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ פֶּן־יָמוּת בַּמִּלְחָמָה וְאִישׁ אַחֵר
30   יְחַלְּלֶנּוּ׃ וּמִי־הָאִישׁ אֲשֶׁר־אֵרַשׂ אִשָּׁה וְלֹא לְקָחָהּ
31   יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ פֶּן־יָמוּת בַּמִּלְחָמָה וְאִישׁ אַחֵר
32   יִקָּחֶנָּה׃ וְיָסְפוּ הַשֹּׁטְרִים לְדַבֵּר אֶל־הָעָם וְאָמְרוּ מִי־
33  הָאִישׁ הַיָּרֵא וְרַךְ הַלֵּבָב יֵלֵךְ וְיָשֹׁב לְבֵיתוֹ וְלֹא
34   יִמַּס אֶת־לְבַב אֶחָיו כִּלְבָבוֹ׃ וְהָיָה כְּכַלֹּת הַשֹּׁטְרִים
35   לְדַבֵּר אֶל־הָעָם וּפָקְדוּ שָׂרֵי צְבָאוֹת בְּרֹאשׁ
36   הָעָם׃ [ס]              כִּי־תִקְרַב אֶל־עִיר לְהִלָּחֵם
37   עָלֶיהָ וְקָרָאתָ אֵלֶיהָ לְשָׁלוֹם׃ וְהָיָה אִם־שָׁלוֹם תַּעַנְךָ
38   וּפָתְחָה לָךְ וְהָיָה כָּל־הָעָם הַנִּמְצָא־בָהּ יִהְיוּ לְךָ
39   לָמַס וַעֲבָדוּךָ׃ וְאִם־לֹא תַשְׁלִים עִמָּךְ וְעָשְׂתָה עִמְּךָ
40   מִלְחָמָה וְצַרְתָּ עָלֶיהָ׃ וּנְתָנָהּ יְהוָה אֱלֹהֶיךָ בְּיָדֶךָ
41   וְהִכִּיתָ אֶת־כָּל־זְכוּרָהּ לְפִי־חָרֶב׃ רַק הַנָּשִׁים וְהַטַּף
42   וְהַבְּהֵמָה וְכֹל אֲשֶׁר יִהְיֶה בָעִיר כָּל־שְׁלָלָהּ תָּבֹז לָךְ
