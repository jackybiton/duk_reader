 1  וְהֵם אַלְפֵי מְנַשֶּׁה׃♢ [ס]             וְלִזְבוּלֻן אָמַר
 2   שְׂמַח זְבוּלֻן בְּצֵאתֶךָ וְיִשָּׂשכָר בְּאֹהָלֶיךָ׃ עַמִּים הַר־
 3  יִקְרָאוּ שָׁם יִזְבְּחוּ זִבְחֵי־צֶדֶק כִּי שֶׁפַע יַמִּים יִינָקוּ
 4   וּשְׂפֻנֵי טְמוּנֵי חוֹל׃ [ס]             וּלְגָד
 5   אָמַר בָּרוּךְ מַרְחִיב גָּד כְּלָבִיא שָׁכֵן וְטָרַף זְרוֹעַ
 6   אַף־קָדְקֹד׃ וַיַּרְא רֵאשִׁית לוֹ כִּי־שָׁם חֶלְקַת מְחֹקֵק
 7   סָפוּן וַיֵּתֵא רָאשֵׁי עָם צִדְקַת יְהוָה עָשָׂה וּמִשְׁפָּטָיו
 8   עִם־יִשְׂרָאֵל׃ [ס]              וּלְדָן
 9   אָמַר דָּן גּוּר אַרְיֵה יְזַנֵּק מִן־הַבָּשָׁן׃ וּלְנַפְתָּלִי אָמַר
10   נַפְתָּלִי שְׂבַע רָצוֹן וּמָלֵא בִּרְכַּת יְהוָה יָם וְדָרוֹם
11   יְרָשָׁה׃ [ס]             וּלְאָשֵׁר אָמַר
12   בָּרוּךְ מִבָּנִים אָשֵׁר יְהִי רְצוּי אֶחָיו וְטֹבֵל בַּשֶּׁמֶן
13   רַגְלוֹ׃ בַּרְזֶל וּנְחֹשֶׁת מִנְעָלֶךָ וּכְיָמֶיךָ דָּבְאֶךָ׃ אֵין כָּאֵל
14   יְשֻׁרוּן רֹכֵב שָׁמַיִם בְּעֶזְרֶךָ וּבְגַאֲוָתוֹ שְׁחָקִים׃  מְעֹנָה
15   אֱלֹהֵי קֶדֶם וּמִתַּחַת זְרֹעֹת עוֹלָם וַיְגָרֶשׁ מִפָּנֶיךָ אוֹיֵב
16   וַיֹּאמֶר הַשְׁמֵד׃ וַיִּשְׁכֹּן יִשְׂרָאֵל בֶּטַח בָּדָד עֵין יַעֲקֹב
17   אֶל־אֶרֶץ דָּגָן וְתִירוֹשׁ אַף־שָׁמָיו יַעַרְפוּ טָל׃ אַשְׁרֶיךָ
18   יִשְׂרָאֵל מִי כָמוֹךָ עַם נוֹשַׁע בַּיהוָה מָגֵן עֶזְרֶךָ
19   וַאֲשֶׁר־חֶרֶב גַּאֲוָתֶךָ וְיִכָּחֲשׁוּ אֹיְבֶיךָ לָךְ וְאַתָּה עַל־
20  בָּמוֹתֵימוֹ תִדְרֹךְ׃ [ס]            ♢ וַיַּעַל מֹשֶׁה
21   מֵעַרְבֹת מוֹאָב אֶל־הַר נְבוֹ רֹאשׁ הַפִּסְגָּה אֲשֶׁר
22   עַל־פְּנֵי יְרֵחוֹ וַיַּרְאֵהוּ יְהוָה אֶת־כָּל־הָאָרֶץ אֶת־
23  הַגִּלְעָד עַד־דָּן׃ וְאֵת כָּל־נַפְתָּלִי וְאֶת־אֶרֶץ אֶפְרַיִם
24   וּמְנַשֶּׁה וְאֵת כָּל־אֶרֶץ יְהוּדָה עַד הַיָּם הָאַחֲרוֹן׃
25   וְאֶת־הַנֶּגֶב וְאֶת־הַכִּכָּר בִּקְעַת יְרֵחוֹ עִיר הַתְּמָרִים
26   עַד־צֹעַר׃ וַיֹּאמֶר יְהוָה אֵלָיו זֹאת הָאָרֶץ אֲשֶׁר
27   נִשְׁבַּעְתִּי לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב לֵאמֹר לְזַרְעֲךָ
28   אֶתְּנֶנָּה הֶרְאִיתִיךָ בְעֵינֶיךָ וְשָׁמָּה לֹא תַעֲבֹר׃ וַיָּמָת
29   שָׁם מֹשֶׁה עֶבֶד־יְהוָה בְּאֶרֶץ מוֹאָב עַל־פִּי יְהוָה׃
30   וַיִּקְבֹּר אֹתוֹ בַגַּי בְּאֶרֶץ מוֹאָב מוּל בֵּית פְּעוֹר וְלֹא־
31  יָדַע אִישׁ אֶת־קְבֻרָתוֹ עַד הַיּוֹם הַזֶּה׃ וּמֹשֶׁה בֶּן־
32  מֵאָה וְעֶשְׂרִים שָׁנָה בְּמֹתוֹ לֹא־כָהֲתָה עֵינוֹ וְלֹא־נָס
33   לֵחֹה׃ וַיִּבְכּוּ בְנֵי יִשְׂרָאֵל אֶת־מֹשֶׁה בְּעַרְבֹת מוֹאָב
34   שְׁלֹשִׁים יוֹם וַיִּתְּמוּ יְמֵי בְכִי אֵבֶל מֹשֶׁה׃ וִיהוֹשֻׁעַ בִּן־
35  נוּן מָלֵא רוּחַ חָכְמָה כִּי־סָמַךְ מֹשֶׁה אֶת־יָדָיו עָלָיו
36   וַיִּשְׁמְעוּ אֵלָיו בְּנֵי־יִשְׂרָאֵל וַיַּעֲשׂוּ כַּאֲשֶׁר צִוָּה יְהוָה
37   אֶת־מֹשֶׁה׃ וְלֹא־קָם נָבִיא עוֹד בְּיִשְׂרָאֵל כְּמֹשֶׁה
38   אֲשֶׁר יְדָעוֹ יְהוָה פָּנִים אֶל־פָּנִים׃ לְכָל־הָאֹתֹת
39   וְהַמּוֹפְתִים אֲשֶׁר שְׁלָחוֹ יְהוָה לַעֲשׂוֹת בְּאֶרֶץ
40   מִצְרָיִם לְפַרְעֹה וּלְכָל־עֲבָדָיו וּלְכָל־אַרְצוֹ׃ וּלְכֹל
41   הַיָּד הַחֲזָקָה וּלְכֹל הַמּוֹרָא הַגָּדוֹל אֲשֶׁר עָשָׂה מֹשֶׁה
42   לְעֵינֵי כָּל־יִשְׂרָאֵל׃ [פ]
