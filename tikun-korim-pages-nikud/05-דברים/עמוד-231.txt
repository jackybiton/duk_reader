 1  וְכָתַב לָהּ סֵפֶר כְּרִיתֻת וְנָתַן בְּיָדָהּ וְשִׁלְּחָהּ מִבֵּיתוֹ
 2   אוֹ כִי יָמוּת הָאִישׁ הָאַחֲרוֹן אֲשֶׁר־לְקָחָהּ לוֹ לְאִשָּׁה׃
 3   לֹא־יוּכַל בַּעְלָהּ הָרִאשׁוֹן אֲשֶׁר־שִׁלְּחָהּ לָשׁוּב
 4   לְקַחְתָּהּ לִהְיוֹת לוֹ לְאִשָּׁה אַחֲרֵי אֲשֶׁר הֻטַּמָּאָה כִּי־
 5  תוֹעֵבָה הִוא לִפְנֵי יְהוָה וְלֹא תַחֲטִיא אֶת־הָאָרֶץ אֲשֶׁר
 6   יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה׃ [ס]              כִּי־
 7  יִקַּח אִישׁ אִשָּׁה חֲדָשָׁה לֹא יֵצֵא בַּצָּבָא וְלֹא־יַעֲבֹר
 8   עָלָיו לְכָל־דָּבָר נָקִי יִהְיֶה לְבֵיתוֹ שָׁנָה אֶחָת וְשִׂמַּח
 9   אֶת־אִשְׁתּוֹ אֲשֶׁר־לָקָח׃ לֹא־יַחֲבֹל רֵחַיִם וָרָכֶב כִּי־
10  נֶפֶשׁ הוּא חֹבֵל׃ [ס]             כִּי־
11  יִמָּצֵא אִישׁ גֹּנֵב נֶפֶשׁ מֵאֶחָיו מִבְּנֵי יִשְׂרָאֵל
12   וְהִתְעַמֶּר־בּוֹ וּמְכָרוֹ וּמֵת הַגַּנָּב הַהוּא וּבִעַרְתָּ הָרָע
13   מִקִּרְבֶּךָ׃ [ס]             הִשָּׁמֶר בְּנֶגַע־הַצָּרַעַת
14   לִשְׁמֹר מְאֹד וְלַעֲשׂוֹת כְּכֹל אֲשֶׁר־יוֹרוּ אֶתְכֶם
15   הַכֹּהֲנִים הַלְוִיִּם כַּאֲשֶׁר צִוִּיתִם תִּשְׁמְרוּ לַעֲשׂוֹת׃
16   זָכוֹר אֵת אֲשֶׁר־עָשָׂה יְהוָה אֱלֹהֶיךָ לְמִרְיָם בַּדֶּרֶךְ
17   בְּצֵאתְכֶם מִמִּצְרָיִם׃ [ס]             כִּי־
18  תַשֶּׁה בְרֵעֲךָ מַשַּׁאת מְאוּמָה לֹא־תָבֹא אֶל־
19  בֵּיתוֹ לַעֲבֹט עֲבֹטוֹ׃ בַּחוּץ תַּעֲמֹד וְהָאִישׁ אֲשֶׁר
20   אַתָּה נֹשֶׁה בוֹ יוֹצִיא אֵלֶיךָ אֶת־הַעֲבוֹט הַחוּצָה׃
21   וְאִם־אִישׁ עָנִי הוּא לֹא תִשְׁכַּב בַּעֲבֹטוֹ׃ הָשֵׁב
22   תָּשִׁיב לוֹ אֶת־הַעֲבוֹט כְּבוֹא הַשֶּׁמֶשׁ וְשָׁכַב
23   בְּשַׂלְמָתוֹ וּבֵרְכֶךָּ וּלְךָ תִּהְיֶה צְדָקָה לִפְנֵי יְהוָה
24   אֱלֹהֶיךָ׃ [ס]              לֹא־תַעֲשֹׁק שָׂכִיר עָנִי
25   וְאֶבְיוֹן מֵאַחֶיךָ אוֹ מִגֵּרְךָ אֲשֶׁר בְּאַרְצְךָ בִּשְׁעָרֶיךָ׃
26   בְּיוֹמוֹ תִתֵּן שְׂכָרוֹ וְלֹא־תָבוֹא עָלָיו הַשֶּׁמֶשׁ כִּי עָנִי
27   הוּא וְאֵלָיו הוּא נֹשֵׂא אֶת־נַפְשׁוֹ וְלֹא־יִקְרָא עָלֶיךָ
28   אֶל־יְהוָה וְהָיָה בְךָ חֵטְא׃ [ס]             לֹא־
29  יוּמְתוּ אָבוֹת עַל־בָּנִים וּבָנִים לֹא־יוּמְתוּ עַל־אָבוֹת
30   אִישׁ בְּחֶטְאוֹ יוּמָתוּ׃ [ס]             לֹא
31   תַטֶּה מִשְׁפַּט גֵּר יָתוֹם וְלֹא תַחֲבֹל בֶּגֶד אַלְמָנָה׃
32   וְזָכַרְתָּ כִּי עֶבֶד הָיִיתָ בְּמִצְרַיִם וַיִּפְדְּךָ יְהוָה אֱלֹהֶיךָ
33   מִשָּׁם עַל־כֵּן אָנֹכִי מְצַוְּךָ לַעֲשׂוֹת אֶת־הַדָּבָר
34   הַזֶּה׃ [ס]             כִּי תִקְצֹר קְצִירְךָ בְשָׂדֶךָ
35   וְשָׁכַחְתָּ עֹמֶר בַּשָּׂדֶה לֹא תָשׁוּב לְקַחְתּוֹ לַגֵּר
36   לַיָּתוֹם וְלָאַלְמָנָה יִהְיֶה לְמַעַן יְבָרֶכְךָ יְהוָה אֱלֹהֶיךָ
37   בְּכֹל מַעֲשֵׂה יָדֶיךָ׃ [ס]             כִּי תַחְבֹּט
38   זֵיתְךָ לֹא תְפַאֵר אַחֲרֶיךָ לַגֵּר לַיָּתוֹם וְלָאַלְמָנָה
39   יִהְיֶה׃ כִּי תִבְצֹר כַּרְמְךָ לֹא תְעוֹלֵל אַחֲרֶיךָ לַגֵּר
40   לַיָּתוֹם וְלָאַלְמָנָה יִהְיֶה׃ וְזָכַרְתָּ כִּי־עֶבֶד הָיִיתָ בְּאֶרֶץ
41   מִצְרָיִם עַל־כֵּן אָנֹכִי מְצַוְּךָ לַעֲשׂוֹת אֶת־הַדָּבָר
42   הַזֶּה׃ [ס]             כִּי־יִהְיֶה רִיב בֵּין אֲנָשִׁים
