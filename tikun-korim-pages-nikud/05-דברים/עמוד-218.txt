 1  וּבְנֵיכֶם וּבְנֹתֵיכֶם וְעַבְדֵיכֶם וְאַמְהֹתֵיכֶם וְהַלֵּוִי אֲשֶׁר
 2   בְּשַׁעֲרֵיכֶם כִּי אֵין לוֹ חֵלֶק וְנַחֲלָה אִתְּכֶם׃ הִשָּׁמֶר
 3   לְךָ פֶּן־תַּעֲלֶה עֹלֹתֶיךָ בְּכָל־מָקוֹם אֲשֶׁר תִּרְאֶה׃
 4   כִּי אִם־בַּמָּקוֹם אֲשֶׁר־יִבְחַר יְהוָה בְּאַחַד שְׁבָטֶיךָ
 5   שָׁם תַּעֲלֶה עֹלֹתֶיךָ וְשָׁם תַּעֲשֶׂה כֹּל אֲשֶׁר אָנֹכִי
 6   מְצַוֶּךָּ׃ רַק בְּכָל־אַוַּת נַפְשְׁךָ תִּזְבַּח  וְאָכַלְתָּ בָשָׂר
 7   כְּבִרְכַּת יְהוָה אֱלֹהֶיךָ אֲשֶׁר נָתַן־לְךָ בְּכָל־שְׁעָרֶיךָ
 8   הַטָּמֵא וְהַטָּהוֹר יֹאכְלֶנּוּ כַּצְּבִי וְכָאַיָּל׃ רַק הַדָּם לֹא
 9   תֹאכֵלוּ עַל־הָאָרֶץ תִּשְׁפְּכֶנּוּ כַּמָּיִם׃ לֹא־תוּכַל
10   לֶאֱכֹל בִּשְׁעָרֶיךָ מַעְשַׂר דְּגָנְךָ וְתִירֹשְׁךָ וְיִצְהָרֶךָ
11   וּבְכֹרֹת בְּקָרְךָ וְצֹאנֶךָ וְכָל־נְדָרֶיךָ אֲשֶׁר תִּדֹּר
12   וְנִדְבֹתֶיךָ וּתְרוּמַת יָדֶךָ׃ כִּי אִם־לִפְנֵי יְהוָה אֱלֹהֶיךָ
13   תֹּאכְלֶנּוּ בַּמָּקוֹם אֲשֶׁר יִבְחַר יְהוָה אֱלֹהֶיךָ בּוֹ
14   אַתָּה וּבִנְךָ וּבִתֶּךָ וְעַבְדְּךָ וַאֲמָתֶךָ וְהַלֵּוִי אֲשֶׁר
15   בִּשְׁעָרֶיךָ וְשָׂמַחְתָּ לִפְנֵי יְהוָה אֱלֹהֶיךָ בְּכֹל מִשְׁלַח
16   יָדֶךָ׃ הִשָּׁמֶר לְךָ פֶּן־תַּעֲזֹב אֶת־הַלֵּוִי כָּל־יָמֶיךָ עַל־
17  אַדְמָתֶךָ׃ [ס]             כִּי־יַרְחִיב יְהוָה אֱלֹהֶיךָ
18   אֶת־גְּבֻלְךָ כַּאֲשֶׁר דִּבֶּר־לָךְ וְאָמַרְתָּ אֹכְלָה בָשָׂר
19   כִּי־תְאַוֶּה נַפְשְׁךָ לֶאֱכֹל בָּשָׂר בְּכָל־אַוַּת נַפְשְׁךָ
20   תֹּאכַל בָּשָׂר׃ כִּי־יִרְחַק מִמְּךָ הַמָּקוֹם אֲשֶׁר יִבְחַר
21   יְהוָה אֱלֹהֶיךָ לָשׂוּם שְׁמוֹ שָׁם וְזָבַחְתָּ מִבְּקָרְךָ
22   וּמִצֹּאנְךָ אֲשֶׁר נָתַן יְהוָה לְךָ כַּאֲשֶׁר צִוִּיתִךָ וְאָכַלְתָּ
23   בִּשְׁעָרֶיךָ בְּכֹל אַוַּת נַפְשֶׁךָ׃ אַךְ כַּאֲשֶׁר יֵאָכֵל אֶת־
24  הַצְּבִי וְאֶת־הָאַיָּל כֵּן תֹּאכְלֶנּוּ הַטָּמֵא וְהַטָּהוֹר יַחְדָּו
25   יֹאכְלֶנּוּ׃ רַק חֲזַק לְבִלְתִּי אֲכֹל הַדָּם כִּי הַדָּם הוּא
26   הַנָּפֶשׁ וְלֹא־תֹאכַל הַנֶּפֶשׁ עִם־הַבָּשָׂר׃ לֹא תֹּאכְלֶנּוּ
27   עַל־הָאָרֶץ תִּשְׁפְּכֶנּוּ כַּמָּיִם׃ לֹא תֹּאכְלֶנּוּ לְמַעַן יִיטַב
28   לְךָ וּלְבָנֶיךָ אַחֲרֶיךָ כִּי־תַעֲשֶׂה הַיָּשָׁר בְּעֵינֵי יְהוָה׃
29   רַק קָדָשֶׁיךָ אֲשֶׁר־יִהְיוּ לְךָ וּנְדָרֶיךָ תִּשָּׂא וּבָאתָ אֶל־
30  הַמָּקוֹם אֲשֶׁר־יִבְחַר יְהוָה׃ וְעָשִׂיתָ עֹלֹתֶיךָ הַבָּשָׂר
31   וְהַדָּם עַל־מִזְבַּח יְהוָה אֱלֹהֶיךָ וְדַם־זְבָחֶיךָ יִשָּׁפֵךְ עַל־
32  מִזְבַּח יְהוָה אֱלֹהֶיךָ וְהַבָּשָׂר תֹּאכֵל׃ שְׁמֹר וְשָׁמַעְתָּ
33   אֵת כָּל־הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוֶּךָּ לְמַעַן
34   יִיטַב לְךָ וּלְבָנֶיךָ אַחֲרֶיךָ עַד־עוֹלָם כִּי תַעֲשֶׂה הַטּוֹב
35   וְהַיָּשָׁר בְּעֵינֵי יְהוָה אֱלֹהֶיךָ׃ [ס]              כִּי־
36  יַכְרִית יְהוָה אֱלֹהֶיךָ אֶת־הַגּוֹיִם אֲשֶׁר אַתָּה בָא־
37  שָׁמָּה לָרֶשֶׁת אוֹתָם מִפָּנֶיךָ וְיָרַשְׁתָּ אֹתָם וְיָשַׁבְתָּ
38   בְּאַרְצָם׃ הִשָּׁמֶר לְךָ פֶּן־תִּנָּקֵשׁ אַחֲרֵיהֶם אַחֲרֵי
39   הִשָּׁמְדָם מִפָּנֶיךָ וּפֶן־תִּדְרֹשׁ לֵאלֹהֵיהֶם לֵאמֹר אֵיכָה
40   יַעַבְדוּ הַגּוֹיִם הָאֵלֶּה אֶת־אֱלֹהֵיהֶם וְאֶעֱשֶׂה־כֵּן גַּם־
41  אָנִי׃ לֹא־תַעֲשֶׂה כֵן לַיהוָה אֱלֹהֶיךָ כִּי כָל־תּוֹעֲבַת
42   יְהוָה אֲשֶׁר שָׂנֵא עָשׂוּ לֵאלֹהֵיהֶם כִּי גַם אֶת־בְּנֵיהֶם
