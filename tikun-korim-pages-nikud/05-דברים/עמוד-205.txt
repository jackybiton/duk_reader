 1  וְאֶת־הָאָרֶץ הַזֹּאת יָרַשְׁנוּ בָּעֵת הַהִוא מֵעֲרֹעֵר
 2   אֲשֶׁר־עַל־נַחַל אַרְנֹן וַחֲצִי הַר־הַגִּלְעָד וְעָרָיו נָתַתִּי
 3   לָראוּבֵנִי וְלַגָּדִי׃ וְיֶתֶר הַגִּלְעָד וְכָל־הַבָּשָׁן מַמְלֶכֶת
 4   עוֹג נָתַתִּי לַחֲצִי שֵׁבֶט הַמְנַשֶּׁה כֹּל חֶבֶל הָאַרְגֹּב
 5   לְכָל־הַבָּשָׁן הַהוּא יִקָּרֵא אֶרֶץ רְפָאִים׃ יָאִיר בֶּן־
 6  מְנַשֶּׁה לָקַח אֶת־כָּל־חֶבֶל אַרְגֹּב עַד־גְּבוּל הַגְּשׁוּרִי
 7   וְהַמַּעֲכָתִי וַיִּקְרָא אֹתָם עַל־שְׁמוֹ אֶת־הַבָּשָׁן חַוּ͏ֹת
 8   יָאִיר עַד הַיּוֹם הַזֶּה׃  וּלְמָכִיר נָתַתִּי אֶת־הַגִּלְעָד׃
 9   וְלָראוּבֵנִי וְלַגָּדִי נָתַתִּי מִן־הַגִּלְעָד וְעַד־נַחַל אַרְנֹן
10   תּוֹךְ הַנַּחַל וּגְבֻל וְעַד יַבֹּק הַנַּחַל גְּבוּל בְּנֵי עַמּוֹן׃
11   וְהָעֲרָבָה וְהַיַּרְדֵּן וּגְבֻל מִכִּנֶּרֶת וְעַד יָם הָעֲרָבָה
12   יָם הַמֶּלַח תַּחַת אַשְׁדֹּת הַפִּסְגָּה מִזְרָחָה׃ וָאֲצַו
13   אֶתְכֶם בָּעֵת הַהִוא לֵאמֹר יְהוָה אֱלֹהֵיכֶם נָתַן
14   לָכֶם אֶת־הָאָרֶץ הַזֹּאת לְרִשְׁתָּהּ חֲלוּצִים תַּעַבְרוּ
15   לִפְנֵי אֲחֵיכֶם בְּנֵי־יִשְׂרָאֵל כָּל־בְּנֵי־חָיִל׃ רַק נְשֵׁיכֶם
16   וְטַפְּכֶם וּמִקְנֵכֶם יָדַעְתִּי כִּי־מִקְנֶה רַב לָכֶם יֵשְׁבוּ
17   בְּעָרֵיכֶם אֲשֶׁר נָתַתִּי לָכֶם׃  עַד אֲשֶׁר־יָנִיחַ יְהוָה
18   לַאֲחֵיכֶם כָּכֶם וְיָרְשׁוּ גַם־הֵם אֶת־הָאָרֶץ אֲשֶׁר
19   יְהוָה אֱלֹהֵיכֶם נֹתֵן לָהֶם בְּעֵבֶר הַיַּרְדֵּן וְשַׁבְתֶּם
20   אִישׁ לִירֻשָּׁתוֹ אֲשֶׁר נָתַתִּי לָכֶם׃ וְאֶת־יְהוֹשׁוּעַ
21   צִוֵּיתִי בָּעֵת הַהִוא לֵאמֹר עֵינֶיךָ הָרֹאֹת אֵת כָּל־
22  אֲשֶׁר עָשָׂה יְהוָה אֱלֹהֵיכֶם לִשְׁנֵי הַמְּלָכִים הָאֵלֶּה
23   כֵּן־יַעֲשֶׂה יְהוָה לְכָל־הַמַּמְלָכוֹת אֲשֶׁר אַתָּה עֹבֵר
24   שָׁמָּה׃ לֹא תִּירָאוּם כִּי יְהוָה אֱלֹהֵיכֶם הוּא הַנִּלְחָם
25   לָכֶם׃ [ס]            וָאֶתְחַנַּן
26   אֶל־יְהוָה בָּעֵת הַהִוא לֵאמֹר׃ אֲדֹנָי יְהוִה אַתָּה
27   הַחִלּוֹתָ לְהַרְאוֹת אֶת־עַבְדְּךָ אֶת־גָּדְלְךָ וְאֶת־יָדְךָ
28   הַחֲזָקָה אֲשֶׁר מִי־אֵל בַּשָּׁמַיִם וּבָאָרֶץ אֲשֶׁר־יַעֲשֶׂה
29   כְמַעֲשֶׂיךָ וְכִגְבוּרֹתֶךָ׃ אֶעְבְּרָה־נָּא וְאֶרְאֶה אֶת־
30  הָאָרֶץ הַטּוֹבָה אֲשֶׁר בְּעֵבֶר הַיַּרְדֵּן הָהָר הַטּוֹב
31   הַזֶּה וְהַלְּבָנֹן׃♢ וַיִּתְעַבֵּר יְהוָה בִּי לְמַעַנְכֶם וְלֹא שָׁמַע
32   אֵלָי וַיֹּאמֶר יְהוָה אֵלַי רַב־לָךְ אַל־תּוֹסֶף דַּבֵּר אֵלַי
33   עוֹד בַּדָּבָר הַזֶּה׃ עֲלֵה  רֹאשׁ הַפִּסְגָּה וְשָׂא עֵינֶיךָ
34   יָמָּה וְצָפֹנָה וְתֵימָנָה וּמִזְרָחָה וּרְאֵה בְעֵינֶיךָ כִּי־לֹא
35   תַעֲבֹר אֶת־הַיַּרְדֵּן הַזֶּה׃ וְצַו אֶת־יְהוֹשֻׁעַ וְחַזְּקֵהוּ
36   וְאַמְּצֵהוּ כִּי־הוּא יַעֲבֹר לִפְנֵי הָעָם הַזֶּה וְהוּא יַנְחִיל
37   אוֹתָם אֶת־הָאָרֶץ אֲשֶׁר תִּרְאֶה׃ וַנֵּשֶׁב בַּגָּיְא מוּל
38   בֵּית פְּעוֹר׃ [פ]
39   וְעַתָּה יִשְׂרָאֵל שְׁמַע אֶל־הַחֻקִּים וְאֶל־הַמִּשְׁפָּטִים
40   אֲשֶׁר אָנֹכִי מְלַמֵּד אֶתְכֶם לַעֲשׂוֹת לְמַעַן תִּחְיוּ וּבָאתֶם
41   וִירִשְׁתֶּם אֶת־הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֵי אֲבֹתֵיכֶם נֹתֵן
42   לָכֶם׃ לֹא תֹסִפוּ עַל־הַדָּבָר אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם
