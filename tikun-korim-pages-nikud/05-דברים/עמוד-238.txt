 1   וַיִּקְרָא מֹשֶׁה אֶל־כָּל־יִשְׂרָאֵל וַיֹּאמֶר אֲלֵהֶם אַתֶּם
 2   רְאִיתֶם אֵת כָּל־אֲשֶׁר עָשָׂה יְהוָה לְעֵינֵיכֶם בְּאֶרֶץ
 3   מִצְרַיִם לְפַרְעֹה וּלְכָל־עֲבָדָיו וּלְכָל־אַרְצוֹ׃ הַמַּסּוֹת
 4   הַגְּדֹלֹת אֲשֶׁר רָאוּ עֵינֶיךָ הָאֹתֹת וְהַמֹּפְתִים הַגְּדֹלִים
 5   הָהֵם׃ וְלֹא־נָתַן יְהוָה לָכֶם לֵב לָדַעַת וְעֵינַיִם לִרְאוֹת
 6   וְאָזְנַיִם לִשְׁמֹעַ עַד הַיּוֹם הַזֶּה׃ וָאוֹלֵךְ אֶתְכֶם אַרְבָּעִים
 7   שָׁנָה בַּמִּדְבָּר לֹא־בָלוּ שַׂלְמֹתֵיכֶם מֵעֲלֵיכֶם וְנַעַלְךָ
 8   לֹא־בָלְתָה מֵעַל רַגְלֶךָ׃ לֶחֶם לֹא אֲכַלְתֶּם וְיַיִן
 9   וְשֵׁכָר לֹא שְׁתִיתֶם לְמַעַן תֵּדְעוּ כִּי אֲנִי יְהוָה
10   אֱלֹהֵיכֶם׃  וַתָּבֹאוּ אֶל־הַמָּקוֹם הַזֶּה וַיֵּצֵא סִיחֹן מֶלֶךְ־
11  חֶשְׁבּוֹן וְעוֹג מֶלֶךְ־הַבָּשָׁן לִקְרָאתֵנוּ לַמִּלְחָמָה
12   וַנַּכֵּם׃ וַנִּקַּח אֶת־אַרְצָם וַנִּתְּנָהּ לְנַחֲלָה לָראוּבֵנִי
13   וְלַגָּדִי וְלַחֲצִי שֵׁבֶט הַמְנַשִּׁי׃ וּשְׁמַרְתֶּם אֶת־דִּבְרֵי
14   הַבְּרִית הַזֹּאת וַעֲשִׂיתֶם אֹתָם לְמַעַן תַּשְׂכִּילוּ אֵת
15   כָּל־אֲשֶׁר תַּעֲשׂוּן׃ [פ]
16  אַתֶּם נִצָּבִים הַיּוֹם כֻּלְּכֶם לִפְנֵי יְהוָה אֱלֹהֵיכֶם
17   רָאשֵׁיכֶם שִׁבְטֵיכֶם זִקְנֵיכֶם וְשֹׁטְרֵיכֶם כֹּל אִישׁ
18   יִשְׂרָאֵל׃ טַפְּכֶם נְשֵׁיכֶם וְגֵרְךָ אֲשֶׁר בְּקֶרֶב מַחֲנֶיךָ
19   מֵחֹטֵב עֵצֶיךָ עַד שֹׁאֵב מֵימֶיךָ׃ לְעָבְרְךָ בִּבְרִית
20   יְהוָה אֱלֹהֶיךָ וּבְאָלָתוֹ אֲשֶׁר יְהוָה אֱלֹהֶיךָ כֹּרֵת
21   עִמְּךָ הַיּוֹם׃♢  לְמַעַן הָקִים־אֹתְךָ הַיּוֹם  לוֹ לְעָם וְהוּא
22   יִהְיֶה־לְּךָ לֵאלֹהִים כַּאֲשֶׁר דִּבֶּר־לָךְ וְכַאֲשֶׁר נִשְׁבַּע
23   לַאֲבֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב׃ וְלֹא אִתְּכֶם
24   לְבַדְּכֶם אָנֹכִי כֹּרֵת אֶת־הַבְּרִית הַזֹּאת וְאֶת־הָאָלָה
25   הַזֹּאת׃ כִּי אֶת־אֲשֶׁר יֶשְׁנוֹ פֹּה עִמָּנוּ עֹמֵד הַיּוֹם
26   לִפְנֵי יְהוָה אֱלֹהֵינוּ וְאֵת אֲשֶׁר אֵינֶנּוּ פֹּה עִמָּנוּ הַיּוֹם׃♢
27    כִּי־אַתֶּם יְדַעְתֶּם אֵת אֲשֶׁר־יָשַׁבְנוּ בְּאֶרֶץ מִצְרָיִם
28   וְאֵת אֲשֶׁר־עָבַרְנוּ בְּקֶרֶב הַגּוֹיִם אֲשֶׁר עֲבַרְתֶּם׃
29   וַתִּרְאוּ אֶת־שִׁקּוּצֵיהֶם וְאֵת גִּלֻּלֵיהֶם עֵץ וָאֶבֶן כֶּסֶף
30   וְזָהָב אֲשֶׁר עִמָּהֶם׃ פֶּן־יֵשׁ בָּכֶם אִישׁ אוֹ־אִשָּׁה אוֹ
31   מִשְׁפָּחָה אוֹ־שֵׁבֶט אֲשֶׁר לְבָבוֹ פֹנֶה הַיּוֹם מֵעִם
32   יְהוָה אֱלֹהֵינוּ לָלֶכֶת לַעֲבֹד אֶת־אֱלֹהֵי הַגּוֹיִם הָהֵם
33   פֶּן־יֵשׁ בָּכֶם שֹׁרֶשׁ פֹּרֶה רֹאשׁ וְלַעֲנָה׃ וְהָיָה בְּשָׁמְעוֹ
34   אֶת־דִּבְרֵי הָאָלָה הַזֹּאת וְהִתְבָּרֵךְ בִּלְבָבוֹ לֵאמֹר
35   שָׁלוֹם יִהְיֶה־לִּי כִּי בִּשְׁרִרוּת לִבִּי אֵלֵךְ לְמַעַן סְפוֹת
36   הָרָוָה אֶת־הַצְּמֵאָה׃ לֹא־יֹאבֶה יְהוָה סְלֹחַ לוֹ כִּי אָז
37   יֶעְשַׁן אַף־יְהוָה וְקִנְאָתוֹ בָּאִישׁ הַהוּא וְרָבְצָה בּוֹ
38   כָּל־הָאָלָה הַכְּתוּבָה בַּסֵּפֶר הַזֶּה וּמָחָה יְהוָה אֶת־
39  שְׁמוֹ מִתַּחַת הַשָּׁמָיִם׃ וְהִבְדִּילוֹ יְהוָה לְרָעָה מִכֹּל
40   שִׁבְטֵי יִשְׂרָאֵל כְּכֹל אָלוֹת הַבְּרִית הַכְּתוּבָה בְּסֵפֶר
41   הַתּוֹרָה הַזֶּה׃ וְאָמַר הַדּוֹר הָאַחֲרוֹן בְּנֵיכֶם אֲשֶׁר
42   יָקוּמוּ מֵאַחֲרֵיכֶם וְהַנָּכְרִי אֲשֶׁר יָבֹא מֵאֶרֶץ רְחוֹקָה
