 1  וּדְבַשׁ כַּאֲשֶׁר דִּבֶּר יְהוָה אֱלֹהֵי־אֲבֹתֶיךָ לָךְ׃ וְהָיָה
 2   בְּעָבְרְכֶם אֶת־הַיַּרְדֵּן תָּקִימוּ אֶת־הָאֲבָנִים הָאֵלֶּה
 3   אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם בְּהַר עֵיבָל וְשַׂדְתָּ
 4   אוֹתָם בַּשִּׂיד׃ וּבָנִיתָ שָּׁם מִזְבֵּחַ לַיהוָה אֱלֹהֶיךָ מִזְבַּח
 5   אֲבָנִים לֹא־תָנִיף עֲלֵיהֶם בַּרְזֶל׃ אֲבָנִים שְׁלֵמוֹת תִּבְנֶה
 6   אֶת־מִזְבַּח יְהוָה אֱלֹהֶיךָ וְהַעֲלִיתָ עָלָיו עוֹלֹת לַיהוָה
 7   אֱלֹהֶיךָ׃ וְזָבַחְתָּ שְׁלָמִים וְאָכַלְתָּ שָּׁם וְשָׂמַחְתָּ לִפְנֵי
 8   יְהוָה אֱלֹהֶיךָ׃ וְכָתַבְתָּ עַל־הָאֲבָנִים אֶת־כָּל־דִּבְרֵי
 9   הַתּוֹרָה הַזֹּאת בַּאֵר הֵיטֵב׃ [ס]             וַיְדַבֵּר
10   מֹשֶׁה וְהַכֹּהֲנִים הַלְוִיִּם אֶל כָּל־יִשְׂרָאֵל לֵאמֹר הַסְכֵּת
11   וּשְׁמַע יִשְׂרָאֵל הַיּוֹם הַזֶּה נִהְיֵיתָ לְעָם לַיהוָה אֱלֹהֶיךָ׃
12   וְשָׁמַעְתָּ בְּקוֹל יְהוָה אֱלֹהֶיךָ וְעָשִׂיתָ אֶת־מִצְו͏ֹתָו וְאֶת־
13  חֻקָּיו אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם׃ [ס]              וַיְצַו
14   מֹשֶׁה אֶת־הָעָם בַּיּוֹם הַהוּא לֵאמֹר׃ אֵלֶּה יַעַמְדוּ
15   לְבָרֵךְ אֶת־הָעָם עַל־הַר גְּרִזִים בְּעָבְרְכֶם אֶת־הַיַּרְדֵּן
16   שִׁמְעוֹן וְלֵוִי וִיהוּדָה וְיִשָּׂשכָר וְיוֹסֵף וּבִנְיָמִן׃ וְאֵלֶּה
17   יַעַמְדוּ עַל־הַקְּלָלָה בְּהַר עֵיבָל רְאוּבֵן גָּד וְאָשֵׁר
18   וּזְבוּלֻן דָּן וְנַפְתָּלִי׃ וְעָנוּ הַלְוִיִּם וְאָמְרוּ אֶל־כָּל־אִישׁ
19   יִשְׂרָאֵל קוֹל רָם׃ [ס]             אָרוּר
20   הָאִישׁ אֲשֶׁר יַעֲשֶׂה פֶסֶל וּמַסֵּכָה תּוֹעֲבַת יְהוָה
21   מַעֲשֵׂה יְדֵי חָרָשׁ וְשָׂם בַּסָּתֶר וְעָנוּ כָל־הָעָם וְאָמְרוּ
22   אָמֵן׃ [ס]             אָרוּר מַקְלֶה אָבִיו וְאִמּוֹ וְאָמַר
23   כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר מַסִּיג גְּבוּל רֵעֵהוּ
24   וְאָמַר כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר מַשְׁגֶּה
25   עִוֵּר בַּדָּרֶךְ וְאָמַר כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר
26   מַטֶּה מִשְׁפַּט גֵּר־יָתוֹם וְאַלְמָנָה וְאָמַר כָּל־הָעָם אָמֵן׃
27   אָרוּר שֹׁכֵב עִם־אֵשֶׁת אָבִיו כִּי גִלָּה כְּנַף אָבִיו וְאָמַר
28   כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר שֹׁכֵב עִם־כָּל־
29  בְּהֵמָה וְאָמַר כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר
30   שֹׁכֵב עִם־אֲחֹתוֹ בַּת־אָבִיו אוֹ בַת־אִמּוֹ וְאָמַר כָּל־הָעָם
31   אָמֵן׃ [ס]             אָרוּר שֹׁכֵב עִם־חֹתַנְתּוֹ וְאָמַר
32   כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר מַכֵּה רֵעֵהוּ
33   בַּסָּתֶר וְאָמַר כָּל־הָעָם אָמֵן׃ [ס]             אָרוּר
34   לֹקֵחַ שֹׁחַד לְהַכּוֹת נֶפֶשׁ דָּם נָקִי וְאָמַר כָּל־הָעָם
35   אָמֵן׃ [ס]             אָרוּר אֲשֶׁר לֹא־יָקִים
36   אֶת־דִּבְרֵי הַתּוֹרָה־הַזֹּאת לַעֲשׂוֹת אוֹתָם וְאָמַר כָּל־
37  הָעָם אָמֵן׃ [פ]
38   וְהָיָה אִם־שָׁמוֹעַ תִּשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ
39   לִשְׁמֹר לַעֲשׂוֹת אֶת־כָּל־מִצְו͏ֹתָיו אֲשֶׁר אָנֹכִי מְצַוְּךָ
40   הַיּוֹם וּנְתָנְךָ יְהוָה אֱלֹהֶיךָ עֶלְיוֹן עַל כָּל־גּוֹיֵי
41   הָאָרֶץ׃ וּבָאוּ עָלֶיךָ כָּל־הַבְּרָכוֹת הָאֵלֶּה וְהִשִּׂיגֻךָ
42   כִּי תִשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ׃ בָּרוּךְ אַתָּה בָּעִיר
