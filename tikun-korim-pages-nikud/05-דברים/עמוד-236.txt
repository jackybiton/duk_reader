 1  וְאֵין מַחֲרִיד׃ יַכְּכָה יְהוָה בִּשְׁחִין מִצְרַיִם וּבַטְּחֹרִים
 2   וּבַגָּרָב וּבֶחָרֶס אֲשֶׁר לֹא־תוּכַל לְהֵרָפֵא׃ יַכְּכָה
 3   יְהוָה בְּשִׁגָּעוֹן וּבְעִוָּרוֹן וּבְתִמְהוֹן לֵבָב׃ וְהָיִיתָ
 4   מְמַשֵּׁשׁ בַּצָּהֳרַיִם כַּאֲשֶׁר יְמַשֵּׁשׁ הַעִוֵּר בָּאֲפֵלָה
 5   וְלֹא תַצְלִיחַ אֶת־דְּרָכֶיךָ וְהָיִיתָ אַךְ עָשׁוּק וְגָזוּל
 6   כָּל־הַיָּמִים וְאֵין מוֹשִׁיעַ׃ אִשָּׁה תְאָרֵשׂ וְאִישׁ אַחֵר
 7   יִשְׁכָּבֶנָּה בַּיִת תִּבְנֶה וְלֹא־תֵשֵׁב בּוֹ כֶּרֶם תִּטַּע וְלֹא
 8   תְחַלְּלֶנּוּ׃ שׁוֹרְךָ טָבוּחַ לְעֵינֶיךָ וְלֹא תֹאכַל מִמֶּנּוּ
 9   חֲמֹרְךָ גָּזוּל מִלְּפָנֶיךָ וְלֹא יָשׁוּב לָךְ צֹאנְךָ נְתֻנוֹת
10   לְאֹיְבֶיךָ וְאֵין לְךָ מוֹשִׁיעַ׃ בָּנֶיךָ וּבְנֹתֶיךָ נְתֻנִים
11   לְעַם אַחֵר וְעֵינֶיךָ רֹאוֹת וְכָלוֹת אֲלֵיהֶם כָּל־הַיּוֹם
12   וְאֵין לְאֵל יָדֶךָ׃ פְּרִי אַדְמָתְךָ וְכָל־יְגִיעֲךָ יֹאכַל
13   עַם אֲשֶׁר לֹא־יָדָעְתָּ וְהָיִיתָ רַק עָשׁוּק וְרָצוּץ כָּל־
14  הַיָּמִים׃ וְהָיִיתָ מְשֻׁגָּע מִמַּרְאֵה עֵינֶיךָ אֲשֶׁר תִּרְאֶה׃
15   יַכְּכָה יְהוָה בִּשְׁחִין רָע עַל־הַבִּרְכַּיִם וְעַל־הַשֹּׁקַיִם
16   אֲשֶׁר לֹא־תוּכַל לְהֵרָפֵא מִכַּף רַגְלְךָ וְעַד קָדְקֳדֶךָ׃
17   יוֹלֵךְ יְהוָה אֹתְךָ וְאֶת־מַלְכְּךָ אֲשֶׁר תָּקִים עָלֶיךָ
18   אֶל־גּוֹי אֲשֶׁר לֹא־יָדַעְתָּ אַתָּה וַאֲבֹתֶיךָ וְעָבַדְתָּ
19   שָּׁם אֱלֹהִים אֲחֵרִים עֵץ וָאָבֶן׃ וְהָיִיתָ לְשַׁמָּה
20   לְמָשָׁל וְלִשְׁנִינָה בְּכֹל הָעַמִּים אֲשֶׁר־יְנַהֶגְךָ יְהוָה
21   שָׁמָּה׃ זֶרַע רַב תּוֹצִיא הַשָּׂדֶה וּמְעַט תֶּאֱסֹף כִּי
22   יַחְסְלֶנּוּ הָאַרְבֶּה׃ כְּרָמִים תִּטַּע וְעָבָדְתָּ וְיַיִן לֹא־
23  תִשְׁתֶּה וְלֹא תֶאֱגֹר כִּי תֹאכְלֶנּוּ הַתֹּלָעַת׃ זֵיתִים
24   יִהְיוּ לְךָ בְּכָל־גְּבוּלֶךָ וְשֶׁמֶן לֹא תָסוּךְ כִּי יִשַּׁל
25   זֵיתֶךָ׃ בָּנִים וּבָנוֹת תּוֹלִיד וְלֹא־יִהְיוּ לָךְ כִּי יֵלְכוּ
26   בַּשֶּׁבִי׃ כָּל־עֵצְךָ וּפְרִי אַדְמָתֶךָ יְיָרֵשׁ הַצְּלָצַל׃ הַגֵּר
27   אֲשֶׁר בְּקִרְבְּךָ יַעֲלֶה עָלֶיךָ מַעְלָה מָּעְלָה וְאַתָּה
28   תֵרֵד מַטָּה מָּטָּה׃ הוּא יַלְוְךָ וְאַתָּה לֹא תַלְוֶנּוּ הוּא
29   יִהְיֶה לְרֹאשׁ וְאַתָּה תִּהְיֶה לְזָנָב׃ וּבָאוּ עָלֶיךָ כָּל־
30  הַקְּלָלוֹת הָאֵלֶּה וּרְדָפוּךָ וְהִשִּׂיגוּךָ עַד הִשָּׁמְדָךְ
31   כִּי־לֹא שָׁמַעְתָּ בְּקוֹל יְהוָה אֱלֹהֶיךָ לִשְׁמֹר מִצְו͏ֹתָיו
32   וְחֻקֹּתָיו אֲשֶׁר צִוָּךְ׃ וְהָיוּ בְךָ לְאוֹת וּלְמוֹפֵת וּבְזַרְעֲךָ
33   עַד־עוֹלָם׃ תַּחַת אֲשֶׁר לֹא־עָבַדְתָּ אֶת־יְהוָה אֱלֹהֶיךָ
34   בְּשִׂמְחָה וּבְטוּב לֵבָב מֵרֹב כֹּל׃ וְעָבַדְתָּ אֶת־אֹיְבֶיךָ
35   אֲשֶׁר יְשַׁלְּחֶנּוּ יְהוָה בָּךְ בְּרָעָב וּבְצָמָא וּבְעֵירֹם
36   וּבְחֹסֶר כֹּל וְנָתַן עֹל בַּרְזֶל עַל־צַוָּארֶךָ עַד
37   הִשְׁמִידוֹ אֹתָךְ׃ יִשָּׂא יְהוָה עָלֶיךָ גּוֹי מֵרָחֹק מִקְצֵה
38   הָאָרֶץ כַּאֲשֶׁר יִדְאֶה הַנָּשֶׁר גּוֹי אֲשֶׁר לֹא־תִשְׁמַע
39   לְשֹׁנוֹ׃ גּוֹי עַז פָּנִים אֲשֶׁר לֹא־יִשָּׂא פָנִים לְזָקֵן וְנַעַר
40   לֹא יָחֹן׃ וְאָכַל פְּרִי בְהֶמְתְּךָ וּפְרִי־אַדְמָתְךָ עַד
41   הִשָּׁמְדָךְ אֲשֶׁר לֹא־יַשְׁאִיר לְךָ דָּגָן תִּירוֹשׁ וְיִצְהָר
42   שְׁגַר אֲלָפֶיךָ וְעַשְׁתְּרֹת צֹאנֶךָ עַד הַאֲבִידוֹ אֹתָךְ׃
