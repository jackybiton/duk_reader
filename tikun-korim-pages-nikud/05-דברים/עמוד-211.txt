 1  וְנָשַׁל גּוֹיִם־רַבִּים  מִפָּנֶיךָ הַחִתִּי וְהַגִּרְגָּשִׁי וְהָאֱמֹרִי
 2   וְהַכְּנַעֲנִי וְהַפְּרִזִּי וְהַחִוִּי וְהַיְבוּסִי שִׁבְעָה גוֹיִם רַבִּים
 3   וַעֲצוּמִים מִמֶּךָּ׃ וּנְתָנָם יְהוָה אֱלֹהֶיךָ לְפָנֶיךָ וְהִכִּיתָם
 4   הַחֲרֵם תַּחֲרִים אֹתָם לֹא־תִכְרֹת לָהֶם בְּרִית וְלֹא
 5   תְחָנֵּם׃ וְלֹא תִתְחַתֵּן בָּם בִּתְּךָ לֹא־תִתֵּן לִבְנוֹ וּבִתּוֹ לֹא־
 6  תִקַּח לִבְנֶךָ׃ כִּי־יָסִיר אֶת־בִּנְךָ מֵאַחֲרַי וְעָבְדוּ אֱלֹהִים
 7   אֲחֵרִים וְחָרָה אַף־יְהוָה בָּכֶם וְהִשְׁמִידְךָ מַהֵר׃ כִּי־
 8  אִם־כֹּה תַעֲשׂוּ לָהֶם מִזְבְּחֹתֵיהֶם תִּתֹּצוּ וּמַצֵּבֹתָם
 9   תְּשַׁבֵּרוּ וַאֲשֵׁירֵהֶם תְּגַדֵּעוּן וּפְסִילֵיהֶם תִּשְׂרְפוּן
10   בָּאֵשׁ׃ כִּי עַם קָדוֹשׁ אַתָּה לַיהוָה אֱלֹהֶיךָ בְּךָ
11   בָּחַר  יְהוָה אֱלֹהֶיךָ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים
12   אֲשֶׁר עַל־פְּנֵי הָאֲדָמָה׃ לֹא מֵרֻבְּכֶם מִכָּל־הָעַמִּים
13   חָשַׁק יְהוָה בָּכֶם וַיִּבְחַר בָּכֶם כִּי־אַתֶּם הַמְעַט
14   מִכָּל־הָעַמִּים׃ כִּי מֵאַהֲבַת יְהוָה אֶתְכֶם וּמִשָּׁמְרוֹ
15   אֶת־הַשְּׁבֻעָה אֲשֶׁר נִשְׁבַּע לַאֲבֹתֵיכֶם הוֹצִיא יְהוָה
16   אֶתְכֶם בְּיָד חֲזָקָה וַיִּפְדְּךָ מִבֵּית עֲבָדִים מִיַּד פַּרְעֹה
17   מֶלֶךְ־מִצְרָיִם׃  וְיָדַעְתָּ כִּי־יְהוָה אֱלֹהֶיךָ הוּא הָאֱלֹהִים
18   הָאֵל הַנֶּאֱמָן שֹׁמֵר הַבְּרִית וְהַחֶסֶד לְאֹהֲבָיו וּלְשֹׁמְרֵי
19   מִצְו͏ֹתָו לְאֶלֶף דּוֹר׃ וּמְשַׁלֵּם לְשֹׂנְאָיו אֶל־פָּנָיו
20   לְהַאֲבִידוֹ לֹא יְאַחֵר לְשֹׂנְאוֹ אֶל־פָּנָיו יְשַׁלֶּם־לוֹ׃
21   וְשָׁמַרְתָּ אֶת־הַמִּצְוָה וְאֶת־הַחֻקִּים וְאֶת־הַמִּשְׁפָּטִים
22   אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם לַעֲשׂוֹתָם׃ [פ]
23  וְהָיָה  עֵקֶב תִּשְׁמְעוּן אֵת הַמִּשְׁפָּטִים הָאֵלֶּה וּשְׁמַרְתֶּם
24   וַעֲשִׂיתֶם אֹתָם וְשָׁמַר יְהוָה אֱלֹהֶיךָ לְךָ אֶת־הַבְּרִית
25   וְאֶת־הַחֶסֶד אֲשֶׁר נִשְׁבַּע לַאֲבֹתֶיךָ׃ וַאֲהֵבְךָ וּבֵרַכְךָ
26   וְהִרְבֶּךָ וּבֵרַךְ פְּרִי־בִטְנְךָ וּפְרִי־אַדְמָתֶךָ דְּגָנְךָ
27   וְתִירֹשְׁךָ וְיִצְהָרֶךָ שְׁגַר־אֲלָפֶיךָ וְעַשְׁתְּרֹת צֹאנֶךָ עַל
28   הָאֲדָמָה אֲשֶׁר־נִשְׁבַּע לַאֲבֹתֶיךָ לָתֶת לָךְ׃ בָּרוּךְ תִּהְיֶה
29   מִכָּל־הָעַמִּים לֹא־יִהְיֶה בְךָ עָקָר וַעֲקָרָה וּבִבְהֶמְתֶּךָ׃
30   וְהֵסִיר יְהוָה מִמְּךָ כָּל־חֹלִי וְכָל־מַדְוֵי מִצְרַיִם הָרָעִים
31   אֲשֶׁר יָדַעְתָּ לֹא יְשִׂימָם בָּךְ וּנְתָנָם בְּכָל־שֹׂנְאֶיךָ׃
32   וְאָכַלְתָּ אֶת־כָּל־הָעַמִּים אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ
33   לֹא־תָחוֹס עֵינְךָ עֲלֵיהֶם וְלֹא תַעֲבֹד אֶת־אֱלֹהֵיהֶם
34   כִּי־מוֹקֵשׁ הוּא לָךְ׃ [ס]             כִּי
35   תֹאמַר בִּלְבָבְךָ רַבִּים הַגּוֹיִם הָאֵלֶּה מִמֶּנִּי אֵיכָה
36   אוּכַל לְהוֹרִישָׁם׃ לֹא תִירָא מֵהֶם זָכֹר תִּזְכֹּר
37   אֵת אֲשֶׁר־עָשָׂה יְהוָה אֱלֹהֶיךָ לְפַרְעֹה וּלְכָל־
38  מִצְרָיִם׃ הַמַּסֹּת הַגְּדֹלֹת אֲשֶׁר־רָאוּ עֵינֶיךָ וְהָאֹתֹת
39   וְהַמֹּפְתִים וְהַיָּד הַחֲזָקָה וְהַזְּרֹעַ הַנְּטוּיָה אֲשֶׁר
40   הוֹצִאֲךָ יְהוָה אֱלֹהֶיךָ כֵּן־יַעֲשֶׂה יְהוָה אֱלֹהֶיךָ לְכָל־
41  הָעַמִּים אֲשֶׁר־אַתָּה יָרֵא מִפְּנֵיהֶם׃ וְגַם אֶת־הַצִּרְעָה
42   יְשַׁלַּח יְהוָה אֱלֹהֶיךָ בָּם עַד־אֲבֹד הַנִּשְׁאָרִים
