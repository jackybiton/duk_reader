 1  וּבַצֹּאן וּבַיַּיִן וּבַשֵּׁכָר וּבְכֹל אֲשֶׁר תִּשְׁאָלְךָ נַפְשֶׁךָ
 2   וְאָכַלְתָּ שָּׁם לִפְנֵי יְהוָה אֱלֹהֶיךָ וְשָׂמַחְתָּ אַתָּה
 3   וּבֵיתֶךָ׃ וְהַלֵּוִי אֲשֶׁר־בִּשְׁעָרֶיךָ לֹא תַעַזְבֶנּוּ כִּי אֵין לוֹ
 4   חֵלֶק וְנַחֲלָה עִמָּךְ׃ [ס]             מִקְצֵה  שָׁלֹשׁ
 5   שָׁנִים תּוֹצִיא אֶת־כָּל־מַעְשַׂר תְּבוּאָתְךָ בַּשָּׁנָה הַהִוא
 6   וְהִנַּחְתָּ בִּשְׁעָרֶיךָ׃ וּבָא הַלֵּוִי כִּי אֵין־לוֹ חֵלֶק וְנַחֲלָה
 7   עִמָּךְ וְהַגֵּר וְהַיָּתוֹם וְהָאַלְמָנָה אֲשֶׁר בִּשְׁעָרֶיךָ
 8   וְאָכְלוּ וְשָׂבֵעוּ לְמַעַן יְבָרֶכְךָ יְהוָה אֱלֹהֶיךָ בְּכָל־
 9  מַעֲשֵׂה יָדְךָ אֲשֶׁר תַּעֲשֶׂה׃ [ס]              מִקֵּץ
10   שֶׁבַע־שָׁנִים תַּעֲשֶׂה שְׁמִטָּה׃ וְזֶה דְּבַר הַשְּׁמִטָּה
11   שָׁמוֹט כָּל־בַּעַל מַשֵּׁה יָדוֹ אֲשֶׁר יַשֶּׁה בְּרֵעֵהוּ לֹא־
12  יִגֹּשׂ אֶת־רֵעֵהוּ וְאֶת־אָחִיו כִּי־קָרָא שְׁמִטָּה לַיהוָה׃
13   אֶת־הַנָּכְרִי תִּגֹּשׂ וַאֲשֶׁר יִהְיֶה לְךָ אֶת־אָחִיךָ תַּשְׁמֵט
14   יָדֶךָ׃ אֶפֶס כִּי לֹא יִהְיֶה־בְּךָ אֶבְיוֹן כִּי־בָרֵךְ יְבָרֶכְךָ
15   יְהוָה בָּאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן־לְךָ נַחֲלָה
16   לְרִשְׁתָּהּ׃ רַק אִם־שָׁמוֹעַ תִּשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ
17   לִשְׁמֹר לַעֲשׂוֹת אֶת־כָּל־הַמִּצְוָה הַזֹּאת אֲשֶׁר אָנֹכִי
18   מְצַוְּךָ הַיּוֹם׃ כִּי־יְהוָה אֱלֹהֶיךָ בֵּרַכְךָ כַּאֲשֶׁר דִּבֶּר־
19  לָךְ וְהַעֲבַטְתָּ גּוֹיִם רַבִּים וְאַתָּה לֹא תַעֲבֹט וּמָשַׁלְתָּ
20   בְּגוֹיִם רַבִּים וּבְךָ לֹא יִמְשֹׁלוּ׃ [ס]             כִּי־
21  יִהְיֶה בְךָ אֶבְיוֹן מֵאַחַד אַחֶיךָ בְּאַחַד שְׁעָרֶיךָ
22   בְּאַרְצְךָ אֲשֶׁר־יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ לֹא תְאַמֵּץ
23   אֶת־לְבָבְךָ וְלֹא תִקְפֹּץ אֶת־יָדְךָ מֵאָחִיךָ הָאֶבְיוֹן׃
24   כִּי־פָתֹחַ תִּפְתַּח אֶת־יָדְךָ לוֹ וְהַעֲבֵט תַּעֲבִיטֶנּוּ דֵּי
25   מַחְסֹרוֹ אֲשֶׁר יֶחְסַר לוֹ׃ הִשָּׁמֶר לְךָ פֶּן־יִהְיֶה דָבָר
26   עִם־לְבָבְךָ בְלִיַּעַל לֵאמֹר קָרְבָה שְׁנַת־הַשֶּׁבַע שְׁנַת
27   הַשְּׁמִטָּה וְרָעָה עֵינְךָ בְּאָחִיךָ הָאֶבְיוֹן וְלֹא תִתֵּן לוֹ
28   וְקָרָא עָלֶיךָ אֶל־יְהוָה וְהָיָה בְךָ חֵטְא׃ נָתוֹן תִּתֵּן לוֹ
29   וְלֹא־יֵרַע לְבָבְךָ בְּתִתְּךָ לוֹ כִּי בִּגְלַל  הַדָּבָר הַזֶּה
30   יְבָרֶכְךָ יְהוָה אֱלֹהֶיךָ בְּכָל־מַעֲשֶׂךָ וּבְכֹל מִשְׁלַח
31   יָדֶךָ׃ כִּי לֹא־יֶחְדַּל אֶבְיוֹן מִקֶּרֶב הָאָרֶץ עַל־כֵּן אָנֹכִי
32   מְצַוְּךָ לֵאמֹר פָּתֹחַ תִּפְתַּח אֶת־יָדְךָ לְאָחִיךָ לַעֲנִיֶּךָ
33   וּלְאֶבְיֹנְךָ בְּאַרְצֶךָ׃ [ס]             כִּי־
34  יִמָּכֵר לְךָ אָחִיךָ הָעִבְרִי אוֹ הָעִבְרִיָּה וַעֲבָדְךָ שֵׁשׁ
35   שָׁנִים וּבַשָּׁנָה הַשְּׁבִיעִת תְּשַׁלְּחֶנּוּ חָפְשִׁי מֵעִמָּךְ׃
36   וְכִי־תְשַׁלְּחֶנּוּ חָפְשִׁי מֵעִמָּךְ לֹא תְשַׁלְּחֶנּוּ רֵיקָם׃
37   הַעֲנֵיק תַּעֲנִיק לוֹ מִצֹּאנְךָ וּמִגָּרְנְךָ וּמִיִּקְבֶךָ אֲשֶׁר
38   בֵּרַכְךָ יְהוָה אֱלֹהֶיךָ תִּתֶּן־לוֹ׃ וְזָכַרְתָּ כִּי עֶבֶד הָיִיתָ
39   בְּאֶרֶץ מִצְרַיִם וַיִּפְדְּךָ יְהוָה אֱלֹהֶיךָ עַל־כֵּן אָנֹכִי
40   מְצַוְּךָ אֶת־הַדָּבָר הַזֶּה הַיּוֹם׃ וְהָיָה כִּי־יֹאמַר אֵלֶיךָ
41   לֹא אֵצֵא מֵעִמָּךְ כִּי אֲהֵבְךָ וְאֶת־בֵּיתֶךָ כִּי־טוֹב
42   לוֹ עִמָּךְ׃ וְלָקַחְתָּ אֶת־הַמַּרְצֵעַ וְנָתַתָּה בְאָזְנוֹ וּבַדֶּלֶת
