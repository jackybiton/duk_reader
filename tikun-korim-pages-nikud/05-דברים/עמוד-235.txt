 1  וּבָרוּךְ אַתָּה בַּשָּׂדֶה׃ בָּרוּךְ פְּרִי־בִטְנְךָ וּפְרִי אַדְמָתְךָ
 2   וּפְרִי בְהֶמְתֶּךָ שְׁגַר אֲלָפֶיךָ וְעַשְׁתְּרוֹת צֹאנֶךָ׃ בָּרוּךְ
 3   טַנְאֲךָ וּמִשְׁאַרְתֶּךָ׃ בָּרוּךְ אַתָּה בְּבֹאֶךָ וּבָרוּךְ אַתָּה
 4   בְּצֵאתֶךָ׃  יִתֵּן יְהוָה אֶת־אֹיְבֶיךָ הַקָּמִים עָלֶיךָ נִגָּפִים
 5   לְפָנֶיךָ בְּדֶרֶךְ אֶחָד יֵצְאוּ אֵלֶיךָ וּבְשִׁבְעָה דְרָכִים
 6   יָנוּסוּ לְפָנֶיךָ׃ יְצַו יְהוָה אִתְּךָ אֶת־הַבְּרָכָה בַּאֲסָמֶיךָ
 7   וּבְכֹל מִשְׁלַח יָדֶךָ וּבֵרַכְךָ בָּאָרֶץ אֲשֶׁר־יְהוָה
 8   אֱלֹהֶיךָ נֹתֵן לָךְ׃ יְקִימְךָ יְהוָה לוֹ לְעַם קָדוֹשׁ
 9   כַּאֲשֶׁר נִשְׁבַּע־לָךְ כִּי תִשְׁמֹר אֶת־מִצְו͏ֹת יְהוָה
10   אֱלֹהֶיךָ וְהָלַכְתָּ בִּדְרָכָיו׃ וְרָאוּ כָּל־עַמֵּי הָאָרֶץ
11   כִּי שֵׁם יְהוָה נִקְרָא עָלֶיךָ וְיָרְאוּ מִמֶּךָּ׃ וְהוֹתִרְךָ
12   יְהוָה לְטוֹבָה בִּפְרִי בִטְנְךָ וּבִפְרִי בְהֶמְתְּךָ וּבִפְרִי
13   אַדְמָתֶךָ עַל הָאֲדָמָה אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֶיךָ
14   לָתֶת לָךְ׃ יִפְתַּח יְהוָה  לְךָ אֶת־אוֹצָרוֹ הַטּוֹב אֶת־
15  הַשָּׁמַיִם לָתֵת מְטַר־אַרְצְךָ בְּעִתּוֹ וּלְבָרֵךְ אֵת כָּל־
16  מַעֲשֵׂה יָדֶךָ וְהִלְוִיתָ גּוֹיִם רַבִּים וְאַתָּה לֹא תִלְוֶה׃
17   וּנְתָנְךָ יְהוָה לְרֹאשׁ וְלֹא לְזָנָב וְהָיִיתָ רַק לְמַעְלָה
18   וְלֹא תִהְיֶה לְמָטָּה כִּי־תִשְׁמַע אֶל־מִצְו͏ֹת  יְהוָה
19   אֱלֹהֶיךָ אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם לִשְׁמֹר וְלַעֲשׂוֹת׃
20   וְלֹא תָסוּר מִכָּל־הַדְּבָרִים אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם
21   הַיּוֹם יָמִין וּשְׂמֹאול לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים
22   לְעָבְדָם׃ [פ]
23   וְהָיָה אִם־לֹא תִשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ לִשְׁמֹר
24   לַעֲשׂוֹת אֶת־כָּל־מִצְו͏ֹתָיו וְחֻקֹּתָיו אֲשֶׁר אָנֹכִי מְצַוְּךָ
25   הַיּוֹם וּבָאוּ עָלֶיךָ כָּל־הַקְּלָלוֹת הָאֵלֶּה וְהִשִּׂיגוּךָ׃
26   אָרוּר אַתָּה בָּעִיר וְאָרוּר אַתָּה בַּשָּׂדֶה׃ אָרוּר
27   טַנְאֲךָ וּמִשְׁאַרְתֶּךָ׃ אָרוּר פְּרִי־בִטְנְךָ וּפְרִי אַדְמָתֶךָ
28   שְׁגַר אֲלָפֶיךָ וְעַשְׁתְּרֹת צֹאנֶךָ׃ אָרוּר אַתָּה בְּבֹאֶךָ
29   וְאָרוּר אַתָּה בְּצֵאתֶךָ׃ יְשַׁלַּח יְהוָה  בְּךָ אֶת־הַמְּאֵרָה
30   אֶת־הַמְּהוּמָה וְאֶת־הַמִּגְעֶרֶת בְּכָל־מִשְׁלַח יָדְךָ
31   אֲשֶׁר תַּעֲשֶׂה עַד הִשָּׁמֶדְךָ וְעַד־אֲבָדְךָ מַהֵר
32   מִפְּנֵי רֹעַ מַעֲלָלֶיךָ אֲשֶׁר עֲזַבְתָּנִי׃ יַדְבֵּק יְהוָה בְּךָ
33   אֶת־הַדָּבֶר עַד כַּלֹּתוֹ אֹתְךָ מֵעַל הָאֲדָמָה אֲשֶׁר־
34  אַתָּה בָא־שָׁמָּה לְרִשְׁתָּהּ׃ יַכְּכָה יְהוָה בַּשַּׁחֶפֶת
35   וּבַקַּדַּחַת וּבַדַּלֶּקֶת וּבַחַרְחֻר וּבַחֶרֶב וּבַשִּׁדָּפוֹן
36   וּבַיֵּרָקוֹן וּרְדָפוּךָ עַד אָבְדֶךָ׃ וְהָיוּ שָׁמֶיךָ אֲשֶׁר עַל־
37  רֹאשְׁךָ נְחֹשֶׁת וְהָאָרֶץ אֲשֶׁר־תַּחְתֶּיךָ בַּרְזֶל׃ יִתֵּן
38   יְהוָה אֶת־מְטַר אַרְצְךָ אָבָק וְעָפָר מִן־הַשָּׁמַיִם יֵרֵד
39   עָלֶיךָ עַד הִשָּׁמְדָךְ׃ יִתֶּנְךָ יְהוָה  נִגָּף לִפְנֵי אֹיְבֶיךָ
40   בְּדֶרֶךְ אֶחָד תֵּצֵא אֵלָיו וּבְשִׁבְעָה דְרָכִים תָּנוּס
41   לְפָנָיו וְהָיִיתָ לְזַעֲוָה לְכֹל מַמְלְכוֹת הָאָרֶץ׃ וְהָיְתָה
42   נִבְלָתְךָ לְמַאֲכָל לְכָל־עוֹף הַשָּׁמַיִם וּלְבֶהֱמַת הָאָרֶץ
