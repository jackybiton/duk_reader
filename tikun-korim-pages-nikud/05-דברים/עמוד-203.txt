 1  וְאֶת־הָעָם צַו לֵאמֹר אַתֶּם עֹבְרִים בִּגְבוּל אֲחֵיכֶם
 2   בְּנֵי־עֵשָׂו הַיֹּשְׁבִים בְּשֵׂעִיר וְיִירְאוּ מִכֶּם וְנִשְׁמַרְתֶּם
 3   מְאֹד׃ אַל־תִּתְגָּרוּ בָם כִּי לֹא־אֶתֵּן לָכֶם מֵאַרְצָם
 4   עַד מִדְרַךְ כַּף־רָגֶל כִּי־יְרֻשָּׁה לְעֵשָׂו נָתַתִּי אֶת־
 5  הַר שֵׂעִיר׃ אֹכֶל תִּשְׁבְּרוּ מֵאִתָּם בַּכֶּסֶף וַאֲכַלְתֶּם
 6   וְגַם־מַיִם תִּכְרוּ מֵאִתָּם בַּכֶּסֶף וּשְׁתִיתֶם׃ כִּי יְהוָה
 7   אֱלֹהֶיךָ בֵּרַכְךָ בְּכֹל מַעֲשֵׂה יָדֶךָ יָדַע לֶכְתְּךָ
 8   אֶת־הַמִּדְבָּר הַגָּדֹל הַזֶּה זֶה  אַרְבָּעִים שָׁנָה יְהוָה
 9   אֱלֹהֶיךָ עִמָּךְ לֹא חָסַרְתָּ דָּבָר׃ וַנַּעֲבֹר מֵאֵת אַחֵינוּ
10   בְנֵי־עֵשָׂו הַיֹּשְׁבִים בְּשֵׂעִיר מִדֶּרֶךְ הָעֲרָבָה מֵאֵילַת
11   וּמֵעֶצְיֹן גָּבֶר [ס]             וַנֵּפֶן וַנַּעֲבֹר דֶּרֶךְ
12   מִדְבַּר מוֹאָב׃ וַיֹּאמֶר יְהוָה אֵלַי אַל־תָּצַר אֶת־מוֹאָב
13   וְאַל־תִּתְגָּר בָּם מִלְחָמָה כִּי לֹא־אֶתֵּן לְךָ מֵאַרְצוֹ
14   יְרֻשָּׁה כִּי לִבְנֵי־לוֹט נָתַתִּי אֶת־עָר יְרֻשָּׁה׃ הָאֵמִים
15   לְפָנִים יָשְׁבוּ בָהּ עַם גָּדוֹל וְרַב וָרָם כָּעֲנָקִים׃
16   רְפָאִים יֵחָשְׁבוּ אַף־הֵם כָּעֲנָקִים וְהַמֹּאָבִים יִקְרְאוּ
17   לָהֶם אֵמִים׃ וּבְשֵׂעִיר יָשְׁבוּ הַחֹרִים לְפָנִים וּבְנֵי
18   עֵשָׂו יִירָשׁוּם וַיַּשְׁמִידוּם מִפְּנֵיהֶם וַיֵּשְׁבוּ תַּחְתָּם
19   כַּאֲשֶׁר עָשָׂה יִשְׂרָאֵל לְאֶרֶץ יְרֻשָּׁתוֹ אֲשֶׁר־נָתַן
20   יְהוָה לָהֶם׃ עַתָּה קֻמוּ וְעִבְרוּ לָכֶם אֶת־נַחַל זָרֶד
21   וַנַּעֲבֹר אֶת־נַחַל זָרֶד׃ וְהַיָּמִים אֲשֶׁר־הָלַכְנוּ  מִקָּדֵשׁ
22   בַּרְנֵעַ עַד אֲשֶׁר־עָבַרְנוּ אֶת־נַחַל זֶרֶד שְׁלֹשִׁים
23   וּשְׁמֹנֶה שָׁנָה עַד־תֹּם כָּל־הַדּוֹר אַנְשֵׁי הַמִּלְחָמָה
24   מִקֶּרֶב הַמַּחֲנֶה כַּאֲשֶׁר נִשְׁבַּע יְהוָה לָהֶם׃ וְגַם יַד־
25  יְהוָה הָיְתָה בָּם לְהֻמָּם מִקֶּרֶב הַמַּחֲנֶה עַד תֻּמָּם׃
26   וַיְהִי כַאֲשֶׁר־תַּמּוּ כָּל־אַנְשֵׁי הַמִּלְחָמָה לָמוּת מִקֶּרֶב
27   הָעָם׃ [ס]             וַיְדַבֵּר יְהוָה אֵלַי לֵאמֹר׃
28   אַתָּה עֹבֵר הַיּוֹם אֶת־גְּבוּל מוֹאָב אֶת־עָר׃ וְקָרַבְתָּ
29   מוּל בְּנֵי עַמּוֹן אַל־תְּצֻרֵם וְאַל־תִּתְגָּר בָּם כִּי לֹא־אֶתֵּן
30   מֵאֶרֶץ בְּנֵי־עַמּוֹן לְךָ יְרֻשָּׁה כִּי לִבְנֵי־לוֹט נְתַתִּיהָ
31   יְרֻשָּׁה׃ אֶרֶץ־רְפָאִים תֵּחָשֵׁב אַף־הִוא רְפָאִים יָשְׁבוּ־
32  בָהּ לְפָנִים וְהָעַמֹּנִים יִקְרְאוּ לָהֶם זַמְזֻמִּים׃ עַם גָּדוֹל
33   וְרַב וָרָם כָּעֲנָקִים וַיַּשְׁמִידֵם יְהוָה מִפְּנֵיהֶם וַיִּירָשֻׁם
34   וַיֵּשְׁבוּ תַחְתָּם׃ כַּאֲשֶׁר עָשָׂה לִבְנֵי עֵשָׂו הַיֹּשְׁבִים
35   בְּשֵׂעִיר אֲשֶׁר הִשְׁמִיד אֶת־הַחֹרִי מִפְּנֵיהֶם וַיִּירָשֻׁם
36   וַיֵּשְׁבוּ תַחְתָּם עַד הַיּוֹם הַזֶּה׃ וְהָעַוִּים הַיֹּשְׁבִים
37   בַּחֲצֵרִים עַד־עַזָּה כַּפְתֹּרִים הַיֹּצְאִים מִכַּפְתֹּר
38   הִשְׁמִידֻם וַיֵּשְׁבוּ תַחְתָּם׃ קוּמוּ סְּעוּ וְעִבְרוּ אֶת־נַחַל
39   אַרְנֹן רְאֵה נָתַתִּי בְיָדְךָ אֶת־סִיחֹן מֶלֶךְ־חֶשְׁבּוֹן הָאֱמֹרִי
40   וְאֶת־אַרְצוֹ הָחֵל רָשׁ וְהִתְגָּר בּוֹ מִלְחָמָה׃ הַיּוֹם הַזֶּה
41   אָחֵל תֵּת פַּחְדְּךָ וְיִרְאָתְךָ עַל־פְּנֵי הָעַמִּים תַּחַת כָּל־
42  הַשָּׁמָיִם אֲשֶׁר יִשְׁמְעוּן שִׁמְעֲךָ וְרָגְזוּ וְחָלוּ מִפָּנֶיךָ׃
