 1  וּמִצְו͏ֹתָיו אֲשֶׁר אָנֹכִי מְצַוֶּךָ אַתָּה וּבִנְךָ וּבֶן־בִּנְךָ כֹּל
 2   יְמֵי חַיֶּיךָ וּלְמַעַן יַאֲרִכֻן יָמֶיךָ׃ וְשָׁמַעְתָּ יִשְׂרָאֵל
 3   וְשָׁמַרְתָּ לַעֲשׂוֹת אֲשֶׁר יִיטַב לְךָ וַאֲשֶׁר תִּרְבּוּן
 4   מְאֹד כַּאֲשֶׁר דִּבֶּר יְהוָה אֱלֹהֵי אֲבֹתֶיךָ לָךְ אֶרֶץ
 5   זָבַת חָלָב וּדְבָשׁ׃ [פ]
 6    שְׁמַע יִשְׂרָאֵל יְהוָה אֱלֹהֵינוּ יְהוָה  אֶחָד׃
 7   וְאָהַבְתָּ אֵת יְהוָה אֱלֹהֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשְׁךָ
 8   וּבְכָל־מְאֹדֶךָ׃ וְהָיוּ הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוְּךָ
 9   הַיּוֹם עַל־לְבָבֶךָ׃ וְשִׁנַּנְתָּם לְבָנֶיךָ וְדִבַּרְתָּ בָּם בְּשִׁבְתְּךָ
10   בְּבֵיתֶךָ וּבְלֶכְתְּךָ בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ׃ וּקְשַׁרְתָּם
11   לְאוֹת עַל־יָדֶךָ וְהָיוּ לְטֹטָפֹת בֵּין עֵינֶיךָ׃ וּכְתַבְתָּם עַל־
12  מְזֻזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ׃ [ס]             וְהָיָה
13   כִּי יְבִיאֲךָ  יְהוָה אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר נִשְׁבַּע
14   לַאֲבֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב לָתֶת לָךְ עָרִים
15   גְּדֹלֹת וְטֹבֹת אֲשֶׁר לֹא־בָנִיתָ׃ וּבָתִּים מְלֵאִים כָּל־
16  טוּב אֲשֶׁר לֹא־מִלֵּאתָ וּבֹרֹת חֲצוּבִים אֲשֶׁר לֹא־
17  חָצַבְתָּ כְּרָמִים וְזֵיתִים אֲשֶׁר לֹא־נָטָעְתָּ וְאָכַלְתָּ
18   וְשָׂבָעְתָּ׃ הִשָּׁמֶר לְךָ פֶּן־תִּשְׁכַּח אֶת־יְהוָה אֲשֶׁר
19   הוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים׃ אֶת־יְהוָה
20   אֱלֹהֶיךָ תִּירָא וְאֹתוֹ תַעֲבֹד וּבִשְׁמוֹ תִּשָּׁבֵעַ׃ לֹא
21   תֵלְכוּן אַחֲרֵי אֱלֹהִים אֲחֵרִים מֵאֱלֹהֵי הָעַמִּים אֲשֶׁר
22   סְבִיבוֹתֵיכֶם׃ כִּי אֵל קַנָּא יְהוָה אֱלֹהֶיךָ בְּקִרְבֶּךָ פֶּן־
23  יֶחֱרֶה אַף־יְהוָה אֱלֹהֶיךָ בָּךְ וְהִשְׁמִידְךָ מֵעַל פְּנֵי
24   הָאֲדָמָה׃ [ס]             לֹא תְנַסּוּ אֶת־יְהוָה
25   אֱלֹהֵיכֶם כַּאֲשֶׁר נִסִּיתֶם בַּמַּסָּה׃ שָׁמוֹר תִּשְׁמְרוּן
26   אֶת־מִצְו͏ֹת יְהוָה אֱלֹהֵיכֶם וְעֵדֹתָיו וְחֻקָּיו אֲשֶׁר צִוָּךְ׃
27   וְעָשִׂיתָ הַיָּשָׁר וְהַטּוֹב בְּעֵינֵי יְהוָה לְמַעַן יִיטַב לָךְ
28   וּבָאתָ וְיָרַשְׁתָּ אֶת־הָאָרֶץ הַטֹּבָה אֲשֶׁר־נִשְׁבַּע יְהוָה
29   לַאֲבֹתֶיךָ׃ לַהֲדֹף אֶת־כָּל־אֹיְבֶיךָ מִפָּנֶיךָ כַּאֲשֶׁר דִּבֶּר
30   יְהוָה׃ [ס]             כִּי־יִשְׁאָלְךָ בִנְךָ מָחָר
31   לֵאמֹר מָה הָעֵדֹת וְהַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר צִוָּה
32   יְהוָה אֱלֹהֵינוּ אֶתְכֶם׃ וְאָמַרְתָּ לְבִנְךָ עֲבָדִים הָיִינוּ
33   לְפַרְעֹה בְּמִצְרָיִם וַיֹּצִיאֵנוּ יְהוָה מִמִּצְרַיִם בְּיָד חֲזָקָה׃
34   וַיִּתֵּן יְהוָה אוֹתֹת וּמֹפְתִים גְּדֹלִים וְרָעִים  בְּמִצְרַיִם
35   בְּפַרְעֹה וּבְכָל־בֵּיתוֹ לְעֵינֵינוּ׃ וְאוֹתָנוּ הוֹצִיא מִשָּׁם
36   לְמַעַן הָבִיא אֹתָנוּ לָתֶת לָנוּ אֶת־הָאָרֶץ אֲשֶׁר נִשְׁבַּע
37   לַאֲבֹתֵינוּ׃ וַיְצַוֵּנוּ יְהוָה לַעֲשׂוֹת אֶת־כָּל־הַחֻקִּים הָאֵלֶּה
38   לְיִרְאָה אֶת־יְהוָה אֱלֹהֵינוּ לְטוֹב לָנוּ כָּל־הַיָּמִים
39   לְחַיֹּתֵנוּ כְּהַיּוֹם הַזֶּה׃ וּצְדָקָה תִּהְיֶה־לָּנוּ כִּי־נִשְׁמֹר
40   לַעֲשׂוֹת אֶת־כָּל־הַמִּצְוָה הַזֹּאת לִפְנֵי יְהוָה אֱלֹהֵינוּ
41   כַּאֲשֶׁר צִוָּנוּ׃ [ס]              כִּי יְבִיאֲךָ יְהוָה
42   אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר־אַתָּה בָא־שָׁמָּה לְרִשְׁתָּהּ
