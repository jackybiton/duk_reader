 1  וְהוֹצִא עָלֶיהָ שֵׁם רָע וְאָמַר אֶת־הָאִשָּׁה הַזֹּאת
 2   לָקַחְתִּי וָאֶקְרַב אֵלֶיהָ וְלֹא־מָצָאתִי לָהּ בְּתוּלִים׃ וְלָקַח
 3   אֲבִי הַנַּעֲרָ וְאִמָּהּ וְהוֹצִיאוּ אֶת־בְּתוּלֵי הַנַּעֲרָ אֶל־זִקְנֵי
 4   הָעִיר הַשָּׁעְרָה׃ וְאָמַר אֲבִי הַנַּעֲרָ אֶל־הַזְּקֵנִים אֶת־בִּתִּי
 5   נָתַתִּי לָאִישׁ הַזֶּה לְאִשָּׁה וַיִּשְׂנָאֶהָ׃ וְהִנֵּה־הוּא שָׂם
 6   עֲלִילֹת דְּבָרִים לֵאמֹר לֹא־מָצָאתִי לְבִתְּךָ בְּתוּלִים
 7   וְאֵלֶּה בְּתוּלֵי בִתִּי וּפָרְשׂוּ הַשִּׂמְלָה לִפְנֵי זִקְנֵי הָעִיר׃
 8   וְלָקְחוּ זִקְנֵי הָעִיר־הַהִוא אֶת־הָאִישׁ וְיִסְּרוּ אֹתוֹ׃ וְעָנְשׁוּ
 9   אֹתוֹ מֵאָה כֶסֶף וְנָתְנוּ לַאֲבִי הַנַּעֲרָה כִּי הוֹצִיא שֵׁם
10   רָע עַל בְּתוּלַת יִשְׂרָאֵל וְלוֹ־תִהְיֶה לְאִשָּׁה לֹא־יוּכַל
11   לְשַׁלְּחָהּ כָּל־יָמָיו׃ [ס]             וְאִם־אֱמֶת
12   הָיָה הַדָּבָר הַזֶּה לֹא־נִמְצְאוּ בְתוּלִים לַנַּעֲרָ׃ וְהוֹצִיאוּ
13   אֶת־הַנַּעֲרָ אֶל־פֶּתַח בֵּית־אָבִיהָ וּסְקָלוּהָ אַנְשֵׁי עִירָהּ
14   בָּאֲבָנִים וָמֵתָה כִּי־עָשְׂתָה נְבָלָה בְּיִשְׂרָאֵל לִזְנוֹת בֵּית
15   אָבִיהָ וּבִעַרְתָּ הָרָע מִקִּרְבֶּךָ׃ [ס]             כִּי־
16  יִמָּצֵא אִישׁ שֹׁכֵב  עִם־אִשָּׁה בְעֻלַת־בַּעַל וּמֵתוּ גַּם־
17  שְׁנֵיהֶם הָאִישׁ הַשֹּׁכֵב עִם־הָאִשָּׁה וְהָאִשָּׁה וּבִעַרְתָּ
18   הָרָע מִיִּשְׂרָאֵל׃ [ס]             כִּי יִהְיֶה
19   נַעֲרָ בְתוּלָה מְאֹרָשָׂה לְאִישׁ וּמְצָאָהּ אִישׁ בָּעִיר
20   וְשָׁכַב עִמָּהּ׃ וְהוֹצֵאתֶם אֶת־שְׁנֵיהֶם אֶל־שַׁעַר  הָעִיר
21   הַהִוא וּסְקַלְתֶּם אֹתָם בָּאֲבָנִים וָמֵתוּ אֶת־הַנַּעֲרָ
22   עַל־דְּבַר אֲשֶׁר לֹא־צָעֲקָה בָעִיר וְאֶת־הָאִישׁ עַל־
23  דְּבַר אֲשֶׁר־עִנָּה אֶת־אֵשֶׁת רֵעֵהוּ וּבִעַרְתָּ הָרָע
24   מִקִּרְבֶּךָ׃ [ס]             וְאִם־בַּשָּׂדֶה יִמְצָא
25   הָאִישׁ אֶת־הַנַּעֲרָ הַמְאֹרָשָׂה וְהֶחֱזִיק־בָּהּ הָאִישׁ
26   וְשָׁכַב עִמָּהּ וּמֵת הָאִישׁ אֲשֶׁר־שָׁכַב עִמָּהּ לְבַדּוֹ׃
27   וְלַנַּעֲרָ לֹא־תַעֲשֶׂה דָבָר אֵין לַנַּעֲרָ חֵטְא מָוֶת כִּי
28   כַּאֲשֶׁר יָקוּם אִישׁ עַל־רֵעֵהוּ וּרְצָחוֹ נֶפֶשׁ כֵּן הַדָּבָר
29   הַזֶּה׃ כִּי בַשָּׂדֶה מְצָאָהּ צָעֲקָה הַנַּעֲרָ הַמְאֹרָשָׂה וְאֵין
30   מוֹשִׁיעַ לָהּ׃ [ס]             כִּי־יִמְצָא אִישׁ
31   נַעֲרָ בְתוּלָה אֲשֶׁר לֹא־אֹרָשָׂה וּתְפָשָׂהּ וְשָׁכַב
32   עִמָּהּ וְנִמְצָאוּ׃ וְנָתַן הָאִישׁ הַשֹּׁכֵב עִמָּהּ לַאֲבִי הַנַּעֲרָ
33   חֲמִשִּׁים כָּסֶף וְלוֹ־תִהְיֶה לְאִשָּׁה תַּחַת אֲשֶׁר עִנָּהּ
34   לֹא־יוּכַל שַׁלְּחָהּ כָּל־יָמָיו׃ [ס]             לֹא־
35  יִקַּח אִישׁ אֶת־אֵשֶׁת אָבִיו וְלֹא יְגַלֶּה כְּנַף
36   אָבִיו׃ [ס]             לֹא־יָבֹא פְצוּעַ־דַּכָּא
37   וּכְרוּת שָׁפְכָה בִּקְהַל יְהוָה׃ [ס]             לֹא־
38  יָבֹא מַמְזֵר בִּקְהַל יְהוָה גַּם דּוֹר עֲשִׂירִי לֹא־יָבֹא לוֹ
39   בִּקְהַל יְהוָה׃ [ס]             לֹא־יָבֹא עַמּוֹנִי
40   וּמוֹאָבִי בִּקְהַל יְהוָה גַּם דּוֹר עֲשִׂירִי לֹא־יָבֹא לָהֶם
41   בִּקְהַל יְהוָה עַד־עוֹלָם׃ עַל־דְּבַר אֲשֶׁר לֹא־קִדְּמוּ
42   אֶתְכֶם בַּלֶּחֶם וּבַמַּיִם בַּדֶּרֶךְ בְּצֵאתְכֶם מִמִּצְרָיִם
