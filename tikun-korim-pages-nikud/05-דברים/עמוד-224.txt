 1  וְהָאִישׁ אֲשֶׁר־יַעֲשֶׂה בְזָדוֹן לְבִלְתִּי שְׁמֹעַ אֶל־הַכֹּהֵן
 2   הָעֹמֵד לְשָׁרֶת שָׁם אֶת־יְהוָה אֱלֹהֶיךָ אוֹ אֶל־הַשֹּׁפֵט
 3   וּמֵת הָאִישׁ הַהוּא וּבִעַרְתָּ הָרָע מִיִּשְׂרָאֵל׃ וְכָל־הָעָם
 4   יִשְׁמְעוּ וְיִרָאוּ וְלֹא יְזִידוּן עוֹד׃ [ס]              כִּי־
 5  תָבֹא אֶל־הָאָרֶץ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ וִירִשְׁתָּהּ
 6   וְיָשַׁבְתָּה בָּהּ וְאָמַרְתָּ אָשִׂימָה עָלַי מֶלֶךְ כְּכָל־הַגּוֹיִם
 7   אֲשֶׁר סְבִיבֹתָי׃ שׂוֹם תָּשִׂים עָלֶיךָ מֶלֶךְ אֲשֶׁר יִבְחַר
 8   יְהוָה אֱלֹהֶיךָ בּוֹ מִקֶּרֶב אַחֶיךָ תָּשִׂים עָלֶיךָ מֶלֶךְ
 9   לֹא תוּכַל לָתֵת עָלֶיךָ אִישׁ נָכְרִי אֲשֶׁר לֹא־אָחִיךָ
10   הוּא׃ רַק לֹא־יַרְבֶּה־לּוֹ סוּסִים וְלֹא־יָשִׁיב אֶת־הָעָם
11   מִצְרַיְמָה לְמַעַן הַרְבּוֹת סוּס וַיהוָה אָמַר לָכֶם לֹא
12   תֹסִפוּן לָשׁוּב בַּדֶּרֶךְ הַזֶּה עוֹד׃ וְלֹא יַרְבֶּה־לּוֹ נָשִׁים
13   וְלֹא יָסוּר לְבָבוֹ וְכֶסֶף וְזָהָב לֹא יַרְבֶּה־לּוֹ מְאֹד׃
14   וְהָיָה כְשִׁבְתּוֹ עַל כִּסֵּא מַמְלַכְתּוֹ וְכָתַב לוֹ אֶת־
15  מִשְׁנֵה הַתּוֹרָה הַזֹּאת עַל־סֵפֶר מִלִּפְנֵי הַכֹּהֲנִים
16   הַלְוִיִּם׃ וְהָיְתָה עִמּוֹ וְקָרָא בוֹ כָּל־יְמֵי חַיָּיו לְמַעַן
17   יִלְמַד לְיִרְאָה אֶת־יְהוָה אֱלֹהָיו לִשְׁמֹר אֶת־כָּל־
18  דִּבְרֵי הַתּוֹרָה הַזֹּאת וְאֶת־הַחֻקִּים הָאֵלֶּה לַעֲשֹׂתָם׃
19   לְבִלְתִּי רוּם־לְבָבוֹ מֵאֶחָיו וּלְבִלְתִּי סוּר מִן־הַמִּצְוָה
20   יָמִין וּשְׂמֹאול לְמַעַן יַאֲרִיךְ יָמִים עַל־מַמְלַכְתּוֹ הוּא
21   וּבָנָיו בְּקֶרֶב יִשְׂרָאֵל׃♢ [ס]              לֹא־
22  יִהְיֶה לַכֹּהֲנִים הַלְוִיִּם כָּל־שֵׁבֶט לֵוִי חֵלֶק וְנַחֲלָה
23   עִם־יִשְׂרָאֵל אִשֵּׁי יְהוָה וְנַחֲלָתוֹ יֹאכֵלוּן׃ וְנַחֲלָה לֹא־
24  יִהְיֶה־לּוֹ בְּקֶרֶב אֶחָיו יְהוָה הוּא נַחֲלָתוֹ כַּאֲשֶׁר דִּבֶּר־
25  לוֹ׃ [ס]             וְזֶה יִהְיֶה מִשְׁפַּט הַכֹּהֲנִים
26   מֵאֵת הָעָם מֵאֵת זֹבְחֵי הַזֶּבַח אִם־שׁוֹר אִם־שֶׂה וְנָתַן
27   לַכֹּהֵן הַזְּרֹעַ וְהַלְּחָיַיִם וְהַקֵּבָה׃ רֵאשִׁית דְּגָנְךָ תִּירֹשְׁךָ
28   וְיִצְהָרֶךָ וְרֵאשִׁית גֵּז צֹאנְךָ תִּתֶּן־לוֹ׃ כִּי בוֹ בָּחַר יְהוָה
29   אֱלֹהֶיךָ מִכָּל־שְׁבָטֶיךָ לַעֲמֹד לְשָׁרֵת בְּשֵׁם־יְהוָה הוּא
30   וּבָנָיו כָּל־הַיָּמִים׃ [ס]              וְכִי־יָבֹא הַלֵּוִי
31   מֵאַחַד שְׁעָרֶיךָ מִכָּל־יִשְׂרָאֵל אֲשֶׁר־הוּא גָּר שָׁם
32   וּבָא בְּכָל־אַוַּת נַפְשׁוֹ אֶל־הַמָּקוֹם אֲשֶׁר־יִבְחַר יְהוָה׃
33   וְשֵׁרֵת בְּשֵׁם יְהוָה אֱלֹהָיו כְּכָל־אֶחָיו הַלְוִיִּם הָעֹמְדִים
34   שָׁם לִפְנֵי יְהוָה׃ חֵלֶק כְּחֵלֶק יֹאכֵלוּ לְבַד מִמְכָּרָיו
35   עַל־הָאָבוֹת׃ [ס]             כִּי אַתָּה בָּא אֶל־הָאָרֶץ
36   אֲשֶׁר־יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ לֹא־תִלְמַד לַעֲשׂוֹת
37   כְּתוֹעֲבֹת הַגּוֹיִם הָהֵם׃ לֹא־יִמָּצֵא בְךָ מַעֲבִיר בְּנוֹ־וּבִתּוֹ
38   בָּאֵשׁ קֹסֵם קְסָמִים מְעוֹנֵן וּמְנַחֵשׁ וּמְכַשֵּׁף׃ וְחֹבֵר חָבֶר
39   וְשֹׁאֵל אוֹב וְיִדְּעֹנִי וְדֹרֵשׁ אֶל־הַמֵּתִים׃ כִּי־תוֹעֲבַת יְהוָה
40   כָּל־עֹשֵׂה אֵלֶּה וּבִגְלַל הַתּוֹעֵבֹת הָאֵלֶּה יְהוָה אֱלֹהֶיךָ
41   מוֹרִישׁ אוֹתָם מִפָּנֶיךָ׃ תָּמִים תִּהְיֶה עִם יְהוָה אֱלֹהֶיךָ׃
42    כִּי  הַגּוֹיִם הָאֵלֶּה אֲשֶׁר אַתָּה יוֹרֵשׁ אוֹתָם אֶל־מְעֹנְנִים
