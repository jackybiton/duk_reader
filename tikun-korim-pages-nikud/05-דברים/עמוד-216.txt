 1  וְלַאֲבִירָם בְּנֵי אֱלִיאָב בֶּן־רְאוּבֵן אֲשֶׁר פָּצְתָה
 2   הָאָרֶץ אֶת־פִּיהָ וַתִּבְלָעֵם וְאֶת־בָּתֵּיהֶם וְאֶת־אָהֳלֵיהֶם
 3   וְאֵת כָּל־הַיְקוּם אֲשֶׁר בְּרַגְלֵיהֶם בְּקֶרֶב כָּל־יִשְׂרָאֵל׃
 4   כִּי עֵינֵיכֶם הָרֹאֹת אֶת־כָּל־מַעֲשֵׂה יְהוָה הַגָּדֹל
 5   אֲשֶׁר עָשָׂה׃ וּשְׁמַרְתֶּם אֶת־כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי
 6   מְצַוְּךָ הַיּוֹם לְמַעַן תֶּחֶזְקוּ וּבָאתֶם וִירִשְׁתֶּם אֶת־
 7  הָאָרֶץ אֲשֶׁר אַתֶּם עֹבְרִים שָׁמָּה לְרִשְׁתָּהּ׃ וּלְמַעַן
 8   תַּאֲרִיכוּ יָמִים עַל־הָאֲדָמָה אֲשֶׁר נִשְׁבַּע יְהוָה
 9   לַאֲבֹתֵיכֶם לָתֵת לָהֶם וּלְזַרְעָם אֶרֶץ זָבַת חָלָב
10   וּדְבָשׁ׃ [ס]              כִּי הָאָרֶץ אֲשֶׁר אַתָּה
11   בָא־שָׁמָּה לְרִשְׁתָּהּ לֹא כְאֶרֶץ מִצְרַיִם הִוא אֲשֶׁר
12   יְצָאתֶם מִשָּׁם אֲשֶׁר תִּזְרַע אֶת־זַרְעֲךָ וְהִשְׁקִיתָ
13   בְרַגְלְךָ כְּגַן הַיָּרָק׃ וְהָאָרֶץ אֲשֶׁר אַתֶּם עֹבְרִים
14   שָׁמָּה לְרִשְׁתָּהּ אֶרֶץ הָרִים וּבְקָעֹת לִמְטַר הַשָּׁמַיִם
15   תִּשְׁתֶּה־מָּיִם׃ אֶרֶץ אֲשֶׁר־יְהוָה אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ
16   תָּמִיד עֵינֵי יְהוָה אֱלֹהֶיךָ בָּהּ מֵרֵשִׁית הַשָּׁנָה וְעַד
17   אַחֲרִית שָׁנָה׃ [ס]             וְהָיָה אִם־
18  שָׁמֹעַ תִּשְׁמְעוּ אֶל־מִצְו͏ֹתַי אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם
19   הַיּוֹם לְאַהֲבָה אֶת־יְהוָה אֱלֹהֵיכֶם וּלְעָבְדוֹ בְּכָל־
20  לְבַבְכֶם וּבְכָל־נַפְשְׁכֶם׃ וְנָתַתִּי מְטַר־אַרְצְכֶם בְּעִתּוֹ
21   יוֹרֶה וּמַלְקוֹשׁ וְאָסַפְתָּ דְגָנֶךָ וְתִירֹשְׁךָ וְיִצְהָרֶךָ׃
22   וְנָתַתִּי עֵשֶׂב בְּשָׂדְךָ לִבְהֶמְתֶּךָ וְאָכַלְתָּ וְשָׂבָעְתָּ׃
23   הִשָּׁמְרוּ לָכֶם פֶּן יִפְתֶּה לְבַבְכֶם וְסַרְתֶּם וַעֲבַדְתֶּם
24   אֱלֹהִים אֲחֵרִים וְהִשְׁתַּחֲוִיתֶם לָהֶם׃ וְחָרָה אַף־יְהוָה
25   בָּכֶם וְעָצַר אֶת־הַשָּׁמַיִם וְלֹא־יִהְיֶה מָטָר וְהָאֲדָמָה
26   לֹא תִתֵּן אֶת־יְבוּלָהּ וַאֲבַדְתֶּם מְהֵרָה מֵעַל הָאָרֶץ
27   הַטֹּבָה אֲשֶׁר יְהוָה נֹתֵן לָכֶם׃ וְשַׂמְתֶּם אֶת־דְּבָרַי אֵלֶּה
28   עַל־לְבַבְכֶם וְעַל־נַפְשְׁכֶם וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל־
29  יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם׃ וְלִמַּדְתֶּם אֹתָם
30   אֶת־בְּנֵיכֶם לְדַבֵּר בָּם בְּשִׁבְתְּךָ בְּבֵיתֶךָ וּבְלֶכְתְּךָ
31   בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ׃ וּכְתַבְתָּם עַל־מְזוּזוֹת
32   בֵּיתֶךָ וּבִשְׁעָרֶיךָ׃ לְמַעַן יִרְבּוּ יְמֵיכֶם וִימֵי בְנֵיכֶם עַל
33   הָאֲדָמָה אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֵיכֶם לָתֵת לָהֶם
34   כִּימֵי הַשָּׁמַיִם עַל־הָאָרֶץ׃ [ס]              כִּי
35   אִם־שָׁמֹר תִּשְׁמְרוּן אֶת־כָּל־הַמִּצְוָה הַזֹּאת אֲשֶׁר
36   אָנֹכִי מְצַוֶּה אֶתְכֶם לַעֲשֹׂתָהּ לְאַהֲבָה אֶת־יְהוָה
37   אֱלֹהֵיכֶם לָלֶכֶת בְּכָל־דְּרָכָיו וּלְדָבְקָה־בוֹ׃ וְהוֹרִישׁ
38   יְהוָה אֶת־כָּל־הַגּוֹיִם הָאֵלֶּה מִלִּפְנֵיכֶם וִירִשְׁתֶּם
39   גּוֹיִם גְּדֹלִים וַעֲצֻמִים מִכֶּם׃ כָּל־הַמָּקוֹם אֲשֶׁר
40   תִּדְרֹךְ כַּף־רַגְלְכֶם בּוֹ לָכֶם יִהְיֶה מִן־הַמִּדְבָּר
41   וְהַלְּבָנוֹן מִן־הַנָּהָר נְהַר־פְּרָת וְעַד הַיָּם הָאַחֲרוֹן
42   יִהְיֶה גְּבֻלְכֶם׃ לֹא־יִתְיַצֵּב אִישׁ בִּפְנֵיכֶם פַּחְדְּכֶם
