 1  וְאֶל־קֹסְמִים יִשְׁמָעוּ וְאַתָּה לֹא כֵן נָתַן לְךָ יְהוָה
 2   אֱלֹהֶיךָ׃ נָבִיא מִקִּרְבְּךָ מֵאַחֶיךָ כָּמֹנִי יָקִים לְךָ יְהוָה
 3   אֱלֹהֶיךָ אֵלָיו תִּשְׁמָעוּן׃ כְּכֹל אֲשֶׁר־שָׁאַלְתָּ מֵעִם
 4   יְהוָה אֱלֹהֶיךָ בְּחֹרֵב בְּיוֹם הַקָּהָל לֵאמֹר לֹא אֹסֵף
 5   לִשְׁמֹעַ אֶת־קוֹל יְהוָה אֱלֹהָי וְאֶת־הָאֵשׁ הַגְּדֹלָה
 6   הַזֹּאת לֹא־אֶרְאֶה עוֹד וְלֹא אָמוּת׃ וַיֹּאמֶר יְהוָה
 7   אֵלָי הֵיטִיבוּ אֲשֶׁר דִּבֵּרוּ׃ נָבִיא אָקִים לָהֶם מִקֶּרֶב
 8   אֲחֵיהֶם כָּמוֹךָ וְנָתַתִּי דְבָרַי בְּפִיו וְדִבֶּר אֲלֵיהֶם אֵת
 9   כָּל־אֲשֶׁר אֲצַוֶּנּוּ׃ וְהָיָה הָאִישׁ אֲשֶׁר לֹא־יִשְׁמַע אֶל־
10  דְּבָרַי אֲשֶׁר יְדַבֵּר בִּשְׁמִי אָנֹכִי אֶדְרֹשׁ מֵעִמּוֹ׃ אַךְ
11   הַנָּבִיא אֲשֶׁר יָזִיד לְדַבֵּר דָּבָר בִּשְׁמִי אֵת אֲשֶׁר לֹא־
12  צִוִּיתִיו לְדַבֵּר וַאֲשֶׁר יְדַבֵּר בְּשֵׁם אֱלֹהִים אֲחֵרִים
13   וּמֵת הַנָּבִיא הַהוּא׃ וְכִי תֹאמַר בִּלְבָבֶךָ אֵיכָה נֵדַע
14   אֶת־הַדָּבָר אֲשֶׁר לֹא־דִבְּרוֹ יְהוָה׃ אֲשֶׁר יְדַבֵּר
15   הַנָּבִיא בְּשֵׁם יְהוָה וְלֹא־יִהְיֶה הַדָּבָר וְלֹא יָבֹא הוּא
16   הַדָּבָר אֲשֶׁר לֹא־דִבְּרוֹ יְהוָה בְּזָדוֹן דִּבְּרוֹ הַנָּבִיא
17   לֹא תָגוּר מִמֶּנּוּ׃ [ס]             כִּי־יַכְרִית
18   יְהוָה אֱלֹהֶיךָ אֶת־הַגּוֹיִם אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ
19   אֶת־אַרְצָם וִירִשְׁתָּם וְיָשַׁבְתָּ בְעָרֵיהֶם וּבְבָתֵּיהֶם׃
20   שָׁלוֹשׁ עָרִים תַּבְדִּיל לָךְ בְּתוֹךְ אַרְצְךָ אֲשֶׁר יְהוָה
21   אֱלֹהֶיךָ נֹתֵן לְךָ לְרִשְׁתָּהּ׃ תָּכִין לְךָ הַדֶּרֶךְ וְשִׁלַּשְׁתָּ
22   אֶת־גְּבוּל אַרְצְךָ אֲשֶׁר יַנְחִילְךָ יְהוָה אֱלֹהֶיךָ וְהָיָה
23   לָנוּס שָׁמָּה כָּל־רֹצֵחַ׃ וְזֶה דְּבַר הָרֹצֵחַ אֲשֶׁר־יָנוּס
24   שָׁמָּה וָחָי אֲשֶׁר יַכֶּה אֶת־רֵעֵהוּ בִּבְלִי־דַעַת וְהוּא
25   לֹא־שֹׂנֵא לוֹ מִתְּמֹל שִׁלְשֹׁם׃ וַאֲשֶׁר יָבֹא אֶת־רֵעֵהוּ
26   בַיַּעַר לַחְטֹב עֵצִים וְנִדְּחָה יָדוֹ בַגַּרְזֶן לִכְרֹת הָעֵץ
27   וְנָשַׁל הַבַּרְזֶל מִן־הָעֵץ וּמָצָא אֶת־רֵעֵהוּ וָמֵת הוּא
28   יָנוּס אֶל־אַחַת הֶעָרִים־הָאֵלֶּה וָחָי׃ פֶּן־יִרְדֹּף גֹּאֵל
29   הַדָּם אַחֲרֵי הָרֹצֵחַ כִּי־יֵחַם לְבָבוֹ וְהִשִּׂיגוֹ כִּי־יִרְבֶּה
30   הַדֶּרֶךְ וְהִכָּהוּ נָפֶשׁ וְלוֹ אֵין מִשְׁפַּט־מָוֶת כִּי לֹא
31   שֹׂנֵא הוּא לוֹ מִתְּמוֹל שִׁלְשׁוֹם׃ עַל־כֵּן אָנֹכִי מְצַוְּךָ
32   לֵאמֹר שָׁלֹשׁ עָרִים תַּבְדִּיל לָךְ׃ וְאִם־יַרְחִיב יְהוָה
33   אֱלֹהֶיךָ אֶת־גְּבֻלְךָ כַּאֲשֶׁר נִשְׁבַּע לַאֲבֹתֶיךָ וְנָתַן
34   לְךָ אֶת־כָּל־הָאָרֶץ אֲשֶׁר דִּבֶּר לָתֵת לַאֲבֹתֶיךָ׃ כִּי־
35  תִשְׁמֹר אֶת־כָּל־הַמִּצְוָה הַזֹּאת לַעֲשֹׂתָהּ אֲשֶׁר אָנֹכִי
36   מְצַוְּךָ הַיּוֹם לְאַהֲבָה אֶת־יְהוָה אֱלֹהֶיךָ וְלָלֶכֶת
37   בִּדְרָכָיו כָּל־הַיָּמִים וְיָסַפְתָּ לְךָ עוֹד שָׁלֹשׁ עָרִים
38   עַל הַשָּׁלֹשׁ הָאֵלֶּה׃ וְלֹא יִשָּׁפֵךְ דָּם נָקִי בְּקֶרֶב
39   אַרְצְךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לְךָ נַחֲלָה וְהָיָה
40   עָלֶיךָ דָּמִים׃ [פ]
41   וְכִי־יִהְיֶה אִישׁ שֹׂנֵא לְרֵעֵהוּ וְאָרַב לוֹ וְקָם עָלָיו
42   וְהִכָּהוּ נֶפֶשׁ וָמֵת וְנָס אֶל־אַחַת הֶעָרִים הָאֵל׃
