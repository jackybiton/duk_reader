 1  וּמוֹרַאֲכֶם יִתֵּן  יְהוָה אֱלֹהֵיכֶם עַל־פְּנֵי כָל־הָאָרֶץ אֲשֶׁר
 2   תִּדְרְכוּ־בָהּ כַּאֲשֶׁר דִּבֶּר לָכֶם׃ [ס]            רְאֵה
 3   אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם בְּרָכָה וּקְלָלָה׃ אֶת־הַבְּרָכָה
 4   אֲשֶׁר תִּשְׁמְעוּ אֶל־מִצְו͏ֹת יְהוָה אֱלֹהֵיכֶם אֲשֶׁר אָנֹכִי
 5   מְצַוֶּה אֶתְכֶם הַיּוֹם׃ וְהַקְּלָלָה אִם־לֹא תִשְׁמְעוּ אֶל־
 6  מִצְו͏ֹת יְהוָה אֱלֹהֵיכֶם וְסַרְתֶּם מִן־הַדֶּרֶךְ אֲשֶׁר אָנֹכִי
 7   מְצַוֶּה אֶתְכֶם הַיּוֹם לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים
 8   אֲשֶׁר לֹא־יְדַעְתֶּם׃ [ס]             וְהָיָה כִּי
 9   יְבִיאֲךָ יְהוָה אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר־אַתָּה בָא־שָׁמָּה
10   לְרִשְׁתָּהּ וְנָתַתָּה אֶת־הַבְּרָכָה עַל־הַר גְּרִזִים וְאֶת־
11  הַקְּלָלָה עַל־הַר עֵיבָל׃ הֲלֹא־הֵמָּה בְּעֵבֶר הַיַּרְדֵּן אַחֲרֵי
12   דֶּרֶךְ מְבוֹא הַשֶּׁמֶשׁ בְּאֶרֶץ הַכְּנַעֲנִי הַיֹּשֵׁב בָּעֲרָבָה
13   מוּל הַגִּלְגָּל אֵצֶל אֵלוֹנֵי מֹרֶה׃ כִּי אַתֶּם עֹבְרִים אֶת־
14  הַיַּרְדֵּן לָבֹא לָרֶשֶׁת אֶת־הָאָרֶץ אֲשֶׁר־יְהוָה אֱלֹהֵיכֶם
15   נֹתֵן לָכֶם וִירִשְׁתֶּם אֹתָהּ וִישַׁבְתֶּם־בָּהּ׃♢ וּשְׁמַרְתֶּם
16   לַעֲשׂוֹת אֵת כָּל־הַחֻקִּים וְאֶת־הַמִּשְׁפָּטִים אֲשֶׁר אָנֹכִי
17   נֹתֵן לִפְנֵיכֶם הַיּוֹם׃ אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר
18   תִּשְׁמְרוּן לַעֲשׂוֹת בָּאָרֶץ אֲשֶׁר נָתַן יְהוָה אֱלֹהֵי
19   אֲבֹתֶיךָ לְךָ לְרִשְׁתָּהּ כָּל־הַיָּמִים אֲשֶׁר־אַתֶּם חַיִּים עַל־
20  הָאֲדָמָה׃ אַבֵּד תְּאַבְּדוּן אֶת־כָּל־הַמְּקֹמוֹת אֲשֶׁר עָבְדוּ־
21  שָׁם הַגּוֹיִם אֲשֶׁר אַתֶּם יֹרְשִׁים אֹתָם אֶת־אֱלֹהֵיהֶם
22   עַל־הֶהָרִים הָרָמִים וְעַל־הַגְּבָעוֹת וְתַחַת כָּל־עֵץ
23   רַעֲנָן׃ וְנִתַּצְתֶּם אֶת־מִזְבְּחֹתָם וְשִׁבַּרְתֶּם אֶת־מַצֵּבֹתָם
24   וַאֲשֵׁרֵיהֶם תִּשְׂרְפוּן בָּאֵשׁ וּפְסִילֵי אֱלֹהֵיהֶם תְּגַדֵּעוּן
25   וְאִבַּדְתֶּם אֶת־שְׁמָם מִן־הַמָּקוֹם הַהוּא׃ לֹא־תַעֲשׂוּן
26   כֵּן לַיהוָה אֱלֹהֵיכֶם׃ כִּי אִם־אֶל־הַמָּקוֹם אֲשֶׁר־יִבְחַר
27   יְהוָה אֱלֹהֵיכֶם מִכָּל־שִׁבְטֵיכֶם לָשׂוּם אֶת־שְׁמוֹ
28   שָׁם לְשִׁכְנוֹ תִדְרְשׁוּ וּבָאתָ שָּׁמָּה׃♢ וַהֲבֵאתֶם שָׁמָּה
29   עֹלֹתֵיכֶם וְזִבְחֵיכֶם וְאֵת מַעְשְׂרֹתֵיכֶם וְאֵת תְּרוּמַת
30   יֶדְכֶם וְנִדְרֵיכֶם וְנִדְבֹתֵיכֶם וּבְכֹרֹת בְּקַרְכֶם וְצֹאנְכֶם׃
31   וַאֲכַלְתֶּם־שָׁם לִפְנֵי יְהוָה אֱלֹהֵיכֶם וּשְׂמַחְתֶּם בְּכֹל
32   מִשְׁלַח יֶדְכֶם אַתֶּם וּבָתֵּיכֶם אֲשֶׁר בֵּרַכְךָ יְהוָה
33   אֱלֹהֶיךָ׃ לֹא תַעֲשׂוּן כְּכֹל אֲשֶׁר אֲנַחְנוּ עֹשִׂים פֹּה
34   הַיּוֹם אִישׁ כָּל־הַיָּשָׁר בְּעֵינָיו׃ כִּי לֹא־בָאתֶם עַד־עָתָּה
35   אֶל־הַמְּנוּחָה וְאֶל־הַנַּחֲלָה אֲשֶׁר־יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ׃
36   וַעֲבַרְתֶּם אֶת־הַיַּרְדֵּן וִישַׁבְתֶּם בָּאָרֶץ אֲשֶׁר־יְהוָה
37   אֱלֹהֵיכֶם מַנְחִיל אֶתְכֶם וְהֵנִיחַ לָכֶם מִכָּל־אֹיְבֵיכֶם
38   מִסָּבִיב וִישַׁבְתֶּם־בֶּטַח׃♢  וְהָיָה הַמָּקוֹם אֲשֶׁר־יִבְחַר
39   יְהוָה אֱלֹהֵיכֶם בּוֹ לְשַׁכֵּן שְׁמוֹ שָׁם שָׁמָּה תָבִיאוּ
40   אֵת כָּל־אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם עוֹלֹתֵיכֶם וְזִבְחֵיכֶם
41   מַעְשְׂרֹתֵיכֶם וּתְרֻמַת יֶדְכֶם וְכֹל מִבְחַר נִדְרֵיכֶם אֲשֶׁר
42   תִּדְּרוּ לַיהוָה׃ וּשְׂמַחְתֶּם לִפְנֵי יְהוָה אֱלֹהֵיכֶם אַתֶּם
