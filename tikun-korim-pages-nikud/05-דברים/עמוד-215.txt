 1  וַיִּהְיוּ שָׁם כַּאֲשֶׁר צִוַּנִי יְהוָה׃ וּבְנֵי יִשְׂרָאֵל נָסְעוּ
 2   מִבְּאֵרֹת בְּנֵי־יַעֲקָן מוֹסֵרָה שָׁם מֵת אַהֲרֹן וַיִּקָּבֵר שָׁם
 3   וַיְכַהֵן אֶלְעָזָר בְּנוֹ תַּחְתָּיו׃ מִשָּׁם נָסְעוּ הַגֻּדְגֹּדָה וּמִן־
 4  הַגֻּדְגֹּדָה יָטְבָתָה אֶרֶץ נַחֲלֵי מָיִם׃ בָּעֵת הַהִוא הִבְדִּיל
 5   יְהוָה אֶת־שֵׁבֶט הַלֵּוִי לָשֵׂאת אֶת־אֲרוֹן בְּרִית־יְהוָה
 6   לַעֲמֹד לִפְנֵי יְהוָה לְשָׁרְתוֹ וּלְבָרֵךְ בִּשְׁמוֹ עַד הַיּוֹם
 7   הַזֶּה׃ עַל־כֵּן לֹא־הָיָה לְלֵוִי חֵלֶק וְנַחֲלָה עִם־אֶחָיו
 8   יְהוָה הוּא נַחֲלָתוֹ כַּאֲשֶׁר דִּבֶּר יְהוָה אֱלֹהֶיךָ לוֹ׃
 9   וְאָנֹכִי עָמַדְתִּי בָהָר כַּיָּמִים הָרִאשֹׁנִים אַרְבָּעִים יוֹם
10   וְאַרְבָּעִים לָיְלָה וַיִּשְׁמַע יְהוָה אֵלַי גַּם בַּפַּעַם הַהִוא
11   לֹא־אָבָה יְהוָה הַשְׁחִיתֶךָ׃ וַיֹּאמֶר יְהוָה אֵלַי קוּם לֵךְ
12   לְמַסַּע לִפְנֵי הָעָם וְיָבֹאוּ וְיִירְשׁוּ אֶת־הָאָרֶץ אֲשֶׁר־
13  נִשְׁבַּעְתִּי לַאֲבֹתָם לָתֵת לָהֶם׃ [פ]
14    וְעַתָּה יִשְׂרָאֵל מָה יְהוָה אֱלֹהֶיךָ שֹׁאֵל מֵעִמָּךְ כִּי
15   אִם־לְיִרְאָה אֶת־יְהוָה אֱלֹהֶיךָ לָלֶכֶת בְּכָל־דְּרָכָיו
16   וּלְאַהֲבָה אֹתוֹ וְלַעֲבֹד אֶת־יְהוָה אֱלֹהֶיךָ בְּכָל־לְבָבְךָ
17   וּבְכָל־נַפְשֶׁךָ׃ לִשְׁמֹר אֶת־מִצְו͏ֹת יְהוָה וְאֶת־חֻקֹּתָיו
18   אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם לְטוֹב לָךְ׃ הֵן לַיהוָה אֱלֹהֶיךָ
19   הַשָּׁמַיִם וּשְׁמֵי הַשָּׁמָיִם הָאָרֶץ וְכָל־אֲשֶׁר־בָּהּ׃
20   רַק בַּאֲבֹתֶיךָ חָשַׁק יְהוָה לְאַהֲבָה אוֹתָם וַיִּבְחַר
21   בְּזַרְעָם אַחֲרֵיהֶם בָּכֶם מִכָּל־הָעַמִּים כַּיּוֹם הַזֶּה׃
22   וּמַלְתֶּם אֵת עָרְלַת לְבַבְכֶם וְעָרְפְּכֶם לֹא תַקְשׁוּ
23   עוֹד׃ כִּי יְהוָה אֱלֹהֵיכֶם הוּא אֱלֹהֵי הָאֱלֹהִים וַאֲדֹנֵי
24   הָאֲדֹנִים הָאֵל הַגָּדֹל הַגִּבֹּר וְהַנּוֹרָא אֲשֶׁר לֹא־יִשָּׂא
25   פָנִים וְלֹא יִקַּח שֹׁחַד׃ עֹשֶׂה מִשְׁפַּט יָתוֹם וְאַלְמָנָה
26   וְאֹהֵב גֵּר לָתֶת לוֹ לֶחֶם וְשִׂמְלָה׃ וַאֲהַבְתֶּם אֶת־הַגֵּר
27   כִּי־גֵרִים הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם׃ אֶת־יְהוָה אֱלֹהֶיךָ
28   תִּירָא אֹתוֹ תַעֲבֹד וּבוֹ תִדְבָּק וּבִשְׁמוֹ תִּשָּׁבֵעַ׃ הוּא
29   תְהִלָּתְךָ וְהוּא אֱלֹהֶיךָ אֲשֶׁר־עָשָׂה אִתְּךָ אֶת־הַגְּדֹלֹת
30   וְאֶת־הַנּוֹרָאֹת הָאֵלֶּה אֲשֶׁר רָאוּ עֵינֶיךָ׃ בְּשִׁבְעִים
31   נֶפֶשׁ יָרְדוּ אֲבֹתֶיךָ מִצְרָיְמָה וְעַתָּה שָׂמְךָ יְהוָה
32   אֱלֹהֶיךָ כְּכוֹכְבֵי הַשָּׁמַיִם לָרֹב׃ וְאָהַבְתָּ אֵת יְהוָה
33   אֱלֹהֶיךָ וְשָׁמַרְתָּ מִשְׁמַרְתּוֹ וְחֻקֹּתָיו וּמִשְׁפָּטָיו
34   וּמִצְו͏ֹתָיו כָּל־הַיָּמִים׃ וִידַעְתֶּם הַיּוֹם כִּי  לֹא אֶת־בְּנֵיכֶם
35   אֲשֶׁר לֹא־יָדְעוּ וַאֲשֶׁר לֹא־רָאוּ אֶת־מוּסַר יְהוָה
36   אֱלֹהֵיכֶם אֶת־גָּדְלוֹ אֶת־יָדוֹ הַחֲזָקָה וּזְרֹעוֹ הַנְּטוּיָה׃
37   וְאֶת־אֹתֹתָיו וְאֶת־מַעֲשָׂיו אֲשֶׁר עָשָׂה בְּתוֹךְ מִצְרָיִם
38   לְפַרְעֹה מֶלֶךְ־מִצְרַיִם וּלְכָל־אַרְצוֹ׃ וַאֲשֶׁר עָשָׂה
39   לְחֵיל מִצְרַיִם לְסוּסָיו וּלְרִכְבּוֹ אֲשֶׁר הֵצִיף אֶת־
40  מֵי יַם־סוּף עַל־פְּנֵיהֶם בְּרָדְפָם אַחֲרֵיכֶם וַיְאַבְּדֵם
41   יְהוָה עַד הַיּוֹם הַזֶּה׃ וַאֲשֶׁר עָשָׂה לָכֶם בַּמִּדְבָּר
42   עַד־בֹּאֲכֶם עַד־הַמָּקוֹם הַזֶּה׃ וַאֲשֶׁר עָשָׂה לְדָתָן
