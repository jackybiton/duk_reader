 1  וּלְבִלְתִּי־בֹא אֶל־הָאָרֶץ הַטּוֹבָה אֲשֶׁר יְהוָה אֱלֹהֶיךָ
 2   נֹתֵן לְךָ נַחֲלָה׃ כִּי אָנֹכִי מֵת בָּאָרֶץ הַזֹּאת אֵינֶנִּי עֹבֵר
 3   אֶת־הַיַּרְדֵּן וְאַתֶּם עֹבְרִים וִירִשְׁתֶּם אֶת־הָאָרֶץ הַטּוֹבָה
 4   הַזֹּאת׃ הִשָּׁמְרוּ לָכֶם פֶּן־תִּשְׁכְּחוּ אֶת־בְּרִית יְהוָה
 5   אֱלֹהֵיכֶם אֲשֶׁר כָּרַת עִמָּכֶם וַעֲשִׂיתֶם לָכֶם פֶּסֶל
 6   תְּמוּנַת כֹּל אֲשֶׁר צִוְּךָ יְהוָה אֱלֹהֶיךָ׃ כִּי יְהוָה אֱלֹהֶיךָ
 7   אֵשׁ אֹכְלָה הוּא אֵל קַנָּא׃ [פ]
 8   כִּי־תוֹלִיד בָּנִים וּבְנֵי בָנִים וְנוֹשַׁנְתֶּם בָּאָרֶץ וְהִשְׁחַתֶּם
 9   וַעֲשִׂיתֶם פֶּסֶל תְּמוּנַת כֹּל וַעֲשִׂיתֶם הָרַע בְּעֵינֵי יְהוָה־
10  אֱלֹהֶיךָ לְהַכְעִיסוֹ׃ הַעִידֹתִי בָכֶם הַיּוֹם אֶת־הַשָּׁמַיִם
11   וְאֶת־הָאָרֶץ כִּי־אָבֹד תֹּאבֵדוּן מַהֵר מֵעַל הָאָרֶץ
12   אֲשֶׁר אַתֶּם עֹבְרִים אֶת־הַיַּרְדֵּן שָׁמָּה לְרִשְׁתָּהּ לֹא־
13  תַאֲרִיכֻן יָמִים עָלֶיהָ כִּי הִשָּׁמֵד תִּשָּׁמֵדוּן׃ וְהֵפִיץ
14   יְהוָה אֶתְכֶם בָּעַמִּים וְנִשְׁאַרְתֶּם מְתֵי מִסְפָּר בַּגּוֹיִם
15   אֲשֶׁר יְנַהֵג יְהוָה אֶתְכֶם שָׁמָּה׃ וַעֲבַדְתֶּם־שָׁם אֱלֹהִים
16   מַעֲשֵׂה יְדֵי אָדָם עֵץ וָאֶבֶן אֲשֶׁר לֹא־יִרְאוּן וְלֹא
17   יִשְׁמְעוּן וְלֹא יֹאכְלוּן וְלֹא יְרִיחֻן׃ וּבִקַּשְׁתֶּם מִשָּׁם
18   אֶת־יְהוָה אֱלֹהֶיךָ וּמָצָאתָ כִּי תִדְרְשֶׁנּוּ בְּכָל־לְבָבְךָ
19   וּבְכָל־נַפְשֶׁךָ׃ בַּצַּר לְךָ וּמְצָאוּךָ כֹּל הַדְּבָרִים הָאֵלֶּה
20   בְּאַחֲרִית הַיָּמִים וְשַׁבְתָּ עַד־יְהוָה אֱלֹהֶיךָ וְשָׁמַעְתָּ
21   בְּקֹלוֹ׃ כִּי אֵל רַחוּם יְהוָה אֱלֹהֶיךָ לֹא יַרְפְּךָ וְלֹא
22   יַשְׁחִיתֶךָ וְלֹא יִשְׁכַּח אֶת־בְּרִית אֲבֹתֶיךָ אֲשֶׁר
23   נִשְׁבַּע לָהֶם׃ כִּי שְׁאַל־נָא לְיָמִים רִאשֹׁנִים אֲשֶׁר־
24  הָיוּ לְפָנֶיךָ לְמִן־הַיּוֹם אֲשֶׁר בָּרָא אֱלֹהִים  אָדָם עַל־
25  הָאָרֶץ וּלְמִקְצֵה הַשָּׁמַיִם וְעַד־קְצֵה הַשָּׁמָיִם הֲנִהְיָה
26   כַּדָּבָר הַגָּדוֹל הַזֶּה אוֹ הֲנִשְׁמַע כָּמֹהוּ׃ הֲשָׁמַע עָם
27   קוֹל אֱלֹהִים מְדַבֵּר מִתּוֹךְ־הָאֵשׁ כַּאֲשֶׁר־שָׁמַעְתָּ
28   אַתָּה וַיֶּחִי׃ אוֹ  הֲנִסָּה אֱלֹהִים לָבוֹא לָקַחַת לוֹ גוֹי
29   מִקֶּרֶב גּוֹי בְּמַסֹּת בְּאֹתֹת וּבְמוֹפְתִים וּבְמִלְחָמָה
30   וּבְיָד חֲזָקָה וּבִזְרוֹעַ נְטוּיָה וּבְמוֹרָאִים גְּדֹלִים כְּכֹל
31   אֲשֶׁר־עָשָׂה לָכֶם יְהוָה אֱלֹהֵיכֶם בְּמִצְרַיִם לְעֵינֶיךָ׃
32   אַתָּה הָרְאֵתָ לָדַעַת כִּי יְהוָה הוּא הָאֱלֹהִים אֵין
33   עוֹד מִלְּבַדּוֹ׃ מִן־הַשָּׁמַיִם הִשְׁמִיעֲךָ אֶת־קֹלוֹ לְיַסְּרֶךָּ
34   וְעַל־הָאָרֶץ הֶרְאֲךָ אֶת־אִשּׁוֹ הַגְּדוֹלָה וּדְבָרָיו
35   שָׁמַעְתָּ מִתּוֹךְ הָאֵשׁ׃ וְתַחַת כִּי אָהַב אֶת־אֲבֹתֶיךָ
36   וַיִּבְחַר בְּזַרְעוֹ אַחֲרָיו וַיּוֹצִאֲךָ בְּפָנָיו בְּכֹחוֹ הַגָּדֹל
37   מִמִּצְרָיִם׃ לְהוֹרִישׁ גּוֹיִם גְּדֹלִים וַעֲצֻמִים מִמְּךָ
38   מִפָּנֶיךָ לַהֲבִיאֲךָ לָתֶת־לְךָ אֶת־אַרְצָם נַחֲלָה כַּיּוֹם
39   הַזֶּה׃ וְיָדַעְתָּ הַיּוֹם וַהֲשֵׁבֹתָ אֶל־לְבָבֶךָ כִּי יְהוָה
40   הוּא הָאֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל־הָאָרֶץ מִתָּחַת
41   אֵין עוֹד׃ וְשָׁמַרְתָּ אֶת־חֻקָּיו וְאֶת־מִצְו͏ֹתָיו אֲשֶׁר
42   אָנֹכִי מְצַוְּךָ הַיּוֹם אֲשֶׁר יִיטַב לְךָ וּלְבָנֶיךָ אַחֲרֶיךָ
