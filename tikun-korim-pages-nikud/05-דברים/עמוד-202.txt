 1  וַיְרַגְּלוּ אֹתָהּ׃ וַיִּקְחוּ בְיָדָם מִפְּרִי הָאָרֶץ וַיּוֹרִדוּ אֵלֵינוּ
 2   וַיָּשִׁבוּ אֹתָנוּ דָבָר וַיֹּאמְרוּ טוֹבָה הָאָרֶץ אֲשֶׁר־יְהוָה
 3   אֱלֹהֵינוּ נֹתֵן לָנוּ׃ וְלֹא אֲבִיתֶם לַעֲלֹת וַתַּמְרוּ אֶת־פִּי
 4   יְהוָה אֱלֹהֵיכֶם׃ וַתֵּרָגְנוּ בְאָהֳלֵיכֶם וַתֹּאמְרוּ בְּשִׂנְאַת
 5   יְהוָה אֹתָנוּ הוֹצִיאָנוּ מֵאֶרֶץ מִצְרָיִם לָתֵת אֹתָנוּ בְּיַד
 6   הָאֱמֹרִי לְהַשְׁמִידֵנוּ׃ אָנָה  אֲנַחְנוּ עֹלִים אַחֵינוּ הֵמַסּוּ
 7   אֶת־לְבָבֵנוּ לֵאמֹר עַם גָּדוֹל וָרָם מִמֶּנּוּ עָרִים גְּדֹלֹת
 8   וּבְצוּרֹת בַּשָּׁמָיִם וְגַם־בְּנֵי עֲנָקִים רָאִינוּ שָׁם׃ וָאֹמַר
 9   אֲלֵכֶם לֹא־תַעַרְצוּן וְלֹא־תִירְאוּן מֵהֶם׃ יְהוָה אֱלֹהֵיכֶם
10   הַהֹלֵךְ לִפְנֵיכֶם הוּא יִלָּחֵם לָכֶם כְּכֹל אֲשֶׁר עָשָׂה
11   אִתְּכֶם בְּמִצְרַיִם לְעֵינֵיכֶם׃ וּבַמִּדְבָּר אֲשֶׁר רָאִיתָ
12   אֲשֶׁר נְשָׂאֲךָ יְהוָה אֱלֹהֶיךָ כַּאֲשֶׁר יִשָּׂא־אִישׁ אֶת־
13  בְּנוֹ בְּכָל־הַדֶּרֶךְ אֲשֶׁר הֲלַכְתֶּם עַד־בֹּאֲכֶם עַד־
14  הַמָּקוֹם הַזֶּה׃ וּבַדָּבָר הַזֶּה אֵינְכֶם מַאֲמִינִם בַּיהוָה
15   אֱלֹהֵיכֶם׃ הַהֹלֵךְ לִפְנֵיכֶם בַּדֶּרֶךְ לָתוּר לָכֶם מָקוֹם
16   לַחֲנֹתְכֶם בָּאֵשׁ  לַיְלָה לַרְאֹתְכֶם בַּדֶּרֶךְ אֲשֶׁר תֵּלְכוּ־
17  בָהּ וּבֶעָנָן יוֹמָם׃ וַיִּשְׁמַע יְהוָה אֶת־קוֹל דִּבְרֵיכֶם
18   וַיִּקְצֹף וַיִּשָּׁבַע לֵאמֹר׃ אִם־יִרְאֶה אִישׁ בָּאֲנָשִׁים
19   הָאֵלֶּה הַדּוֹר הָרָע הַזֶּה אֵת הָאָרֶץ הַטּוֹבָה אֲשֶׁר
20   נִשְׁבַּעְתִּי לָתֵת לַאֲבֹתֵיכֶם׃ זוּלָתִי כָּלֵב בֶּן־יְפֻנֶּה הוּא
21   יִרְאֶנָּה וְלוֹ־אֶתֵּן אֶת־הָאָרֶץ אֲשֶׁר דָּרַךְ־בָּהּ וּלְבָנָיו
22   יַעַן אֲשֶׁר מִלֵּא אַחֲרֵי יְהוָה׃ גַּם־בִּי הִתְאַנַּף יְהוָה
23   בִּגְלַלְכֶם לֵאמֹר גַּם־אַתָּה לֹא־תָבֹא שָׁם׃ יְהוֹשֻׁעַ בִּן־
24  נוּן הָעֹמֵד לְפָנֶיךָ הוּא יָבֹא שָׁמָּה אֹתוֹ חַזֵּק כִּי־הוּא
25   יַנְחִלֶנָּה אֶת־יִשְׂרָאֵל׃  וְטַפְּכֶם אֲשֶׁר אֲמַרְתֶּם לָבַז
26   יִהְיֶה וּבְנֵיכֶם אֲשֶׁר לֹא־יָדְעוּ הַיּוֹם טוֹב וָרָע הֵמָּה
27   יָבֹאוּ שָׁמָּה וְלָהֶם אֶתְּנֶנָּה וְהֵם יִירָשׁוּהָ׃ וְאַתֶּם פְּנוּ
28   לָכֶם וּסְעוּ הַמִּדְבָּרָה דֶּרֶךְ יַם־סוּף׃ וַתַּעֲנוּ  וַתֹּאמְרוּ
29   אֵלַי חָטָאנוּ לַיהוָה אֲנַחְנוּ נַעֲלֶה וְנִלְחַמְנוּ כְּכֹל אֲשֶׁר־
30  צִוָּנוּ יְהוָה אֱלֹהֵינוּ וַתַּחְגְּרוּ אִישׁ אֶת־כְּלֵי מִלְחַמְתּוֹ
31   וַתָּהִינוּ לַעֲלֹת הָהָרָה׃ וַיֹּאמֶר יְהוָה אֵלַי אֱמֹר לָהֶם
32   לֹא תַעֲלוּ וְלֹא־תִלָּחֲמוּ כִּי אֵינֶנִּי בְּקִרְבְּכֶם וְלֹא תִּנָּגְפוּ
33   לִפְנֵי אֹיְבֵיכֶם׃ וָאֲדַבֵּר אֲלֵיכֶם וְלֹא שְׁמַעְתֶּם וַתַּמְרוּ
34   אֶת־פִּי יְהוָה וַתָּזִדוּ וַתַּעֲלוּ הָהָרָה׃ וַיֵּצֵא הָאֱמֹרִי הַיֹּשֵׁב
35   בָּהָר הַהוּא לִקְרַאתְכֶם וַיִּרְדְּפוּ אֶתְכֶם כַּאֲשֶׁר
36   תַּעֲשֶׂינָה הַדְּבֹרִים וַיַּכְּתוּ אֶתְכֶם בְּשֵׂעִיר עַד־חָרְמָה׃
37   וַתָּשֻׁבוּ וַתִּבְכּוּ לִפְנֵי יְהוָה וְלֹא־שָׁמַע יְהוָה בְּקֹלְכֶם
38   וְלֹא הֶאֱזִין אֲלֵיכֶם׃ וַתֵּשְׁבוּ בְקָדֵשׁ יָמִים רַבִּים
39   כַּיָּמִים אֲשֶׁר יְשַׁבְתֶּם׃ וַנֵּפֶן וַנִּסַּע הַמִּדְבָּרָה דֶּרֶךְ
40   יַם־סוּף כַּאֲשֶׁר דִּבֶּר יְהוָה אֵלָי וַנָּסָב אֶת־הַר־שֵׂעִיר
41   יָמִים רַבִּים׃ [ס]              וַיֹּאמֶר יְהוָה אֵלַי
42   לֵאמֹר׃ רַב־לָכֶם סֹב אֶת־הָהָר הַזֶּה פְּנוּ לָכֶם צָפֹנָה׃
