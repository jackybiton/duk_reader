 1  וְאֶעֱשֶׂה אוֹתְךָ לְגוֹי־עָצוּם וָרָב מִמֶּנּוּ׃ וָאֵפֶן וָאֵרֵד מִן־
 2  הָהָר וְהָהָר בֹּעֵר בָּאֵשׁ וּשְׁנֵי לוּחֹת הַבְּרִית עַל שְׁתֵּי
 3   יָדָי׃ וָאֵרֶא וְהִנֵּה חֲטָאתֶם לַיהוָה אֱלֹהֵיכֶם עֲשִׂיתֶם
 4   לָכֶם עֵגֶל מַסֵּכָה סַרְתֶּם מַהֵר מִן־הַדֶּרֶךְ אֲשֶׁר־צִוָּה
 5   יְהוָה אֶתְכֶם׃ וָאֶתְפֹּשׂ בִּשְׁנֵי הַלֻּחֹת וָאַשְׁלִכֵם מֵעַל
 6   שְׁתֵּי יָדָי וָאֲשַׁבְּרֵם לְעֵינֵיכֶם׃ וָאֶתְנַפַּל לִפְנֵי יְהוָה
 7   כָּרִאשֹׁנָה אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה לֶחֶם לֹא
 8   אָכַלְתִּי וּמַיִם לֹא שָׁתִיתִי עַל כָּל־חַטַּאתְכֶם אֲשֶׁר
 9   חֲטָאתֶם לַעֲשׂוֹת הָרַע בְּעֵינֵי יְהוָה לְהַכְעִיסוֹ׃ כִּי
10   יָגֹרְתִּי מִפְּנֵי הָאַף וְהַחֵמָה אֲשֶׁר קָצַף יְהוָה עֲלֵיכֶם
11   לְהַשְׁמִיד אֶתְכֶם וַיִּשְׁמַע יְהוָה אֵלַי גַּם בַּפַּעַם הַהִוא׃
12   וּבְאַהֲרֹן הִתְאַנַּף יְהוָה מְאֹד לְהַשְׁמִידוֹ וָאֶתְפַּלֵּל
13   גַּם־בְּעַד אַהֲרֹן בָּעֵת הַהִוא׃ וְאֶת־חַטַּאתְכֶם אֲשֶׁר־
14  עֲשִׂיתֶם אֶת־הָעֵגֶל לָקַחְתִּי וָאֶשְׂרֹף אֹתוֹ  בָּאֵשׁ
15   וָאֶכֹּת אֹתוֹ טָחוֹן הֵיטֵב עַד אֲשֶׁר־דַּק לְעָפָר וָאַשְׁלִךְ
16   אֶת־עֲפָרוֹ אֶל־הַנַּחַל הַיֹּרֵד מִן־הָהָר׃ וּבְתַבְעֵרָה
17   וּבְמַסָּה וּבְקִבְרֹת הַתַּאֲוָה מַקְצִפִים הֱיִיתֶם אֶת־יְהוָה׃
18   וּבִשְׁלֹחַ יְהוָה אֶתְכֶם מִקָּדֵשׁ בַּרְנֵעַ לֵאמֹר עֲלוּ
19   וּרְשׁוּ אֶת־הָאָרֶץ אֲשֶׁר נָתַתִּי לָכֶם וַתַּמְרוּ אֶת־פִּי
20   יְהוָה אֱלֹהֵיכֶם וְלֹא הֶאֱמַנְתֶּם לוֹ וְלֹא שְׁמַעְתֶּם
21   בְּקֹלוֹ׃ מַמְרִים הֱיִיתֶם עִם־יְהוָה מִיּוֹם דַּעְתִּי אֶתְכֶם׃
22   וָאֶתְנַפַּל לִפְנֵי יְהוָה אֵת אַרְבָּעִים הַיּוֹם וְאֶת־אַרְבָּעִים
23   הַלַּיְלָה אֲשֶׁר הִתְנַפָּלְתִּי כִּי־אָמַר יְהוָה לְהַשְׁמִיד
24   אֶתְכֶם׃ וָאֶתְפַּלֵּל אֶל־יְהוָה וָאֹמַר אֲדֹנָי יְהוִה אַל־
25  תַּשְׁחֵת עַמְּךָ וְנַחֲלָתְךָ אֲשֶׁר פָּדִיתָ בְּגָדְלֶךָ אֲשֶׁר־
26  הוֹצֵאתָ מִמִּצְרַיִם בְּיָד חֲזָקָה׃ זְכֹר לַעֲבָדֶיךָ לְאַבְרָהָם
27   לְיִצְחָק וּלְיַעֲקֹב אַל־תֵּפֶן אֶל־קְשִׁי הָעָם הַזֶּה וְאֶל־
28  רִשְׁעוֹ וְאֶל־חַטָּאתוֹ׃ פֶּן־יֹאמְרוּ הָאָרֶץ אֲשֶׁר הוֹצֵאתָנוּ
29   מִשָּׁם מִבְּלִי יְכֹלֶת יְהוָה לַהֲבִיאָם אֶל־הָאָרֶץ אֲשֶׁר־
30  דִּבֶּר לָהֶם וּמִשִּׂנְאָתוֹ אוֹתָם הוֹצִיאָם לַהֲמִתָם
31   בַּמִּדְבָּר׃ וְהֵם עַמְּךָ וְנַחֲלָתֶךָ אֲשֶׁר הוֹצֵאתָ בְּכֹחֲךָ
32   הַגָּדֹל וּבִזְרֹעֲךָ הַנְּטוּיָה׃ [פ]
33    בָּעֵת הַהִוא אָמַר יְהוָה אֵלַי פְּסָל־לְךָ שְׁנֵי־לוּחֹת
34   אֲבָנִים כָּרִאשֹׁנִים וַעֲלֵה אֵלַי הָהָרָה וְעָשִׂיתָ לְּךָ
35   אֲרוֹן עֵץ׃ וְאֶכְתֹּב עַל־הַלֻּחֹת אֶת־הַדְּבָרִים אֲשֶׁר
36   הָיוּ עַל־הַלֻּחֹת הָרִאשֹׁנִים אֲשֶׁר שִׁבַּרְתָּ וְשַׂמְתָּם
37   בָּאָרוֹן׃ וָאַעַשׂ אֲרוֹן עֲצֵי שִׁטִּים וָאֶפְסֹל שְׁנֵי־לֻחֹת
38   אֲבָנִים כָּרִאשֹׁנִים וָאַעַל הָהָרָה וּשְׁנֵי הַלֻּחֹת בְּיָדִי׃
39   וַיִּכְתֹּב עַל־הַלֻּחֹת כַּמִּכְתָּב הָרִאשׁוֹן אֵת עֲשֶׂרֶת
40   הַדְּבָרִים אֲשֶׁר דִּבֶּר יְהוָה אֲלֵיכֶם בָּהָר מִתּוֹךְ
41   הָאֵשׁ בְּיוֹם הַקָּהָל וַיִּתְּנֵם יְהוָה אֵלָי׃ וָאֵפֶן וָאֵרֵד
42   מִן־הָהָר וָאָשִׂם אֶת־הַלֻּחֹת בָּאָרוֹן אֲשֶׁר עָשִׂיתִי
