 1  וְהַנָּשִׁים וְהַטַּף וְגֵרְךָ אֲשֶׁר בִּשְׁעָרֶיךָ לְמַעַן
 2   יִשְׁמְעוּ וּלְמַעַן יִלְמְדוּ וְיָרְאוּ אֶת־יְהוָה אֱלֹהֵיכֶם
 3   וְשָׁמְרוּ לַעֲשׂוֹת אֶת־כָּל־דִּבְרֵי הַתּוֹרָה הַזֹּאת׃
 4   וּבְנֵיהֶם אֲשֶׁר לֹא־יָדְעוּ יִשְׁמְעוּ וְלָמְדוּ לְיִרְאָה
 5   אֶת־יְהוָה אֱלֹהֵיכֶם כָּל־הַיָּמִים אֲשֶׁר אַתֶּם חַיִּים
 6   עַל־הָאֲדָמָה אֲשֶׁר אַתֶּם עֹבְרִים אֶת־הַיַּרְדֵּן שָׁמָּה
 7   לְרִשְׁתָּהּ׃♢ [פ]
 8    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה הֵן קָרְבוּ יָמֶיךָ לָמוּת קְרָא
 9   אֶת־יְהוֹשֻׁעַ וְהִתְיַצְּבוּ בְּאֹהֶל מוֹעֵד וַאֲצַוֶּנּוּ וַיֵּלֶךְ
10   מֹשֶׁה וִיהוֹשֻׁעַ וַיִּתְיַצְּבוּ בְּאֹהֶל מוֹעֵד׃ וַיֵּרָא יְהוָה
11   בָּאֹהֶל בְּעַמּוּד עָנָן וַיַּעֲמֹד עַמּוּד הֶעָנָן עַל־פֶּתַח
12   הָאֹהֶל׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה הִנְּךָ שֹׁכֵב עִם־
13  אֲבֹתֶיךָ וְקָם הָעָם הַזֶּה וְזָנָה  אַחֲרֵי  אֱלֹהֵי נֵכַר־הָאָרֶץ
14   אֲשֶׁר הוּא בָא־שָׁמָּה בְּקִרְבּוֹ וַעֲזָבַנִי וְהֵפֵר אֶת־
15  בְּרִיתִי אֲשֶׁר כָּרַתִּי אִתּוֹ׃ וְחָרָה אַפִּי בוֹ בַיּוֹם־הַהוּא
16   וַעֲזַבְתִּים וְהִסְתַּרְתִּי פָנַי מֵהֶם וְהָיָה לֶאֱכֹל וּמְצָאֻהוּ
17   רָעוֹת רַבּוֹת וְצָרוֹת וְאָמַר בַּיּוֹם הַהוּא הֲלֹא עַל כִּי־
18  אֵין אֱלֹהַי בְּקִרְבִּי מְצָאוּנִי הָרָעוֹת הָאֵלֶּה׃ וְאָנֹכִי
19   הַסְתֵּר אַסְתִּיר פָּנַי בַּיּוֹם הַהוּא עַל כָּל־הָרָעָה
20   אֲשֶׁר עָשָׂה כִּי פָנָה אֶל־אֱלֹהִים אֲחֵרִים׃ וְעַתָּה
21   כִּתְבוּ לָכֶם אֶת־הַשִּׁירָה הַזֹּאת וְלַמְּדָהּ אֶת־בְּנֵי־
22  יִשְׂרָאֵל שִׂימָהּ בְּפִיהֶם לְמַעַן תִּהְיֶה־לִּי הַשִּׁירָה
23   הַזֹּאת לְעֵד בִּבְנֵי יִשְׂרָאֵל׃  כִּי־אֲבִיאֶנּוּ אֶל־הָאֲדָמָה
24   אֲשֶׁר־נִשְׁבַּעְתִּי לַאֲבֹתָיו זָבַת חָלָב וּדְבַשׁ וְאָכַל
25   וְשָׂבַע וְדָשֵׁן וּפָנָה אֶל־אֱלֹהִים אֲחֵרִים וַעֲבָדוּם
26   וְנִאֲצוּנִי וְהֵפֵר אֶת־בְּרִיתִי׃ וְהָיָה כִּי־תִמְצֶאןָ אֹתוֹ
27   רָעוֹת רַבּוֹת וְצָרוֹת וְעָנְתָה הַשִּׁירָה הַזֹּאת לְפָנָיו
28   לְעֵד כִּי לֹא תִשָּׁכַח מִפִּי זַרְעוֹ כִּי יָדַעְתִּי אֶת־יִצְרוֹ
29   אֲשֶׁר הוּא עֹשֶׂה הַיּוֹם בְּטֶרֶם אֲבִיאֶנּוּ אֶל־הָאָרֶץ
30   אֲשֶׁר נִשְׁבַּעְתִּי׃ וַיִּכְתֹּב מֹשֶׁה אֶת־הַשִּׁירָה הַזֹּאת
31   בַּיּוֹם הַהוּא וַיְלַמְּדָהּ אֶת־בְּנֵי יִשְׂרָאֵל׃ וַיְצַו אֶת־
32  יְהוֹשֻׁעַ בִּן־נוּן וַיֹּאמֶר חֲזַק וֶאֱמָץ כִּי אַתָּה תָּבִיא
33   אֶת־בְּנֵי יִשְׂרָאֵל אֶל־הָאָרֶץ אֲשֶׁר־נִשְׁבַּעְתִּי לָהֶם
34   וְאָנֹכִי אֶהְיֶה עִמָּךְ׃ וַיְהִי  כְּכַלּוֹת מֹשֶׁה לִכְתֹּב אֶת־
35  דִּבְרֵי הַתּוֹרָה־הַזֹּאת עַל־סֵפֶר עַד תֻּמָּם׃  וַיְצַו מֹשֶׁה
36   אֶת־הַלְוִיִּם נֹשְׂאֵי אֲרוֹן בְּרִית־יְהוָה לֵאמֹר׃ לָקֹחַ
37   אֵת סֵפֶר הַתּוֹרָה הַזֶּה וְשַׂמְתֶּם אֹתוֹ מִצַּד אֲרוֹן
38   בְּרִית־יְהוָה אֱלֹהֵיכֶם וְהָיָה־שָׁם בְּךָ לְעֵד׃ כִּי אָנֹכִי
39   יָדַעְתִּי אֶת־מֶרְיְךָ וְאֶת־עָרְפְּךָ הַקָּשֶׁה הֵן בְּעוֹדֶנִּי
40   חַי עִמָּכֶם הַיּוֹם מַמְרִים הֱיִתֶם עִם־יְהוָה וְאַף כִּי־
41  אַחֲרֵי מוֹתִי׃  הַקְהִילוּ אֵלַי אֶת־כָּל־זִקְנֵי שִׁבְטֵיכֶם
42   וְשֹׁטְרֵיכֶם וַאֲדַבְּרָה בְאָזְנֵיהֶם אֵת הַדְּבָרִים הָאֵלֶּה
