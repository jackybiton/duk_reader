1וְהַנִּסְתָּרִיםמִפָּנֶיךָ׃לֹאתַעֲרֹץמִפְּנֵיהֶםכִּי־יְהוָה
2אֱלֹהֶיךָבְּקִרְבֶּךָאֵלגָּדוֹלוְנוֹרָא׃♢וְנָשַׁליְהוָהאֱלֹהֶיךָ
3אֶת־הַגּוֹיִםהָאֵלמִפָּנֶיךָמְעַטמְעָטלֹאתוּכַלכַּלֹּתָם
4מַהֵרפֶּן־תִּרְבֶּהעָלֶיךָחַיַּתהַשָּׂדֶה׃וּנְתָנָםיְהוָה
5אֱלֹהֶיךָלְפָנֶיךָוְהָמָםמְהוּמָהגְדֹלָהעַדהִשָּׁמְדָם׃
6וְנָתַןמַלְכֵיהֶםבְּיָדֶךָוְהַאֲבַדְתָּאֶת־שְׁמָםמִתַּחַת
7הַשָּׁמָיִםלֹא־יִתְיַצֵּבאִישׁבְּפָנֶיךָעַדהִשְׁמִדְךָאֹתָם׃
8פְּסִילֵיאֱלֹהֵיהֶםתִּשְׂרְפוּןבָּאֵשׁלֹא־תַחְמֹדכֶּסֶף
9וְזָהָבעֲלֵיהֶםוְלָקַחְתָּלָךְפֶּןתִּוָּקֵשׁבּוֹכִּיתוֹעֲבַת
10יְהוָהאֱלֹהֶיךָהוּא׃וְלֹא־תָבִיאתוֹעֵבָהאֶל־בֵּיתֶךָ
11וְהָיִיתָחֵרֶםכָּמֹהוּשַׁקֵּץתְּשַׁקְּצֶנּוּוְתַעֵבתְּתַעֲבֶנּוּ
12כִּי־חֵרֶםהוּא׃
13כָּל־הַמִּצְוָהאֲשֶׁראָנֹכִימְצַוְּךָהַיּוֹםתִּשְׁמְרוּן
14לַעֲשׂוֹתלְמַעַןתִּחְיוּןוּרְבִיתֶםוּבָאתֶםוִירִשְׁתֶּםאֶת־
15הָאָרֶץאֲשֶׁר־נִשְׁבַּעיְהוָהלַאֲבֹתֵיכֶם׃וְזָכַרְתָּאֶת־
16כָּל־הַדֶּרֶךְאֲשֶׁרהוֹלִיכְךָיְהוָהאֱלֹהֶיךָזֶהאַרְבָּעִים
17שָׁנָהבַּמִּדְבָּרלְמַעַןעַנֹּתְךָלְנַסֹּתְךָלָדַעַתאֶת־אֲשֶׁר
18בִּלְבָבְךָהֲתִשְׁמֹרמִצְו͏ֹתָואִם־לֹא׃וַיְעַנְּךָוַיַּרְעִבֶךָ
19וַיַּאֲכִלְךָאֶת־הַמָּןאֲשֶׁרלֹא־יָדַעְתָּוְלֹאיָדְעוּןאֲבֹתֶיךָ
20לְמַעַןהוֹדִיעֲךָכִּילֹאעַל־הַלֶּחֶםלְבַדּוֹיִחְיֶההָאָדָם
21כִּיעַל־כָּל־מוֹצָאפִי־יְהוָהיִחְיֶההָאָדָם׃שִׂמְלָתְךָ
22לֹאבָלְתָהמֵעָלֶיךָוְרַגְלְךָלֹאבָצֵקָהזֶהאַרְבָּעִים
23שָׁנָה׃♢וְיָדַעְתָּעִם־לְבָבֶךָכִּיכַּאֲשֶׁריְיַסֵּראִישׁאֶת־
24בְּנוֹיְהוָהאֱלֹהֶיךָמְיַסְּרֶךָּ׃וְשָׁמַרְתָּאֶת־מִצְו͏ֹתיְהוָה
25אֱלֹהֶיךָלָלֶכֶתבִּדְרָכָיווּלְיִרְאָהאֹתוֹ׃כִּייְהוָה
26אֱלֹהֶיךָמְבִיאֲךָאֶל־אֶרֶץטוֹבָהאֶרֶץנַחֲלֵימָיִם

29אֶרֶץאֲשֶׁרלֹאבְמִסְכֵּנֻתתֹּאכַל־בָּהּלֶחֶםלֹא־
30תֶחְסַרכֹּלבָּהּאֶרֶץאֲשֶׁראֲבָנֶיהָבַרְזֶלוּמֵהֲרָרֶיהָ
31תַּחְצֹבנְחֹשֶׁת׃וְאָכַלְתָּוְשָׂבָעְתָּוּבֵרַכְתָּאֶת־יְהוָה
32אֱלֹהֶיךָעַל־הָאָרֶץהַטֹּבָהאֲשֶׁרנָתַן־לָךְ׃♢הִשָּׁמֶרלְךָ
33פֶּן־תִּשְׁכַּחאֶת־יְהוָהאֱלֹהֶיךָלְבִלְתִּישְׁמֹרמִצְו͏ֹתָיו
34וּמִשְׁפָּטָיווְחֻקֹּתָיואֲשֶׁראָנֹכִימְצַוְּךָהַיּוֹם׃פֶּן־

37לְךָיִרְבֶּה׃וְרָםלְבָבֶךָוְשָׁכַחְתָּאֶת־יְהוָהאֱלֹהֶיךָ
38הַמּוֹצִיאֲךָמֵאֶרֶץמִצְרַיִםמִבֵּיתעֲבָדִים׃הַמּוֹלִיכְךָ
