 1  וְרָאוּ אֶת־מַכּוֹת הָאָרֶץ הַהִוא וְאֶת־תַּחֲלֻאֶיהָ אֲשֶׁר־
 2  חִלָּה יְהוָה בָּהּ׃ גָּפְרִית וָמֶלַח שְׂרֵפָה כָל־אַרְצָהּ לֹא
 3   תִזָּרַע וְלֹא תַצְמִחַ וְלֹא־יַעֲלֶה בָהּ כָּל־עֵשֶׂב
 4   כְּמַהְפֵּכַת סְדֹם וַעֲמֹרָה אַדְמָה וּצְבוֺיִם אֲשֶׁר הָפַךְ
 5   יְהוָה בְּאַפּוֹ וּבַחֲמָתוֹ׃ וְאָמְרוּ כָּל־הַגּוֹיִם עַל־מֶה עָשָׂה
 6   יְהוָה כָּכָה לָאָרֶץ הַזֹּאת מֶה חֳרִי הָאַף הַגָּדוֹל הַזֶּה׃
 7   וְאָמְרוּ עַל אֲשֶׁר עָזְבוּ אֶת־בְּרִית יְהוָה אֱלֹהֵי אֲבֹתָם
 8   אֲשֶׁר כָּרַת עִמָּם בְּהוֹצִיאוֹ אֹתָם מֵאֶרֶץ מִצְרָיִם׃
 9   וַיֵּלְכוּ וַיַּעַבְדוּ אֱלֹהִים אֲחֵרִים וַיִּשְׁתַּחֲווּ לָהֶם
10   אֱלֹהִים אֲשֶׁר לֹא־יְדָעוּם וְלֹא חָלַק לָהֶם׃ וַיִּחַר־אַף
11   יְהוָה בָּאָרֶץ הַהִוא לְהָבִיא עָלֶיהָ אֶת־כָּל־הַקְּלָלָה
12   הַכְּתוּבָה בַּסֵּפֶר הַזֶּה׃ וַיִּתְּשֵׁם יְהוָה מֵעַל אַדְמָתָם
13   בְּאַף וּבְחֵמָה וּבְקֶצֶף גָּדוֹל וַיַּשְׁלִכֵם אֶל־אֶרֶץ
14   אַחֶרֶת כַּיּוֹם הַזֶּה׃ הַנִּסְתָּרֹת לַיהוָה אֱלֹהֵינוּ וְהַנִּגְלֹת
15   לָנוּ וּלְבָנֵינוּ עַד־עוֹלָם לַעֲשׂוֹת אֶת־כָּל־דִּבְרֵי
16   הַתּוֹרָה הַזֹּאת׃♢ [ס]              וְהָיָה כִי־יָבֹאוּ
17   עָלֶיךָ כָּל־הַדְּבָרִים הָאֵלֶּה הַבְּרָכָה וְהַקְּלָלָה אֲשֶׁר
18   נָתַתִּי לְפָנֶיךָ וַהֲשֵׁבֹתָ אֶל־לְבָבֶךָ בְּכָל־הַגּוֹיִם אֲשֶׁר
19   הִדִּיחֲךָ יְהוָה אֱלֹהֶיךָ שָׁמָּה׃ וְשַׁבְתָּ עַד־יְהוָה אֱלֹהֶיךָ
20   וְשָׁמַעְתָּ בְקֹלוֹ כְּכֹל אֲשֶׁר־אָנֹכִי מְצַוְּךָ הַיּוֹם אַתָּה
21   וּבָנֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשֶׁךָ׃ וְשָׁב יְהוָה אֱלֹהֶיךָ
22   אֶת־שְׁבוּתְךָ וְרִחֲמֶךָ וְשָׁב וְקִבֶּצְךָ מִכָּל־הָעַמִּים
23   אֲשֶׁר הֱפִיצְךָ יְהוָה אֱלֹהֶיךָ שָׁמָּה׃ אִם־יִהְיֶה נִדַּחֲךָ
24   בִּקְצֵה הַשָּׁמָיִם מִשָּׁם יְקַבֶּצְךָ יְהוָה אֱלֹהֶיךָ וּמִשָּׁם
25   יִקָּחֶךָ׃ וֶהֱבִיאֲךָ יְהוָה אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר־יָרְשׁוּ
26   אֲבֹתֶיךָ וִירִשְׁתָּהּ וְהֵיטִבְךָ וְהִרְבְּךָ מֵאֲבֹתֶיךָ׃ וּמָל
27   יְהוָה אֱלֹהֶיךָ אֶת־לְבָבְךָ וְאֶת־לְבַב זַרְעֶךָ לְאַהֲבָה
28   אֶת־יְהוָה אֱלֹהֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשְׁךָ לְמַעַן
29   חַיֶּיךָ׃  וְנָתַן יְהוָה אֱלֹהֶיךָ אֵת כָּל־הָאָלוֹת הָאֵלֶּה עַל־
30  אֹיְבֶיךָ וְעַל־שֹׂנְאֶיךָ אֲשֶׁר רְדָפוּךָ׃ וְאַתָּה תָשׁוּב
31   וְשָׁמַעְתָּ בְּקוֹל יְהוָה וְעָשִׂיתָ אֶת־כָּל־מִצְו͏ֹתָיו אֲשֶׁר
32   אָנֹכִי מְצַוְּךָ הַיּוֹם׃ וְהוֹתִירְךָ יְהוָה אֱלֹהֶיךָ בְּכֹל  מַעֲשֵׂה
33   יָדֶךָ בִּפְרִי בִטְנְךָ וּבִפְרִי בְהֶמְתְּךָ וּבִפְרִי אַדְמָתְךָ
34   לְטֹבָה כִּי  יָשׁוּב יְהוָה לָשׂוּשׂ עָלֶיךָ לְטוֹב כַּאֲשֶׁר־
35  שָׂשׂ עַל־אֲבֹתֶיךָ׃ כִּי תִשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ
36   לִשְׁמֹר מִצְו͏ֹתָיו וְחֻקֹּתָיו הַכְּתוּבָה בְּסֵפֶר הַתּוֹרָה
37   הַזֶּה כִּי תָשׁוּב אֶל־יְהוָה אֱלֹהֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־
38  נַפְשֶׁךָ׃ [ס]              כִּי הַמִּצְוָה הַזֹּאת אֲשֶׁר
39   אָנֹכִי מְצַוְּךָ הַיּוֹם לֹא־נִפְלֵאת הִוא מִמְּךָ וְלֹא רְחֹקָה
40   הִוא׃ לֹא בַשָּׁמַיִם הִוא לֵאמֹר מִי יַעֲלֶה־לָּנוּ הַשָּׁמַיְמָה
41   וְיִקָּחֶהָ לָּנוּ וְיַשְׁמִעֵנוּ אֹתָהּ וְנַעֲשֶׂנָּה׃ וְלֹא־מֵעֵבֶר לַיָּם
42   הִוא לֵאמֹר מִי יַעֲבָר־לָנוּ אֶל־עֵבֶר הַיָּם וְיִקָּחֶהָ לָּנוּ
