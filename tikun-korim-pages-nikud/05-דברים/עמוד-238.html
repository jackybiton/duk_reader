10אֱלֹהֵיכֶם׃וַתָּבֹאוּאֶל־הַמָּקוֹםהַזֶּהוַיֵּצֵאסִיחֹןמֶלֶךְ־
11חֶשְׁבּוֹןוְעוֹגמֶלֶךְ־הַבָּשָׁןלִקְרָאתֵנוּלַמִּלְחָמָה
12וַנַּכֵּם׃וַנִּקַּחאֶת־אַרְצָםוַנִּתְּנָהּלְנַחֲלָהלָראוּבֵנִי
13וְלַגָּדִיוְלַחֲצִישֵׁבֶטהַמְנַשִּׁי׃וּשְׁמַרְתֶּםאֶת־דִּבְרֵי

17רָאשֵׁיכֶםשִׁבְטֵיכֶםזִקְנֵיכֶםוְשֹׁטְרֵיכֶםכֹּלאִישׁ
18יִשְׂרָאֵל׃טַפְּכֶםנְשֵׁיכֶםוְגֵרְךָאֲשֶׁרבְּקֶרֶבמַחֲנֶיךָ

20יְהוָהאֱלֹהֶיךָוּבְאָלָתוֹאֲשֶׁריְהוָהאֱלֹהֶיךָכֹּרֵת

25הַזֹּאת׃כִּיאֶת־אֲשֶׁריֶשְׁנוֹפֹּהעִמָּנוּעֹמֵדהַיּוֹם
26לִפְנֵייְהוָהאֱלֹהֵינוּוְאֵתאֲשֶׁראֵינֶנּוּפֹּהעִמָּנוּהַיּוֹם׃♢
27כִּי־אַתֶּםיְדַעְתֶּםאֵתאֲשֶׁר־יָשַׁבְנוּבְּאֶרֶץמִצְרָיִם

30וְזָהָבאֲשֶׁרעִמָּהֶם׃פֶּן־יֵשׁבָּכֶםאִישׁאוֹ־אִשָּׁהאוֹ
31מִשְׁפָּחָהאוֹ־שֵׁבֶטאֲשֶׁרלְבָבוֹפֹנֶההַיּוֹםמֵעִם
32יְהוָהאֱלֹהֵינוּלָלֶכֶתלַעֲבֹדאֶת־אֱלֹהֵיהַגּוֹיִםהָהֵם

34אֶת־דִּבְרֵיהָאָלָההַזֹּאתוְהִתְבָּרֵךְבִּלְבָבוֹלֵאמֹר
