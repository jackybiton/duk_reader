 1  וְלֹא תִגְרְעוּ מִמֶּנּוּ לִשְׁמֹר אֶת־מִצְו͏ֹת יְהוָה אֱלֹהֵיכֶם
 2   אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם׃ עֵינֵיכֶם הָרֹאוֹת אֵת אֲשֶׁר־
 3  עָשָׂה יְהוָה בְּבַעַל פְּעוֹר כִּי כָל־הָאִישׁ אֲשֶׁר הָלַךְ
 4   אַחֲרֵי בַעַל־פְּעוֹר הִשְׁמִידוֹ יְהוָה אֱלֹהֶיךָ מִקִּרְבֶּךָ׃
 5   וְאַתֶּם הַדְּבֵקִים בַּיהוָה אֱלֹהֵיכֶם חַיִּים כֻּלְּכֶם הַיּוֹם׃♢
 6    רְאֵה  לִמַּדְתִּי אֶתְכֶם חֻקִּים וּמִשְׁפָּטִים כַּאֲשֶׁר צִוַּנִי
 7   יְהוָה אֱלֹהָי לַעֲשׂוֹת כֵּן בְּקֶרֶב הָאָרֶץ אֲשֶׁר אַתֶּם
 8   בָּאִים שָׁמָּה לְרִשְׁתָּהּ׃ וּשְׁמַרְתֶּם וַעֲשִׂיתֶם כִּי הִוא
 9   חָכְמַתְכֶם וּבִינַתְכֶם לְעֵינֵי הָעַמִּים אֲשֶׁר יִשְׁמְעוּן
10   אֵת כָּל־הַחֻקִּים הָאֵלֶּה וְאָמְרוּ רַק עַם־חָכָם וְנָבוֹן
11   הַגּוֹי הַגָּדוֹל הַזֶּה׃ כִּי מִי־גוֹי גָּדוֹל אֲשֶׁר־לוֹ אֱלֹהִים
12   קְרֹבִים אֵלָיו כַּיהוָה אֱלֹהֵינוּ בְּכָל־קָרְאֵנוּ אֵלָיו׃ וּמִי
13   גּוֹי גָּדוֹל אֲשֶׁר־לוֹ חֻקִּים וּמִשְׁפָּטִים צַדִּיקִם כְּכֹל
14   הַתּוֹרָה הַזֹּאת אֲשֶׁר אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם׃♢ רַק
15   הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן־תִּשְׁכַּח אֶת־
16  הַדְּבָרִים אֲשֶׁר־רָאוּ עֵינֶיךָ וּפֶן־יָסוּרוּ מִלְּבָבְךָ כֹּל
17   יְמֵי חַיֶּיךָ וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ׃ יוֹם אֲשֶׁר
18   עָמַדְתָּ לִפְנֵי יְהוָה אֱלֹהֶיךָ בְּחֹרֵב בֶּאֱמֹר יְהוָה אֵלַי
19   הַקְהֶל־לִי אֶת־הָעָם וְאַשְׁמִעֵם אֶת־דְּבָרָי אֲשֶׁר
20   יִלְמְדוּן לְיִרְאָה אֹתִי כָּל־הַיָּמִים אֲשֶׁר הֵם חַיִּים עַל־
21  הָאֲדָמָה וְאֶת־בְּנֵיהֶם יְלַמֵּדוּן׃ וַתִּקְרְבוּן וַתַּעַמְדוּן
22   תַּחַת הָהָר וְהָהָר בֹּעֵר בָּאֵשׁ עַד־לֵב הַשָּׁמַיִם
23   חֹשֶׁךְ עָנָן וַעֲרָפֶל׃ וַיְדַבֵּר יְהוָה אֲלֵיכֶם מִתּוֹךְ הָאֵשׁ
24   קוֹל דְּבָרִים אַתֶּם שֹׁמְעִים וּתְמוּנָה אֵינְכֶם רֹאִים
25   זוּלָתִי קוֹל׃ וַיַּגֵּד לָכֶם אֶת־בְּרִיתוֹ אֲשֶׁר צִוָּה אֶתְכֶם
26   לַעֲשׂוֹת עֲשֶׂרֶת הַדְּבָרִים וַיִּכְתְּבֵם עַל־שְׁנֵי לֻחוֹת
27   אֲבָנִים׃ וְאֹתִי צִוָּה יְהוָה בָּעֵת הַהִוא לְלַמֵּד אֶתְכֶם
28   חֻקִּים וּמִשְׁפָּטִים לַעֲשֹׂתְכֶם אֹתָם בָּאָרֶץ אֲשֶׁר
29   אַתֶּם עֹבְרִים שָׁמָּה לְרִשְׁתָּהּ׃ וְנִשְׁמַרְתֶּם מְאֹד
30   לְנַפְשֹׁתֵיכֶם כִּי לֹא רְאִיתֶם כָּל־תְּמוּנָה בְּיוֹם דִּבֶּר
31   יְהוָה אֲלֵיכֶם בְּחֹרֵב מִתּוֹךְ הָאֵשׁ׃ פֶּן־תַּשְׁחִתוּן
32   וַעֲשִׂיתֶם לָכֶם פֶּסֶל תְּמוּנַת כָּל־סָמֶל תַּבְנִית זָכָר
33   אוֹ נְקֵבָה׃ תַּבְנִית כָּל־בְּהֵמָה אֲשֶׁר בָּאָרֶץ תַּבְנִית כָּל־
34  צִפּוֹר כָּנָף אֲשֶׁר תָּעוּף בַּשָּׁמָיִם׃ תַּבְנִית כָּל־רֹמֵשׂ
35   בָּאֲדָמָה תַּבְנִית כָּל־דָּגָה אֲשֶׁר־בַּמַּיִם מִתַּחַת לָאָרֶץ׃
36   וּפֶן־תִּשָּׂא עֵינֶיךָ הַשָּׁמַיְמָה וְרָאִיתָ אֶת־הַשֶּׁמֶשׁ וְאֶת־
37  הַיָּרֵחַ וְאֶת־הַכּוֹכָבִים כֹּל צְבָא הַשָּׁמַיִם וְנִדַּחְתָּ
38   וְהִשְׁתַּחֲוִיתָ לָהֶם וַעֲבַדְתָּם אֲשֶׁר חָלַק יְהוָה אֱלֹהֶיךָ
39   אֹתָם לְכֹל הָעַמִּים תַּחַת כָּל־הַשָּׁמָיִם׃ וְאֶתְכֶם
40   לָקַח יְהוָה וַיּוֹצִא אֶתְכֶם מִכּוּר הַבַּרְזֶל מִמִּצְרָיִם
41   לִהְיוֹת לוֹ לְעַם נַחֲלָה כַּיּוֹם הַזֶּה׃ וַיהוָה הִתְאַנַּף־
42  בִּי עַל־דִּבְרֵיכֶם וַיִּשָּׁבַע לְבִלְתִּי עָבְרִי אֶת־הַיַּרְדֵּן
