 1  וְיַשְׁמִעֵנוּ אֹתָהּ וְנַעֲשֶׂנָּה׃ כִּי־קָרוֹב אֵלֶיךָ הַדָּבָר מְאֹד
 2   בְּפִיךָ וּבִלְבָבְךָ לַעֲשֹׂתוֹ׃ [ס]              רְאֵה
 3   נָתַתִּי לְפָנֶיךָ הַיּוֹם אֶת־הַחַיִּים וְאֶת־הַטּוֹב וְאֶת־
 4  הַמָּוֶת וְאֶת־הָרָע׃ אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם לְאַהֲבָה
 5   אֶת־יְהוָה אֱלֹהֶיךָ לָלֶכֶת בִּדְרָכָיו וְלִשְׁמֹר מִצְו͏ֹתָיו
 6   וְחֻקֹּתָיו וּמִשְׁפָּטָיו וְחָיִיתָ וְרָבִיתָ וּבֵרַכְךָ יְהוָה
 7   אֱלֹהֶיךָ בָּאָרֶץ אֲשֶׁר־אַתָּה בָא־שָׁמָּה לְרִשְׁתָּהּ׃
 8   וְאִם־יִפְנֶה לְבָבְךָ וְלֹא תִשְׁמָע וְנִדַּחְתָּ וְהִשְׁתַּחֲוִיתָ
 9   לֵאלֹהִים אֲחֵרִים וַעֲבַדְתָּם׃ הִגַּדְתִּי לָכֶם הַיּוֹם
10   כִּי אָבֹד תֹּאבֵדוּן לֹא־תַאֲרִיכֻן יָמִים עַל־הָאֲדָמָה
11   אֲשֶׁר אַתָּה עֹבֵר אֶת־הַיַּרְדֵּן לָבוֹא שָׁמָּה לְרִשְׁתָּהּ׃
12   הַעִדֹתִי בָכֶם הַיּוֹם אֶת־הַשָּׁמַיִם וְאֶת־הָאָרֶץ הַחַיִּים
13   וְהַמָּוֶת נָתַתִּי לְפָנֶיךָ הַבְּרָכָה וְהַקְּלָלָה וּבָחַרְתָּ
14   בַּחַיִּים לְמַעַן תִּחְיֶה אַתָּה וְזַרְעֶךָ׃ לְאַהֲבָה אֶת־
15  יְהוָה אֱלֹהֶיךָ לִשְׁמֹעַ בְּקֹלוֹ וּלְדָבְקָה־בוֹ כִּי הוּא
16   חַיֶּיךָ וְאֹרֶךְ יָמֶיךָ לָשֶׁבֶת עַל־הָאֲדָמָה אֲשֶׁר
17   נִשְׁבַּע יְהוָה לַאֲבֹתֶיךָ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב
18   לָתֵת לָהֶם׃ [פ]
19  וַיֵּלֶךְ מֹשֶׁה וַיְדַבֵּר אֶת־הַדְּבָרִים הָאֵלֶּה אֶל־כָּל־
20  יִשְׂרָאֵל׃ וַיֹּאמֶר אֲלֵהֶם בֶּן־מֵאָה וְעֶשְׂרִים שָׁנָה
21   אָנֹכִי הַיּוֹם לֹא־אוּכַל עוֹד לָצֵאת וְלָבוֹא וַיהוָה
22   אָמַר אֵלַי לֹא תַעֲבֹר אֶת־הַיַּרְדֵּן הַזֶּה׃ יְהוָה אֱלֹהֶיךָ
23   הוּא  עֹבֵר לְפָנֶיךָ הוּא־יַשְׁמִיד אֶת־הַגּוֹיִם הָאֵלֶּה
24   מִלְּפָנֶיךָ וִירִשְׁתָּם יְהוֹשֻׁעַ הוּא עֹבֵר לְפָנֶיךָ כַּאֲשֶׁר
25   דִּבֶּר יְהוָה׃♢  וְעָשָׂה יְהוָה לָהֶם כַּאֲשֶׁר עָשָׂה לְסִיחוֹן
26   וּלְעוֹג מַלְכֵי הָאֱמֹרִי וּלְאַרְצָם אֲשֶׁר הִשְׁמִיד אֹתָם׃
27   וּנְתָנָם יְהוָה לִפְנֵיכֶם וַעֲשִׂיתֶם לָהֶם כְּכָל־הַמִּצְוָה
28   אֲשֶׁר צִוִּיתִי אֶתְכֶם׃ חִזְקוּ וְאִמְצוּ אַל־תִּירְאוּ וְאַל־
29  תַּעַרְצוּ מִפְּנֵיהֶם כִּי  יְהוָה אֱלֹהֶיךָ הוּא הַהֹלֵךְ עִמָּךְ
30   לֹא יַרְפְּךָ וְלֹא יַעַזְבֶךָּ׃♢ [ס]              וַיִּקְרָא
31   מֹשֶׁה לִיהוֹשֻׁעַ וַיֹּאמֶר אֵלָיו לְעֵינֵי כָל־יִשְׂרָאֵל
32   חֲזַק וֶאֱמָץ כִּי אַתָּה תָּבוֹא אֶת־הָעָם הַזֶּה אֶל־הָאָרֶץ
33   אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתָם לָתֵת לָהֶם וְאַתָּה
34   תַּנְחִילֶנָּה אוֹתָם׃ וַיהוָה הוּא  הַהֹלֵךְ לְפָנֶיךָ הוּא
35   יִהְיֶה עִמָּךְ לֹא יַרְפְּךָ וְלֹא יַעַזְבֶךָּ לֹא תִירָא וְלֹא
36   תֵחָת׃ וַיִּכְתֹּב מֹשֶׁה אֶת־הַתּוֹרָה הַזֹּאת וַיִּתְּנָהּ אֶל־
37  הַכֹּהֲנִים בְּנֵי לֵוִי הַנֹּשְׂאִים אֶת־אֲרוֹן בְּרִית יְהוָה
38   וְאֶל־כָּל־זִקְנֵי יִשְׂרָאֵל׃  וַיְצַו מֹשֶׁה אוֹתָם לֵאמֹר מִקֵּץ
39   שֶׁבַע שָׁנִים בְּמֹעֵד שְׁנַת הַשְּׁמִטָּה בְּחַג הַסֻּכּוֹת׃
40   בְּבוֹא כָל־יִשְׂרָאֵל לֵרָאוֹת אֶת־פְּנֵי יְהוָה אֱלֹהֶיךָ
41   בַּמָּקוֹם אֲשֶׁר יִבְחָר תִּקְרָא אֶת־הַתּוֹרָה הַזֹּאת נֶגֶד
42   כָּל־יִשְׂרָאֵל בְּאָזְנֵיהֶם׃ הַקְהֵל אֶת־הָעָם הָאֲנָשִׁים
