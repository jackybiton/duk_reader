 1  וְנַעֲרֹתֶיהָ הֹלְכֹת עַל־יַד הַיְאֹר וַתֵּרֶא אֶת־הַתֵּבָה בְּתוֹךְ
 2   הַסּוּף וַתִּשְׁלַח אֶת־אֲמָתָהּ וַתִּקָּחֶהָ׃ וַתִּפְתַּח וַתִּרְאֵהוּ אֶת־
 3  הַיֶּלֶד וְהִנֵּה־נַעַר בֹּכֶה וַתַּחְמֹל עָלָיו וַתֹּאמֶר מִיַּלְדֵי
 4   הָעִבְרִים זֶה׃ וַתֹּאמֶר אֲחֹתוֹ אֶל־בַּת־פַּרְעֹה הַאֵלֵךְ
 5   וְקָרָאתִי לָךְ אִשָּׁה מֵינֶקֶת מִן הָעִבְרִיֹּת וְתֵינִק לָךְ אֶת־
 6  הַיָּלֶד׃ וַתֹּאמֶר־לָהּ בַּת־פַּרְעֹה לֵכִי וַתֵּלֶךְ הָעַלְמָה
 7   וַתִּקְרָא אֶת־אֵם הַיָּלֶד׃ וַתֹּאמֶר לָהּ בַּת־פַּרְעֹה הֵילִיכִי
 8   אֶת־הַיֶּלֶד הַזֶּה וְהֵינִקִהוּ לִי וַאֲנִי אֶתֵּן אֶת־שְׂכָרֵךְ
 9   וַתִּקַּח הָאִשָּׁה הַיֶּלֶד וַתְּנִיקֵהוּ׃ וַיִּגְדַּל הַיֶּלֶד וַתְּבִאֵהוּ
10   לְבַת־פַּרְעֹה וַיְהִי־לָהּ לְבֵן וַתִּקְרָא שְׁמוֹ מֹשֶׁה וַתֹּאמֶר
11   כִּי מִן־הַמַּיִם מְשִׁיתִהוּ׃  וַיְהִי  בַּיָּמִים הָהֵם וַיִּגְדַּל מֹשֶׁה
12   וַיֵּצֵא אֶל־אֶחָיו וַיַּרְא בְּסִבְלֹתָם וַיַּרְא אִישׁ מִצְרִי מַכֶּה
13   אִישׁ־עִבְרִי מֵאֶחָיו׃ וַיִּפֶן כֹּה וָכֹה וַיַּרְא כִּי אֵין אִישׁ
14   וַיַּךְ אֶת־הַמִּצְרִי וַיִּטְמְנֵהוּ בַּחוֹל׃ וַיֵּצֵא בַּיּוֹם הַשֵּׁנִי
15   וְהִנֵּה שְׁנֵי־אֲנָשִׁים עִבְרִים נִצִּים וַיֹּאמֶר לָרָשָׁע לָמָּה
16   תַכֶּה רֵעֶךָ׃ וַיֹּאמֶר מִי שָׂמְךָ לְאִישׁ שַׂר וְשֹׁפֵט עָלֵינוּ
17   הַלְהָרְגֵנִי אַתָּה אֹמֵר כַּאֲשֶׁר הָרַגְתָּ אֶת־הַמִּצְרִי וַיִּירָא
18   מֹשֶׁה וַיֹּאמַר אָכֵן נוֹדַע הַדָּבָר׃ וַיִּשְׁמַע פַּרְעֹה אֶת־
19  הַדָּבָר הַזֶּה וַיְבַקֵּשׁ לַהֲרֹג אֶת־מֹשֶׁה וַיִּבְרַח מֹשֶׁה
20   מִפְּנֵי פַרְעֹה וַיֵּשֶׁב בְּאֶרֶץ־מִדְיָן וַיֵּשֶׁב עַל־הַבְּאֵר׃
21   וּלְכֹהֵן מִדְיָן שֶׁבַע בָּנוֹת וַתָּבֹאנָה וַתִּדְלֶנָה וַתְּמַלֶּאנָה
22   אֶת־הָרְהָטִים לְהַשְׁקוֹת צֹאן אֲבִיהֶן׃ וַיָּבֹאוּ הָרֹעִים
23   וַיְגָרְשׁוּם וַיָּקָם מֹשֶׁה וַיּוֹשִׁעָן וַיַּשְׁקְ אֶת־צֹאנָם׃
24   וַתָּבֹאנָה אֶל־רְעוּאֵל אֲבִיהֶן וַיֹּאמֶר מַדּוּעַ מִהַרְתֶּן
25   בֹּא הַיּוֹם׃ וַתֹּאמַרְןָ אִישׁ מִצְרִי הִצִּילָנוּ מִיַּד הָרֹעִים
26   וְגַם־דָּלֹה דָלָה לָנוּ וַיַּשְׁקְ אֶת־הַצֹּאן׃ וַיֹּאמֶר אֶל־בְּנֹתָיו
27   וְאַיּוֹ לָמָּה זֶּה עֲזַבְתֶּן אֶת־הָאִישׁ קִרְאֶן לוֹ וְיֹאכַל
28   לָחֶם׃ וַיּוֹאֶל מֹשֶׁה לָשֶׁבֶת אֶת־הָאִישׁ וַיִּתֵּן אֶת־צִפֹּרָה
29   בִתּוֹ לְמֹשֶׁה׃ וַתֵּלֶד בֵּן וַיִּקְרָא אֶת־שְׁמוֹ גֵּרְשֹׁם כִּי
30   אָמַר גֵּר הָיִיתִי בְּאֶרֶץ נָכְרִיָּה׃ [פ]
31   וַיְהִי בַיָּמִים הָרַבִּים הָהֵם וַיָּמָת מֶלֶךְ מִצְרַיִם וַיֵּאָנְחוּ
32   בְנֵי־יִשְׂרָאֵל מִן־הָעֲבֹדָה וַיִּזְעָקוּ וַתַּעַל שַׁוְעָתָם אֶל־
33  הָאֱלֹהִים מִן־הָעֲבֹדָה׃ וַיִּשְׁמַע אֱלֹהִים אֶת־נַאֲקָתָם
34   וַיִּזְכֹּר אֱלֹהִים אֶת־בְּרִיתוֹ אֶת־אַבְרָהָם אֶת־יִצְחָק
35   וְאֶת־יַעֲקֹב׃ וַיַּרְא אֱלֹהִים אֶת־בְּנֵי יִשְׂרָאֵל וַיֵּדַע
36   אֱלֹהִים׃ [ס]              וּמֹשֶׁה הָיָה רֹעֶה אֶת־צֹאן
37   יִתְרוֹ חֹתְנוֹ כֹּהֵן מִדְיָן וַיִּנְהַג אֶת־הַצֹּאן אַחַר הַמִּדְבָּר
38   וַיָּבֹא אֶל־הַר הָאֱלֹהִים חֹרֵבָה׃ וַיֵּרָא מַלְאַךְ יְהוָה אֵלָיו
39   בְּלַבַּת־אֵשׁ מִתּוֹךְ הַסְּנֶה וַיַּרְא וְהִנֵּה הַסְּנֶה בֹּעֵר בָּאֵשׁ
40   וְהַסְּנֶה אֵינֶנּוּ אֻכָּל׃ וַיֹּאמֶר מֹשֶׁה אָסֻרָה־נָּא וְאֶרְאֶה
41   אֶת־הַמַּרְאֶה הַגָּדֹל הַזֶּה מַדּוּעַ לֹא־יִבְעַר הַסְּנֶה׃ וַיַּרְא
42   יְהוָה כִּי סָר לִרְאוֹת וַיִּקְרָא אֵלָיו אֱלֹהִים מִתּוֹךְ הַסְּנֶה
