 1  וְיִשְׁאֲלוּ אִישׁ  מֵאֵת רֵעֵהוּ וְאִשָּׁה מֵאֵת רְעוּתָהּ כְּלֵי־
 2  כֶסֶף וּכְלֵי זָהָב׃ וַיִּתֵּן יְהוָה אֶת־חֵן הָעָם בְּעֵינֵי מִצְרָיִם
 3   גַּם  הָאִישׁ מֹשֶׁה גָּדוֹל מְאֹד בְּאֶרֶץ מִצְרַיִם בְּעֵינֵי
 4   עַבְדֵי־פַרְעֹה וּבְעֵינֵי הָעָם׃ [ס]              וַיֹּאמֶר
 5   מֹשֶׁה כֹּה אָמַר יְהוָה כַּחֲצֹת הַלַּיְלָה אֲנִי יוֹצֵא בְּתוֹךְ
 6   מִצְרָיִם׃ וּמֵת כָּל־בְּכוֹר בְּאֶרֶץ מִצְרַיִם מִבְּכוֹר
 7   פַּרְעֹה הַיֹּשֵׁב עַל־כִּסְאוֹ עַד בְּכוֹר הַשִּׁפְחָה אֲשֶׁר
 8   אַחַר הָרֵחָיִם וְכֹל בְּכוֹר בְּהֵמָה׃ וְהָיְתָה צְעָקָה
 9   גְדֹלָה בְּכָל־אֶרֶץ מִצְרָיִם אֲשֶׁר כָּמֹהוּ לֹא נִהְיָתָה
10   וְכָמֹהוּ לֹא תֹסִף׃ וּלְכֹל  בְּנֵי יִשְׂרָאֵל לֹא יֶחֱרַץ־כֶּלֶב
11   לְשֹׁנוֹ לְמֵאִישׁ וְעַד־בְּהֵמָה לְמַעַן תֵּדְעוּן אֲשֶׁר יַפְלֶה
12   יְהוָה בֵּין מִצְרַיִם וּבֵין יִשְׂרָאֵל׃ וְיָרְדוּ כָל־עֲבָדֶיךָ
13   אֵלֶּה אֵלַי וְהִשְׁתַּחֲווּ־לִי לֵאמֹר צֵא אַתָּה וְכָל־הָעָם
14   אֲשֶׁר־בְּרַגְלֶיךָ וְאַחֲרֵי־כֵן אֵצֵא וַיֵּצֵא מֵעִם־פַּרְעֹה
15   בָּחֳרִי־אָף׃ [ס]             וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה
16   לֹא־יִשְׁמַע אֲלֵיכֶם פַּרְעֹה לְמַעַן רְבוֹת מוֹפְתַי בְּאֶרֶץ
17   מִצְרָיִם׃ וּמֹשֶׁה וְאַהֲרֹן עָשׂוּ אֶת־כָּל־הַמֹּפְתִים הָאֵלֶּה
18   לִפְנֵי פַרְעֹה וַיְחַזֵּק יְהוָה אֶת־לֵב פַּרְעֹה וְלֹא־שִׁלַּח
19   אֶת־בְּנֵי־יִשְׂרָאֵל מֵאַרְצוֹ׃ [ס]             וַיֹּאמֶר
20   יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן בְּאֶרֶץ מִצְרַיִם לֵאמֹר׃
21   הַחֹדֶשׁ הַזֶּה לָכֶם רֹאשׁ חֳדָשִׁים רִאשׁוֹן הוּא לָכֶם
22   לְחָדְשֵׁי הַשָּׁנָה׃ דַּבְּרוּ אֶל־כָּל־עֲדַת יִשְׂרָאֵל לֵאמֹר
23   בֶּעָשֹׂר לַחֹדֶשׁ הַזֶּה וְיִקְחוּ לָהֶם אִישׁ שֶׂה לְבֵית־
24  אָבֹת שֶׂה לַבָּיִת׃ וְאִם־יִמְעַט הַבַּיִת מִהְיוֹת מִשֶּׂה
25   וְלָקַח הוּא וּשְׁכֵנוֹ הַקָּרֹב אֶל־בֵּיתוֹ בְּמִכְסַת נְפָשֹׁת
26   אִישׁ לְפִי אָכְלוֹ תָּכֹסּוּ עַל־הַשֶּׂה׃ שֶׂה תָמִים זָכָר
27   בֶּן־שָׁנָה יִהְיֶה לָכֶם מִן־הַכְּבָשִׂים וּמִן־הָעִזִּים תִּקָּחוּ׃
28   וְהָיָה לָכֶם לְמִשְׁמֶרֶת עַד אַרְבָּעָה עָשָׂר יוֹם
29   לַחֹדֶשׁ הַזֶּה וְשָׁחֲטוּ אֹתוֹ כֹּל קְהַל עֲדַת־יִשְׂרָאֵל
30   בֵּין הָעַרְבָּיִם׃ וְלָקְחוּ מִן־הַדָּם וְנָתְנוּ עַל־שְׁתֵּי הַמְּזוּזֹת
31   וְעַל־הַמַּשְׁקוֹף עַל הַבָּתִּים אֲשֶׁר־יֹאכְלוּ אֹתוֹ בָּהֶם׃
32   וְאָכְלוּ אֶת־הַבָּשָׂר בַּלַּיְלָה הַזֶּה צְלִי־אֵשׁ וּמַצּוֹת עַל־
33  מְרֹרִים יֹאכְלֻהוּ׃ אַל־תֹּאכְלוּ מִמֶּנּוּ נָא וּבָשֵׁל מְבֻשָּׁל
34   בַּמָּיִם כִּי אִם־צְלִי־אֵשׁ רֹאשׁוֹ עַל־כְּרָעָיו וְעַל־קִרְבּוֹ׃
35   וְלֹא־תוֹתִירוּ מִמֶּנּוּ עַד־בֹּקֶר וְהַנֹּתָר מִמֶּנּוּ עַד־בֹּקֶר
36   בָּאֵשׁ תִּשְׂרֹפוּ׃ וְכָכָה תֹּאכְלוּ אֹתוֹ מָתְנֵיכֶם חֲגֻרִים
37   נַעֲלֵיכֶם בְּרַגְלֵיכֶם וּמַקֶּלְכֶם בְּיֶדְכֶם וַאֲכַלְתֶּם אֹתוֹ
38   בְּחִפָּזוֹן פֶּסַח הוּא לַיהוָה׃ וְעָבַרְתִּי בְאֶרֶץ־מִצְרַיִם
39   בַּלַּיְלָה הַזֶּה וְהִכֵּיתִי כָל־בְּכוֹר בְּאֶרֶץ מִצְרַיִם
40   מֵאָדָם וְעַד־בְּהֵמָה וּבְכָל־אֱלֹהֵי מִצְרַיִם אֶעֱשֶׂה
41   שְׁפָטִים אֲנִי יְהוָה׃ וְהָיָה הַדָּם לָכֶם לְאֹת עַל הַבָּתִּים
42   אֲשֶׁר אַתֶּם שָׁם וְרָאִיתִי אֶת־הַדָּם וּפָסַחְתִּי עֲלֵכֶם
