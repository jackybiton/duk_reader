 1  וַיִּרְאוּ שֹׁטְרֵי בְנֵי־יִשְׂרָאֵל אֹתָם בְּרָע לֵאמֹר לֹא־תִגְרְעוּ
 2   מִלִּבְנֵיכֶם דְּבַר־יוֹם בְּיוֹמוֹ׃ וַיִּפְגְּעוּ אֶת־מֹשֶׁה וְאֶת־אַהֲרֹן
 3   נִצָּבִים לִקְרָאתָם בְּצֵאתָם מֵאֵת פַּרְעֹה׃ וַיֹּאמְרוּ אֲלֵהֶם
 4   יֵרֶא יְהוָה עֲלֵיכֶם וְיִשְׁפֹּט אֲשֶׁר הִבְאַשְׁתֶּם אֶת־רֵיחֵנוּ
 5   בְּעֵינֵי פַרְעֹה וּבְעֵינֵי עֲבָדָיו לָתֶת־חֶרֶב בְּיָדָם לְהָרְגֵנוּ׃
 6    וַיָּשָׁב מֹשֶׁה אֶל־יְהוָה וַיֹּאמַר אֲדֹנָי לָמָה הֲרֵעֹתָה לָעָם
 7   הַזֶּה לָמָּה זֶּה שְׁלַחְתָּנִי׃ וּמֵאָז בָּאתִי אֶל־פַּרְעֹה לְדַבֵּר
 8   בִּשְׁמֶךָ הֵרַע לָעָם הַזֶּה וְהַצֵּל לֹא־הִצַּלְתָּ אֶת־עַמֶּךָ׃
 9   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה עַתָּה תִרְאֶה אֲשֶׁר אֶעֱשֶׂה
10   לְפַרְעֹה כִּי בְיָד חֲזָקָה יְשַׁלְּחֵם וּבְיָד חֲזָקָה יְגָרְשֵׁם
11   מֵאַרְצוֹ׃ [ס]            וַיְדַבֵּר אֱלֹהִים אֶל־מֹשֶׁה
12   וַיֹּאמֶר אֵלָיו אֲנִי יְהוָה׃ וָאֵרָא אֶל־אַבְרָהָם אֶל־יִצְחָק
13   וְאֶל־יַעֲקֹב בְּאֵל שַׁדָּי וּשְׁמִי יְהוָה לֹא נוֹדַעְתִּי לָהֶם׃
14   וְגַם הֲקִמֹתִי אֶת־בְּרִיתִי אִתָּם לָתֵת לָהֶם אֶת־אֶרֶץ
15   כְּנָעַן אֵת אֶרֶץ מְגֻרֵיהֶם אֲשֶׁר־גָּרוּ בָהּ׃ וְגַם  אֲנִי שָׁמַעְתִּי
16   אֶת־נַאֲקַת בְּנֵי יִשְׂרָאֵל אֲשֶׁר מִצְרַיִם מַעֲבִדִים אֹתָם
17   וָאֶזְכֹּר אֶת־בְּרִיתִי׃♢ לָכֵן אֱמֹר לִבְנֵי־יִשְׂרָאֵל אֲנִי יְהוָה
18   וְהוֹצֵאתִי אֶתְכֶם מִתַּחַת סִבְלֹת מִצְרַיִם וְהִצַּלְתִּי אֶתְכֶם
19   מֵעֲבֹדָתָם וְגָאַלְתִּי אֶתְכֶם בִּזְרוֹעַ נְטוּיָה וּבִשְׁפָטִים
20   גְּדֹלִים׃ וְלָקַחְתִּי אֶתְכֶם לִי לְעָם וְהָיִיתִי לָכֶם לֵאלֹהִים
21   וִידַעְתֶּם כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם הַמּוֹצִיא אֶתְכֶם מִתַּחַת
22   סִבְלוֹת מִצְרָיִם׃ וְהֵבֵאתִי אֶתְכֶם אֶל־הָאָרֶץ אֲשֶׁר
23   נָשָׂאתִי אֶת־יָדִי לָתֵת אֹתָהּ לְאַבְרָהָם לְיִצְחָק וּלְיַעֲקֹב
24   וְנָתַתִּי אֹתָהּ לָכֶם מוֹרָשָׁה אֲנִי יְהוָה׃ וַיְדַבֵּר מֹשֶׁה כֵּן
25   אֶל־בְּנֵי יִשְׂרָאֵל וְלֹא שָׁמְעוּ אֶל־מֹשֶׁה מִקֹּצֶר רוּחַ
26   וּמֵעֲבֹדָה קָשָׁה׃♢ [פ]
27   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ בֹּא דַבֵּר אֶל־פַּרְעֹה
28   מֶלֶךְ מִצְרָיִם וִישַׁלַּח אֶת־בְּנֵי־יִשְׂרָאֵל מֵאַרְצוֹ׃
29   וַיְדַבֵּר מֹשֶׁה לִפְנֵי יְהוָה לֵאמֹר הֵן בְּנֵי־יִשְׂרָאֵל
30   לֹא־שָׁמְעוּ אֵלַי וְאֵיךְ יִשְׁמָעֵנִי פַרְעֹה וַאֲנִי עֲרַל
31   שְׂפָתָיִם׃ [פ]
32   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן וַיְצַוֵּם אֶל־בְּנֵי
33   יִשְׂרָאֵל וְאֶל־פַּרְעֹה מֶלֶךְ מִצְרָיִם לְהוֹצִיא אֶת־בְּנֵי־
34  יִשְׂרָאֵל מֵאֶרֶץ מִצְרָיִם׃♢ [ס]              אֵלֶּה רָאשֵׁי
35   בֵית־אֲבֹתָם בְּנֵי רְאוּבֵן בְּכֹר יִשְׂרָאֵל חֲנוֹךְ וּפַלּוּא
36   חֶצְרֹן וְכַרְמִי אֵלֶּה מִשְׁפְּחֹת רְאוּבֵן׃ וּבְנֵי שִׁמְעוֹן
37   יְמוּאֵל וְיָמִין וְאֹהַד וְיָכִין וְצֹחַר וְשָׁאוּל בֶּן־הַכְּנַעֲנִית
38   אֵלֶּה מִשְׁפְּחֹת שִׁמְעוֹן׃ וְאֵלֶּה שְׁמוֹת בְּנֵי־לֵוִי לְתֹלְדֹתָם
39   גֵּרְשׁוֹן וּקְהָת וּמְרָרִי וּשְׁנֵי חַיֵּי לֵוִי שֶׁבַע וּשְׁלֹשִׁים
40   וּמְאַת שָׁנָה׃ בְּנֵי גֵרְשׁוֹן לִבְנִי וְשִׁמְעִי לְמִשְׁפְּחֹתָם׃
41   וּבְנֵי קְהָת עַמְרָם וְיִצְהָר וְחֶבְרוֹן וְעֻזִּיאֵל וּשְׁנֵי חַיֵּי
42   קְהָת שָׁלֹשׁ וּשְׁלֹשִׁים וּמְאַת שָׁנָה׃ וּבְנֵי מְרָרִי מַחְלִי
