 1  וּשְׁמַע בְּקֹלוֹ אַל־תַּמֵּר בּוֹ כִּי לֹא יִשָּׂא לְפִשְׁעֲכֶם כִּי
 2   שְׁמִי בְּקִרְבּוֹ׃ כִּי אִם־שָׁמוֹעַ תִּשְׁמַע בְּקֹלוֹ וְעָשִׂיתָ כֹּל
 3   אֲשֶׁר אֲדַבֵּר וְאָיַבְתִּי אֶת־אֹיְבֶיךָ וְצַרְתִּי אֶת־צֹרְרֶיךָ׃ כִּי־
 4  יֵלֵךְ מַלְאָכִי לְפָנֶיךָ וֶהֱבִיאֲךָ אֶל־הָאֱמֹרִי וְהַחִתִּי וְהַפְּרִזִּי
 5   וְהַכְּנַעֲנִי הַחִוִּי וְהַיְבוּסִי וְהִכְחַדְתִּיו׃ לֹא־תִשְׁתַּחֲוֶה
 6   לֵאלֹהֵיהֶם וְלֹא תָעָבְדֵם וְלֹא תַעֲשֶׂה כְּמַעֲשֵׂיהֶם כִּי
 7   הָרֵס תְּהָרְסֵם וְשַׁבֵּר תְּשַׁבֵּר מַצֵּבֹתֵיהֶם׃ וַעֲבַדְתֶּם אֵת
 8   יְהוָה אֱלֹהֵיכֶם וּבֵרַךְ אֶת־לַחְמְךָ וְאֶת־מֵימֶיךָ וַהֲסִרֹתִי
 9   מַחֲלָה מִקִּרְבֶּךָ׃ [ס]              לֹא תִהְיֶה
10   מְשַׁכֵּלָה וַעֲקָרָה בְּאַרְצֶךָ אֶת־מִסְפַּר יָמֶיךָ אֲמַלֵּא׃
11   אֶת־אֵימָתִי אֲשַׁלַּח לְפָנֶיךָ וְהַמֹּתִי אֶת־כָּל־הָעָם אֲשֶׁר
12   תָּבֹא בָּהֶם וְנָתַתִּי אֶת־כָּל־אֹיְבֶיךָ אֵלֶיךָ עֹרֶף׃ וְשָׁלַחְתִּי
13   אֶת־הַצִּרְעָה לְפָנֶיךָ וְגֵרְשָׁה אֶת־הַחִוִּי אֶת־הַכְּנַעֲנִי וְאֶת־
14  הַחִתִּי מִלְּפָנֶיךָ׃ לֹא אֲגָרְשֶׁנּוּ מִפָּנֶיךָ בְּשָׁנָה אֶחָת פֶּן־
15  תִּהְיֶה הָאָרֶץ שְׁמָמָה וְרַבָּה עָלֶיךָ חַיַּת הַשָּׂדֶה׃ מְעַט
16   מְעַט אֲגָרְשֶׁנּוּ מִפָּנֶיךָ עַד אֲשֶׁר תִּפְרֶה וְנָחַלְתָּ אֶת־
17  הָאָרֶץ׃ וְשַׁתִּי אֶת־גְּבֻלְךָ מִיַּם־סוּף וְעַד־יָם פְּלִשְׁתִּים
18   וּמִמִּדְבָּר עַד־הַנָּהָר כִּי  אֶתֵּן בְּיֶדְכֶם אֵת יֹשְׁבֵי הָאָרֶץ
19   וְגֵרַשְׁתָּמוֹ מִפָּנֶיךָ׃ לֹא־תִכְרֹת לָהֶם וְלֵאלֹהֵיהֶם בְּרִית׃
20   לֹא יֵשְׁבוּ בְּאַרְצְךָ פֶּן־יַחֲטִיאוּ אֹתְךָ לִי כִּי תַעֲבֹד אֶת־
21  אֱלֹהֵיהֶם כִּי־יִהְיֶה לְךָ לְמוֹקֵשׁ׃ [פ]
22   וְאֶל־מֹשֶׁה אָמַר עֲלֵה אֶל־יְהוָה אַתָּה וְאַהֲרֹן נָדָב
23   וַאֲבִיהוּא וְשִׁבְעִים מִזִּקְנֵי יִשְׂרָאֵל וְהִשְׁתַּחֲוִיתֶם
24   מֵרָחֹק׃ וְנִגַּשׁ מֹשֶׁה לְבַדּוֹ אֶל־יְהוָה וְהֵם לֹא יִגָּשׁוּ
25   וְהָעָם לֹא יַעֲלוּ עִמּוֹ׃ וַיָּבֹא מֹשֶׁה וַיְסַפֵּר לָעָם אֵת כָּל־
26  דִּבְרֵי יְהוָה וְאֵת כָּל־הַמִּשְׁפָּטִים וַיַּעַן כָּל־הָעָם קוֹל
27   אֶחָד וַיֹּאמְרוּ כָּל־הַדְּבָרִים אֲשֶׁר־דִּבֶּר יְהוָה נַעֲשֶׂה׃
28   וַיִּכְתֹּב מֹשֶׁה אֵת כָּל־דִּבְרֵי יְהוָה וַיַּשְׁכֵּם בַּבֹּקֶר וַיִּבֶן
29   מִזְבֵּחַ תַּחַת הָהָר וּשְׁתֵּים עֶשְׂרֵה מַצֵּבָה לִשְׁנֵים
30   עָשָׂר שִׁבְטֵי יִשְׂרָאֵל׃ וַיִּשְׁלַח אֶת־נַעֲרֵי בְּנֵי יִשְׂרָאֵל
31   וַיַּעֲלוּ עֹלֹת וַיִּזְבְּחוּ זְבָחִים שְׁלָמִים לַיהוָה פָּרִים׃ וַיִּקַּח
32   מֹשֶׁה חֲצִי הַדָּם וַיָּשֶׂם בָּאַגָּנֹת וַחֲצִי הַדָּם זָרַק עַל־
33  הַמִּזְבֵּחַ׃ וַיִּקַּח סֵפֶר הַבְּרִית וַיִּקְרָא בְּאָזְנֵי הָעָם וַיֹּאמְרוּ
34   כֹּל אֲשֶׁר־דִּבֶּר יְהוָה נַעֲשֶׂה וְנִשְׁמָע׃ וַיִּקַּח מֹשֶׁה אֶת־
35  הַדָּם וַיִּזְרֹק עַל־הָעָם וַיֹּאמֶר הִנֵּה דַם־הַבְּרִית אֲשֶׁר
36   כָּרַת יְהוָה עִמָּכֶם עַל כָּל־הַדְּבָרִים הָאֵלֶּה׃ וַיַּעַל
37   מֹשֶׁה וְאַהֲרֹן נָדָב וַאֲבִיהוּא וְשִׁבְעִים מִזִּקְנֵי יִשְׂרָאֵל׃
38   וַיִּרְאוּ אֵת אֱלֹהֵי יִשְׂרָאֵל וְתַחַת רַגְלָיו כְּמַעֲשֵׂה
39   לִבְנַת הַסַּפִּיר וּכְעֶצֶם הַשָּׁמַיִם לָטֹהַר׃ וְאֶל־אֲצִילֵי
40   בְּנֵי יִשְׂרָאֵל לֹא שָׁלַח יָדוֹ וַיֶּחֱזוּ אֶת־הָאֱלֹהִים וַיֹּאכְלוּ
41   וַיִּשְׁתּוּ׃ [ס]             וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה
42   עֲלֵה אֵלַי הָהָרָה וֶהְיֵה־שָׁם וְאֶתְּנָה לְךָ אֶת־לֻחֹת הָאֶבֶן
