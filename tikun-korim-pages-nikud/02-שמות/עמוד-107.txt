 1  וּמַחְתֹּתֶיהָ זָהָב טָהוֹר׃ כִּכָּר זָהָב טָהוֹר עָשָׂה אֹתָהּ
 2   וְאֵת כָּל־כֵּלֶיהָ׃ [פ]
 3   וַיַּעַשׂ אֶת־מִזְבַּח הַקְּטֹרֶת עֲצֵי שִׁטִּים אַמָּה אָרְכּוֹ
 4   וְאַמָּה רָחְבּוֹ רָבוּעַ וְאַמָּתַיִם קֹמָתוֹ מִמֶּנּוּ הָיוּ קַרְנֹתָיו׃
 5   וַיְצַף אֹתוֹ זָהָב טָהוֹר אֶת־גַּגּוֹ וְאֶת־קִירֹתָיו סָבִיב
 6   וְאֶת־קַרְנֹתָיו וַיַּעַשׂ לוֹ זֵר זָהָב סָבִיב׃ וּשְׁתֵּי טַבְּעֹת
 7   זָהָב עָשָׂה־לוֹ  מִתַּחַת לְזֵרוֹ עַל שְׁתֵּי צַלְעֹתָיו עַל
 8   שְׁנֵי צִדָּיו לְבָתִּים לְבַדִּים לָשֵׂאת אֹתוֹ בָּהֶם׃ וַיַּעַשׂ
 9   אֶת־הַבַּדִּים עֲצֵי שִׁטִּים וַיְצַף אֹתָם זָהָב׃ וַיַּעַשׂ אֶת־
10  שֶׁמֶן הַמִּשְׁחָה קֹדֶשׁ וְאֶת־קְטֹרֶת הַסַּמִּים טָהוֹר
11   מַעֲשֵׂה רֹקֵחַ׃ [ס]              וַיַּעַשׂ
12   אֶת־מִזְבַּח הָעֹלָה עֲצֵי שִׁטִּים חָמֵשׁ אַמּוֹת אָרְכּוֹ
13   וְחָמֵשׁ־אַמּוֹת רָחְבּוֹ רָבוּעַ וְשָׁלֹשׁ אַמּוֹת קֹמָתוֹ׃
14   וַיַּעַשׂ קַרְנֹתָיו עַל אַרְבַּע פִּנֹּתָיו מִמֶּנּוּ הָיוּ קַרְנֹתָיו
15   וַיְצַף אֹתוֹ נְחֹשֶׁת׃ וַיַּעַשׂ אֶת־כָּל־כְּלֵי הַמִּזְבֵּחַ אֶת־
16  הַסִּירֹת וְאֶת־הַיָּעִים וְאֶת־הַמִּזְרָקֹת אֶת־הַמִּזְלָגֹת וְאֶת־
17  הַמַּחְתֹּת כָּל־כֵּלָיו עָשָׂה נְחֹשֶׁת׃ וַיַּעַשׂ לַמִּזְבֵּחַ מִכְבָּר
18   מַעֲשֵׂה רֶשֶׁת נְחֹשֶׁת תַּחַת כַּרְכֻּבּוֹ מִלְּמַטָּה עַד־
19  חֶצְיוֹ׃ וַיִּצֹק אַרְבַּע טַבָּעֹת בְּאַרְבַּע הַקְּצָו‍ֹת לְמִכְבַּר
20   הַנְּחֹשֶׁת בָּתִּים לַבַּדִּים׃ וַיַּעַשׂ אֶת־הַבַּדִּים עֲצֵי שִׁטִּים
21   וַיְצַף אֹתָם נְחֹשֶׁת׃ וַיָּבֵא אֶת־הַבַּדִּים בַּטַּבָּעֹת עַל
22   צַלְעֹת הַמִּזְבֵּחַ לָשֵׂאת אֹתוֹ בָּהֶם נְבוּב לֻחֹת עָשָׂה
23   אֹתוֹ׃ [ס]             וַיַּעַשׂ אֵת הַכִּיּוֹר נְחֹשֶׁת
24   וְאֵת כַּנּוֹ נְחֹשֶׁת בְּמַרְאֹת הַצֹּבְאֹת אֲשֶׁר צָבְאוּ פֶּתַח
25   אֹהֶל מוֹעֵד׃ [ס]             וַיַּעַשׂ אֶת־הֶחָצֵר
26   לִפְאַת  נֶגֶב תֵּימָנָה קַלְעֵי הֶחָצֵר שֵׁשׁ מָשְׁזָר מֵאָה
27   בָּאַמָּה׃ עַמּוּדֵיהֶם עֶשְׂרִים וְאַדְנֵיהֶם עֶשְׂרִים נְחֹשֶׁת
28   וָוֵי הָעַמּוּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃ וְלִפְאַת צָפוֹן מֵאָה
29   בָאַמָּה עַמּוּדֵיהֶם עֶשְׂרִים וְאַדְנֵיהֶם עֶשְׂרִים נְחֹשֶׁת
30   וָוֵי הָעַמּוּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃ וְלִפְאַת־יָם קְלָעִים
31   חֲמִשִּׁים בָּאַמָּה עַמּוּדֵיהֶם עֲשָׂרָה וְאַדְנֵיהֶם עֲשָׂרָה
32   וָוֵי הָעַמֻּדִים וַחֲשׁוּקֵיהֶם כָּסֶף׃ וְלִפְאַת קֵדְמָה מִזְרָחָה
33   חֲמִשִּׁים אַמָּה׃ קְלָעִים חֲמֵשׁ־עֶשְׂרֵה אַמָּה אֶל־הַכָּתֵף
34   עַמּוּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃ וְלַכָּתֵף הַשֵּׁנִית
35   מִזֶּה וּמִזֶּה לְשַׁעַר הֶחָצֵר קְלָעִים חֲמֵשׁ עֶשְׂרֵה
36   אַמָּה עַמֻּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃ כָּל־קַלְעֵי
37   הֶחָצֵר סָבִיב שֵׁשׁ מָשְׁזָר׃ וְהָאֲדָנִים לָעַמֻּדִים נְחֹשֶׁת
38   וָוֵי הָעַמּוּדִים וַחֲשׁוּקֵיהֶם כֶּסֶף וְצִפּוּי רָאשֵׁיהֶם כָּסֶף
39   וְהֵם מְחֻשָּׁקִים כֶּסֶף כֹּל עַמֻּדֵי הֶחָצֵר׃  וּמָסַךְ שַׁעַר
40   הֶחָצֵר מַעֲשֵׂה רֹקֵם תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי
41   וְשֵׁשׁ מָשְׁזָר וְעֶשְׂרִים אַמָּה אֹרֶךְ וְקוֹמָה בְרֹחַב
42   חָמֵשׁ אַמּוֹת לְעֻמַּת קַלְעֵי הֶחָצֵר׃ וְעַמֻּדֵיהֶם אַרְבָּעָה
