 1  וַיִּקְרָא לְזִקְנֵי הָעָם וַיָּשֶׂם לִפְנֵיהֶם אֵת כָּל־הַדְּבָרִים
 2   הָאֵלֶּה אֲשֶׁר צִוָּהוּ יְהוָה׃ וַיַּעֲנוּ כָל־הָעָם יַחְדָּו וַיֹּאמְרוּ
 3   כֹּל אֲשֶׁר־דִּבֶּר יְהוָה נַעֲשֶׂה וַיָּשֶׁב מֹשֶׁה אֶת־דִּבְרֵי
 4   הָעָם אֶל־יְהוָה׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה הִנֵּה אָנֹכִי
 5   בָּא אֵלֶיךָ בְּעַב הֶעָנָן בַּעֲבוּר יִשְׁמַע הָעָם בְּדַבְּרִי
 6   עִמָּךְ וְגַם־בְּךָ יַאֲמִינוּ לְעוֹלָם וַיַּגֵּד מֹשֶׁה אֶת־דִּבְרֵי
 7   הָעָם אֶל־יְהוָה׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה לֵךְ אֶל־הָעָם
 8   וְקִדַּשְׁתָּם הַיּוֹם וּמָחָר וְכִבְּסוּ שִׂמְלֹתָם׃ וְהָיוּ נְכֹנִים
 9   לַיּוֹם הַשְּׁלִישִׁי כִּי  בַּיּוֹם הַשְּׁלִשִׁי יֵרֵד יְהוָה לְעֵינֵי
10   כָל־הָעָם עַל־הַר סִינָי׃ וְהִגְבַּלְתָּ אֶת־הָעָם סָבִיב
11   לֵאמֹר הִשָּׁמְרוּ לָכֶם עֲלוֹת בָּהָר וּנְגֹעַ בְּקָצֵהוּ כָּל־
12  הַנֹּגֵעַ בָּהָר מוֹת יוּמָת׃ לֹא־תִגַּע בּוֹ יָד כִּי־סָקוֹל יִסָּקֵל
13   אוֹ־יָרֹה יִיָּרֶה אִם־בְּהֵמָה אִם־אִישׁ לֹא יִחְיֶה בִּמְשֹׁךְ
14   הַיֹּבֵל הֵמָּה יַעֲלוּ בָהָר׃ וַיֵּרֶד מֹשֶׁה מִן־הָהָר אֶל־
15  הָעָם וַיְקַדֵּשׁ אֶת־הָעָם וַיְכַבְּסוּ שִׂמְלֹתָם׃ וַיֹּאמֶר
16   אֶל־הָעָם הֱיוּ נְכֹנִים לִשְׁלֹשֶׁת יָמִים אַל־תִּגְּשׁוּ אֶל־
17  אִשָּׁה׃ וַיְהִי בַיּוֹם הַשְּׁלִישִׁי בִּהְיֹת הַבֹּקֶר וַיְהִי קֹלֹת
18   וּבְרָקִים וְעָנָן כָּבֵד עַל־הָהָר וְקֹל שֹׁפָר חָזָק מְאֹד
19   וַיֶּחֱרַד כָּל־הָעָם אֲשֶׁר בַּמַּחֲנֶה׃ וַיּוֹצֵא מֹשֶׁה אֶת־הָעָם
20   לִקְרַאת הָאֱלֹהִים מִן־הַמַּחֲנֶה וַיִּתְיַצְּבוּ בְּתַחְתִּית
21   הָהָר׃ וְהַר סִינַי עָשַׁן כֻּלּוֹ מִפְּנֵי אֲשֶׁר יָרַד עָלָיו
22   יְהוָה בָּאֵשׁ וַיַּעַל עֲשָׁנוֹ כְּעֶשֶׁן הַכִּבְשָׁן וַיֶּחֱרַד כָּל־
23  הָהָר מְאֹד׃ וַיְהִי קוֹל הַשֹּׁפָר הוֹלֵךְ וְחָזֵק מְאֹד מֹשֶׁה
24   יְדַבֵּר וְהָאֱלֹהִים יַעֲנֶנּוּ בְקוֹל׃  וַיֵּרֶד יְהוָה עַל־הַר סִינַי
25   אֶל־רֹאשׁ הָהָר וַיִּקְרָא יְהוָה לְמֹשֶׁה אֶל־רֹאשׁ הָהָר
26   וַיַּעַל מֹשֶׁה׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה רֵד הָעֵד בָּעָם
27   פֶּן־יֶהֶרְסוּ אֶל־יְהוָה לִרְאוֹת וְנָפַל מִמֶּנּוּ רָב׃ וְגַם
28   הַכֹּהֲנִים הַנִּגָּשִׁים אֶל־יְהוָה יִתְקַדָּשׁוּ פֶּן־יִפְרֹץ בָּהֶם
29   יְהוָה׃ וַיֹּאמֶר מֹשֶׁה אֶל־יְהוָה לֹא־יוּכַל הָעָם לַעֲלֹת
30   אֶל־הַר סִינָי כִּי־אַתָּה הַעֵדֹתָה בָּנוּ לֵאמֹר הַגְבֵּל אֶת־
31  הָהָר וְקִדַּשְׁתּוֹ׃ וַיֹּאמֶר אֵלָיו יְהוָה לֶךְ־רֵד וְעָלִיתָ
32   אַתָּה וְאַהֲרֹן עִמָּךְ וְהַכֹּהֲנִים וְהָעָם אַל־יֶהֶרְסוּ לַעֲלֹת
33   אֶל־יְהוָה פֶּן־יִפְרָץ־בָּם׃ וַיֵּרֶד מֹשֶׁה אֶל־הָעָם
34   וַיֹּאמֶר אֲלֵהֶם׃ [ס]             וַיְדַבֵּר אֱלֹהִים אֵת כָּל־
35  הַדְּבָרִים הָאֵלֶּה לֵאמֹר׃ [ס]             אָנֹכִי
36   יְהוָה אֱלֹהֶיךָ אֲשֶׁר הוֹצֵאתִיךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית
37   עֲבָדִים לֹא יִהְיֶה לְךָ אֱלֹהִים אֲחֵרִים עַל־פָּנַי לֹא
38  תַעֲשֶׂה־לְךָ פֶסֶל  וְכָל־תְּמוּנָה אֲשֶׁר בַּשָּׁמַיִם  מִמַּעַל
39   וַאֲשֶׁר בָּאָרֶץ מִתַּחַת וַאֲשֶׁר בַּמַּיִם  מִתַּחַת לָאָרֶץ לֹא־
40  תִשְׁתַּחֲוֶה לָהֶם וְלֹא תָעָבְדֵם כִּי אָנֹכִי יְהֹוָה אֱלֹהֶיךָ
41   אֵל קַנָּא פֹּקֵד עֲוֹן אָבֹת עַל־בָּנִים עַל־שִׁלֵּשִׁים
42   וְעַל־רִבֵּעִים לְשֹׂנְאָי וְעֹשֶׂה חֶסֶד לַאֲלָפִים לְאֹהֲבַי
