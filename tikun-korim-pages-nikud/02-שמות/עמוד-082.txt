 1  וַיִּקַּח יִתְרוֹ חֹתֵן מֹשֶׁה עֹלָה וּזְבָחִים לֵאלֹהִים וַיָּבֹא
 2   אַהֲרֹן וְכֹל  זִקְנֵי יִשְׂרָאֵל לֶאֱכָל־לֶחֶם עִם־חֹתֵן מֹשֶׁה
 3   לִפְנֵי הָאֱלֹהִים׃♢  וַיְהִי מִמָּחֳרָת וַיֵּשֶׁב מֹשֶׁה לִשְׁפֹּט אֶת־
 4  הָעָם וַיַּעֲמֹד הָעָם עַל־מֹשֶׁה מִן־הַבֹּקֶר עַד־הָעָרֶב׃
 5   וַיַּרְא חֹתֵן מֹשֶׁה אֵת כָּל־אֲשֶׁר־הוּא עֹשֶׂה לָעָם
 6   וַיֹּאמֶר מָה־הַדָּבָר הַזֶּה אֲשֶׁר אַתָּה עֹשֶׂה לָעָם מַדּוּעַ
 7   אַתָּה יוֹשֵׁב לְבַדֶּךָ וְכָל־הָעָם נִצָּב עָלֶיךָ מִן־בֹּקֶר עַד־
 8  עָרֶב׃ וַיֹּאמֶר מֹשֶׁה לְחֹתְנוֹ כִּי־יָבֹא אֵלַי הָעָם לִדְרֹשׁ
 9   אֱלֹהִים׃ כִּי־יִהְיֶה לָהֶם דָּבָר בָּא אֵלַי וְשָׁפַטְתִּי בֵּין
10   אִישׁ וּבֵין רֵעֵהוּ וְהוֹדַעְתִּי אֶת־חֻקֵּי הָאֱלֹהִים וְאֶת־
11  תּוֹרֹתָיו׃ וַיֹּאמֶר חֹתֵן מֹשֶׁה אֵלָיו לֹא־טוֹב הַדָּבָר
12   אֲשֶׁר אַתָּה עֹשֶׂה׃ נָבֹל תִּבֹּל גַּם־אַתָּה גַּם־הָעָם הַזֶּה
13   אֲשֶׁר עִמָּךְ כִּי־כָבֵד מִמְּךָ הַדָּבָר לֹא־תוּכַל עֲשֹׂהוּ
14   לְבַדֶּךָ׃ עַתָּה שְׁמַע בְּקֹלִי אִיעָצְךָ וִיהִי אֱלֹהִים עִמָּךְ
15   הֱיֵה אַתָּה לָעָם מוּל הָאֱלֹהִים וְהֵבֵאתָ אַתָּה אֶת־
16  הַדְּבָרִים אֶל־הָאֱלֹהִים׃ וְהִזְהַרְתָּה אֶתְהֶם אֶת־הַחֻקִּים
17   וְאֶת־הַתּוֹרֹת וְהוֹדַעְתָּ לָהֶם אֶת־הַדֶּרֶךְ יֵלְכוּ בָהּ וְאֶת־
18  הַמַּעֲשֶׂה אֲשֶׁר יַעֲשׂוּן׃ וְאַתָּה תֶחֱזֶה מִכָּל־הָעָם אַנְשֵׁי־
19  חַיִל יִרְאֵי אֱלֹהִים אַנְשֵׁי אֱמֶת שֹׂנְאֵי בָצַע וְשַׂמְתָּ
20   עֲלֵהֶם שָׂרֵי אֲלָפִים שָׂרֵי מֵאוֹת שָׂרֵי חֲמִשִּׁים
21   וְשָׂרֵי עֲשָׂרֹת׃ וְשָׁפְטוּ אֶת־הָעָם בְּכָל־עֵת וְהָיָה כָּל־
22  הַדָּבָר הַגָּדֹל יָבִיאוּ אֵלֶיךָ וְכָל־הַדָּבָר הַקָּטֹן יִשְׁפְּטוּ־
23  הֵם וְהָקֵל מֵעָלֶיךָ וְנָשְׂאוּ אִתָּךְ׃ אִם אֶת־הַדָּבָר הַזֶּה
24   תַּעֲשֶׂה וְצִוְּךָ אֱלֹהִים וְיָכָלְתָּ עֲמֹד וְגַם כָּל־הָעָם הַזֶּה
25   עַל־מְקֹמוֹ יָבֹא בְשָׁלוֹם׃  וַיִּשְׁמַע מֹשֶׁה לְקוֹל חֹתְנוֹ
26   וַיַּעַשׂ כֹּל אֲשֶׁר אָמָר׃ וַיִּבְחַר מֹשֶׁה אַנְשֵׁי־חַיִל מִכָּל־
27  יִשְׂרָאֵל וַיִּתֵּן אֹתָם רָאשִׁים עַל־הָעָם שָׂרֵי אֲלָפִים
28   שָׂרֵי מֵאוֹת שָׂרֵי חֲמִשִּׁים וְשָׂרֵי עֲשָׂרֹת׃ וְשָׁפְטוּ
29   אֶת־הָעָם בְּכָל־עֵת אֶת־הַדָּבָר הַקָּשֶׁה יְבִיאוּן אֶל־
30  מֹשֶׁה וְכָל־הַדָּבָר הַקָּטֹן יִשְׁפּוּטוּ הֵם׃ וַיְשַׁלַּח מֹשֶׁה
31   אֶת־חֹתְנוֹ וַיֵּלֶךְ לוֹ אֶל־אַרְצוֹ׃ [פ]
32    בַּחֹדֶשׁ הַשְּׁלִישִׁי לְצֵאת בְּנֵי־יִשְׂרָאֵל מֵאֶרֶץ מִצְרָיִם
33   בַּיּוֹם הַזֶּה בָּאוּ מִדְבַּר סִינָי׃ וַיִּסְעוּ מֵרְפִידִים וַיָּבֹאוּ
34   מִדְבַּר סִינַי וַיַּחֲנוּ בַּמִּדְבָּר וַיִּחַן־שָׁם יִשְׂרָאֵל נֶגֶד
35   הָהָר׃ וּמֹשֶׁה עָלָה אֶל־הָאֱלֹהִים וַיִּקְרָא אֵלָיו יְהוָה
36   מִן־הָהָר לֵאמֹר כֹּה תֹאמַר לְבֵית יַעֲקֹב וְתַגֵּיד לִבְנֵי
37   יִשְׂרָאֵל׃ אַתֶּם רְאִיתֶם אֲשֶׁר עָשִׂיתִי לְמִצְרָיִם וָאֶשָּׂא
38   אֶתְכֶם עַל־כַּנְפֵי נְשָׁרִים וָאָבִא אֶתְכֶם אֵלָי׃ וְעַתָּה
39   אִם־שָׁמוֹעַ תִּשְׁמְעוּ בְּקֹלִי וּשְׁמַרְתֶּם אֶת־בְּרִיתִי
40   וִהְיִיתֶם לִי סְגֻלָּה מִכָּל־הָעַמִּים כִּי־לִי כָּל־הָאָרֶץ׃
41   וְאַתֶּם תִּהְיוּ־לִי מַמְלֶכֶת כֹּהֲנִים וְגוֹי קָדוֹשׁ אֵלֶּה
42   הַדְּבָרִים אֲשֶׁר תְּדַבֵּר אֶל־בְּנֵי יִשְׂרָאֵל׃  וַיָּבֹא מֹשֶׁה
