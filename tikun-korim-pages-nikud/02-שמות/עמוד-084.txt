 1  וּלְשֹׁמְרֵי מִצְו‍ֹתָי׃ [ס]             לֹא תִשָּׂא אֶת־
 2  שֵׁם־יְהוָה אֱלֹהֶיךָ לַשָּׁוְא כִּי לֹא יְנַקֶּה יְהוָה אֵת
 3   אֲשֶׁר־יִשָּׂא אֶת־שְׁמוֹ לַשָּׁוְא׃ [פ]
 4   זָכוֹר אֶת־יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ שֵׁשֶׁת יָמִים תַּעֲבֹד
 5   וְעָשִׂיתָ כָל־מְלַאכְתֶּךָ וְיוֹם הַשְּׁבִיעִי שַׁבָּת  לַיהֹוָה
 6   אֱלֹהֶיךָ לֹא תַעֲשֶׂה כָל־מְלָאכָה אַתָּה וּבִנְךָ וּבִתֶּךָ
 7   עַבְדְּךָ וַאֲמָתְךָ וּבְהֶמְתֶּךָ וְגֵרְךָ אֲשֶׁר בִּשְׁעָרֶיךָ
 8  כִּי שֵׁשֶׁת־יָמִים עָשָׂה יְהֹוָה אֶת־הַשָּׁמַיִם וְאֶת־
 9  הָאָרֶץ אֶת־הַיָּם וְאֶת־כָּל־אֲשֶׁר־בָּם וַיָּנַח בַּיּוֹם
10   הַשְּׁבִיעִי עַל־כֵּן בֵּרַךְ יְהוָה אֶת־יוֹם הַשַּׁבָּת
11   וַיְקַדְּשֵׁהוּ׃ [ס]             כַּבֵּד אֶת־אָבִיךָ וְאֶת־אִמֶּךָ
12   לְמַעַן יַאֲרִכוּן יָמֶיךָ עַל הָאֲדָמָה אֲשֶׁר־יְהוָה אֱלֹהֶיךָ
13   נֹתֵן לָךְ׃ [ס]             לֹא תִּרְצָח: [ס]             לֹא
14  תִּנְאָף: [ס]            לֹא תִּגְנֹב:[ס]             לֹא־
15  תַעֲנֶה בְרֵעֲךָ עֵד שָׁקֶר׃ [ס]             לֹא
16   תַחְמֹד בֵּית רֵעֶךָ [ס]             לֹא־
17  תַחְמֹד אֵשֶׁת רֵעֶךָ וְעַבְדּוֹ וַאֲמָתוֹ וְשׁוֹרוֹ וַחֲמֹרוֹ
18   וְכֹל אֲשֶׁר לְרֵעֶךָ׃ [פ]
19    וְכָל־הָעָם רֹאִים אֶת־הַקּוֹלֹת וְאֶת־הַלַּפִּידִם וְאֵת קוֹל
20   הַשֹּׁפָר וְאֶת־הָהָר עָשֵׁן וַיַּרְא הָעָם וַיָּנֻעוּ וַיַּעַמְדוּ
21   מֵרָחֹק׃ וַיֹּאמְרוּ אֶל־מֹשֶׁה דַּבֵּר־אַתָּה עִמָּנוּ וְנִשְׁמָעָה
22   וְאַל־יְדַבֵּר עִמָּנוּ אֱלֹהִים פֶּן־נָמוּת׃ וַיֹּאמֶר מֹשֶׁה
23   אֶל־הָעָם אַל־תִּירָאוּ כִּי לְבַעֲבוּר נַסּוֹת אֶתְכֶם בָּא
24   הָאֱלֹהִים וּבַעֲבוּר תִּהְיֶה יִרְאָתוֹ עַל־פְּנֵיכֶם לְבִלְתִּי
25   תֶחֱטָאוּ׃ וַיַּעֲמֹד הָעָם מֵרָחֹק וּמֹשֶׁה נִגַּשׁ אֶל־הָעֲרָפֶל
26   אֲשֶׁר־שָׁם הָאֱלֹהִים׃ [ס]              וַיֹּאמֶר יְהוָה
27   אֶל־מֹשֶׁה כֹּה תֹאמַר אֶל־בְּנֵי יִשְׂרָאֵל אַתֶּם רְאִיתֶם
28   כִּי מִן־הַשָּׁמַיִם דִּבַּרְתִּי עִמָּכֶם׃ לֹא תַעֲשׂוּן אִתִּי אֱלֹהֵי
29   כֶסֶף וֵאלֹהֵי זָהָב לֹא תַעֲשׂוּ לָכֶם׃ מִזְבַּח אֲדָמָה
30   תַּעֲשֶׂה־לִּי וְזָבַחְתָּ עָלָיו אֶת־עֹלֹתֶיךָ וְאֶת־שְׁלָמֶיךָ
31   אֶת־צֹאנְךָ וְאֶת־בְּקָרֶךָ בְּכָל־הַמָּקוֹם אֲשֶׁר אַזְכִּיר
32   אֶת־שְׁמִי אָבוֹא אֵלֶיךָ וּבֵרַכְתִּיךָ׃ וְאִם־מִזְבַּח אֲבָנִים
33   תַּעֲשֶׂה־לִּי לֹא־תִבְנֶה אֶתְהֶן גָּזִית כִּי חַרְבְּךָ הֵנַפְתָּ
34   עָלֶיהָ וַתְּחַלְלֶהָ׃ וְלֹא־תַעֲלֶה בְמַעֲלֹת עַל־מִזְבְּחִי
35   אֲשֶׁר לֹא־תִגָּלֶה עֶרְוָתְךָ עָלָיו׃ [פ]
36  וְאֵלֶּה הַמִּשְׁפָּטִים אֲשֶׁר תָּשִׂים לִפְנֵיהֶם׃ כִּי תִקְנֶה
37   עֶבֶד עִבְרִי שֵׁשׁ שָׁנִים יַעֲבֹד וּבַשְּׁבִעִת יֵצֵא לַחָפְשִׁי
38   חִנָּם׃ אִם־בְּגַפּוֹ יָבֹא בְּגַפּוֹ יֵצֵא אִם־בַּעַל אִשָּׁה הוּא
39   וְיָצְאָה אִשְׁתּוֹ עִמּוֹ׃ אִם־אֲדֹנָיו יִתֶּן־לוֹ אִשָּׁה וְיָלְדָה־
40  לוֹ בָנִים אוֹ בָנוֹת הָאִשָּׁה וִילָדֶיהָ תִּהְיֶה לַאדֹנֶיהָ
41   וְהוּא יֵצֵא בְגַפּוֹ׃ וְאִם־אָמֹר יֹאמַר הָעֶבֶד אָהַבְתִּי
42   אֶת־אֲדֹנִי אֶת־אִשְׁתִּי וְאֶת־בָּנָי לֹא אֵצֵא חָפְשִׁי׃
