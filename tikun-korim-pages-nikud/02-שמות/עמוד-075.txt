 1  וַתֶּחֱזַק מִצְרַיִם עַל־הָעָם לְמַהֵר לְשַׁלְּחָם מִן־הָאָרֶץ כִּי
 2   אָמְרוּ כֻּלָּנוּ מֵתִים׃ וַיִּשָּׂא הָעָם אֶת־בְּצֵקוֹ טֶרֶם יֶחְמָץ
 3   מִשְׁאֲרֹתָם צְרֻרֹת בְּשִׂמְלֹתָם עַל־שִׁכְמָם׃ וּבְנֵי־יִשְׂרָאֵל
 4   עָשׂוּ כִּדְבַר מֹשֶׁה וַיִּשְׁאֲלוּ מִמִּצְרַיִם כְּלֵי־כֶסֶף וּכְלֵי
 5   זָהָב וּשְׂמָלֹת׃ וַיהוָה נָתַן אֶת־חֵן הָעָם בְּעֵינֵי מִצְרַיִם
 6   וַיַּשְׁאִלוּם וַיְנַצְּלוּ אֶת־מִצְרָיִם׃ [פ]
 7   וַיִּסְעוּ בְנֵי־יִשְׂרָאֵל מֵרַעְמְסֵס סֻכֹּתָה כְּשֵׁשׁ־מֵאוֹת
 8   אֶלֶף רַגְלִי הַגְּבָרִים לְבַד מִטָּף׃ וְגַם־עֵרֶב רַב עָלָה
 9   אִתָּם וְצֹאן וּבָקָר מִקְנֶה כָּבֵד מְאֹד׃ וַיֹּאפוּ אֶת־
10  הַבָּצֵק אֲשֶׁר הוֹצִיאוּ מִמִּצְרַיִם עֻגֹת מַצּוֹת כִּי לֹא
11   חָמֵץ כִּי־גֹרְשׁוּ מִמִּצְרַיִם וְלֹא יָכְלוּ לְהִתְמַהְמֵהַּ וְגַם־
12  צֵדָה לֹא־עָשׂוּ לָהֶם׃ וּמוֹשַׁב בְּנֵי יִשְׂרָאֵל אֲשֶׁר
13   יָשְׁבוּ בְּמִצְרָיִם שְׁלֹשִׁים שָׁנָה וְאַרְבַּע מֵאוֹת שָׁנָה׃ וַיְהִי
14   מִקֵּץ שְׁלֹשִׁים שָׁנָה וְאַרְבַּע מֵאוֹת שָׁנָה וַיְהִי
15   בְּעֶצֶם הַיּוֹם הַזֶּה יָצְאוּ כָּל־צִבְאוֹת יְהוָה מֵאֶרֶץ
16   מִצְרָיִם׃ לֵיל שִׁמֻּרִים הוּא לַיהוָה לְהוֹצִיאָם מֵאֶרֶץ
17   מִצְרָיִם הוּא־הַלַּיְלָה הַזֶּה לַיהוָה שִׁמֻּרִים לְכָל־בְּנֵי
18   יִשְׂרָאֵל לְדֹרֹתָם׃ [פ]
19   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה וְאַהֲרֹן זֹאת חֻקַּת הַפָּסַח
20   כָּל־בֶּן־נֵכָר לֹא־יֹאכַל בּוֹ׃ וְכָל־עֶבֶד אִישׁ מִקְנַת־
21  כָּסֶף וּמַלְתָּה אֹתוֹ אָז יֹאכַל בּוֹ׃ תּוֹשָׁב וְשָׂכִיר לֹא־
22  יֹאכַל בּוֹ׃ בְּבַיִת אֶחָד יֵאָכֵל לֹא־תוֹצִיא מִן־הַבַּיִת
23   מִן־הַבָּשָׂר חוּצָה וְעֶצֶם לֹא תִשְׁבְּרוּ־בוֹ׃ כָּל־עֲדַת
24   יִשְׂרָאֵל יַעֲשׂוּ אֹתוֹ׃ וְכִי־יָגוּר אִתְּךָ גֵּר וְעָשָׂה פֶסַח
25   לַיהוָה הִמּוֹל לוֹ כָל־זָכָר וְאָז יִקְרַב לַעֲשֹׂתוֹ וְהָיָה
26   כְּאֶזְרַח הָאָרֶץ וְכָל־עָרֵל לֹא־יֹאכַל בּוֹ׃ תּוֹרָה אַחַת
27   יִהְיֶה לָאֶזְרָח וְלַגֵּר הַגָּר בְּתוֹכְכֶם׃ וַיַּעֲשׂוּ כָּל־בְּנֵי
28   יִשְׂרָאֵל כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה וְאֶת־אַהֲרֹן
29   כֵּן עָשׂוּ׃ [ס]             וַיְהִי בְּעֶצֶם הַיּוֹם
30   הַזֶּה הוֹצִיא יְהוָה אֶת־בְּנֵי יִשְׂרָאֵל מֵאֶרֶץ מִצְרַיִם
31   עַל־צִבְאֹתָם׃ [פ]
32    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ קַדֶּשׁ־לִי כָל־בְּכוֹר
33   פֶּטֶר כָּל־רֶחֶם בִּבְנֵי יִשְׂרָאֵל בָּאָדָם וּבַבְּהֵמָה לִי הוּא׃
34   וַיֹּאמֶר מֹשֶׁה אֶל־הָעָם זָכוֹר אֶת־הַיּוֹם הַזֶּה אֲשֶׁר
35   יְצָאתֶם מִמִּצְרַיִם מִבֵּית עֲבָדִים כִּי בְּחֹזֶק יָד הוֹצִיא
36   יְהוָה אֶתְכֶם מִזֶּה וְלֹא יֵאָכֵל חָמֵץ׃ הַיּוֹם אַתֶּם יֹצְאִים
37   בְּחֹדֶשׁ הָאָבִיב׃ וְהָיָה כִי־יְבִיאֲךָ יְהוָה אֶל־אֶרֶץ הַכְּנַעֲנִי
38   וְהַחִתִּי וְהָאֱמֹרִי וְהַחִוִּי וְהַיְבוּסִי אֲשֶׁר נִשְׁבַּע לַאֲבֹתֶיךָ
39   לָתֶת לָךְ אֶרֶץ זָבַת חָלָב וּדְבָשׁ וְעָבַדְתָּ אֶת־הָעֲבֹדָה
40   הַזֹּאת בַּחֹדֶשׁ הַזֶּה׃ שִׁבְעַת יָמִים תֹּאכַל מַצֹּת וּבַיּוֹם
41   הַשְּׁבִיעִי חַג לַיהוָה׃ מַצּוֹת יֵאָכֵל אֵת שִׁבְעַת הַיָּמִים
42   וְלֹא־יֵרָאֶה לְךָ חָמֵץ וְלֹא־יֵרָאֶה לְךָ שְׂאֹר בְּכָל־גְּבֻלֶךָ׃
