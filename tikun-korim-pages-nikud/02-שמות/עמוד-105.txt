 1  וַחֲמִשִּׁים לֻלָאֹת עָשָׂה בִּקְצֵה הַיְרִיעָה אֲשֶׁר
 2   בַּמַּחְבֶּרֶת הַשֵּׁנִית מַקְבִּילֹת הַלֻּלָאֹת אַחַת אֶל־אֶחָת׃
 3   וַיַּעַשׂ חֲמִשִּׁים קַרְסֵי זָהָב וַיְחַבֵּר אֶת־הַיְרִיעֹת אַחַת אֶל־
 4  אַחַת בַּקְּרָסִים וַיְהִי הַמִּשְׁכָּן אֶחָד׃ [פ]
 5   וַיַּעַשׂ יְרִיעֹת עִזִּים לְאֹהֶל עַל־הַמִּשְׁכָּן עַשְׁתֵּי־עֶשְׂרֵה
 6   יְרִיעֹת עָשָׂה אֹתָם׃ אֹרֶךְ הַיְרִיעָה הָאַחַת שְׁלֹשִׁים
 7   בָּאַמָּה וְאַרְבַּע אַמּוֹת רֹחַב הַיְרִיעָה הָאֶחָת מִדָּה אַחַת
 8   לְעַשְׁתֵּי עֶשְׂרֵה יְרִיעֹת׃ וַיְחַבֵּר אֶת־חֲמֵשׁ הַיְרִיעֹת
 9   לְבָד וְאֶת־שֵׁשׁ הַיְרִיעֹת לְבָד׃ וַיַּעַשׂ לֻלָאֹת חֲמִשִּׁים
10   עַל שְׂפַת הַיְרִיעָה הַקִּיצֹנָה בַּמַּחְבָּרֶת וַחֲמִשִּׁים לֻלָאֹת
11   עָשָׂה עַל־שְׂפַת הַיְרִיעָה הַחֹבֶרֶת הַשֵּׁנִית׃ וַיַּעַשׂ
12   קַרְסֵי נְחֹשֶׁת חֲמִשִּׁים לְחַבֵּר אֶת־הָאֹהֶל לִהְיֹת אֶחָד׃
13   וַיַּעַשׂ מִכְסֶה לָאֹהֶל עֹרֹת אֵילִם מְאָדָּמִים וּמִכְסֵה
14   עֹרֹת תְּחָשִׁים מִלְמָעְלָה׃ [ס]              וַיַּעַשׂ
15   אֶת־הַקְּרָשִׁים לַמִּשְׁכָּן עֲצֵי שִׁטִּים עֹמְדִים׃ עֶשֶׂר
16   אַמֹּת אֹרֶךְ הַקָּרֶשׁ וְאַמָּה וַחֲצִי הָאַמָּה רֹחַב הַקֶּרֶשׁ
17   הָאֶחָד׃ שְׁתֵּי יָדֹת לַקֶּרֶשׁ הָאֶחָד מְשֻׁלָּבֹת אַחַת
18   אֶל־אֶחָת כֵּן עָשָׂה לְכֹל קַרְשֵׁי הַמִּשְׁכָּן׃ וַיַּעַשׂ אֶת־
19  הַקְּרָשִׁים לַמִּשְׁכָּן עֶשְׂרִים קְרָשִׁים לִפְאַת נֶגֶב
20   תֵּימָנָה׃ וְאַרְבָּעִים אַדְנֵי־כֶסֶף עָשָׂה תַּחַת עֶשְׂרִים
21   הַקְּרָשִׁים שְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד לִשְׁתֵּי
22   יְדֹתָיו וּשְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד לִשְׁתֵּי יְדֹתָיו׃
23   וּלְצֶלַע הַמִּשְׁכָּן הַשֵּׁנִית לִפְאַת צָפוֹן עָשָׂה עֶשְׂרִים
24   קְרָשִׁים׃ וְאַרְבָּעִים אַדְנֵיהֶם כָּסֶף שְׁנֵי אֲדָנִים תַּחַת
25   הַקֶּרֶשׁ הָאֶחָד וּשְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד׃
26   וּלְיַרְכְּתֵי הַמִּשְׁכָּן יָמָּה עָשָׂה שִׁשָּׁה קְרָשִׁים׃ וּשְׁנֵי
27   קְרָשִׁים עָשָׂה לִמְקֻצְעֹת הַמִּשְׁכָּן בַּיַּרְכָתָיִם׃ וְהָיוּ
28   תוֹאֲמִם מִלְּמַטָּה וְיַחְדָּו יִהְיוּ תַמִּים אֶל־רֹאשׁוֹ אֶל־
29  הַטַּבַּעַת הָאֶחָת כֵּן עָשָׂה לִשְׁנֵיהֶם לִשְׁנֵי הַמִּקְצֹעֹת׃
30   וְהָיוּ שְׁמֹנָה קְרָשִׁים וְאַדְנֵיהֶם כֶּסֶף שִׁשָּׁה עָשָׂר
31   אֲדָנִים שְׁנֵי אֲדָנִים שְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד׃
32   וַיַּעַשׂ בְּרִיחֵי עֲצֵי שִׁטִּים חֲמִשָּׁה לְקַרְשֵׁי צֶלַע־
33  הַמִּשְׁכָּן הָאֶחָת׃ וַחֲמִשָּׁה בְרִיחִם לְקַרְשֵׁי צֶלַע־
34  הַמִּשְׁכָּן הַשֵּׁנִית וַחֲמִשָּׁה בְרִיחִם לְקַרְשֵׁי הַמִּשְׁכָּן
35   לַיַּרְכָתַיִם יָמָּה׃ וַיַּעַשׂ אֶת־הַבְּרִיחַ הַתִּיכֹן לִבְרֹחַ
36   בְּתוֹךְ הַקְּרָשִׁים מִן־הַקָּצֶה אֶל־הַקָּצֶה׃ וְאֶת־הַקְּרָשִׁים
37   צִפָּה זָהָב וְאֶת־טַבְּעֹתָם עָשָׂה זָהָב בָּתִּים לַבְּרִיחִם
38   וַיְצַף אֶת־הַבְּרִיחִם זָהָב׃ וַיַּעַשׂ אֶת־הַפָּרֹכֶת תְּכֵלֶת
39   וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר מַעֲשֵׂה חֹשֵׁב
40   עָשָׂה אֹתָהּ כְּרֻבִים׃ וַיַּעַשׂ לָהּ אַרְבָּעָה עַמּוּדֵי שִׁטִּים
41   וַיְצַפֵּם זָהָב וָוֵיהֶם זָהָב וַיִּצֹק לָהֶם אַרְבָּעָה אַדְנֵי־
42  כָסֶף׃ וַיַּעַשׂ מָסָךְ לְפֶתַח הָאֹהֶל תְּכֵלֶת וְאַרְגָּמָן
