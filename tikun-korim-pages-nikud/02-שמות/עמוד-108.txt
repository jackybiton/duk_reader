 1  וְאַדְנֵיהֶם אַרְבָּעָה נְחֹשֶׁת וָוֵיהֶם כֶּסֶף וְצִפּוּי רָאשֵׁיהֶם
 2   וַחֲשֻׁקֵיהֶם כָּסֶף׃ וְכָל־הַיְתֵדֹת לַמִּשְׁכָּן וְלֶחָצֵר סָבִיב
 3   נְחֹשֶׁת׃ [ס]            אֵלֶּה פְקוּדֵי
 4   הַמִּשְׁכָּן מִשְׁכַּן הָעֵדֻת אֲשֶׁר פֻּקַּד עַל־פִּי מֹשֶׁה
 5   עֲבֹדַת הַלְוִיִּם בְּיַד אִיתָמָר בֶּן־אַהֲרֹן הַכֹּהֵן׃ וּבְצַלְאֵל
 6   בֶּן־אוּרִי בֶן־חוּר לְמַטֵּה יְהוּדָה עָשָׂה אֵת כָּל־אֲשֶׁר־
 7  צִוָּה יְהוָה אֶת־מֹשֶׁה׃ וְאִתּוֹ אָהֳלִיאָב בֶּן־אֲחִיסָמָךְ
 8   לְמַטֵּה־דָן חָרָשׁ וְחֹשֵׁב וְרֹקֵם בַּתְּכֵלֶת וּבָאַרְגָּמָן
 9   וּבְתוֹלַעַת הַשָּׁנִי וּבַשֵּׁשׁ׃♢ [ס]             כָּל־
10  הַזָּהָב הֶעָשׂוּי לַמְּלָאכָה בְּכֹל מְלֶאכֶת הַקֹּדֶשׁ וַיְהִי
11   זְהַב הַתְּנוּפָה תֵּשַׁע וְעֶשְׂרִים כִּכָּר וּשְׁבַע מֵאוֹת
12   וּשְׁלֹשִׁים שֶׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ׃ וְכֶסֶף פְּקוּדֵי
13   הָעֵדָה מְאַת כִּכָּר וְאֶלֶף וּשְׁבַע מֵאוֹת וַחֲמִשָּׁה
14   וְשִׁבְעִים שֶׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ׃ בֶּקַע לַגֻּלְגֹּלֶת
15   מַחֲצִית הַשֶּׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ לְכֹל הָעֹבֵר עַל־
16  הַפְּקֻדִים מִבֶּן עֶשְׂרִים שָׁנָה וָמַעְלָה לְשֵׁשׁ־מֵאוֹת
17   אֶלֶף וּשְׁלֹשֶׁת אֲלָפִים וַחֲמֵשׁ מֵאוֹת וַחֲמִשִּׁים׃ וַיְהִי
18   מְאַת כִּכַּר הַכֶּסֶף לָצֶקֶת אֵת אַדְנֵי הַקֹּדֶשׁ וְאֵת
19   אַדְנֵי הַפָּרֹכֶת מְאַת אֲדָנִים לִמְאַת הַכִּכָּר כִּכָּר
20   לָאָדֶן׃♢ וְאֶת־הָאֶלֶף וּשְׁבַע הַמֵּאוֹת וַחֲמִשָּׁה וְשִׁבְעִים
21   עָשָׂה וָוִים לָעַמּוּדִים וְצִפָּה רָאשֵׁיהֶם וְחִשַּׁק אֹתָם׃
22   וּנְחֹשֶׁת הַתְּנוּפָה שִׁבְעִים כִּכָּר וְאַלְפַּיִם וְאַרְבַּע־
23  מֵאוֹת שָׁקֶל׃ וַיַּעַשׂ בָּהּ אֶת־אַדְנֵי פֶּתַח אֹהֶל מוֹעֵד
24   וְאֵת מִזְבַּח הַנְּחֹשֶׁת וְאֶת־מִכְבַּר הַנְּחֹשֶׁת אֲשֶׁר־לוֹ
25   וְאֵת כָּל־כְּלֵי הַמִּזְבֵּחַ׃ וְאֶת־אַדְנֵי הֶחָצֵר סָבִיב וְאֶת־
26  אַדְנֵי שַׁעַר הֶחָצֵר וְאֵת כָּל־יִתְדֹת הַמִּשְׁכָּן וְאֶת־
27  כָּל־יִתְדֹת הֶחָצֵר סָבִיב׃ וּמִן־הַתְּכֵלֶת וְהָאַרְגָּמָן
28   וְתוֹלַעַת הַשָּׁנִי עָשׂוּ בִגְדֵי־שְׂרָד לְשָׁרֵת בַּקֹּדֶשׁ
29   וַיַּעֲשׂוּ אֶת־בִּגְדֵי הַקֹּדֶשׁ אֲשֶׁר לְאַהֲרֹן כַּאֲשֶׁר
30   צִוָּה יְהוָה אֶת־מֹשֶׁה׃♢ [פ]
31    וַיַּעַשׂ אֶת־הָאֵפֹד זָהָב תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי
32   וְשֵׁשׁ מָשְׁזָר׃ וַיְרַקְּעוּ אֶת־פַּחֵי הַזָּהָב וְקִצֵּץ פְּתִילִם
33   לַעֲשׂוֹת בְּתוֹךְ הַתְּכֵלֶת וּבְתוֹךְ הָאַרְגָּמָן וּבְתוֹךְ
34   תּוֹלַעַת הַשָּׁנִי וּבְתוֹךְ הַשֵּׁשׁ מַעֲשֵׂה חֹשֵׁב׃ כְּתֵפֹת
35   עָשׂוּ־לוֹ חֹבְרֹת עַל־שְׁנֵי קְצוֹתָיו חֻבָּר׃ וְחֵשֶׁב
36   אֲפֻדָּתוֹ אֲשֶׁר עָלָיו מִמֶּנּוּ הוּא כְּמַעֲשֵׂהוּ זָהָב תְּכֵלֶת
37   וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר כַּאֲשֶׁר צִוָּה
38   יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיַּעֲשׂוּ
39   אֶת־אַבְנֵי הַשֹּׁהַם מֻסַבֹּת מִשְׁבְּצֹת זָהָב מְפֻתָּחֹת
40   פִּתּוּחֵי חוֹתָם עַל־שְׁמוֹת בְּנֵי יִשְׂרָאֵל׃ וַיָּשֶׂם אֹתָם
41   עַל כִּתְפֹת הָאֵפֹד אַבְנֵי זִכָּרוֹן לִבְנֵי יִשְׂרָאֵל כַּאֲשֶׁר
42   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [פ]
