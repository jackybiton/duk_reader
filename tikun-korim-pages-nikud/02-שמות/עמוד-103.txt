 1  וְהֵשִׁיב מֹשֶׁה אֶת־הַמַּסְוֶה עַל־פָּנָיו עַד־בֹּאוֹ לְדַבֵּר
 2   אִתּוֹ׃ [ס]            וַיַּקְהֵל מֹשֶׁה אֶת־כָּל־עֲדַת
 3   בְּנֵי יִשְׂרָאֵל וַיֹּאמֶר אֲלֵהֶם אֵלֶּה הַדְּבָרִים אֲשֶׁר־צִוָּה
 4   יְהוָה לַעֲשֹׂת אֹתָם׃ שֵׁשֶׁת יָמִים תֵּעָשֶׂה מְלָאכָה
 5   וּבַיּוֹם הַשְּׁבִיעִי יִהְיֶה לָכֶם קֹדֶשׁ שַׁבַּת שַׁבָּתוֹן
 6   לַיהוָה כָּל־הָעֹשֶׂה בוֹ מְלָאכָה יוּמָת׃ לֹא־תְבַעֲרוּ אֵשׁ
 7   בְּכֹל מֹשְׁבֹתֵיכֶם בְּיוֹם הַשַּׁבָּת׃♢ [פ]
 8   וַיֹּאמֶר מֹשֶׁה אֶל־כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל לֵאמֹר זֶה
 9   הַדָּבָר אֲשֶׁר־צִוָּה יְהוָה לֵאמֹר׃ קְחוּ מֵאִתְּכֶם תְּרוּמָה
10   לַיהוָה כֹּל נְדִיב לִבּוֹ יְבִיאֶהָ אֵת תְּרוּמַת יְהוָה זָהָב
11   וָכֶסֶף וּנְחֹשֶׁת׃ וּתְכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ
12   וְעִזִּים׃ וְעֹרֹת אֵילִם מְאָדָּמִים וְעֹרֹת תְּחָשִׁים וַעֲצֵי
13   שִׁטִּים׃ וְשֶׁמֶן לַמָּאוֹר וּבְשָׂמִים לְשֶׁמֶן הַמִּשְׁחָה
14   וְלִקְטֹרֶת הַסַּמִּים׃ וְאַבְנֵי־שֹׁהַם וְאַבְנֵי מִלֻּאִים לָאֵפוֹד
15   וְלַחֹשֶׁן׃ וְכָל־חֲכַם־לֵב בָּכֶם יָבֹאוּ וְיַעֲשׂוּ אֵת כָּל־אֲשֶׁר
16   צִוָּה יְהוָה׃♢ אֶת־הַמִּשְׁכָּן אֶת־אָהֳלוֹ וְאֶת־מִכְסֵהוּ אֶת־
17  קְרָסָיו וְאֶת־קְרָשָׁיו אֶת־בְּרִיחָו אֶת־עַמֻּדָיו וְאֶת־אֲדָנָיו׃
18   אֶת־הָאָרֹן וְאֶת־בַּדָּיו אֶת־הַכַּפֹּרֶת וְאֵת פָּרֹכֶת הַמָּסָךְ׃
19   אֶת־הַשֻּׁלְחָן וְאֶת־בַּדָּיו וְאֶת־כָּל־כֵּלָיו וְאֵת לֶחֶם
20   הַפָּנִים׃ וְאֶת־מְנֹרַת הַמָּאוֹר וְאֶת־כֵּלֶיהָ וְאֶת־נֵרֹתֶיהָ וְאֵת
21   שֶׁמֶן הַמָּאוֹר׃ וְאֶת־מִזְבַּח הַקְּטֹרֶת וְאֶת־בַּדָּיו וְאֵת שֶׁמֶן
22   הַמִּשְׁחָה וְאֵת קְטֹרֶת הַסַּמִּים וְאֶת־מָסַךְ הַפֶּתַח לְפֶתַח
23   הַמִּשְׁכָּן׃ אֵת  מִזְבַּח הָעֹלָה וְאֶת־מִכְבַּר הַנְּחֹשֶׁת אֲשֶׁר־
24  לוֹ אֶת־בַּדָּיו וְאֶת־כָּל־כֵּלָיו אֶת־הַכִּיֹּר וְאֶת־כַּנּוֹ׃ אֵת
25   קַלְעֵי הֶחָצֵר אֶת־עַמֻּדָיו וְאֶת־אֲדָנֶיהָ וְאֵת מָסַךְ שַׁעַר
26   הֶחָצֵר׃ אֶת־יִתְדֹת הַמִּשְׁכָּן וְאֶת־יִתְדֹת הֶחָצֵר וְאֶת־
27  מֵיתְרֵיהֶם׃ אֶת־בִּגְדֵי הַשְּׂרָד לְשָׁרֵת בַּקֹּדֶשׁ אֶת־בִּגְדֵי
28   הַקֹּדֶשׁ לְאַהֲרֹן הַכֹּהֵן וְאֶת־בִּגְדֵי בָנָיו לְכַהֵן׃ וַיֵּצְאוּ
29   כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל מִלִּפְנֵי מֹשֶׁה׃♢  וַיָּבֹאוּ כָּל־אִישׁ
30   אֲשֶׁר־נְשָׂאוֹ לִבּוֹ וְכֹל אֲשֶׁר נָדְבָה רוּחוֹ אֹתוֹ הֵבִיאוּ
31   אֶת־תְּרוּמַת יְהוָה לִמְלֶאכֶת אֹהֶל מוֹעֵד וּלְכָל־עֲבֹדָתוֹ
32   וּלְבִגְדֵי הַקֹּדֶשׁ׃ וַיָּבֹאוּ הָאֲנָשִׁים עַל־הַנָּשִׁים כֹּל
33   נְדִיב לֵב הֵבִיאוּ חָח וָנֶזֶם וְטַבַּעַת וְכוּמָז כָּל־כְּלִי
34   זָהָב וְכָל־אִישׁ אֲשֶׁר הֵנִיף תְּנוּפַת זָהָב לַיהוָה׃ וְכָל־
35  אִישׁ אֲשֶׁר־נִמְצָא אִתּוֹ תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי
36   וְשֵׁשׁ וְעִזִּים וְעֹרֹת אֵילִם מְאָדָּמִים וְעֹרֹת תְּחָשִׁים
37   הֵבִיאוּ׃ כָּל־מֵרִים תְּרוּמַת כֶּסֶף וּנְחֹשֶׁת הֵבִיאוּ אֵת
38   תְּרוּמַת יְהוָה וְכֹל אֲשֶׁר נִמְצָא אִתּוֹ עֲצֵי שִׁטִּים
39   לְכָל־מְלֶאכֶת הָעֲבֹדָה הֵבִיאוּ׃ וְכָל־אִשָּׁה חַכְמַת־
40  לֵב בְּיָדֶיהָ טָווּ וַיָּבִיאוּ מַטְוֶה אֶת־הַתְּכֵלֶת וְאֶת־
41  הָאַרְגָּמָן אֶת־תּוֹלַעַת הַשָּׁנִי וְאֶת־הַשֵּׁשׁ׃ וְכָל־הַנָּשִׁים
42   אֲשֶׁר נָשָׂא לִבָּן אֹתָנָה בְּחָכְמָה טָווּ אֶת־הָעִזִּים׃
