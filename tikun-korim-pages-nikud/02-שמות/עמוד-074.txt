 1  וְלֹא־יִהְיֶה בָכֶם נֶגֶף לְמַשְׁחִית בְּהַכֹּתִי בְּאֶרֶץ מִצְרָיִם׃
 2   וְהָיָה הַיּוֹם הַזֶּה לָכֶם לְזִכָּרוֹן וְחַגֹּתֶם אֹתוֹ חַג לַיהוָה
 3   לְדֹרֹתֵיכֶם חֻקַּת עוֹלָם תְּחָגֻּהוּ׃ שִׁבְעַת יָמִים מַצּוֹת
 4   תֹּאכֵלוּ אַךְ בַּיּוֹם הָרִאשׁוֹן תַּשְׁבִּיתוּ שְּׂאֹר מִבָּתֵּיכֶם
 5   כִּי  כָּל־אֹכֵל חָמֵץ וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מִיִּשְׂרָאֵל
 6   מִיּוֹם הָרִאשֹׁן עַד־יוֹם הַשְּׁבִעִי׃ וּבַיּוֹם הָרִאשׁוֹן מִקְרָא־
 7  קֹדֶשׁ וּבַיּוֹם הַשְּׁבִיעִי מִקְרָא־קֹדֶשׁ יִהְיֶה לָכֶם כָּל־
 8  מְלָאכָה לֹא־יֵעָשֶׂה בָהֶם אַךְ אֲשֶׁר יֵאָכֵל לְכָל־נֶפֶשׁ
 9   הוּא לְבַדּוֹ יֵעָשֶׂה לָכֶם׃ וּשְׁמַרְתֶּם אֶת־הַמַּצּוֹת כִּי
10   בְּעֶצֶם הַיּוֹם הַזֶּה הוֹצֵאתִי אֶת־צִבְאוֹתֵיכֶם מֵאֶרֶץ
11   מִצְרָיִם וּשְׁמַרְתֶּם אֶת־הַיּוֹם הַזֶּה לְדֹרֹתֵיכֶם חֻקַּת
12   עוֹלָם׃ בָּרִאשֹׁן בְּאַרְבָּעָה עָשָׂר יוֹם לַחֹדֶשׁ בָּעֶרֶב
13   תֹּאכְלוּ מַצֹּת עַד יוֹם הָאֶחָד וְעֶשְׂרִים לַחֹדֶשׁ בָּעָרֶב׃
14   שִׁבְעַת יָמִים שְׂאֹר לֹא יִמָּצֵא בְּבָתֵּיכֶם כִּי  כָּל־אֹכֵל
15   מַחְמֶצֶת וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעֲדַת יִשְׂרָאֵל
16   בַּגֵּר וּבְאֶזְרַח הָאָרֶץ׃ כָּל־מַחְמֶצֶת לֹא תֹאכֵלוּ בְּכֹל
17   מוֹשְׁבֹתֵיכֶם תֹּאכְלוּ מַצּוֹת׃ [פ]
18   וַיִּקְרָא מֹשֶׁה לְכָל־זִקְנֵי יִשְׂרָאֵל וַיֹּאמֶר אֲלֵהֶם
19   מִשְׁכוּ וּקְחוּ לָכֶם צֹאן לְמִשְׁפְּחֹתֵיכֶם וְשַׁחֲטוּ
20   הַפָּסַח׃ וּלְקַחְתֶּם אֲגֻדַּת אֵזוֹב וּטְבַלְתֶּם בַּדָּם אֲשֶׁר־
21  בַּסַּף וְהִגַּעְתֶּם אֶל־הַמַּשְׁקוֹף וְאֶל־שְׁתֵּי הַמְּזוּזֹת מִן־
22  הַדָּם אֲשֶׁר בַּסָּף וְאַתֶּם לֹא תֵצְאוּ אִישׁ מִפֶּתַח־בֵּיתוֹ
23   עַד־בֹּקֶר׃ וְעָבַר יְהוָה לִנְגֹּף אֶת־מִצְרַיִם וְרָאָה אֶת־
24  הַדָּם עַל־הַמַּשְׁקוֹף וְעַל שְׁתֵּי הַמְּזוּזֹת וּפָסַח יְהוָה
25   עַל־הַפֶּתַח וְלֹא יִתֵּן הַמַּשְׁחִית לָבֹא אֶל־בָּתֵּיכֶם
26   לִנְגֹּף׃ וּשְׁמַרְתֶּם אֶת־הַדָּבָר הַזֶּה לְחָק־לְךָ וּלְבָנֶיךָ
27   עַד־עוֹלָם׃ וְהָיָה כִּי־תָבֹאוּ אֶל־הָאָרֶץ אֲשֶׁר יִתֵּן יְהוָה
28   לָכֶם כַּאֲשֶׁר דִּבֵּר וּשְׁמַרְתֶּם אֶת־הָעֲבֹדָה הַזֹּאת׃
29   וְהָיָה כִּי־יֹאמְרוּ אֲלֵיכֶם בְּנֵיכֶם מָה הָעֲבֹדָה הַזֹּאת
30   לָכֶם׃ וַאֲמַרְתֶּם זֶבַח־פֶּסַח הוּא לַיהוָה אֲשֶׁר פָּסַח
31   עַל־בָּתֵּי בְנֵי־יִשְׂרָאֵל בְּמִצְרַיִם בְּנָגְפּוֹ אֶת־מִצְרַיִם
32   וְאֶת־בָּתֵּינוּ הִצִּיל וַיִּקֹּד הָעָם וַיִּשְׁתַּחֲווּ׃ וַיֵּלְכוּ וַיַּעֲשׂוּ
33   בְּנֵי יִשְׂרָאֵל כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה וְאַהֲרֹן כֵּן
34   עָשׂוּ׃ [ס]              וַיְהִי  בַּחֲצִי הַלַּיְלָה וַיהוָה
35   הִכָּה כָל־בְּכוֹר בְּאֶרֶץ מִצְרַיִם מִבְּכֹר פַּרְעֹה הַיֹּשֵׁב
36   עַל־כִּסְאוֹ עַד בְּכוֹר הַשְּׁבִי אֲשֶׁר בְּבֵית הַבּוֹר וְכֹל
37   בְּכוֹר בְּהֵמָה׃ וַיָּקָם פַּרְעֹה לַיְלָה הוּא וְכָל־עֲבָדָיו
38   וְכָל־מִצְרַיִם וַתְּהִי צְעָקָה גְדֹלָה בְּמִצְרָיִם כִּי־אֵין בַּיִת
39   אֲשֶׁר אֵין־שָׁם מֵת׃ וַיִּקְרָא לְמֹשֶׁה וּלְאַהֲרֹן לַיְלָה
40   וַיֹּאמֶר קוּמוּ צְּאוּ מִתּוֹךְ עַמִּי גַּם־אַתֶּם גַּם־בְּנֵי יִשְׂרָאֵל
41   וּלְכוּ עִבְדוּ אֶת־יְהוָה כְּדַבֶּרְכֶם׃ גַּם־צֹאנְכֶם גַּם־
42  בְּקַרְכֶם קְחוּ כַּאֲשֶׁר דִּבַּרְתֶּם וָלֵכוּ וּבֵרַכְתֶּם גַּם־אֹתִי׃
