 1  הַבָּאִים אַחֲרֵיהֶם בַּיָּם לֹא־נִשְׁאַר בָּהֶם עַד־אֶחָד׃ וּבְנֵי יִשְׂרָאֵל הָלְכוּ
 2   בַיַּבָּשָׁה בְּתוֹךְ הַיָּם וְהַמַּיִם לָהֶם חֹמָה מִימִינָם וּמִשְּׂמֹאלָם׃ וַיּוֹשַׁע
 3   יְהוָה בַּיּוֹם הַהוּא אֶת־יִשְׂרָאֵל מִיַּד מִצְרָיִם וַיַּרְא יִשְׂרָאֵל אֶת־מִצְרַיִם
 4   מֵת עַל־שְׂפַת הַיָּם׃ וַיַּרְא יִשְׂרָאֵל אֶת־הַיָּד הַגְּדֹלָה אֲשֶׁר עָשָׂה יְהוָה
 5   בְּמִצְרַיִם וַיִּירְאוּ הָעָם אֶת־יְהוָה וַיַּאֲמִינוּ בַּיהוָה וּבְמֹשֶׁה עַבְדּוֹ׃
 6  
 7   אָז יָשִׁיר־מֹשֶׁה וּבְנֵי יִשְׂרָאֵל אֶת־הַשִּׁירָה הַזֹּאת לַיהוָה וַיֹּאמְרוּ
 8   לֵאמֹר [ס]             אָשִׁירָה לַיהוָה כִּי־גָאֹה גָּאָה [ס]             סוּס
 9   וְרֹכְבוֹ רָמָה בַיָּם׃ [ס]             עָזִּי וְזִמְרָת יָהּ וַיְהִי־לִי
10   לִישׁוּעָה [ס]             זֶה אֵלִי וְאַנְוֵהוּ [ס]             אֱלֹהֵי
11   אָבִי וַאֲרֹמְמֶנְהוּ׃ [ס]             יְהוָה אִישׁ מִלְחָמָה יְהוָה
12   שְׁמוֹ׃ [ס]             מַרְכְּבֹת פַּרְעֹה וְחֵילוֹ יָרָה בַיָּם [ס]             וּמִבְחַר
13   שָׁלִשָׁיו טֻבְּעוּ בְיַם־סוּף׃ [ס]             תְּהֹמֹת יְכַסְיֻמוּ יָרְדוּ בִמְצוֹלֹת כְּמוֹ־
14   אָבֶן׃ [ס]             יְמִינְךָ יְהוָה נֶאְדָּרִי בַּכֹּחַ [ס]             יְמִינְךָ
15   יְהוָה תִּרְעַץ אוֹיֵב׃ [ס]             וּבְרֹב גְּאוֹנְךָ תַּהֲרֹס
16   קָמֶיךָ [ס]             תְּשַׁלַּח חֲרֹנְךָ יֹאכְלֵמוֹ כַּקַּשׁ׃ [ס]             וּבְרוּחַ
17   אַפֶּיךָ נֶעֶרְמוּ מַיִם [ס]             נִצְּבוּ כְמוֹ־נֵד
18   נֹזְלִים [ס]             קָפְאוּ תְהֹמֹת בְּלֶב־יָם׃ [ס]             אָמַר
19   אוֹיֵב אֶרְדֹּף אַשִּׂיג [ס]             אֲחַלֵּק שָׁלָל תִּמְלָאֵמוֹ
20   נַפְשִׁי [ס]             אָרִיק חַרְבִּי תּוֹרִישֵׁמוֹ יָדִי׃ [ס]             נָשַׁפְתָּ
21   בְרוּחֲךָ כִּסָּמוֹ יָם [ס]             צָלְלוּ כַּעוֹפֶרֶת בְּמַיִם
22   אַדִּירִים׃ [ס]             מִי־כָמֹכָה בָּאֵלִם יְהוָה [ס]             מִי
23   כָּמֹכָה נֶאְדָּר בַּקֹּדֶשׁ [ס]             נוֹרָא תְהִלֹּת עֹשֵׂה
24   פֶלֶא׃ [ס]             נָטִיתָ יְמִינְךָ תִּבְלָעֵמוֹ אָרֶץ׃ [ס]             נָחִיתָ
25   בְחַסְדְּךָ עַם־זוּ גָּאָלְתָּ [ס]             נֵהַלְתָּ בְעָזְּךָ אֶל־נְוֵה
26   קָדְשֶׁךָ׃ [ס]             שָׁמְעוּ עַמִּים יִרְגָּזוּן [ס]             חִיל
27   אָחַז יֹשְׁבֵי פְּלָשֶׁת׃ [ס]             אָז נִבְהֲלוּ אַלּוּפֵי
28   אֱדוֹם [ס]             אֵילֵי מוֹאָב יֹאחֲזֵמוֹ רָעַד [ס]             נָמֹגוּ
29   כֹּל יֹשְׁבֵי כְנָעַן׃ [ס]             תִּפֹּל עֲלֵיהֶם אֵימָתָה
30   וָפַחַד [ס]             בִּגְדֹל זְרוֹעֲךָ יִדְּמוּ כָּאָבֶן [ס]             עַד־
31   יַעֲבֹר עַמְּךָ יְהוָה [ס]             עַד־יַעֲבֹר עַם־זוּ
32   קָנִיתָ׃ [ס]             תְּבִאֵמוֹ וְתִטָּעֵמוֹ בְּהַר נַחֲלָתְךָ [ס]             מָכוֹן
33   לְשִׁבְתְּךָ פָּעַלְתָּ יְהוָה [ס]             מִקְּדָשׁ אֲדֹנָי כּוֹנְנוּ
34   יָדֶיךָ׃ [ס]             יְהוָה  יִמְלֹךְ לְעֹלָם וָעֶד׃ [ס]             כִּי
35   בָא סוּס פַּרְעֹה בְּרִכְבּוֹ וּבְפָרָשָׁיו בַּיָּם [ס]             וַיָּשֶׁב יְהוָה עֲלֵהֶם
36   אֶת־מֵי הַיָּם [ס]             וּבְנֵי יִשְׂרָאֵל הָלְכוּ בַיַּבָּשָׁה בְּתוֹךְ הַיָּם׃
37  
38   וַתִּקַּח מִרְיָם הַנְּבִיאָה אֲחוֹת אַהֲרֹן אֶת־הַתֹּף בְּיָדָהּ וַתֵּצֶאןָ כָל־הַנָּשִׁים
39   אַחֲרֶיהָ בְּתֻפִּים וּבִמְחֹלֹת׃ וַתַּעַן לָהֶם מִרְיָם שִׁירוּ לַיהוָה כִּי־גָאֹה גָּאָה
40   סוּס וְרֹכְבוֹ רָמָה בַיָּם׃ [ס]             וַיַּסַּע מֹשֶׁה אֶת־יִשְׂרָאֵל מִיַּם־סוּף
41   וַיֵּצְאוּ אֶל־מִדְבַּר־שׁוּר וַיֵּלְכוּ שְׁלֹשֶׁת־יָמִים בַּמִּדְבָּר וְלֹא־מָצְאוּ מָיִם׃
42   וַיָּבֹאוּ מָרָתָה וְלֹא יָכְלוּ לִשְׁתֹּת מַיִם מִמָּרָה כִּי מָרִים הֵם עַל־כֵּן קָרָא־שְׁמָהּ מָרָה׃
