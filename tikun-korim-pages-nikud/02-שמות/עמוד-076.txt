 1  וְהִגַּדְתָּ לְבִנְךָ בַּיּוֹם הַהוּא לֵאמֹר בַּעֲבוּר זֶה עָשָׂה
 2   יְהוָה לִי בְּצֵאתִי מִמִּצְרָיִם׃ וְהָיָה לְךָ לְאוֹת עַל־יָדְךָ
 3   וּלְזִכָּרוֹן בֵּין עֵינֶיךָ לְמַעַן תִּהְיֶה תּוֹרַת יְהוָה בְּפִיךָ כִּי
 4   בְּיָד חֲזָקָה הוֹצִאֲךָ יְהוָה מִמִּצְרָיִם׃ וְשָׁמַרְתָּ אֶת־הַחֻקָּה
 5   הַזֹּאת לְמוֹעֲדָהּ מִיָּמִים יָמִימָה׃ [פ]
 6   וְהָיָה כִּי־יְבִאֲךָ יְהוָה אֶל־אֶרֶץ הַכְּנַעֲנִי כַּאֲשֶׁר
 7   נִשְׁבַּע לְךָ וְלַאֲבֹתֶיךָ וּנְתָנָהּ לָךְ׃ וְהַעֲבַרְתָּ כָל־פֶּטֶר־
 8  רֶחֶם לַיהוָה וְכָל־פֶּטֶר  שֶׁגֶר בְּהֵמָה אֲשֶׁר יִהְיֶה לְךָ
 9   הַזְּכָרִים לַיהוָה׃ וְכָל־פֶּטֶר חֲמֹר תִּפְדֶּה בְשֶׂה וְאִם־
10  לֹא תִפְדֶּה וַעֲרַפְתּוֹ וְכֹל בְּכוֹר אָדָם בְּבָנֶיךָ תִּפְדֶּה׃
11    וְהָיָה כִּי־יִשְׁאָלְךָ בִנְךָ מָחָר לֵאמֹר מַה־זֹּאת וְאָמַרְתָּ
12   אֵלָיו בְּחֹזֶק יָד הוֹצִיאָנוּ יְהוָה מִמִּצְרַיִם מִבֵּית
13   עֲבָדִים׃ וַיְהִי כִּי־הִקְשָׁה פַרְעֹה לְשַׁלְּחֵנוּ וַיַּהֲרֹג
14   יְהוָה כָּל־בְּכוֹר בְּאֶרֶץ מִצְרַיִם מִבְּכֹר אָדָם וְעַד־
15  בְּכוֹר בְּהֵמָה עַל־כֵּן אֲנִי זֹבֵחַ לַיהוָה כָּל־פֶּטֶר רֶחֶם
16   הַזְּכָרִים וְכָל־בְּכוֹר בָּנַי אֶפְדֶּה׃ וְהָיָה לְאוֹת עַל־
17  יָדְכָה וּלְטוֹטָפֹת בֵּין עֵינֶיךָ כִּי בְּחֹזֶק יָד הוֹצִיאָנוּ
18   יְהוָה מִמִּצְרָיִם׃ [ס]            וַיְהִי
19   בְּשַׁלַּח פַּרְעֹה אֶת־הָעָם וְלֹא־נָחָם אֱלֹהִים דֶּרֶךְ
20   אֶרֶץ פְּלִשְׁתִּים כִּי קָרוֹב הוּא כִּי  אָמַר אֱלֹהִים פֶּן־
21  יִנָּחֵם הָעָם בִּרְאֹתָם מִלְחָמָה וְשָׁבוּ מִצְרָיְמָה׃ וַיַּסֵּב
22   אֱלֹהִים  אֶת־הָעָם דֶּרֶךְ הַמִּדְבָּר יַם־סוּף וַחֲמֻשִׁים
23   עָלוּ בְנֵי־יִשְׂרָאֵל מֵאֶרֶץ מִצְרָיִם׃ וַיִּקַּח מֹשֶׁה אֶת־
24  עַצְמוֹת יוֹסֵף עִמּוֹ כִּי הַשְׁבֵּעַ הִשְׁבִּיעַ אֶת־בְּנֵי
25   יִשְׂרָאֵל לֵאמֹר פָּקֹד יִפְקֹד אֱלֹהִים אֶתְכֶם וְהַעֲלִיתֶם
26   אֶת־עַצְמֹתַי מִזֶּה אִתְּכֶם׃ וַיִּסְעוּ מִסֻּכֹּת וַיַּחֲנוּ בְאֵתָם
27   בִּקְצֵה הַמִּדְבָּר׃ וַיהוָה הֹלֵךְ לִפְנֵיהֶם יוֹמָם בְּעַמּוּד
28   עָנָן לַנְחֹתָם הַדֶּרֶךְ וְלַיְלָה בְּעַמּוּד אֵשׁ לְהָאִיר לָהֶם
29   לָלֶכֶת יוֹמָם וָלָיְלָה׃ לֹא־יָמִישׁ עַמּוּד הֶעָנָן יוֹמָם
30   וְעַמּוּד הָאֵשׁ לָיְלָה לִפְנֵי הָעָם׃♢ [פ]
31   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
32   וְיָשֻׁבוּ וְיַחֲנוּ לִפְנֵי פִּי הַחִירֹת בֵּין מִגְדֹּל וּבֵין הַיָּם
33   לִפְנֵי בַּעַל צְפֹן נִכְחוֹ תַחֲנוּ עַל־הַיָּם׃ וְאָמַר פַּרְעֹה
34   לִבְנֵי יִשְׂרָאֵל נְבֻכִים הֵם בָּאָרֶץ סָגַר עֲלֵיהֶם
35   הַמִּדְבָּר׃ וְחִזַּקְתִּי אֶת־לֵב־פַּרְעֹה וְרָדַף אַחֲרֵיהֶם
36   וְאִכָּבְדָה בְּפַרְעֹה וּבְכָל־חֵילוֹ וְיָדְעוּ מִצְרַיִם כִּי־
37  אֲנִי יְהוָה וַיַּעֲשׂוּ־כֵן׃♢ וַיֻּגַּד לְמֶלֶךְ מִצְרַיִם כִּי בָרַח
38   הָעָם וַיֵּהָפֵךְ לְבַב פַּרְעֹה וַעֲבָדָיו אֶל־הָעָם וַיֹּאמְרוּ
39   מַה־זֹּאת עָשִׂינוּ כִּי־שִׁלַּחְנוּ אֶת־יִשְׂרָאֵל מֵעָבְדֵנוּ׃
40   וַיֶּאְסֹר אֶת־רִכְבּוֹ וְאֶת־עַמּוֹ לָקַח עִמּוֹ׃ וַיִּקַּח שֵׁשׁ־
41  מֵאוֹת רֶכֶב בָּחוּר וְכֹל רֶכֶב מִצְרָיִם וְשָׁלִשִׁם
42   עַל־כֻּלּוֹ׃ וַיְחַזֵּק יְהוָה אֶת־לֵב פַּרְעֹה מֶלֶךְ מִצְרַיִם
