 1  וּבִבְקָרֵנוּ נֵלֵךְ כִּי חַג־יְהוָה לָנוּ׃ וַיֹּאמֶר אֲלֵהֶם יְהִי כֵן
 2   יְהוָה עִמָּכֶם כַּאֲשֶׁר אֲשַׁלַּח אֶתְכֶם וְאֶת־טַפְּכֶם רְאוּ
 3   כִּי רָעָה נֶגֶד פְּנֵיכֶם׃ לֹא כֵן לְכוּ־נָא הַגְּבָרִים וְעִבְדוּ
 4   אֶת־יְהוָה כִּי אֹתָהּ אַתֶּם מְבַקְשִׁים וַיְגָרֶשׁ אֹתָם מֵאֵת
 5   פְּנֵי פַרְעֹה׃♢ [ס]              וַיֹּאמֶר יְהוָה
 6   אֶל־מֹשֶׁה נְטֵה יָדְךָ עַל־אֶרֶץ מִצְרַיִם בָּאַרְבֶּה וְיַעַל
 7   עַל־אֶרֶץ מִצְרָיִם וְיֹאכַל אֶת־כָּל־עֵשֶׂב הָאָרֶץ אֵת
 8   כָּל־אֲשֶׁר הִשְׁאִיר הַבָּרָד׃ וַיֵּט מֹשֶׁה אֶת־מַטֵּהוּ עַל־
 9  אֶרֶץ מִצְרַיִם וַיהוָה נִהַג רוּחַ־קָדִים בָּאָרֶץ כָּל־הַיּוֹם
10   הַהוּא וְכָל־הַלָּיְלָה הַבֹּקֶר הָיָה וְרוּחַ הַקָּדִים נָשָׂא
11   אֶת־הָאַרְבֶּה׃ וַיַּעַל הָאַרְבֶּה עַל כָּל־אֶרֶץ מִצְרַיִם
12   וַיָּנַח בְּכֹל גְּבוּל מִצְרָיִם כָּבֵד מְאֹד לְפָנָיו לֹא־הָיָה
13   כֵן אַרְבֶּה כָּמֹהוּ וְאַחֲרָיו לֹא יִהְיֶה־כֵּן׃ וַיְכַס אֶת־עֵין
14   כָּל־הָאָרֶץ וַתֶּחְשַׁךְ הָאָרֶץ וַיֹּאכַל אֶת־כָּל־עֵשֶׂב
15   הָאָרֶץ וְאֵת כָּל־פְּרִי הָעֵץ אֲשֶׁר הוֹתִיר הַבָּרָד
16   וְלֹא־נוֹתַר כָּל־יֶרֶק בָּעֵץ וּבְעֵשֶׂב הַשָּׂדֶה בְּכָל־
17  אֶרֶץ מִצְרָיִם׃ וַיְמַהֵר פַּרְעֹה לִקְרֹא לְמֹשֶׁה וּלְאַהֲרֹן
18   וַיֹּאמֶר חָטָאתִי לַיהוָה אֱלֹהֵיכֶם וְלָכֶם׃ וְעַתָּה שָׂא נָא
19   חַטָּאתִי אַךְ הַפַּעַם וְהַעְתִּירוּ לַיהוָה אֱלֹהֵיכֶם וְיָסֵר
20   מֵעָלַי רַק אֶת־הַמָּוֶת הַזֶּה׃ וַיֵּצֵא מֵעִם פַּרְעֹה וַיֶּעְתַּר
21   אֶל־יְהוָה׃ וַיַּהֲפֹךְ יְהוָה רוּחַ־יָם חָזָק מְאֹד וַיִּשָּׂא אֶת־
22  הָאַרְבֶּה וַיִּתְקָעֵהוּ יָמָּה סּוּף לֹא נִשְׁאַר אַרְבֶּה אֶחָד
23   בְּכֹל גְּבוּל מִצְרָיִם׃ וַיְחַזֵּק יְהוָה אֶת־לֵב פַּרְעֹה וְלֹא
24   שִׁלַּח אֶת־בְּנֵי יִשְׂרָאֵל׃ [פ]
25   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה נְטֵה יָדְךָ עַל־הַשָּׁמַיִם וִיהִי
26   חֹשֶׁךְ עַל־אֶרֶץ מִצְרָיִם וְיָמֵשׁ חֹשֶׁךְ׃ וַיֵּט מֹשֶׁה אֶת־
27  יָדוֹ עַל־הַשָּׁמָיִם וַיְהִי חֹשֶׁךְ־אֲפֵלָה בְּכָל־אֶרֶץ מִצְרַיִם
28   שְׁלֹשֶׁת יָמִים׃ לֹא־רָאוּ אִישׁ אֶת־אָחִיו וְלֹא־קָמוּ אִישׁ
29   מִתַּחְתָּיו שְׁלֹשֶׁת יָמִים וּלְכָל־בְּנֵי יִשְׂרָאֵל הָיָה אוֹר
30   בְּמוֹשְׁבֹתָם׃  וַיִּקְרָא פַרְעֹה אֶל־מֹשֶׁה וַיֹּאמֶר לְכוּ
31   עִבְדוּ אֶת־יְהוָה רַק צֹאנְכֶם וּבְקַרְכֶם יֻצָּג גַּם־טַפְּכֶם
32   יֵלֵךְ עִמָּכֶם׃ וַיֹּאמֶר מֹשֶׁה גַּם־אַתָּה תִּתֵּן בְּיָדֵנוּ זְבָחִים
33   וְעֹלֹת וְעָשִׂינוּ לַיהוָה אֱלֹהֵינוּ׃ וְגַם־מִקְנֵנוּ יֵלֵךְ עִמָּנוּ לֹא
34   תִשָּׁאֵר פַּרְסָה כִּי מִמֶּנּוּ נִקַּח לַעֲבֹד אֶת־יְהוָה אֱלֹהֵינוּ
35   וַאֲנַחְנוּ לֹא־נֵדַע מַה־נַּעֲבֹד אֶת־יְהוָה עַד־בֹּאֵנוּ שָׁמָּה׃
36   וַיְחַזֵּק יְהוָה אֶת־לֵב פַּרְעֹה וְלֹא אָבָה לְשַׁלְּחָם׃ וַיֹּאמֶר־
37  לוֹ פַרְעֹה לֵךְ מֵעָלָי הִשָּׁמֶר לְךָ אַל־תֹּסֶף רְאוֹת פָּנַי
38   כִּי בְּיוֹם רְאֹתְךָ פָנַי תָּמוּת׃ וַיֹּאמֶר מֹשֶׁה כֵּן דִּבַּרְתָּ
39   לֹא־אֹסִף עוֹד רְאוֹת פָּנֶיךָ׃ [פ]
40   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה עוֹד נֶגַע אֶחָד אָבִיא עַל־פַּרְעֹה
41   וְעַל־מִצְרַיִם אַחֲרֵי־כֵן יְשַׁלַּח אֶתְכֶם מִזֶּה כְּשַׁלְּחוֹ
42   כָּלָה גָּרֵשׁ יְגָרֵשׁ אֶתְכֶם מִזֶּה׃ דַּבֶּר־נָא בְּאָזְנֵי הָעָם
