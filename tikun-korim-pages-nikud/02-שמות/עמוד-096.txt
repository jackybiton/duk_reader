 1  וּמֵאֲשֶׁר לְבָנָיו׃ וְהָיָה לְאַהֲרֹן וּלְבָנָיו לְחָק־עוֹלָם מֵאֵת
 2   בְּנֵי יִשְׂרָאֵל כִּי תְרוּמָה הוּא וּתְרוּמָה יִהְיֶה מֵאֵת בְּנֵי־
 3  יִשְׂרָאֵל מִזִּבְחֵי שַׁלְמֵיהֶם תְּרוּמָתָם לַיהוָה׃ וּבִגְדֵי
 4   הַקֹּדֶשׁ אֲשֶׁר לְאַהֲרֹן יִהְיוּ לְבָנָיו אַחֲרָיו לְמָשְׁחָה
 5   בָהֶם וּלְמַלֵּא־בָם אֶת־יָדָם׃ שִׁבְעַת יָמִים יִלְבָּשָׁם הַכֹּהֵן
 6   תַּחְתָּיו מִבָּנָיו אֲשֶׁר יָבֹא אֶל־אֹהֶל מוֹעֵד לְשָׁרֵת
 7   בַּקֹּדֶשׁ׃ וְאֵת אֵיל הַמִּלֻּאִים תִּקָּח וּבִשַּׁלְתָּ אֶת־בְּשָׂרוֹ
 8   בְּמָקֹם קָדֹשׁ׃ וְאָכַל אַהֲרֹן וּבָנָיו אֶת־בְּשַׂר הָאַיִל וְאֶת־
 9  הַלֶּחֶם אֲשֶׁר בַּסָּל פֶּתַח אֹהֶל מוֹעֵד׃ וְאָכְלוּ אֹתָם
10   אֲשֶׁר כֻּפַּר בָּהֶם לְמַלֵּא אֶת־יָדָם לְקַדֵּשׁ אֹתָם וְזָר לֹא־
11  יֹאכַל כִּי־קֹדֶשׁ הֵם׃ וְאִם־יִוָּתֵר מִבְּשַׂר הַמִּלֻּאִים וּמִן־
12  הַלֶּחֶם עַד־הַבֹּקֶר וְשָׂרַפְתָּ אֶת־הַנּוֹתָר בָּאֵשׁ לֹא יֵאָכֵל
13   כִּי־קֹדֶשׁ הוּא׃ וְעָשִׂיתָ לְאַהֲרֹן וּלְבָנָיו כָּכָה כְּכֹל אֲשֶׁר־
14  צִוִּיתִי אֹתָכָה שִׁבְעַת יָמִים תְּמַלֵּא יָדָם׃ וּפַר חַטָּאת
15   תַּעֲשֶׂה לַיּוֹם עַל־הַכִּפֻּרִים וְחִטֵּאתָ עַל־הַמִּזְבֵּחַ בְּכַפֶּרְךָ
16   עָלָיו וּמָשַׁחְתָּ אֹתוֹ לְקַדְּשׁוֹ׃ שִׁבְעַת יָמִים תְּכַפֵּר עַל־
17  הַמִּזְבֵּחַ וְקִדַּשְׁתָּ אֹתוֹ וְהָיָה הַמִּזְבֵּחַ קֹדֶשׁ קָדָשִׁים
18   כָּל־הַנֹּגֵעַ בַּמִּזְבֵּחַ יִקְדָּשׁ׃ [ס]              וְזֶה
19   אֲשֶׁר תַּעֲשֶׂה עַל־הַמִּזְבֵּחַ כְּבָשִׂים בְּנֵי־שָׁנָה שְׁנַיִם
20   לַיּוֹם תָּמִיד׃ אֶת־הַכֶּבֶשׂ הָאֶחָד תַּעֲשֶׂה בַבֹּקֶר וְאֵת
21   הַכֶּבֶשׂ הַשֵּׁנִי תַּעֲשֶׂה בֵּין הָעַרְבָּיִם׃ וְעִשָּׂרֹן סֹלֶת
22   בָּלוּל בְּשֶׁמֶן כָּתִית רֶבַע הַהִין וְנֵסֶךְ רְבִיעִת הַהִין
23   יָיִן לַכֶּבֶשׂ הָאֶחָד׃ וְאֵת הַכֶּבֶשׂ הַשֵּׁנִי תַּעֲשֶׂה בֵּין
24   הָעַרְבָּיִם כְּמִנְחַת הַבֹּקֶר וּכְנִסְכָּהּ תַּעֲשֶׂה־לָּהּ לְרֵיחַ
25   נִיחֹחַ אִשֶּׁה לַיהוָה׃ עֹלַת תָּמִיד לְדֹרֹתֵיכֶם פֶּתַח אֹהֶל־
26  מוֹעֵד לִפְנֵי יְהוָה אֲשֶׁר אִוָּעֵד לָכֶם שָׁמָּה לְדַבֵּר
27   אֵלֶיךָ שָׁם׃ וְנֹעַדְתִּי שָׁמָּה לִבְנֵי יִשְׂרָאֵל וְנִקְדַּשׁ
28   בִּכְבֹדִי׃ וְקִדַּשְׁתִּי אֶת־אֹהֶל מוֹעֵד וְאֶת־הַמִּזְבֵּחַ וְאֶת־
29  אַהֲרֹן וְאֶת־בָּנָיו אֲקַדֵּשׁ לְכַהֵן לִי׃ וְשָׁכַנְתִּי בְּתוֹךְ בְּנֵי
30   יִשְׂרָאֵל וְהָיִיתִי לָהֶם לֵאלֹהִים׃ וְיָדְעוּ כִּי אֲנִי יְהוָה
31   אֱלֹהֵיהֶם אֲשֶׁר הוֹצֵאתִי אֹתָם מֵאֶרֶץ מִצְרַיִם לְשָׁכְנִי
32   בְתוֹכָם אֲנִי יְהוָה אֱלֹהֵיהֶם׃ [פ]
33    וְעָשִׂיתָ מִזְבֵּחַ מִקְטַר קְטֹרֶת עֲצֵי שִׁטִּים תַּעֲשֶׂה
34   אֹתוֹ׃ אַמָּה אָרְכּוֹ וְאַמָּה רָחְבּוֹ רָבוּעַ יִהְיֶה וְאַמָּתַיִם
35   קֹמָתוֹ מִמֶּנּוּ קַרְנֹתָיו׃ וְצִפִּיתָ אֹתוֹ זָהָב טָהוֹר אֶת־
36  גַּגּוֹ וְאֶת־קִירֹתָיו סָבִיב וְאֶת־קַרְנֹתָיו וְעָשִׂיתָ לּוֹ זֵר
37   זָהָב סָבִיב׃ וּשְׁתֵּי טַבְּעֹת זָהָב תַּעֲשֶׂה־לּוֹ  מִתַּחַת
38   לְזֵרוֹ עַל שְׁתֵּי צַלְעֹתָיו תַּעֲשֶׂה עַל־שְׁנֵי צִדָּיו וְהָיָה
39   לְבָתִּים לְבַדִּים לָשֵׂאת אֹתוֹ בָּהֵמָּה׃ וְעָשִׂיתָ אֶת־
40  הַבַּדִּים עֲצֵי שִׁטִּים וְצִפִּיתָ אֹתָם זָהָב׃ וְנָתַתָּה
41   אֹתוֹ לִפְנֵי הַפָּרֹכֶת אֲשֶׁר עַל־אֲרֹן הָעֵדֻת לִפְנֵי
42   הַכַּפֹּרֶת אֲשֶׁר עַל־הָעֵדֻת אֲשֶׁר אִוָּעֵד לְךָ שָׁמָּה׃
