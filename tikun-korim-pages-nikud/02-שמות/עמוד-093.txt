 1  וּבָנָיו מֵעֶרֶב עַד־בֹּקֶר לִפְנֵי יְהוָה חֻקַּת עוֹלָם לְדֹרֹתָם
 2   מֵאֵת בְּנֵי יִשְׂרָאֵל׃ [ס]             וְאַתָּה
 3   הַקְרֵב אֵלֶיךָ אֶת־אַהֲרֹן אָחִיךָ וְאֶת־בָּנָיו אִתּוֹ מִתּוֹךְ
 4   בְּנֵי יִשְׂרָאֵל לְכַהֲנוֹ־לִי אַהֲרֹן נָדָב וַאֲבִיהוּא אֶלְעָזָר
 5   וְאִיתָמָר בְּנֵי אַהֲרֹן׃ וְעָשִׂיתָ בִגְדֵי־קֹדֶשׁ לְאַהֲרֹן אָחִיךָ
 6   לְכָבוֹד וּלְתִפְאָרֶת׃ וְאַתָּה תְּדַבֵּר אֶל־כָּל־חַכְמֵי־לֵב
 7   אֲשֶׁר מִלֵּאתִיו רוּחַ חָכְמָה וְעָשׂוּ אֶת־בִּגְדֵי אַהֲרֹן
 8   לְקַדְּשׁוֹ לְכַהֲנוֹ־לִי׃ וְאֵלֶּה הַבְּגָדִים אֲשֶׁר יַעֲשׂוּ
 9   חֹשֶׁן וְאֵפוֹד וּמְעִיל וּכְתֹנֶת תַּשְׁבֵּץ מִצְנֶפֶת וְאַבְנֵט
10   וְעָשׂוּ בִגְדֵי־קֹדֶשׁ לְאַהֲרֹן אָחִיךָ וּלְבָנָיו לְכַהֲנוֹ־לִי׃
11   וְהֵם יִקְחוּ אֶת־הַזָּהָב וְאֶת־הַתְּכֵלֶת וְאֶת־הָאַרְגָּמָן
12   וְאֶת־תּוֹלַעַת הַשָּׁנִי וְאֶת־הַשֵּׁשׁ׃♢ [פ]
13   וְעָשׂוּ אֶת־הָאֵפֹד זָהָב תְּכֵלֶת וְאַרְגָּמָן תּוֹלַעַת שָׁנִי
14   וְשֵׁשׁ מָשְׁזָר מַעֲשֵׂה חֹשֵׁב׃ שְׁתֵּי כְתֵפֹת חֹבְרֹת יִהְיֶה־
15  לּוֹ אֶל־שְׁנֵי קְצוֹתָיו וְחֻבָּר׃ וְחֵשֶׁב אֲפֻדָּתוֹ אֲשֶׁר עָלָיו
16   כְּמַעֲשֵׂהוּ מִמֶּנּוּ יִהְיֶה זָהָב תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת
17   שָׁנִי וְשֵׁשׁ מָשְׁזָר׃ וְלָקַחְתָּ אֶת־שְׁתֵּי אַבְנֵי־שֹׁהַם
18   וּפִתַּחְתָּ עֲלֵיהֶם שְׁמוֹת בְּנֵי יִשְׂרָאֵל׃♢ שִׁשָּׁה מִשְּׁמֹתָם
19   עַל הָאֶבֶן הָאֶחָת וְאֶת־שְׁמוֹת הַשִּׁשָּׁה הַנּוֹתָרִים
20   עַל־הָאֶבֶן הַשֵּׁנִית כְּתוֹלְדֹתָם׃ מַעֲשֵׂה חָרַשׁ אֶבֶן
21   פִּתּוּחֵי חֹתָם תְּפַתַּח אֶת־שְׁתֵּי הָאֲבָנִים עַל־שְׁמֹת בְּנֵי
22   יִשְׂרָאֵל מֻסַבֹּת מִשְׁבְּצוֹת זָהָב תַּעֲשֶׂה אֹתָם׃ וְשַׂמְתָּ
23   אֶת־שְׁתֵּי הָאֲבָנִים עַל כִּתְפֹת הָאֵפֹד אַבְנֵי זִכָּרֹן
24   לִבְנֵי יִשְׂרָאֵל וְנָשָׂא אַהֲרֹן אֶת־שְׁמוֹתָם לִפְנֵי יְהוָה
25   עַל־שְׁתֵּי כְתֵפָיו לְזִכָּרֹן׃♢ [ס]              וְעָשִׂיתָ
26   מִשְׁבְּצֹת זָהָב׃ וּשְׁתֵּי שַׁרְשְׁרֹת זָהָב טָהוֹר מִגְבָּלֹת
27   תַּעֲשֶׂה אֹתָם מַעֲשֵׂה עֲבֹת וְנָתַתָּה אֶת־שַׁרְשְׁרֹת
28   הָעֲבֹתֹת עַל־הַמִּשְׁבְּצֹת׃ [ס]             וְעָשִׂיתָ
29   חֹשֶׁן מִשְׁפָּט מַעֲשֵׂה חֹשֵׁב כְּמַעֲשֵׂה אֵפֹד תַּעֲשֶׂנּוּ
30   זָהָב תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר
31   תַּעֲשֶׂה אֹתוֹ׃ רָבוּעַ יִהְיֶה כָּפוּל זֶרֶת אָרְכּוֹ וְזֶרֶת
32   רָחְבּוֹ׃ וּמִלֵּאתָ בוֹ מִלֻּאַת אֶבֶן אַרְבָּעָה טוּרִים אָבֶן
33   טוּר אֹדֶם פִּטְדָה וּבָרֶקֶת הַטּוּר הָאֶחָד׃ וְהַטּוּר הַשֵּׁנִי
34   נֹפֶךְ סַפִּיר וְיָהֲלֹם׃ וְהַטּוּר הַשְּׁלִישִׁי לֶשֶׁם שְׁבוֹ
35   וְאַחְלָמָה׃ וְהַטּוּר הָרְבִיעִי תַּרְשִׁישׁ וְשֹׁהַם וְיָשְׁפֵה
36   מְשֻׁבָּצִים זָהָב יִהְיוּ בְּמִלּוּאֹתָם׃ וְהָאֲבָנִים תִּהְיֶיןָ עַל־
37  שְׁמֹת בְּנֵי־יִשְׂרָאֵל שְׁתֵּים עֶשְׂרֵה עַל־שְׁמֹתָם פִּתּוּחֵי
38   חוֹתָם אִישׁ עַל־שְׁמוֹ תִּהְיֶיןָ לִשְׁנֵי עָשָׂר שָׁבֶט׃
39   וְעָשִׂיתָ עַל־הַחֹשֶׁן שַׁרְשֹׁת גַּבְלֻת מַעֲשֵׂה עֲבֹת זָהָב
40   טָהוֹר׃ וְעָשִׂיתָ עַל־הַחֹשֶׁן שְׁתֵּי טַבְּעוֹת זָהָב וְנָתַתָּ אֶת־
41  שְׁתֵּי הַטַּבָּעוֹת עַל־שְׁנֵי קְצוֹת הַחֹשֶׁן׃ וְנָתַתָּה אֶת־שְׁתֵּי
42   עֲבֹתֹת הַזָּהָב עַל־שְׁתֵּי הַטַּבָּעֹת אֶל־קְצוֹת הַחֹשֶׁן׃
