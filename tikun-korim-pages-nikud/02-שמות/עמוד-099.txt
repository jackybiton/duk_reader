 1  וַיֹּאמֶר אֲלֵהֶם אַהֲרֹן פָּרְקוּ נִזְמֵי הַזָּהָב אֲשֶׁר בְּאָזְנֵי
 2   נְשֵׁיכֶם בְּנֵיכֶם וּבְנֹתֵיכֶם וְהָבִיאוּ אֵלָי׃ וַיִּתְפָּרְקוּ כָּל־
 3  הָעָם אֶת־נִזְמֵי הַזָּהָב אֲשֶׁר בְּאָזְנֵיהֶם וַיָּבִיאוּ אֶל־
 4  אַהֲרֹן׃ וַיִּקַּח מִיָּדָם וַיָּצַר אֹתוֹ בַּחֶרֶט וַיַּעֲשֵׂהוּ עֵגֶל
 5   מַסֵּכָה וַיֹּאמְרוּ אֵלֶּה אֱלֹהֶיךָ יִשְׂרָאֵל אֲשֶׁר הֶעֱלוּךָ
 6   מֵאֶרֶץ מִצְרָיִם׃ וַיַּרְא אַהֲרֹן וַיִּבֶן מִזְבֵּחַ לְפָנָיו וַיִּקְרָא
 7   אַהֲרֹן וַיֹּאמַר חַג לַיהוָה מָחָר׃ וַיַּשְׁכִּימוּ מִמָּחֳרָת
 8   וַיַּעֲלוּ עֹלֹת וַיַּגִּשׁוּ שְׁלָמִים וַיֵּשֶׁב הָעָם לֶאֱכֹל
 9   וְשָׁתוֹ וַיָּקֻמוּ לְצַחֵק׃ [פ]
10   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֶךְ־רֵד כִּי שִׁחֵת עַמְּךָ אֲשֶׁר
11   הֶעֱלֵיתָ מֵאֶרֶץ מִצְרָיִם׃ סָרוּ מַהֵר מִן־הַדֶּרֶךְ אֲשֶׁר
12   צִוִּיתִם עָשׂוּ לָהֶם עֵגֶל מַסֵּכָה וַיִּשְׁתַּחֲווּ־לוֹ וַיִּזְבְּחוּ־
13  לוֹ וַיֹּאמְרוּ אֵלֶּה אֱלֹהֶיךָ יִשְׂרָאֵל אֲשֶׁר הֶעֱלוּךָ
14   מֵאֶרֶץ מִצְרָיִם׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה רָאִיתִי אֶת־
15  הָעָם הַזֶּה וְהִנֵּה עַם־קְשֵׁה־עֹרֶף הוּא׃ וְעַתָּה הַנִּיחָה
16   לִּי וְיִחַר־אַפִּי בָהֶם וַאֲכַלֵּם וְאֶעֱשֶׂה אוֹתְךָ לְגוֹי גָּדוֹל׃
17   וַיְחַל מֹשֶׁה אֶת־פְּנֵי יְהוָה אֱלֹהָיו וַיֹּאמֶר לָמָה יְהוָה
18   יֶחֱרֶה אַפְּךָ בְּעַמֶּךָ אֲשֶׁר הוֹצֵאתָ מֵאֶרֶץ מִצְרַיִם
19   בְּכֹחַ גָּדוֹל וּבְיָד חֲזָקָה׃ לָמָּה יֹאמְרוּ מִצְרַיִם לֵאמֹר
20   בְּרָעָה הוֹצִיאָם לַהֲרֹג אֹתָם בֶּהָרִים וּלְכַלֹּתָם מֵעַל
21   פְּנֵי הָאֲדָמָה שׁוּב מֵחֲרוֹן אַפֶּךָ וְהִנָּחֵם עַל־הָרָעָה
22   לְעַמֶּךָ׃ זְכֹר לְאַבְרָהָם לְיִצְחָק וּלְיִשְׂרָאֵל עֲבָדֶיךָ
23   אֲשֶׁר נִשְׁבַּעְתָּ לָהֶם בָּךְ וַתְּדַבֵּר אֲלֵהֶם אַרְבֶּה אֶת־
24  זַרְעֲכֶם כְּכוֹכְבֵי הַשָּׁמָיִם וְכָל־הָאָרֶץ הַזֹּאת אֲשֶׁר
25   אָמַרְתִּי אֶתֵּן לְזַרְעֲכֶם וְנָחֲלוּ לְעֹלָם׃ וַיִּנָּחֶם יְהוָה עַל־
26  הָרָעָה אֲשֶׁר דִּבֶּר לַעֲשׂוֹת לְעַמּוֹ׃ [פ]
27   וַיִּפֶן וַיֵּרֶד מֹשֶׁה מִן־הָהָר וּשְׁנֵי לֻחֹת הָעֵדֻת בְּיָדוֹ
28   לֻחֹת כְּתֻבִים מִשְּׁנֵי עֶבְרֵיהֶם מִזֶּה וּמִזֶּה הֵם כְּתֻבִים׃
29   וְהַלֻּחֹת מַעֲשֵׂה אֱלֹהִים הֵמָּה וְהַמִּכְתָּב מִכְתַּב אֱלֹהִים
30   הוּא חָרוּת עַל־הַלֻּחֹת׃ וַיִּשְׁמַע יְהוֹשֻׁעַ אֶת־קוֹל
31   הָעָם בְּרֵעֹה וַיֹּאמֶר אֶל־מֹשֶׁה קוֹל מִלְחָמָה בַּמַּחֲנֶה׃
32   וַיֹּאמֶר אֵין קוֹל עֲנוֹת גְּבוּרָה וְאֵין קוֹל עֲנוֹת חֲלוּשָׁה
33   קוֹל עַנּוֹת אָנֹכִי שֹׁמֵעַ׃ וַיְהִי כַּאֲשֶׁר קָרַב אֶל־הַמַּחֲנֶה
34   וַיַּרְא אֶת־הָעֵגֶל וּמְחֹלֹת וַיִּחַר־אַף מֹשֶׁה וַיַּשְׁלֵךְ
35   מִיָּדָו אֶת־הַלֻּחֹת וַיְשַׁבֵּר אֹתָם תַּחַת הָהָר׃ וַיִּקַּח אֶת־
36  הָעֵגֶל אֲשֶׁר עָשׂוּ וַיִּשְׂרֹף בָּאֵשׁ וַיִּטְחַן עַד אֲשֶׁר־דָּק
37   וַיִּזֶר עַל־פְּנֵי הַמַּיִם וַיַּשְׁקְ אֶת־בְּנֵי יִשְׂרָאֵל׃ וַיֹּאמֶר
38   מֹשֶׁה אֶל־אַהֲרֹן מֶה־עָשָׂה לְךָ הָעָם הַזֶּה כִּי־הֵבֵאתָ
39   עָלָיו חֲטָאָה גְדֹלָה׃ וַיֹּאמֶר אַהֲרֹן אַל־יִחַר אַף אֲדֹנִי
40   אַתָּה יָדַעְתָּ אֶת־הָעָם כִּי בְרָע הוּא׃ וַיֹּאמְרוּ לִי עֲשֵׂה־
41  לָנוּ אֱלֹהִים אֲשֶׁר יֵלְכוּ לְפָנֵינוּ כִּי־זֶה  מֹשֶׁה הָאִישׁ
42   אֲשֶׁר הֶעֱלָנוּ מֵאֶרֶץ מִצְרַיִם לֹא יָדַעְנוּ מֶה־הָיָה לוֹ׃
