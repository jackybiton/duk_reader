 1  וַיַּעַשׂ אֶת־הַחֹשֶׁן מַעֲשֵׂה חֹשֵׁב כְּמַעֲשֵׂה אֵפֹד זָהָב
 2   תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר׃ רָבוּעַ
 3   הָיָה כָּפוּל עָשׂוּ אֶת־הַחֹשֶׁן זֶרֶת אָרְכּוֹ וְזֶרֶת רָחְבּוֹ
 4   כָּפוּל׃ וַיְמַלְאוּ־בוֹ אַרְבָּעָה טוּרֵי אָבֶן טוּר אֹדֶם
 5   פִּטְדָה וּבָרֶקֶת הַטּוּר הָאֶחָד׃ וְהַטּוּר הַשֵּׁנִי נֹפֶךְ סַפִּיר
 6   וְיָהֲלֹם׃ וְהַטּוּר הַשְּׁלִישִׁי לֶשֶׁם שְׁבוֹ וְאַחְלָמָה׃ וְהַטּוּר
 7   הָרְבִיעִי תַּרְשִׁישׁ שֹׁהַם וְיָשְׁפֵה מוּסַבֹּת מִשְׁבְּצֹת
 8   זָהָב בְּמִלֻּאֹתָם׃ וְהָאֲבָנִים עַל־שְׁמֹת בְּנֵי־יִשְׂרָאֵל
 9   הֵנָּה שְׁתֵּים עֶשְׂרֵה עַל־שְׁמֹתָם פִּתּוּחֵי חֹתָם אִישׁ
10   עַל־שְׁמוֹ לִשְׁנֵים עָשָׂר שָׁבֶט׃ וַיַּעֲשׂוּ עַל־הַחֹשֶׁן
11   שַׁרְשְׁרֹת גַּבְלֻת מַעֲשֵׂה עֲבֹת זָהָב טָהוֹר׃ וַיַּעֲשׂוּ
12   שְׁתֵּי מִשְׁבְּצֹת זָהָב וּשְׁתֵּי טַבְּעֹת זָהָב וַיִּתְּנוּ אֶת־שְׁתֵּי
13   הַטַּבָּעֹת עַל־שְׁנֵי קְצוֹת הַחֹשֶׁן׃ וַיִּתְּנוּ שְׁתֵּי הָעֲבֹתֹת
14   הַזָּהָב עַל־שְׁתֵּי הַטַּבָּעֹת עַל־קְצוֹת הַחֹשֶׁן׃ וְאֵת שְׁתֵּי
15   קְצוֹת שְׁתֵּי הָעֲבֹתֹת נָתְנוּ עַל־שְׁתֵּי הַמִּשְׁבְּצֹת וַיִּתְּנֻם
16   עַל־כִּתְפֹת הָאֵפֹד אֶל־מוּל פָּנָיו׃ וַיַּעֲשׂוּ שְׁתֵּי טַבְּעֹת
17   זָהָב וַיָּשִׂימוּ עַל־שְׁנֵי קְצוֹת הַחֹשֶׁן עַל־שְׂפָתוֹ אֲשֶׁר
18   אֶל־עֵבֶר הָאֵפֹד בָּיְתָה׃ וַיַּעֲשׂוּ שְׁתֵּי טַבְּעֹת זָהָב
19   וַיִּתְּנֻם עַל־שְׁתֵּי כִתְפֹת הָאֵפֹד מִלְּמַטָּה מִמּוּל פָּנָיו
20   לְעֻמַּת מַחְבַּרְתּוֹ מִמַּעַל לְחֵשֶׁב הָאֵפֹד׃ וַיִּרְכְּסוּ אֶת־
21  הַחֹשֶׁן מִטַּבְּעֹתָיו אֶל־טַבְּעֹת הָאֵפֹד בִּפְתִיל תְּכֵלֶת
22   לִהְיֹת עַל־חֵשֶׁב הָאֵפֹד וְלֹא־יִזַּח הַחֹשֶׁן מֵעַל הָאֵפֹד
23   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [פ]
24    וַיַּעַשׂ אֶת־מְעִיל הָאֵפֹד מַעֲשֵׂה אֹרֵג כְּלִיל תְּכֵלֶת׃
25   וּפִי־הַמְּעִיל בְּתוֹכוֹ כְּפִי תַחְרָא שָׂפָה לְפִיו סָבִיב
26   לֹא יִקָּרֵעַ׃ וַיַּעֲשׂוּ עַל־שׁוּלֵי הַמְּעִיל רִמּוֹנֵי תְּכֵלֶת
27   וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי מָשְׁזָר׃ וַיַּעֲשׂוּ פַעֲמֹנֵי זָהָב
28   טָהוֹר וַיִּתְּנוּ אֶת־הַפַּעֲמֹנִים בְּתוֹךְ הָרִמֹּנִים עַל־שׁוּלֵי
29   הַמְּעִיל סָבִיב בְּתוֹךְ הָרִמֹּנִים׃ פַּעֲמֹן וְרִמֹּן פַּעֲמֹן
30   וְרִמֹּן עַל־שׁוּלֵי הַמְּעִיל סָבִיב לְשָׁרֵת כַּאֲשֶׁר צִוָּה
31   יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיַּעֲשׂוּ
32   אֶת־הַכָּתְנֹת שֵׁשׁ מַעֲשֵׂה אֹרֵג לְאַהֲרֹן וּלְבָנָיו׃ וְאֵת
33   הַמִּצְנֶפֶת שֵׁשׁ וְאֶת־פַּאֲרֵי הַמִּגְבָּעֹת שֵׁשׁ וְאֶת־מִכְנְסֵי
34   הַבָּד שֵׁשׁ מָשְׁזָר׃ וְאֶת־הָאַבְנֵט שֵׁשׁ מָשְׁזָר וּתְכֵלֶת
35   וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי מַעֲשֵׂה רֹקֵם כַּאֲשֶׁר צִוָּה
36   יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיַּעֲשׂוּ
37   אֶת־צִיץ נֵזֶר־הַקֹּדֶשׁ זָהָב טָהוֹר וַיִּכְתְּבוּ עָלָיו מִכְתַּב
38   פִּתּוּחֵי חוֹתָם קֹדֶשׁ לַיהוָה׃ וַיִּתְּנוּ עָלָיו פְּתִיל תְּכֵלֶת
39   לָתֵת עַל־הַמִּצְנֶפֶת מִלְמָעְלָה כַּאֲשֶׁר צִוָּה יְהוָה אֶת־
40  מֹשֶׁה׃ [ס]             וַתֵּכֶל כָּל־עֲבֹדַת
41   מִשְׁכַּן אֹהֶל מוֹעֵד וַיַּעֲשׂוּ בְּנֵי יִשְׂרָאֵל כְּכֹל אֲשֶׁר
42   צִוָּה יְהוָה אֶת־מֹשֶׁה כֵּן עָשׂוּ׃ [פ]
