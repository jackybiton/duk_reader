 1  וּבֵין קֹדֶשׁ הַקֳּדָשִׁים׃ וְנָתַתָּ אֶת־הַכַּפֹּרֶת עַל אֲרוֹן
 2   הָעֵדֻת בְּקֹדֶשׁ הַקֳּדָשִׁים׃ וְשַׂמְתָּ אֶת־הַשֻּׁלְחָן מִחוּץ
 3   לַפָּרֹכֶת וְאֶת־הַמְּנֹרָה נֹכַח הַשֻּׁלְחָן עַל צֶלַע הַמִּשְׁכָּן
 4   תֵּימָנָה וְהַשֻּׁלְחָן תִּתֵּן עַל־צֶלַע צָפוֹן׃ וְעָשִׂיתָ מָסָךְ
 5   לְפֶתַח הָאֹהֶל תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ
 6   מָשְׁזָר מַעֲשֵׂה רֹקֵם׃ וְעָשִׂיתָ לַמָּסָךְ חֲמִשָּׁה עַמּוּדֵי
 7   שִׁטִּים וְצִפִּיתָ אֹתָם זָהָב וָוֵיהֶם זָהָב וְיָצַקְתָּ לָהֶם
 8   חֲמִשָּׁה אַדְנֵי נְחֹשֶׁת׃ [ס]              וְעָשִׂיתָ אֶת־
 9  הַמִּזְבֵּחַ עֲצֵי שִׁטִּים חָמֵשׁ אַמּוֹת אֹרֶךְ וְחָמֵשׁ אַמּוֹת
10   רֹחַב רָבוּעַ יִהְיֶה הַמִּזְבֵּחַ וְשָׁלֹשׁ אַמּוֹת קֹמָתוֹ׃ וְעָשִׂיתָ
11   קַרְנֹתָיו עַל אַרְבַּע פִּנֹּתָיו מִמֶּנּוּ תִּהְיֶיןָ קַרְנֹתָיו וְצִפִּיתָ
12   אֹתוֹ נְחֹשֶׁת׃ וְעָשִׂיתָ סִּירֹתָיו לְדַשְּׁנוֹ וְיָעָיו וּמִזְרְקֹתָיו
13   וּמִזְלְגֹתָיו וּמַחְתֹּתָיו לְכָל־כֵּלָיו תַּעֲשֶׂה נְחֹשֶׁת׃ וְעָשִׂיתָ
14   לּוֹ מִכְבָּר מַעֲשֵׂה רֶשֶׁת נְחֹשֶׁת וְעָשִׂיתָ עַל־הָרֶשֶׁת
15   אַרְבַּע טַבְּעֹת נְחֹשֶׁת עַל אַרְבַּע קְצוֹתָיו׃ וְנָתַתָּה
16   אֹתָהּ תַּחַת כַּרְכֹּב הַמִּזְבֵּחַ מִלְּמָטָּה וְהָיְתָה הָרֶשֶׁת
17   עַד חֲצִי הַמִּזְבֵּחַ׃ וְעָשִׂיתָ בַדִּים לַמִּזְבֵּחַ בַּדֵּי עֲצֵי
18   שִׁטִּים וְצִפִּיתָ אֹתָם נְחֹשֶׁת׃ וְהוּבָא אֶת־בַּדָּיו בַּטַּבָּעֹת
19   וְהָיוּ הַבַּדִּים עַל־שְׁתֵּי צַלְעֹת הַמִּזְבֵּחַ בִּשְׂאֵת אֹתוֹ׃
20   נְבוּב לֻחֹת תַּעֲשֶׂה אֹתוֹ כַּאֲשֶׁר הֶרְאָה אֹתְךָ בָּהָר כֵּן
21   יַעֲשׂוּ׃ [ס]              וְעָשִׂיתָ אֵת חֲצַר הַמִּשְׁכָּן
22   לִפְאַת נֶגֶב־תֵּימָנָה קְלָעִים לֶחָצֵר שֵׁשׁ מָשְׁזָר מֵאָה
23   בָאַמָּה אֹרֶךְ לַפֵּאָה הָאֶחָת׃ וְעַמֻּדָיו עֶשְׂרִים וְאַדְנֵיהֶם
24   עֶשְׂרִים נְחֹשֶׁת וָוֵי הָעַמֻּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃ וְכֵן
25   לִפְאַת צָפוֹן בָּאֹרֶךְ קְלָעִים מֵאָה אֹרֶךְ וְעַמֻּדָו עֶשְׂרִים
26   וְאַדְנֵיהֶם עֶשְׂרִים נְחֹשֶׁת וָוֵי הָעַמֻּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃
27   וְרֹחַב הֶחָצֵר לִפְאַת־יָם קְלָעִים חֲמִשִּׁים אַמָּה עַמֻּדֵיהֶם
28   עֲשָׂרָה וְאַדְנֵיהֶם עֲשָׂרָה׃ וְרֹחַב הֶחָצֵר לִפְאַת קֵדְמָה
29   מִזְרָחָה חֲמִשִּׁים אַמָּה׃ וַחֲמֵשׁ עֶשְׂרֵה אַמָּה קְלָעִים
30   לַכָּתֵף עַמֻּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃ וְלַכָּתֵף
31   הַשֵּׁנִית חֲמֵשׁ עֶשְׂרֵה קְלָעִים עַמֻּדֵיהֶם שְׁלֹשָׁה
32   וְאַדְנֵיהֶם שְׁלֹשָׁה׃ וּלְשַׁעַר הֶחָצֵר מָסָךְ  עֶשְׂרִים אַמָּה
33   תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר מַעֲשֵׂה
34   רֹקֵם עַמֻּדֵיהֶם אַרְבָּעָה וְאַדְנֵיהֶם אַרְבָּעָה׃  כָּל־עַמּוּדֵי
35   הֶחָצֵר סָבִיב מְחֻשָּׁקִים כֶּסֶף וָוֵיהֶם כָּסֶף וְאַדְנֵיהֶם
36   נְחֹשֶׁת׃ אֹרֶךְ הֶחָצֵר מֵאָה בָאַמָּה וְרֹחַב  חֲמִשִּׁים
37   בַּחֲמִשִּׁים וְקֹמָה חָמֵשׁ אַמּוֹת שֵׁשׁ מָשְׁזָר וְאַדְנֵיהֶם
38   נְחֹשֶׁת׃ לְכֹל כְּלֵי הַמִּשְׁכָּן בְּכֹל עֲבֹדָתוֹ וְכָל־יְתֵדֹתָיו
39   וְכָל־יִתְדֹת הֶחָצֵר נְחֹשֶׁת׃ [ס]            וְאַתָּה
40   תְּצַוֶּה  אֶת־בְּנֵי יִשְׂרָאֵל וְיִקְחוּ אֵלֶיךָ שֶׁמֶן זַיִת זָךְ
41   כָּתִית לַמָּאוֹר לְהַעֲלֹת נֵר תָּמִיד׃ בְּאֹהֶל מוֹעֵד
42   מִחוּץ לַפָּרֹכֶת אֲשֶׁר עַל־הָעֵדֻת יַעֲרֹךְ אֹתוֹ אַהֲרֹן
