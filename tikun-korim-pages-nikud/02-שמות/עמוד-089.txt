 1  וְהַתּוֹרָה וְהַמִּצְוָה אֲשֶׁר כָּתַבְתִּי לְהוֹרֹתָם׃ וַיָּקָם מֹשֶׁה
 2   וִיהוֹשֻׁעַ מְשָׁרְתוֹ וַיַּעַל מֹשֶׁה אֶל־הַר הָאֱלֹהִים׃ וְאֶל־
 3  הַזְּקֵנִים אָמַר שְׁבוּ־לָנוּ בָזֶה עַד אֲשֶׁר־נָשׁוּב אֲלֵיכֶם
 4   וְהִנֵּה אַהֲרֹן וְחוּר עִמָּכֶם מִי־בַעַל דְּבָרִים יִגַּשׁ אֲלֵהֶם׃
 5    וַיַּעַל מֹשֶׁה אֶל־הָהָר וַיְכַס הֶעָנָן אֶת־הָהָר׃ וַיִּשְׁכֹּן כְּבוֹד־
 6  יְהוָה עַל־הַר סִינַי וַיְכַסֵּהוּ הֶעָנָן שֵׁשֶׁת יָמִים וַיִּקְרָא
 7   אֶל־מֹשֶׁה בַּיּוֹם הַשְּׁבִיעִי מִתּוֹךְ הֶעָנָן׃ וּמַרְאֵה כְּבוֹד
 8   יְהוָה כְּאֵשׁ אֹכֶלֶת בְּרֹאשׁ הָהָר לְעֵינֵי בְּנֵי יִשְׂרָאֵל׃
 9   וַיָּבֹא מֹשֶׁה בְּתוֹךְ הֶעָנָן וַיַּעַל אֶל־הָהָר וַיְהִי מֹשֶׁה
10   בָּהָר אַרְבָּעִים יוֹם וְאַרְבָּעִים לָיְלָה׃ [פ]
11  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
12   וְיִקְחוּ־לִי תְּרוּמָה מֵאֵת כָּל־אִישׁ אֲשֶׁר יִדְּבֶנּוּ לִבּוֹ
13   תִּקְחוּ אֶת־תְּרוּמָתִי׃ וְזֹאת הַתְּרוּמָה אֲשֶׁר תִּקְחוּ מֵאִתָּם
14   זָהָב וָכֶסֶף וּנְחֹשֶׁת׃ וּתְכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי
15   וְשֵׁשׁ וְעִזִּים׃ וְעֹרֹת אֵילִם מְאָדָּמִים וְעֹרֹת תְּחָשִׁים
16   וַעֲצֵי שִׁטִּים׃♢ שֶׁמֶן לַמָּאֹר בְּשָׂמִים לְשֶׁמֶן הַמִּשְׁחָה
17   וְלִקְטֹרֶת הַסַּמִּים׃ אַבְנֵי־שֹׁהַם וְאַבְנֵי מִלֻּאִים לָאֵפֹד
18   וְלַחֹשֶׁן׃ וְעָשׂוּ לִי מִקְדָּשׁ וְשָׁכַנְתִּי בְּתוֹכָם׃ כְּכֹל אֲשֶׁר
19   אֲנִי מַרְאֶה אוֹתְךָ אֵת תַּבְנִית הַמִּשְׁכָּן וְאֵת תַּבְנִית כָּל־
20  כֵּלָיו וְכֵן תַּעֲשׂוּ׃♢ [ס]             וְעָשׂוּ אֲרוֹן
21   עֲצֵי שִׁטִּים אַמָּתַיִם וָחֵצִי אָרְכּוֹ וְאַמָּה וָחֵצִי רָחְבּוֹ וְאַמָּה
22   וָחֵצִי קֹמָתוֹ׃ וְצִפִּיתָ אֹתוֹ זָהָב טָהוֹר מִבַּיִת וּמִחוּץ
23   תְּצַפֶּנּוּ וְעָשִׂיתָ עָלָיו זֵר זָהָב סָבִיב׃ וְיָצַקְתָּ לּוֹ אַרְבַּע
24   טַבְּעֹת זָהָב וְנָתַתָּה עַל אַרְבַּע פַּעֲמֹתָיו וּשְׁתֵּי טַבָּעֹת
25   עַל־צַלְעוֹ הָאֶחָת וּשְׁתֵּי טַבָּעֹת עַל־צַלְעוֹ הַשֵּׁנִית׃
26   וְעָשִׂיתָ בַדֵּי עֲצֵי שִׁטִּים וְצִפִּיתָ אֹתָם זָהָב׃ וְהֵבֵאתָ אֶת־
27  הַבַּדִּים בַּטַּבָּעֹת עַל צַלְעֹת הָאָרֹן לָשֵׂאת אֶת־הָאָרֹן
28   בָּהֶם׃ בְּטַבְּעֹת הָאָרֹן יִהְיוּ הַבַּדִּים לֹא יָסֻרוּ מִמֶּנּוּ׃ וְנָתַתָּ
29   אֶל־הָאָרֹן אֵת הָעֵדֻת אֲשֶׁר אֶתֵּן אֵלֶיךָ׃♢  וְעָשִׂיתָ כַפֹּרֶת
30   זָהָב טָהוֹר אַמָּתַיִם וָחֵצִי אָרְכָּהּ וְאַמָּה וָחֵצִי רָחְבָּהּ׃
31   וְעָשִׂיתָ שְׁנַיִם כְּרֻבִים זָהָב מִקְשָׁה תַּעֲשֶׂה אֹתָם מִשְּׁנֵי
32   קְצוֹת הַכַּפֹּרֶת׃ וַעֲשֵׂה כְּרוּב אֶחָד מִקָּצָה מִזֶּה וּכְרוּב־
33  אֶחָד מִקָּצָה מִזֶּה מִן־הַכַּפֹּרֶת תַּעֲשׂוּ אֶת־הַכְּרֻבִים עַל־
34  שְׁנֵי קְצוֹתָיו׃ וְהָיוּ הַכְּרֻבִים פֹּרְשֵׂי כְנָפַיִם לְמַעְלָה
35   סֹכְכִים בְּכַנְפֵיהֶם עַל־הַכַּפֹּרֶת וּפְנֵיהֶם אִישׁ אֶל־אָחִיו
36   אֶל־הַכַּפֹּרֶת יִהְיוּ פְּנֵי הַכְּרֻבִים׃ וְנָתַתָּ אֶת־הַכַּפֹּרֶת עַל־
37  הָאָרֹן מִלְמָעְלָה וְאֶל־הָאָרֹן תִּתֵּן אֶת־הָעֵדֻת אֲשֶׁר אֶתֵּן
38   אֵלֶיךָ׃ וְנוֹעַדְתִּי לְךָ שָׁם וְדִבַּרְתִּי אִתְּךָ מֵעַל הַכַּפֹּרֶת
39   מִבֵּין שְׁנֵי הַכְּרֻבִים אֲשֶׁר עַל־אֲרוֹן הָעֵדֻת אֵת כָּל־
40  אֲשֶׁר אֲצַוֶּה אוֹתְךָ אֶל־בְּנֵי יִשְׂרָאֵל׃ [פ]
41   וְעָשִׂיתָ שֻׁלְחָן עֲצֵי שִׁטִּים אַמָּתַיִם אָרְכּוֹ וְאַמָּה
42   רָחְבּוֹ וְאַמָּה וָחֵצִי קֹמָתוֹ׃ וְצִפִּיתָ אֹתוֹ זָהָב טָהוֹר
