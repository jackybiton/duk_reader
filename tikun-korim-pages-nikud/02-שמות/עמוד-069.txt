 1  וַיִּצְעַק מֹשֶׁה אֶל־יְהוָה עַל־דְּבַר הַצְפַרְדְּעִים אֲשֶׁר־
 2  שָׂם לְפַרְעֹה׃ וַיַּעַשׂ יְהוָה כִּדְבַר מֹשֶׁה וַיָּמֻתוּ
 3   הַצְפַרְדְּעִים מִן־הַבָּתִּים מִן־הַחֲצֵרֹת וּמִן־הַשָּׂדֹת׃
 4   וַיִּצְבְּרוּ אֹתָם חֳמָרִם חֳמָרִם וַתִּבְאַשׁ הָאָרֶץ׃ וַיַּרְא
 5   פַּרְעֹה כִּי הָיְתָה הָרְוָחָה וְהַכְבֵּד אֶת־לִבּוֹ וְלֹא שָׁמַע
 6   אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר יְהוָה׃ [ס]             וַיֹּאמֶר
 7   יְהוָה אֶל־מֹשֶׁה אֱמֹר אֶל־אַהֲרֹן נְטֵה אֶת־מַטְּךָ וְהַךְ
 8   אֶת־עֲפַר הָאָרֶץ וְהָיָה לְכִנִּם בְּכָל־אֶרֶץ מִצְרָיִם׃
 9   וַיַּעֲשׂוּ־כֵן וַיֵּט אַהֲרֹן אֶת־יָדוֹ בְמַטֵּהוּ וַיַּךְ אֶת־עֲפַר
10   הָאָרֶץ וַתְּהִי הַכִּנָּם בָּאָדָם וּבַבְּהֵמָה כָּל־עֲפַר הָאָרֶץ
11   הָיָה כִנִּים בְּכָל־אֶרֶץ מִצְרָיִם׃ וַיַּעֲשׂוּ־כֵן הַחַרְטֻמִּים
12   בְּלָטֵיהֶם לְהוֹצִיא אֶת־הַכִּנִּים וְלֹא יָכֹלוּ וַתְּהִי הַכִּנָּם
13   בָּאָדָם וּבַבְּהֵמָה׃ וַיֹּאמְרוּ הַחַרְטֻמִּם אֶל־פַּרְעֹה אֶצְבַּע
14   אֱלֹהִים הִוא וַיֶּחֱזַק לֵב־פַּרְעֹה וְלֹא־שָׁמַע אֲלֵהֶם
15   כַּאֲשֶׁר דִּבֶּר יְהוָה׃ [ס]             וַיֹּאמֶר יְהוָה
16   אֶל־מֹשֶׁה הַשְׁכֵּם בַּבֹּקֶר וְהִתְיַצֵּב לִפְנֵי פַרְעֹה הִנֵּה
17   יוֹצֵא הַמָּיְמָה וְאָמַרְתָּ אֵלָיו כֹּה אָמַר יְהוָה שַׁלַּח עַמִּי
18   וְיַעַבְדֻנִי׃ כִּי אִם־אֵינְךָ מְשַׁלֵּחַ אֶת־עַמִּי הִנְנִי מַשְׁלִיחַ
19   בְּךָ וּבַעֲבָדֶיךָ וּבְעַמְּךָ וּבְבָתֶּיךָ אֶת־הֶעָרֹב וּמָלְאוּ
20   בָּתֵּי מִצְרַיִם אֶת־הֶעָרֹב וְגַם הָאֲדָמָה אֲשֶׁר־הֵם עָלֶיהָ׃
21   וְהִפְלֵיתִי בַיּוֹם הַהוּא אֶת־אֶרֶץ גֹּשֶׁן אֲשֶׁר עַמִּי עֹמֵד
22   עָלֶיהָ לְבִלְתִּי הֱיוֹת־שָׁם עָרֹב לְמַעַן תֵּדַע כִּי אֲנִי
23   יְהוָה בְּקֶרֶב הָאָרֶץ׃  וְשַׂמְתִּי פְדֻת בֵּין עַמִּי וּבֵין עַמֶּךָ
24   לְמָחָר יִהְיֶה הָאֹת הַזֶּה׃ וַיַּעַשׂ יְהוָה כֵּן וַיָּבֹא עָרֹב
25   כָּבֵד בֵּיתָה פַרְעֹה וּבֵית עֲבָדָיו וּבְכָל־אֶרֶץ מִצְרַיִם
26   תִּשָּׁחֵת הָאָרֶץ מִפְּנֵי הֶעָרֹב׃ וַיִּקְרָא פַרְעֹה אֶל־מֹשֶׁה
27   וּלְאַהֲרֹן וַיֹּאמֶר לְכוּ זִבְחוּ לֵאלֹהֵיכֶם בָּאָרֶץ׃ וַיֹּאמֶר
28   מֹשֶׁה לֹא נָכוֹן לַעֲשׂוֹת כֵּן כִּי תּוֹעֲבַת מִצְרַיִם נִזְבַּח
29   לַיהוָה אֱלֹהֵינוּ הֵן נִזְבַּח אֶת־תּוֹעֲבַת מִצְרַיִם לְעֵינֵיהֶם
30   וְלֹא יִסְקְלֻנוּ׃ דֶּרֶךְ שְׁלֹשֶׁת יָמִים נֵלֵךְ בַּמִּדְבָּר וְזָבַחְנוּ
31   לַיהוָה אֱלֹהֵינוּ כַּאֲשֶׁר יֹאמַר אֵלֵינוּ׃ וַיֹּאמֶר פַּרְעֹה
32   אָנֹכִי אֲשַׁלַּח אֶתְכֶם וּזְבַחְתֶּם לַיהוָה אֱלֹהֵיכֶם בַּמִּדְבָּר
33   רַק הַרְחֵק לֹא־תַרְחִיקוּ לָלֶכֶת הַעְתִּירוּ בַּעֲדִי׃ וַיֹּאמֶר
34   מֹשֶׁה הִנֵּה אָנֹכִי יוֹצֵא מֵעִמָּךְ וְהַעְתַּרְתִּי אֶל־יְהוָה וְסָר
35   הֶעָרֹב מִפַּרְעֹה מֵעֲבָדָיו וּמֵעַמּוֹ מָחָר רַק אַל־יֹסֵף
36   פַּרְעֹה הָתֵל לְבִלְתִּי שַׁלַּח אֶת־הָעָם לִזְבֹּחַ לַיהוָה׃
37   וַיֵּצֵא מֹשֶׁה מֵעִם פַּרְעֹה וַיֶּעְתַּר אֶל־יְהוָה׃ וַיַּעַשׂ יְהוָה
38   כִּדְבַר מֹשֶׁה וַיָּסַר הֶעָרֹב מִפַּרְעֹה מֵעֲבָדָיו וּמֵעַמּוֹ
39   לֹא נִשְׁאַר אֶחָד׃ וַיַּכְבֵּד פַּרְעֹה אֶת־לִבּוֹ גַּם בַּפַּעַם
40   הַזֹּאת וְלֹא שִׁלַּח אֶת־הָעָם׃ [פ]
41   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה בֹּא אֶל־פַּרְעֹה וְדִבַּרְתָּ אֵלָיו
42   כֹּה־אָמַר יְהוָה אֱלֹהֵי הָעִבְרִים שַׁלַּח אֶת־עַמִּי
