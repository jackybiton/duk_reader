 1  וְעָשִׂיתָ לּוֹ זֵר זָהָב סָבִיב׃ וְעָשִׂיתָ לּוֹ מִסְגֶּרֶת טֹפַח
 2   סָבִיב וְעָשִׂיתָ זֵר־זָהָב לְמִסְגַּרְתּוֹ סָבִיב׃ וְעָשִׂיתָ לּוֹ
 3   אַרְבַּע טַבְּעֹת זָהָב וְנָתַתָּ אֶת־הַטַּבָּעֹת עַל אַרְבַּע
 4   הַפֵּאֹת אֲשֶׁר לְאַרְבַּע רַגְלָיו׃ לְעֻמַּת הַמִּסְגֶּרֶת תִּהְיֶיןָ
 5   הַטַּבָּעֹת לְבָתִּים לְבַדִּים לָשֵׂאת אֶת־הַשֻּׁלְחָן׃ וְעָשִׂיתָ
 6   אֶת־הַבַּדִּים עֲצֵי שִׁטִּים וְצִפִּיתָ אֹתָם זָהָב וְנִשָּׂא־בָם אֶת־
 7  הַשֻּׁלְחָן׃ וְעָשִׂיתָ קְּעָרֹתָיו וְכַפֹּתָיו וּקְשׂוֹתָיו וּמְנַקִּיֹּתָיו
 8   אֲשֶׁר יֻסַּךְ בָּהֵן זָהָב טָהוֹר תַּעֲשֶׂה אֹתָם׃ וְנָתַתָּ עַל־
 9  הַשֻּׁלְחָן לֶחֶם פָּנִים לְפָנַי תָּמִיד׃ [פ]
10    וְעָשִׂיתָ מְנֹרַת זָהָב טָהוֹר מִקְשָׁה תֵּעָשֶׂה הַמְּנוֹרָה
11   יְרֵכָהּ וְקָנָהּ גְּבִיעֶיהָ כַּפְתֹּרֶיהָ וּפְרָחֶיהָ מִמֶּנָּה יִהְיוּ׃
12   וְשִׁשָּׁה קָנִים יֹצְאִים מִצִּדֶּיהָ שְׁלֹשָׁה  קְנֵי מְנֹרָה מִצִּדָּהּ
13   הָאֶחָד וּשְׁלֹשָׁה קְנֵי מְנֹרָה מִצִּדָּהּ הַשֵּׁנִי׃ שְׁלֹשָׁה
14   גְבִעִים מְשֻׁקָּדִים בַּקָּנֶה הָאֶחָד כַּפְתֹּר וָפֶרַח וּשְׁלֹשָׁה
15   גְבִעִים מְשֻׁקָּדִים בַּקָּנֶה הָאֶחָד כַּפְתֹּר וָפָרַח כֵּן
16   לְשֵׁשֶׁת הַקָּנִים הַיֹּצְאִים מִן־הַמְּנֹרָה׃ וּבַמְּנֹרָה אַרְבָּעָה
17   גְבִעִים מְשֻׁקָּדִים כַּפְתֹּרֶיהָ וּפְרָחֶיהָ׃ וְכַפְתֹּר תַּחַת שְׁנֵי
18   הַקָּנִים מִמֶּנָּה וְכַפְתֹּר תַּחַת שְׁנֵי הַקָּנִים מִמֶּנָּה וְכַפְתֹּר
19   תַּחַת־שְׁנֵי הַקָּנִים מִמֶּנָּה לְשֵׁשֶׁת הַקָּנִים הַיֹּצְאִים מִן־
20  הַמְּנֹרָה׃ כַּפְתֹּרֵיהֶם וּקְנֹתָם מִמֶּנָּה יִהְיוּ כֻּלָּהּ מִקְשָׁה
21   אַחַת זָהָב טָהוֹר׃ וְעָשִׂיתָ אֶת־נֵרֹתֶיהָ שִׁבְעָה וְהֶעֱלָה
22   אֶת־נֵרֹתֶיהָ וְהֵאִיר עַל־עֵבֶר פָּנֶיהָ׃ וּמַלְקָחֶיהָ וּמַחְתֹּתֶיהָ
23   זָהָב טָהוֹר׃ כִּכָּר זָהָב טָהוֹר יַעֲשֶׂה אֹתָהּ אֵת כָּל־
24  הַכֵּלִים הָאֵלֶּה׃ וּרְאֵה וַעֲשֵׂה בְּתַבְנִיתָם אֲשֶׁר־אַתָּה
25   מָרְאֶה בָּהָר׃ [ס]             וְאֶת־הַמִּשְׁכָּן
26   תַּעֲשֶׂה עֶשֶׂר יְרִיעֹת שֵׁשׁ מָשְׁזָר וּתְכֵלֶת וְאַרְגָּמָן
27   וְתֹלַעַת שָׁנִי כְּרֻבִים מַעֲשֵׂה חֹשֵׁב תַּעֲשֶׂה אֹתָם׃ אֹרֶךְ
28   הַיְרִיעָה הָאַחַת שְׁמֹנֶה וְעֶשְׂרִים בָּאַמָּה וְרֹחַב אַרְבַּע
29   בָּאַמָּה הַיְרִיעָה הָאֶחָת מִדָּה אַחַת לְכָל־הַיְרִיעֹת׃
30   חֲמֵשׁ הַיְרִיעֹת תִּהְיֶיןָ חֹבְרֹת אִשָּׁה אֶל־אֲחֹתָהּ וְחָמֵשׁ
31   יְרִיעֹת חֹבְרֹת אִשָּׁה אֶל־אֲחֹתָהּ׃ וְעָשִׂיתָ לֻלְאֹת תְּכֵלֶת
32   עַל שְׂפַת הַיְרִיעָה הָאֶחָת מִקָּצָה בַּחֹבָרֶת וְכֵן תַּעֲשֶׂה
33   בִּשְׂפַת הַיְרִיעָה הַקִּיצוֹנָה בַּמַּחְבֶּרֶת הַשֵּׁנִית׃ חֲמִשִּׁים
34   לֻלָאֹת תַּעֲשֶׂה בַּיְרִיעָה הָאֶחָת וַחֲמִשִּׁים לֻלָאֹת
35   תַּעֲשֶׂה בִּקְצֵה הַיְרִיעָה אֲשֶׁר בַּמַּחְבֶּרֶת הַשֵּׁנִית
36   מַקְבִּילֹת הַלֻּלָאֹת אִשָּׁה אֶל־אֲחֹתָהּ׃ וְעָשִׂיתָ חֲמִשִּׁים
37   קַרְסֵי זָהָב וְחִבַּרְתָּ אֶת־הַיְרִיעֹת אִשָּׁה אֶל־אֲחֹתָהּ
38   בַּקְּרָסִים וְהָיָה הַמִּשְׁכָּן אֶחָד׃ וְעָשִׂיתָ יְרִיעֹת עִזִּים
39   לְאֹהֶל עַל־הַמִּשְׁכָּן עַשְׁתֵּי־עֶשְׂרֵה יְרִיעֹת תַּעֲשֶׂה
40   אֹתָם׃ אֹרֶךְ  הַיְרִיעָה הָאַחַת שְׁלֹשִׁים בָּאַמָּה וְרֹחַב
41   אַרְבַּע בָּאַמָּה הַיְרִיעָה הָאֶחָת מִדָּה אַחַת לְעַשְׁתֵּי
42   עֶשְׂרֵה יְרִיעֹת׃ וְחִבַּרְתָּ אֶת־חֲמֵשׁ הַיְרִיעֹת לְבָד
