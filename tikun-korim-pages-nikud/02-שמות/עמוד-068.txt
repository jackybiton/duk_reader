 1  וְנִצַּבְתָּ לִקְרָאתוֹ עַל־שְׂפַת הַיְאֹר וְהַמַּטֶּה אֲשֶׁר־נֶהְפַּךְ
 2   לְנָחָשׁ תִּקַּח בְּיָדֶךָ׃ וְאָמַרְתָּ אֵלָיו יְהוָה אֱלֹהֵי הָעִבְרִים
 3   שְׁלָחַנִי אֵלֶיךָ לֵאמֹר שַׁלַּח אֶת־עַמִּי וְיַעַבְדֻנִי בַּמִּדְבָּר
 4   וְהִנֵּה לֹא־שָׁמַעְתָּ עַד־כֹּה׃ כֹּה אָמַר יְהוָה בְּזֹאת תֵּדַע
 5   כִּי אֲנִי יְהוָה הִנֵּה אָנֹכִי מַכֶּה  בַּמַּטֶּה אֲשֶׁר־בְּיָדִי עַל־
 6  הַמַּיִם אֲשֶׁר בַּיְאֹר וְנֶהֶפְכוּ לְדָם׃ וְהַדָּגָה אֲשֶׁר־בַּיְאֹר
 7   תָּמוּת וּבָאַשׁ הַיְאֹר וְנִלְאוּ מִצְרַיִם לִשְׁתּוֹת מַיִם מִן־
 8  הַיְאֹר׃ [ס]             וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה
 9   אֱמֹר אֶל־אַהֲרֹן קַח מַטְּךָ וּנְטֵה־יָדְךָ עַל־מֵימֵי מִצְרַיִם
10   עַל־נַהֲרֹתָם  עַל־יְאֹרֵיהֶם וְעַל־אַגְמֵיהֶם וְעַל כָּל־מִקְוֵה
11   מֵימֵיהֶם וְיִהְיוּ־דָם וְהָיָה דָם בְּכָל־אֶרֶץ מִצְרַיִם
12   וּבָעֵצִים וּבָאֲבָנִים׃ וַיַּעֲשׂוּ־כֵן מֹשֶׁה וְאַהֲרֹן כַּאֲשֶׁר  צִוָּה
13   יְהוָה וַיָּרֶם בַּמַּטֶּה וַיַּךְ אֶת־הַמַּיִם אֲשֶׁר בַּיְאֹר לְעֵינֵי
14   פַרְעֹה וּלְעֵינֵי עֲבָדָיו וַיֵּהָפְכוּ כָּל־הַמַּיִם אֲשֶׁר־בַּיְאֹר
15   לְדָם׃ וְהַדָּגָה אֲשֶׁר־בַּיְאֹר מֵתָה וַיִּבְאַשׁ הַיְאֹר וְלֹא־
16  יָכְלוּ מִצְרַיִם לִשְׁתּוֹת מַיִם מִן־הַיְאֹר וַיְהִי הַדָּם בְּכָל־
17  אֶרֶץ מִצְרָיִם׃ וַיַּעֲשׂוּ־כֵן חַרְטֻמֵּי מִצְרַיִם בְּלָטֵיהֶם
18   וַיֶּחֱזַק לֵב־פַּרְעֹה וְלֹא־שָׁמַע אֲלֵהֶם כַּאֲשֶׁר דִּבֶּר
19   יְהוָה׃ וַיִּפֶן פַּרְעֹה וַיָּבֹא אֶל־בֵּיתוֹ וְלֹא־שָׁת לִבּוֹ גַּם־
20  לָזֹאת׃ וַיַּחְפְּרוּ כָל־מִצְרַיִם סְבִיבֹת הַיְאֹר מַיִם לִשְׁתּוֹת
21   כִּי לֹא יָכְלוּ לִשְׁתֹּת מִמֵּימֵי הַיְאֹר׃ וַיִּמָּלֵא שִׁבְעַת
22   יָמִים אַחֲרֵי הַכּוֹת־יְהוָה אֶת־הַיְאֹר׃ [פ]
23   וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה בֹּא אֶל־פַּרְעֹה וְאָמַרְתָּ אֵלָיו
24   כֹּה אָמַר יְהוָה שַׁלַּח אֶת־עַמִּי וְיַעַבְדֻנִי׃ וְאִם־מָאֵן אַתָּה
25   לְשַׁלֵּחַ הִנֵּה אָנֹכִי נֹגֵף אֶת־כָּל־גְּבוּלְךָ בַּצְפַרְדְּעִים׃
26   וְשָׁרַץ הַיְאֹר צְפַרְדְּעִים וְעָלוּ וּבָאוּ בְּבֵיתֶךָ וּבַחֲדַר
27   מִשְׁכָּבְךָ וְעַל־מִטָּתֶךָ וּבְבֵית עֲבָדֶיךָ וּבְעַמֶּךָ וּבְתַנּוּרֶיךָ
28   וּבְמִשְׁאֲרוֹתֶיךָ׃ וּבְכָה וּבְעַמְּךָ וּבְכָל־עֲבָדֶיךָ יַעֲלוּ
29   הַצְפַרְדְּעִים׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה אֱמֹר אֶל־אַהֲרֹן
30   נְטֵה אֶת־יָדְךָ בְּמַטֶּךָ עַל־הַנְּהָרֹת עַל־הַיְאֹרִים וְעַל־
31  הָאֲגַמִּים וְהַעַל אֶת־הַצְפַרְדְּעִים עַל־אֶרֶץ מִצְרָיִם׃
32   וַיֵּט אַהֲרֹן אֶת־יָדוֹ עַל מֵימֵי מִצְרָיִם וַתַּעַל הַצְּפַרְדֵּעַ
33   וַתְּכַס אֶת־אֶרֶץ מִצְרָיִם׃ וַיַּעֲשׂוּ־כֵן הַחַרְטֻמִּים בְּלָטֵיהֶם
34   וַיַּעֲלוּ אֶת־הַצְפַרְדְּעִים עַל־אֶרֶץ מִצְרָיִם׃ וַיִּקְרָא פַרְעֹה
35   לְמֹשֶׁה וּלְאַהֲרֹן וַיֹּאמֶר הַעְתִּירוּ אֶל־יְהוָה וְיָסֵר
36   הַצְפַרְדְּעִים מִמֶּנִּי וּמֵעַמִּי וַאֲשַׁלְּחָה אֶת־הָעָם וְיִזְבְּחוּ
37   לַיהוָה׃ וַיֹּאמֶר מֹשֶׁה לְפַרְעֹה הִתְפָּאֵר עָלַי לְמָתַי
38   אַעְתִּיר לְךָ וְלַעֲבָדֶיךָ וּלְעַמְּךָ לְהַכְרִית הַצְפַרְדְּעִים
39   מִמְּךָ וּמִבָּתֶּיךָ רַק בַּיְאֹר תִּשָּׁאַרְנָה׃ וַיֹּאמֶר לְמָחָר
40   וַיֹּאמֶר כִּדְבָרְךָ לְמַעַן תֵּדַע כִּי־אֵין כַּיהוָה אֱלֹהֵינוּ׃
41    וְסָרוּ הַצְפַרְדְּעִים מִמְּךָ וּמִבָּתֶּיךָ וּמֵעֲבָדֶיךָ וּמֵעַמֶּךָ
42   רַק בַּיְאֹר תִּשָּׁאַרְנָה׃ וַיֵּצֵא מֹשֶׁה וְאַהֲרֹן מֵעִם פַּרְעֹה
