1וְאַדְנֵיהֶםאַרְבָּעָהנְחֹשֶׁתוָוֵיהֶםכֶּסֶףוְצִפּוּירָאשֵׁיהֶם
2וַחֲשֻׁקֵיהֶםכָּסֶף׃וְכָל־הַיְתֵדֹתלַמִּשְׁכָּןוְלֶחָצֵרסָבִיב
3נְחֹשֶׁת׃אֵלֶּה פְקוּדֵי
4הַמִּשְׁכָּןמִשְׁכַּןהָעֵדֻתאֲשֶׁרפֻּקַּדעַל־פִּימֹשֶׁה
5עֲבֹדַתהַלְוִיִּםבְּיַדאִיתָמָרבֶּן־אַהֲרֹןהַכֹּהֵן׃וּבְצַלְאֵל
6בֶּן־אוּרִיבֶן־חוּרלְמַטֵּהיְהוּדָהעָשָׂהאֵתכָּל־אֲשֶׁר־
7צִוָּהיְהוָהאֶת־מֹשֶׁה׃וְאִתּוֹאָהֳלִיאָבבֶּן־אֲחִיסָמָךְ
8לְמַטֵּה־דָןחָרָשׁוְחֹשֵׁבוְרֹקֵםבַּתְּכֵלֶתוּבָאַרְגָּמָן
9וּבְתוֹלַעַת הַשָּׁנִי וּבַשֵּׁשׁ׃♢כָּל־
10הַזָּהָבהֶעָשׂוּילַמְּלָאכָהבְּכֹלמְלֶאכֶתהַקֹּדֶשׁוַיְהִי
11זְהַבהַתְּנוּפָהתֵּשַׁעוְעֶשְׂרִיםכִּכָּרוּשְׁבַעמֵאוֹת
12וּשְׁלֹשִׁיםשֶׁקֶלבְּשֶׁקֶלהַקֹּדֶשׁ׃וְכֶסֶףפְּקוּדֵי
13הָעֵדָהמְאַתכִּכָּרוְאֶלֶףוּשְׁבַעמֵאוֹתוַחֲמִשָּׁה
14וְשִׁבְעִיםשֶׁקֶלבְּשֶׁקֶלהַקֹּדֶשׁ׃בֶּקַעלַגֻּלְגֹּלֶת
15מַחֲצִיתהַשֶּׁקֶלבְּשֶׁקֶלהַקֹּדֶשׁלְכֹלהָעֹבֵרעַל־
16הַפְּקֻדִיםמִבֶּןעֶשְׂרִיםשָׁנָהוָמַעְלָהלְשֵׁשׁ־מֵאוֹת
17אֶלֶףוּשְׁלֹשֶׁתאֲלָפִיםוַחֲמֵשׁמֵאוֹתוַחֲמִשִּׁים׃וַיְהִי
18מְאַתכִּכַּרהַכֶּסֶףלָצֶקֶתאֵתאַדְנֵיהַקֹּדֶשׁוְאֵת
19אַדְנֵיהַפָּרֹכֶתמְאַתאֲדָנִיםלִמְאַתהַכִּכָּרכִּכָּר
20לָאָדֶן׃♢וְאֶת־הָאֶלֶףוּשְׁבַעהַמֵּאוֹתוַחֲמִשָּׁהוְשִׁבְעִים
21עָשָׂהוָוִיםלָעַמּוּדִיםוְצִפָּהרָאשֵׁיהֶםוְחִשַּׁקאֹתָם׃
22וּנְחֹשֶׁתהַתְּנוּפָהשִׁבְעִיםכִּכָּרוְאַלְפַּיִםוְאַרְבַּע־
23מֵאוֹתשָׁקֶל׃וַיַּעַשׂבָּהּאֶת־אַדְנֵיפֶּתַחאֹהֶלמוֹעֵד
24וְאֵתמִזְבַּחהַנְּחֹשֶׁתוְאֶת־מִכְבַּרהַנְּחֹשֶׁתאֲשֶׁר־לוֹ
25וְאֵתכָּל־כְּלֵיהַמִּזְבֵּחַ׃וְאֶת־אַדְנֵיהֶחָצֵרסָבִיבוְאֶת־
26אַדְנֵישַׁעַרהֶחָצֵרוְאֵתכָּל־יִתְדֹתהַמִּשְׁכָּןוְאֶת־
27כָּל־יִתְדֹתהֶחָצֵרסָבִיב׃וּמִן־הַתְּכֵלֶתוְהָאַרְגָּמָן
28וְתוֹלַעַתהַשָּׁנִיעָשׂוּבִגְדֵי־שְׂרָדלְשָׁרֵתבַּקֹּדֶשׁ
29וַיַּעֲשׂוּאֶת־בִּגְדֵיהַקֹּדֶשׁאֲשֶׁרלְאַהֲרֹןכַּאֲשֶׁר

32וְשֵׁשׁמָשְׁזָר׃וַיְרַקְּעוּאֶת־פַּחֵיהַזָּהָבוְקִצֵּץפְּתִילִם
33לַעֲשׂוֹתבְּתוֹךְהַתְּכֵלֶתוּבְתוֹךְהָאַרְגָּמָןוּבְתוֹךְ
34תּוֹלַעַתהַשָּׁנִיוּבְתוֹךְהַשֵּׁשׁמַעֲשֵׂהחֹשֵׁב׃כְּתֵפֹת
35עָשׂוּ־לוֹחֹבְרֹתעַל־שְׁנֵיקְצוֹתָיוחֻבָּר׃וְחֵשֶׁב
36אֲפֻדָּתוֹאֲשֶׁרעָלָיומִמֶּנּוּהוּאכְּמַעֲשֵׂהוּזָהָבתְּכֵלֶת
37וְאַרְגָּמָןוְתוֹלַעַתשָׁנִיוְשֵׁשׁמָשְׁזָרכַּאֲשֶׁרצִוָּה
38יְהוָה אֶת־מֹשֶׁה׃וַיַּעֲשׂוּ
39אֶת־אַבְנֵיהַשֹּׁהַםמֻסַבֹּתמִשְׁבְּצֹתזָהָבמְפֻתָּחֹת
40פִּתּוּחֵיחוֹתָםעַל־שְׁמוֹתבְּנֵייִשְׂרָאֵל׃וַיָּשֶׂםאֹתָם
