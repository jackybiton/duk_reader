 1  וָאֹמַר לָהֶם לְמִי זָהָב הִתְפָּרָקוּ וַיִּתְּנוּ־לִי וָאַשְׁלִכֵהוּ
 2   בָאֵשׁ וַיֵּצֵא הָעֵגֶל הַזֶּה׃ וַיַּרְא מֹשֶׁה אֶת־הָעָם כִּי פָרֻעַ
 3   הוּא כִּי־פְרָעֹה אַהֲרֹן לְשִׁמְצָה בְּקָמֵיהֶם׃ וַיַּעֲמֹד מֹשֶׁה
 4   בְּשַׁעַר הַמַּחֲנֶה וַיֹּאמֶר מִי לַיהוָה אֵלָי וַיֵּאָסְפוּ אֵלָיו
 5   כָּל־בְּנֵי לֵוִי׃ וַיֹּאמֶר לָהֶם כֹּה־אָמַר יְהוָה אֱלֹהֵי יִשְׂרָאֵל
 6   שִׂימוּ אִישׁ־חַרְבּוֹ עַל־יְרֵכוֹ עִבְרוּ וָשׁוּבוּ מִשַּׁעַר לָשַׁעַר
 7   בַּמַּחֲנֶה וְהִרְגוּ אִישׁ־אֶת־אָחִיו וְאִישׁ אֶת־רֵעֵהוּ
 8   וְאִישׁ אֶת־קְרֹבוֹ׃ וַיַּעֲשׂוּ בְנֵי־לֵוִי כִּדְבַר מֹשֶׁה וַיִּפֹּל
 9   מִן־הָעָם בַּיּוֹם הַהוּא כִּשְׁלֹשֶׁת אַלְפֵי אִישׁ׃ וַיֹּאמֶר
10   מֹשֶׁה מִלְאוּ יֶדְכֶם הַיּוֹם לַיהוָה כִּי אִישׁ בִּבְנוֹ וּבְאָחִיו
11   וְלָתֵת עֲלֵיכֶם הַיּוֹם בְּרָכָה׃ וַיְהִי מִמָּחֳרָת וַיֹּאמֶר מֹשֶׁה
12   אֶל־הָעָם אַתֶּם חֲטָאתֶם חֲטָאָה גְדֹלָה וְעַתָּה אֶעֱלֶה
13   אֶל־יְהוָה אוּלַי אֲכַפְּרָה בְּעַד חַטַּאתְכֶם׃ וַיָּשָׁב מֹשֶׁה
14   אֶל־יְהוָה וַיֹּאמַר אָנָּא חָטָא הָעָם הַזֶּה חֲטָאָה גְדֹלָה
15   וַיַּעֲשׂוּ לָהֶם אֱלֹהֵי זָהָב׃ וְעַתָּה אִם־תִּשָּׂא חַטָּאתָם
16   וְאִם־אַיִן מְחֵנִי נָא מִסִּפְרְךָ אֲשֶׁר כָּתָבְתָּ׃ וַיֹּאמֶר יְהוָה
17   אֶל־מֹשֶׁה מִי אֲשֶׁר חָטָא־לִי אֶמְחֶנּוּ מִסִּפְרִי׃ וְעַתָּה
18   לֵךְ  נְחֵה אֶת־הָעָם אֶל אֲשֶׁר־דִּבַּרְתִּי לָךְ הִנֵּה מַלְאָכִי
19   יֵלֵךְ לְפָנֶיךָ וּבְיוֹם פָּקְדִי וּפָקַדְתִּי עֲלֵהֶם חַטָּאתָם׃
20   וַיִּגֹּף יְהוָה אֶת־הָעָם עַל אֲשֶׁר עָשׂוּ אֶת־הָעֵגֶל אֲשֶׁר
21   עָשָׂה אַהֲרֹן׃ [ס]             וַיְדַבֵּר יְהוָה אֶל־
22  מֹשֶׁה לֵךְ עֲלֵה מִזֶּה אַתָּה וְהָעָם אֲשֶׁר הֶעֱלִיתָ מֵאֶרֶץ
23   מִצְרָיִם אֶל־הָאָרֶץ אֲשֶׁר נִשְׁבַּעְתִּי לְאַבְרָהָם לְיִצְחָק
24   וּלְיַעֲקֹב לֵאמֹר לְזַרְעֲךָ אֶתְּנֶנָּה׃ וְשָׁלַחְתִּי לְפָנֶיךָ
25   מַלְאָךְ וְגֵרַשְׁתִּי אֶת־הַכְּנַעֲנִי הָאֱמֹרִי וְהַחִתִּי וְהַפְּרִזִּי
26   הַחִוִּי וְהַיְבוּסִי׃ אֶל־אֶרֶץ זָבַת חָלָב וּדְבָשׁ כִּי לֹא
27   אֶעֱלֶה בְּקִרְבְּךָ כִּי עַם־קְשֵׁה־עֹרֶף אַתָּה פֶּן־אֲכֶלְךָ
28   בַּדָּרֶךְ׃ וַיִּשְׁמַע הָעָם אֶת־הַדָּבָר הָרָע הַזֶּה וַיִּתְאַבָּלוּ
29   וְלֹא־שָׁתוּ אִישׁ עֶדְיוֹ עָלָיו׃ וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה
30   אֱמֹר אֶל־בְּנֵי־יִשְׂרָאֵל אַתֶּם עַם־קְשֵׁה־עֹרֶף רֶגַע
31   אֶחָד אֶעֱלֶה בְקִרְבְּךָ וְכִלִּיתִיךָ וְעַתָּה הוֹרֵד עֶדְיְךָ
32   מֵעָלֶיךָ וְאֵדְעָה מָה אֶעֱשֶׂה־לָּךְ׃ וַיִּתְנַצְּלוּ בְנֵי־יִשְׂרָאֵל
33   אֶת־עֶדְיָם מֵהַר חוֹרֵב׃ וּמֹשֶׁה יִקַּח אֶת־הָאֹהֶל וְנָטָה־לוֹ
34   מִחוּץ לַמַּחֲנֶה הַרְחֵק מִן־הַמַּחֲנֶה וְקָרָא לוֹ אֹהֶל מוֹעֵד
35   וְהָיָה כָּל־מְבַקֵּשׁ יְהוָה יֵצֵא אֶל־אֹהֶל מוֹעֵד אֲשֶׁר
36   מִחוּץ לַמַּחֲנֶה׃ וְהָיָה כְּצֵאת מֹשֶׁה אֶל־הָאֹהֶל יָקוּמוּ
37   כָּל־הָעָם וְנִצְּבוּ אִישׁ פֶּתַח אָהֳלוֹ וְהִבִּיטוּ אַחֲרֵי מֹשֶׁה
38   עַד־בֹּאוֹ הָאֹהֱלָה׃ וְהָיָה כְּבֹא מֹשֶׁה הָאֹהֱלָה יֵרֵד
39   עַמּוּד הֶעָנָן וְעָמַד פֶּתַח הָאֹהֶל וְדִבֶּר עִם־מֹשֶׁה׃ וְרָאָה
40   כָל־הָעָם אֶת־עַמּוּד הֶעָנָן עֹמֵד פֶּתַח הָאֹהֶל וְקָם כָּל־
41  הָעָם וְהִשְׁתַּחֲווּ אִישׁ פֶּתַח אָהֳלוֹ׃ וְדִבֶּר יְהוָה אֶל־
42  מֹשֶׁה פָּנִים אֶל־פָּנִים כַּאֲשֶׁר יְדַבֵּר אִישׁ אֶל־רֵעֵהוּ
