 1  וַיַּעֲשׂוּ־כֵן בְּנֵי יִשְׂרָאֵל וַיִּלְקְטוּ הַמַּרְבֶּה וְהַמַּמְעִיט׃
 2   וַיָּמֹדּוּ בָעֹמֶר וְלֹא הֶעְדִּיף הַמַּרְבֶּה וְהַמַּמְעִיט לֹא
 3   הֶחְסִיר אִישׁ לְפִי־אָכְלוֹ לָקָטוּ׃ וַיֹּאמֶר מֹשֶׁה אֲלֵהֶם
 4   אִישׁ אַל־יוֹתֵר מִמֶּנּוּ עַד־בֹּקֶר׃ וְלֹא־שָׁמְעוּ אֶל־מֹשֶׁה
 5   וַיּוֹתִרוּ אֲנָשִׁים מִמֶּנּוּ עַד־בֹּקֶר וַיָּרֻם תּוֹלָעִים וַיִּבְאַשׁ
 6   וַיִּקְצֹף עֲלֵהֶם מֹשֶׁה׃ וַיִּלְקְטוּ אֹתוֹ בַּבֹּקֶר בַּבֹּקֶר
 7   אִישׁ כְּפִי אָכְלוֹ וְחַם הַשֶּׁמֶשׁ וְנָמָס׃ וַיְהִי  בַּיּוֹם
 8   הַשִּׁשִּׁי לָקְטוּ לֶחֶם מִשְׁנֶה שְׁנֵי הָעֹמֶר לָאֶחָד וַיָּבֹאוּ
 9   כָּל־נְשִׂיאֵי הָעֵדָה וַיַּגִּידוּ לְמֹשֶׁה׃ וַיֹּאמֶר אֲלֵהֶם הוּא
10   אֲשֶׁר דִּבֶּר יְהוָה שַׁבָּתוֹן שַׁבַּת־קֹדֶשׁ לַיהוָה מָחָר
11   אֵת אֲשֶׁר־תֹּאפוּ אֵפוּ וְאֵת אֲשֶׁר־תְּבַשְּׁלוּ בַּשֵּׁלוּ
12   וְאֵת כָּל־הָעֹדֵף הַנִּיחוּ לָכֶם לְמִשְׁמֶרֶת עַד־הַבֹּקֶר׃
13   וַיַּנִּיחוּ אֹתוֹ עַד־הַבֹּקֶר כַּאֲשֶׁר צִוָּה מֹשֶׁה וְלֹא
14   הִבְאִישׁ וְרִמָּה לֹא־הָיְתָה בּוֹ׃ וַיֹּאמֶר מֹשֶׁה אִכְלֻהוּ
15   הַיּוֹם כִּי־שַׁבָּת הַיּוֹם לַיהוָה הַיּוֹם לֹא תִמְצָאֻהוּ
16   בַּשָּׂדֶה׃ שֵׁשֶׁת יָמִים תִּלְקְטֻהוּ וּבַיּוֹם הַשְּׁבִיעִי שַׁבָּת
17   לֹא יִהְיֶה־בּוֹ׃ וַיְהִי בַּיּוֹם הַשְּׁבִיעִי יָצְאוּ מִן־הָעָם
18   לִלְקֹט וְלֹא מָצָאוּ׃ [ס]             וַיֹּאמֶר יְהוָה
19   אֶל־מֹשֶׁה עַד־אָנָה מֵאַנְתֶּם לִשְׁמֹר מִצְו‍ֹתַי וְתוֹרֹתָי׃
20   רְאוּ כִּי־יְהוָה נָתַן לָכֶם הַשַּׁבָּת עַל־כֵּן הוּא נֹתֵן לָכֶם
21   בַּיּוֹם הַשִּׁשִּׁי לֶחֶם יוֹמָיִם שְׁבוּ  אִישׁ תַּחְתָּיו אַל־
22  יֵצֵא אִישׁ מִמְּקֹמוֹ בַּיּוֹם הַשְּׁבִיעִי׃ וַיִּשְׁבְּתוּ הָעָם
23   בַּיּוֹם הַשְּׁבִעִי׃ וַיִּקְרְאוּ בֵית־יִשְׂרָאֵל אֶת־שְׁמוֹ מָן
24   וְהוּא כְּזֶרַע גַּד לָבָן וְטַעְמוֹ כְּצַפִּיחִת בִּדְבָשׁ׃ וַיֹּאמֶר
25   מֹשֶׁה זֶה הַדָּבָר אֲשֶׁר צִוָּה יְהוָה מְלֹא הָעֹמֶר מִמֶּנּוּ
26   לְמִשְׁמֶרֶת לְדֹרֹתֵיכֶם לְמַעַן  יִרְאוּ אֶת־הַלֶּחֶם אֲשֶׁר
27   הֶאֱכַלְתִּי אֶתְכֶם בַּמִּדְבָּר בְּהוֹצִיאִי אֶתְכֶם מֵאֶרֶץ
28   מִצְרָיִם׃ וַיֹּאמֶר מֹשֶׁה אֶל־אַהֲרֹן קַח צִנְצֶנֶת אַחַת
29   וְתֶן־שָׁמָּה מְלֹא־הָעֹמֶר מָן וְהַנַּח אֹתוֹ לִפְנֵי יְהוָה
30   לְמִשְׁמֶרֶת לְדֹרֹתֵיכֶם׃ כַּאֲשֶׁר צִוָּה יְהוָה אֶל־מֹשֶׁה
31   וַיַּנִּיחֵהוּ אַהֲרֹן לִפְנֵי הָעֵדֻת לְמִשְׁמָרֶת׃ וּבְנֵי יִשְׂרָאֵל
32   אָכְלוּ אֶת־הַמָּן אַרְבָּעִים שָׁנָה עַד־בֹּאָם אֶל־אֶרֶץ
33   נוֹשָׁבֶת אֶת־הַמָּן אָכְלוּ עַד־בֹּאָם אֶל־קְצֵה אֶרֶץ כְּנָעַן׃
34   וְהָעֹמֶר עֲשִׂרִית הָאֵיפָה הוּא׃ [פ]
35    וַיִּסְעוּ כָּל־עֲדַת בְּנֵי־יִשְׂרָאֵל מִמִּדְבַּר־סִין לְמַסְעֵיהֶם
36   עַל־פִּי יְהוָה וַיַּחֲנוּ בִּרְפִידִים וְאֵין מַיִם לִשְׁתֹּת הָעָם׃
37   וַיָּרֶב הָעָם עִם־מֹשֶׁה וַיֹּאמְרוּ תְּנוּ־לָנוּ מַיִם וְנִשְׁתֶּה
38   וַיֹּאמֶר לָהֶם מֹשֶׁה מַה־תְּרִיבוּן עִמָּדִי מַה־תְּנַסּוּן אֶת־
39  יְהוָה׃ וַיִּצְמָא שָׁם הָעָם לַמַּיִם וַיָּלֶן הָעָם עַל־מֹשֶׁה
40   וַיֹּאמֶר לָמָּה זֶּה הֶעֱלִיתָנוּ מִמִּצְרַיִם לְהָמִית אֹתִי
41   וְאֶת־בָּנַי וְאֶת־מִקְנַי בַּצָּמָא׃ וַיִּצְעַק מֹשֶׁה אֶל־יְהוָה
42   לֵאמֹר מָה אֶעֱשֶׂה לָעָם הַזֶּה עוֹד מְעַט וּסְקָלֻנִי׃
