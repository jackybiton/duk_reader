 1  וַתִּהֲלַךְ אֵשׁ אָרְצָה וַיַּמְטֵר יְהוָה בָּרָד עַל־אֶרֶץ
 2   מִצְרָיִם׃ וַיְהִי בָרָד וְאֵשׁ מִתְלַקַּחַת בְּתוֹךְ הַבָּרָד
 3   כָּבֵד מְאֹד אֲשֶׁר לֹא־הָיָה כָמֹהוּ בְּכָל־אֶרֶץ מִצְרַיִם
 4   מֵאָז הָיְתָה לְגוֹי׃ וַיַּךְ הַבָּרָד בְּכָל־אֶרֶץ מִצְרַיִם אֵת
 5   כָּל־אֲשֶׁר בַּשָּׂדֶה מֵאָדָם וְעַד־בְּהֵמָה וְאֵת כָּל־עֵשֶׂב
 6   הַשָּׂדֶה הִכָּה הַבָּרָד וְאֶת־כָּל־עֵץ הַשָּׂדֶה שִׁבֵּר׃ רַק
 7   בְּאֶרֶץ גֹּשֶׁן אֲשֶׁר־שָׁם בְּנֵי יִשְׂרָאֵל לֹא הָיָה בָּרָד׃
 8   וַיִּשְׁלַח פַּרְעֹה וַיִּקְרָא לְמֹשֶׁה וּלְאַהֲרֹן וַיֹּאמֶר אֲלֵהֶם
 9   חָטָאתִי הַפָּעַם יְהוָה הַצַּדִּיק וַאֲנִי וְעַמִּי הָרְשָׁעִים׃
10   הַעְתִּירוּ אֶל־יְהוָה וְרַב מִהְיֹת קֹלֹת אֱלֹהִים וּבָרָד
11   וַאֲשַׁלְּחָה אֶתְכֶם וְלֹא תֹסִפוּן לַעֲמֹד׃ וַיֹּאמֶר אֵלָיו
12   מֹשֶׁה כְּצֵאתִי אֶת־הָעִיר אֶפְרֹשׂ אֶת־כַּפַּי אֶל־יְהוָה
13   הַקֹּלוֹת יֶחְדָּלוּן וְהַבָּרָד לֹא יִהְיֶה־עוֹד לְמַעַן תֵּדַע כִּי
14   לַיהוָה הָאָרֶץ׃ וְאַתָּה וַעֲבָדֶיךָ יָדַעְתִּי כִּי טֶרֶם תִּירְאוּן
15   מִפְּנֵי יְהוָה אֱלֹהִים׃ וְהַפִּשְׁתָּה וְהַשְּׂעֹרָה נֻכָּתָה כִּי
16   הַשְּׂעֹרָה אָבִיב וְהַפִּשְׁתָּה גִּבְעֹל׃ וְהַחִטָּה וְהַכֻּסֶּמֶת
17   לֹא נֻכּוּ כִּי אֲפִילֹת הֵנָּה׃  וַיֵּצֵא מֹשֶׁה מֵעִם פַּרְעֹה אֶת־
18  הָעִיר וַיִּפְרֹשׂ כַּפָּיו אֶל־יְהוָה וַיַּחְדְּלוּ הַקֹּלוֹת וְהַבָּרָד
19   וּמָטָר לֹא־נִתַּךְ אָרְצָה׃ וַיַּרְא פַּרְעֹה כִּי־חָדַל הַמָּטָר
20   וְהַבָּרָד וְהַקֹּלֹת וַיֹּסֶף לַחֲטֹא וַיַּכְבֵּד לִבּוֹ הוּא וַעֲבָדָיו׃
21   וַיֶּחֱזַק לֵב פַּרְעֹה וְלֹא שִׁלַּח אֶת־בְּנֵי יִשְׂרָאֵל כַּאֲשֶׁר
22   דִּבֶּר יְהוָה בְּיַד־מֹשֶׁה׃ [פ]
23  וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה בֹּא אֶל־פַּרְעֹה כִּי־אֲנִי
24   הִכְבַּדְתִּי אֶת־לִבּוֹ וְאֶת־לֵב עֲבָדָיו לְמַעַן שִׁתִי אֹתֹתַי
25   אֵלֶּה בְּקִרְבּוֹ׃ וּלְמַעַן תְּסַפֵּר בְּאָזְנֵי בִנְךָ וּבֶן־בִּנְךָ אֵת
26   אֲשֶׁר הִתְעַלַּלְתִּי בְּמִצְרַיִם וְאֶת־אֹתֹתַי אֲשֶׁר־שַׂמְתִּי
27   בָם וִידַעְתֶּם כִּי־אֲנִי יְהוָה׃ וַיָּבֹא מֹשֶׁה וְאַהֲרֹן אֶל־
28  פַּרְעֹה וַיֹּאמְרוּ אֵלָיו כֹּה־אָמַר יְהוָה אֱלֹהֵי הָעִבְרִים
29   עַד־מָתַי מֵאַנְתָּ לֵעָנֹת מִפָּנָי שַׁלַּח עַמִּי וְיַעַבְדֻנִי׃♢ כִּי
30   אִם־מָאֵן אַתָּה לְשַׁלֵּחַ אֶת־עַמִּי הִנְנִי מֵבִיא מָחָר
31   אַרְבֶּה בִּגְבֻלֶךָ׃ וְכִסָּה אֶת־עֵין הָאָרֶץ וְלֹא יוּכַל
32   לִרְאֹת אֶת־הָאָרֶץ וְאָכַל  אֶת־יֶתֶר הַפְּלֵטָה הַנִּשְׁאֶרֶת
33   לָכֶם מִן־הַבָּרָד וְאָכַל אֶת־כָּל־הָעֵץ הַצֹּמֵחַ לָכֶם מִן־
34  הַשָּׂדֶה׃ וּמָלְאוּ בָתֶּיךָ וּבָתֵּי כָל־עֲבָדֶיךָ וּבָתֵּי כָל־
35  מִצְרַיִם אֲשֶׁר לֹא־רָאוּ אֲבֹתֶיךָ וַאֲבוֹת אֲבֹתֶיךָ מִיּוֹם
36   הֱיוֹתָם עַל־הָאֲדָמָה עַד הַיּוֹם הַזֶּה וַיִּפֶן וַיֵּצֵא מֵעִם
37   פַּרְעֹה׃♢ וַיֹּאמְרוּ עַבְדֵי פַרְעֹה אֵלָיו עַד־מָתַי יִהְיֶה זֶה
38   לָנוּ לְמוֹקֵשׁ שַׁלַּח אֶת־הָאֲנָשִׁים וְיַעַבְדוּ אֶת־יְהוָה
39   אֱלֹהֵיהֶם הֲטֶרֶם תֵּדַע כִּי אָבְדָה מִצְרָיִם׃ וַיּוּשַׁב אֶת־
40  מֹשֶׁה וְאֶת־אַהֲרֹן אֶל־פַּרְעֹה וַיֹּאמֶר אֲלֵהֶם לְכוּ
41   עִבְדוּ אֶת־יְהוָה אֱלֹהֵיכֶם מִי וָמִי הַהֹלְכִים׃ וַיֹּאמֶר
42   מֹשֶׁה בִּנְעָרֵינוּ וּבִזְקֵנֵינוּ נֵלֵךְ בְּבָנֵינוּ וּבִבְנוֹתֵנוּ בְּצֹאנֵנוּ
