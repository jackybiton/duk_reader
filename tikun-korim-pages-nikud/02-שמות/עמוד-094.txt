 1  וְאֵת שְׁתֵּי קְצוֹת שְׁתֵּי הָעֲבֹתֹת תִּתֵּן עַל־שְׁתֵּי
 2   הַמִּשְׁבְּצוֹת וְנָתַתָּה עַל־כִּתְפוֹת הָאֵפֹד אֶל־מוּל פָּנָיו׃
 3   וְעָשִׂיתָ שְׁתֵּי טַבְּעוֹת זָהָב וְשַׂמְתָּ אֹתָם עַל־שְׁנֵי קְצוֹת
 4   הַחֹשֶׁן עַל־שְׂפָתוֹ אֲשֶׁר אֶל־עֵבֶר הָאֵפֹד בָּיְתָה׃
 5   וְעָשִׂיתָ שְׁתֵּי טַבְּעוֹת זָהָב וְנָתַתָּה אֹתָם עַל־שְׁתֵּי
 6   כִתְפוֹת הָאֵפוֹד מִלְּמַטָּה מִמּוּל פָּנָיו לְעֻמַּת מַחְבַּרְתּוֹ
 7   מִמַּעַל לְחֵשֶׁב הָאֵפוֹד׃ וְיִרְכְּסוּ אֶת־הַחֹשֶׁן מִטַּבְּעֹתָו
 8   אֶל־טַבְּעֹת הָאֵפוֹד בִּפְתִיל תְּכֵלֶת לִהְיוֹת עַל־חֵשֶׁב
 9   הָאֵפוֹד וְלֹא־יִזַּח הַחֹשֶׁן מֵעַל הָאֵפוֹד׃ וְנָשָׂא אַהֲרֹן
10   אֶת־שְׁמוֹת בְּנֵי־יִשְׂרָאֵל בְּחֹשֶׁן הַמִּשְׁפָּט עַל־לִבּוֹ
11   בְּבֹאוֹ אֶל־הַקֹּדֶשׁ לְזִכָּרֹן לִפְנֵי־יְהוָה תָּמִיד׃ וְנָתַתָּ אֶל־חֹשֶׁן
12   הַמִּשְׁפָּט אֶת־הָאוּרִים וְאֶת־הַתֻּמִּים וְהָיוּ עַל־לֵב אַהֲרֹן
13   בְּבֹאוֹ לִפְנֵי יְהוָה וְנָשָׂא אַהֲרֹן אֶת־מִשְׁפַּט בְּנֵי־יִשְׂרָאֵל
14   עַל־לִבּוֹ לִפְנֵי יְהוָה תָּמִיד׃ [ס]              וְעָשִׂיתָ
15   אֶת־מְעִיל הָאֵפוֹד כְּלִיל תְּכֵלֶת׃ וְהָיָה פִי־רֹאשׁוֹ
16   בְּתוֹכוֹ שָׂפָה יִהְיֶה לְפִיו סָבִיב מַעֲשֵׂה אֹרֵג כְּפִי
17   תַחְרָא יִהְיֶה־לּוֹ לֹא יִקָּרֵעַ׃ וְעָשִׂיתָ עַל־שׁוּלָיו רִמֹּנֵי
18   תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי עַל־שׁוּלָיו סָבִיב וּפַעֲמֹנֵי
19   זָהָב בְּתוֹכָם סָבִיב׃ פַּעֲמֹן זָהָב וְרִמּוֹן פַּעֲמֹן זָהָב
20   וְרִמּוֹן עַל־שׁוּלֵי הַמְּעִיל סָבִיב׃ וְהָיָה עַל־אַהֲרֹן לְשָׁרֵת
21   וְנִשְׁמַע קוֹלוֹ בְּבֹאוֹ אֶל־הַקֹּדֶשׁ לִפְנֵי יְהוָה וּבְצֵאתוֹ
22   וְלֹא יָמוּת׃ [ס]             וְעָשִׂיתָ צִּיץ זָהָב טָהוֹר
23   וּפִתַּחְתָּ עָלָיו פִּתּוּחֵי חֹתָם קֹדֶשׁ לַיהוָה׃ וְשַׂמְתָּ אֹתוֹ
24   עַל־פְּתִיל תְּכֵלֶת וְהָיָה עַל־הַמִּצְנָפֶת אֶל־מוּל פְּנֵי־
25  הַמִּצְנֶפֶת יִהְיֶה׃ וְהָיָה עַל־מֵצַח אַהֲרֹן וְנָשָׂא אַהֲרֹן אֶת־
26  עֲו‍ֹן הַקֳּדָשִׁים אֲשֶׁר יַקְדִּישׁוּ בְּנֵי יִשְׂרָאֵל לְכָל־מַתְּנֹת
27   קָדְשֵׁיהֶם וְהָיָה עַל־מִצְחוֹ תָּמִיד לְרָצוֹן לָהֶם לִפְנֵי
28   יְהוָה׃ וְשִׁבַּצְתָּ הַכְּתֹנֶת שֵׁשׁ וְעָשִׂיתָ מִצְנֶפֶת שֵׁשׁ
29   וְאַבְנֵט תַּעֲשֶׂה מַעֲשֵׂה רֹקֵם׃ וְלִבְנֵי אַהֲרֹן תַּעֲשֶׂה
30   כֻתֳּנֹת וְעָשִׂיתָ לָהֶם אַבְנֵטִים וּמִגְבָּעוֹת תַּעֲשֶׂה לָהֶם
31   לְכָבוֹד וּלְתִפְאָרֶת׃ וְהִלְבַּשְׁתָּ אֹתָם אֶת־אַהֲרֹן אָחִיךָ
32   וְאֶת־בָּנָיו אִתּוֹ וּמָשַׁחְתָּ אֹתָם וּמִלֵּאתָ אֶת־יָדָם וְקִדַּשְׁתָּ
33   אֹתָם וְכִהֲנוּ לִי׃ וַעֲשֵׂה לָהֶם מִכְנְסֵי־בָד לְכַסּוֹת בְּשַׂר
34   עֶרְוָה מִמָּתְנַיִם וְעַד־יְרֵכַיִם יִהְיוּ׃ וְהָיוּ עַל־אַהֲרֹן וְעַל־
35  בָּנָיו בְּבֹאָם  אֶל־אֹהֶל מוֹעֵד אוֹ בְגִשְׁתָּם אֶל־הַמִּזְבֵּחַ
36   לְשָׁרֵת בַּקֹּדֶשׁ וְלֹא־יִשְׂאוּ עָו‍ֹן וָמֵתוּ חֻקַּת עוֹלָם לוֹ
37   וּלְזַרְעוֹ אַחֲרָיו׃ [ס]              וְזֶה הַדָּבָר אֲשֶׁר־
38  תַּעֲשֶׂה לָהֶם לְקַדֵּשׁ אֹתָם לְכַהֵן לִי לְקַח פַּר אֶחָד
39   בֶּן־בָּקָר וְאֵילִם שְׁנַיִם תְּמִימִם׃ וְלֶחֶם מַצּוֹת וְחַלֹּת
40   מַצֹּת בְּלוּלֹת בַּשֶּׁמֶן וּרְקִיקֵי מַצּוֹת מְשֻׁחִים בַּשָּׁמֶן
41   סֹלֶת חִטִּים תַּעֲשֶׂה אֹתָם׃ וְנָתַתָּ אוֹתָם עַל־סַל אֶחָד
42   וְהִקְרַבְתָּ אֹתָם בַּסָּל וְאֶת־הַפָּר וְאֵת שְׁנֵי הָאֵילִם׃
