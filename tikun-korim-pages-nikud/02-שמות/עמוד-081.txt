 1  וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה עֲבֹר לִפְנֵי הָעָם וְקַח אִתְּךָ
 2   מִזִּקְנֵי יִשְׂרָאֵל וּמַטְּךָ אֲשֶׁר הִכִּיתָ בּוֹ אֶת־הַיְאֹר קַח
 3   בְּיָדְךָ וְהָלָכְתָּ׃ הִנְנִי עֹמֵד לְפָנֶיךָ שָּׁם  עַל־הַצּוּר בְּחֹרֵב
 4   וְהִכִּיתָ בַצּוּר וְיָצְאוּ מִמֶּנּוּ מַיִם וְשָׁתָה הָעָם וַיַּעַשׂ כֵּן
 5   מֹשֶׁה לְעֵינֵי זִקְנֵי יִשְׂרָאֵל׃ וַיִּקְרָא שֵׁם הַמָּקוֹם מַסָּה
 6   וּמְרִיבָה עַל־רִיב  בְּנֵי יִשְׂרָאֵל וְעַל נַסֹּתָם אֶת־יְהוָה
 7   לֵאמֹר הֲיֵשׁ יְהוָה בְּקִרְבֵּנוּ אִם־אָיִן׃ [פ]
 8   וַיָּבֹא עֲמָלֵק וַיִּלָּחֶם עִם־יִשְׂרָאֵל בִּרְפִידִם׃ וַיֹּאמֶר
 9   מֹשֶׁה אֶל־יְהוֹשֻׁעַ בְּחַר־לָנוּ אֲנָשִׁים וְצֵא הִלָּחֵם
10   בַּעֲמָלֵק מָחָר אָנֹכִי נִצָּב עַל־רֹאשׁ הַגִּבְעָה וּמַטֵּה
11   הָאֱלֹהִים בְּיָדִי׃ וַיַּעַשׂ יְהוֹשֻׁעַ כַּאֲשֶׁר אָמַר־לוֹ מֹשֶׁה
12   לְהִלָּחֵם בַּעֲמָלֵק וּמֹשֶׁה אַהֲרֹן וְחוּר עָלוּ רֹאשׁ
13   הַגִּבְעָה׃ וְהָיָה כַּאֲשֶׁר יָרִים מֹשֶׁה יָדוֹ וְגָבַר יִשְׂרָאֵל
14   וְכַאֲשֶׁר יָנִיחַ יָדוֹ וְגָבַר עֲמָלֵק׃ וִידֵי מֹשֶׁה כְּבֵדִים
15   וַיִּקְחוּ־אֶבֶן וַיָּשִׂימוּ תַחְתָּיו וַיֵּשֶׁב עָלֶיהָ וְאַהֲרֹן וְחוּר
16   תָּמְכוּ בְיָדָיו מִזֶּה אֶחָד וּמִזֶּה אֶחָד וַיְהִי יָדָיו אֱמוּנָה
17   עַד־בֹּא הַשָּׁמֶשׁ׃ וַיַּחֲלֹשׁ יְהוֹשֻׁעַ אֶת־עֲמָלֵק וְאֶת־
18  עַמּוֹ לְפִי־חָרֶב׃ [פ]
19    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה כְּתֹב זֹאת זִכָּרוֹן בַּסֵּפֶר
20   וְשִׂים בְּאָזְנֵי יְהוֹשֻׁעַ כִּי־מָחֹה אֶמְחֶה אֶת־זֵכֶר עֲמָלֵק
21   מִתַּחַת הַשָּׁמָיִם׃ וַיִּבֶן מֹשֶׁה מִזְבֵּחַ וַיִּקְרָא שְׁמוֹ יְהוָה
22    נִסִּי׃ וַיֹּאמֶר כִּי־יָד עַל־כֵּס יָהּ מִלְחָמָה לַיהוָה בַּעֲמָלֵק
23   מִדֹּר דֹּר׃ [פ]
24  וַיִּשְׁמַע יִתְרוֹ כֹהֵן מִדְיָן חֹתֵן מֹשֶׁה אֵת כָּל־אֲשֶׁר
25   עָשָׂה אֱלֹהִים לְמֹשֶׁה וּלְיִשְׂרָאֵל עַמּוֹ כִּי־הוֹצִיא
26   יְהוָה אֶת־יִשְׂרָאֵל מִמִּצְרָיִם׃ וַיִּקַּח יִתְרוֹ חֹתֵן מֹשֶׁה אֶת־
27  צִפֹּרָה אֵשֶׁת מֹשֶׁה אַחַר שִׁלּוּחֶיהָ׃ וְאֵת שְׁנֵי בָנֶיהָ
28   אֲשֶׁר שֵׁם הָאֶחָד גֵּרְשֹׁם כִּי אָמַר גֵּר הָיִיתִי בְּאֶרֶץ
29   נָכְרִיָּה׃ וְשֵׁם הָאֶחָד אֱלִיעֶזֶר כִּי־אֱלֹהֵי אָבִי בְּעֶזְרִי
30   וַיַּצִּלֵנִי מֵחֶרֶב פַּרְעֹה׃♢ וַיָּבֹא יִתְרוֹ חֹתֵן מֹשֶׁה וּבָנָיו
31   וְאִשְׁתּוֹ אֶל־מֹשֶׁה אֶל־הַמִּדְבָּר אֲשֶׁר־הוּא חֹנֶה שָׁם
32   הַר הָאֱלֹהִים׃ וַיֹּאמֶר אֶל־מֹשֶׁה אֲנִי חֹתֶנְךָ יִתְרוֹ בָּא
33   אֵלֶיךָ וְאִשְׁתְּךָ וּשְׁנֵי בָנֶיהָ עִמָּהּ׃ וַיֵּצֵא מֹשֶׁה לִקְרַאת
34   חֹתְנוֹ וַיִּשְׁתַּחוּ וַיִּשַּׁק־לוֹ וַיִּשְׁאֲלוּ אִישׁ־לְרֵעֵהוּ לְשָׁלוֹם
35   וַיָּבֹאוּ הָאֹהֱלָה׃ וַיְסַפֵּר מֹשֶׁה לְחֹתְנוֹ אֵת כָּל־אֲשֶׁר
36   עָשָׂה יְהוָה לְפַרְעֹה וּלְמִצְרַיִם עַל אוֹדֹת יִשְׂרָאֵל אֵת
37   כָּל־הַתְּלָאָה אֲשֶׁר מְצָאָתַם בַּדֶּרֶךְ וַיַּצִּלֵם יְהוָה׃♢ וַיִּחַדְּ
38   יִתְרוֹ עַל כָּל־הַטּוֹבָה אֲשֶׁר־עָשָׂה יְהוָה לְיִשְׂרָאֵל
39   אֲשֶׁר הִצִּילוֹ מִיַּד מִצְרָיִם׃ וַיֹּאמֶר יִתְרוֹ בָּרוּךְ יְהוָה
40   אֲשֶׁר הִצִּיל אֶתְכֶם מִיַּד מִצְרַיִם וּמִיַּד פַּרְעֹה אֲשֶׁר
41   הִצִּיל אֶת־הָעָם מִתַּחַת יַד־מִצְרָיִם׃ עַתָּה יָדַעְתִּי כִּי־
42  גָדוֹל יְהוָה מִכָּל־הָאֱלֹהִים כִּי בַדָּבָר אֲשֶׁר זָדוּ עֲלֵיהֶם׃
