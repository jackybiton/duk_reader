 1  וְהַנְּשִׂאִם הֵבִיאוּ אֵת אַבְנֵי הַשֹּׁהַם וְאֵת אַבְנֵי הַמִּלֻּאִים
 2   לָאֵפוֹד וְלַחֹשֶׁן׃ וְאֶת־הַבֹּשֶׂם וְאֶת־הַשָּׁמֶן לְמָאוֹר
 3   וּלְשֶׁמֶן הַמִּשְׁחָה וְלִקְטֹרֶת הַסַּמִּים׃ כָּל־אִישׁ וְאִשָּׁה
 4   אֲשֶׁר נָדַב לִבָּם אֹתָם לְהָבִיא לְכָל־הַמְּלָאכָה אֲשֶׁר
 5   צִוָּה יְהוָה לַעֲשׂוֹת בְּיַד־מֹשֶׁה הֵבִיאוּ בְנֵי־יִשְׂרָאֵל
 6   נְדָבָה לַיהוָה׃ [פ]
 7    וַיֹּאמֶר מֹשֶׁה אֶל־בְּנֵי יִשְׂרָאֵל רְאוּ קָרָא יְהוָה בְּשֵׁם
 8   בְּצַלְאֵל בֶּן־אוּרִי בֶן־חוּר לְמַטֵּה יְהוּדָה׃ וַיְמַלֵּא אֹתוֹ
 9   רוּחַ אֱלֹהִים בְּחָכְמָה בִּתְבוּנָה וּבְדַעַת וּבְכָל־מְלָאכָה׃
10   וְלַחְשֹׁב מַחֲשָׁבֹת לַעֲשֹׂת בַּזָּהָב וּבַכֶּסֶף וּבַנְּחֹשֶׁת׃
11   וּבַחֲרֹשֶׁת אֶבֶן לְמַלֹּאת וּבַחֲרֹשֶׁת עֵץ לַעֲשׂוֹת
12   בְּכָל־מְלֶאכֶת מַחֲשָׁבֶת׃ וּלְהוֹרֹת נָתַן בְּלִבּוֹ הוּא
13   וְאָהֳלִיאָב בֶּן־אֲחִיסָמָךְ לְמַטֵּה־דָן׃ מִלֵּא אֹתָם חָכְמַת־
14  לֵב לַעֲשׂוֹת כָּל־מְלֶאכֶת חָרָשׁ  וְחֹשֵׁב וְרֹקֵם בַּתְּכֵלֶת
15   וּבָאַרְגָּמָן בְּתוֹלַעַת הַשָּׁנִי וּבַשֵּׁשׁ וְאֹרֵג עֹשֵׂי כָּל־
16  מְלָאכָה וְחֹשְׁבֵי מַחֲשָׁבֹת׃ וְעָשָׂה בְצַלְאֵל וְאָהֳלִיאָב
17   וְכֹל  אִישׁ חֲכַם־לֵב אֲשֶׁר נָתַן יְהוָה חָכְמָה וּתְבוּנָה
18   בָּהֵמָּה לָדַעַת לַעֲשֹׂת אֶת־כָּל־מְלֶאכֶת עֲבֹדַת
19   הַקֹּדֶשׁ לְכֹל אֲשֶׁר־צִוָּה יְהוָה׃ וַיִּקְרָא מֹשֶׁה אֶל־
20  בְּצַלְאֵל וְאֶל־אָהֳלִיאָב וְאֶל כָּל־אִישׁ חֲכַם־לֵב
21   אֲשֶׁר נָתַן יְהוָה חָכְמָה בְּלִבּוֹ כֹּל אֲשֶׁר נְשָׂאוֹ לִבּוֹ
22   לְקָרְבָה אֶל־הַמְּלָאכָה לַעֲשֹׂת אֹתָהּ׃ וַיִּקְחוּ מִלִּפְנֵי
23   מֹשֶׁה אֵת כָּל־הַתְּרוּמָה אֲשֶׁר הֵבִיאוּ בְּנֵי יִשְׂרָאֵל
24   לִמְלֶאכֶת עֲבֹדַת הַקֹּדֶשׁ לַעֲשֹׂת אֹתָהּ וְהֵם הֵבִיאוּ
25   אֵלָיו עוֹד נְדָבָה בַּבֹּקֶר בַּבֹּקֶר׃ וַיָּבֹאוּ כָּל־הַחֲכָמִים
26   הָעֹשִׂים אֵת כָּל־מְלֶאכֶת הַקֹּדֶשׁ אִישׁ־אִישׁ
27   מִמְּלַאכְתּוֹ אֲשֶׁר־הֵמָּה עֹשִׂים׃ וַיֹּאמְרוּ אֶל־מֹשֶׁה
28   לֵּאמֹר מַרְבִּים הָעָם לְהָבִיא מִדֵּי הָעֲבֹדָה לַמְּלָאכָה
29   אֲשֶׁר־צִוָּה יְהוָה לַעֲשֹׂת אֹתָהּ׃ וַיְצַו מֹשֶׁה וַיַּעֲבִירוּ
30   קוֹל בַּמַּחֲנֶה לֵאמֹר אִישׁ וְאִשָּׁה אַל־יַעֲשׂוּ־עוֹד
31   מְלָאכָה לִתְרוּמַת הַקֹּדֶשׁ וַיִּכָּלֵא הָעָם מֵהָבִיא׃
32   וְהַמְּלָאכָה הָיְתָה דַיָּם לְכָל־הַמְּלָאכָה לַעֲשׂוֹת אֹתָהּ
33   וְהוֹתֵר׃ [ס]              וַיַּעֲשׂוּ כָל־חֲכַם־לֵב בְּעֹשֵׂי
34   הַמְּלָאכָה אֶת־הַמִּשְׁכָּן עֶשֶׂר יְרִיעֹת שֵׁשׁ מָשְׁזָר
35   וּתְכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי כְּרֻבִים מַעֲשֵׂה חֹשֵׁב
36   עָשָׂה אֹתָם׃ אֹרֶךְ הַיְרִיעָה הָאַחַת שְׁמֹנֶה וְעֶשְׂרִים
37   בָּאַמָּה וְרֹחַב אַרְבַּע בָּאַמָּה הַיְרִיעָה הָאֶחָת מִדָּה אַחַת
38   לְכָל־הַיְרִיעֹת׃ וַיְחַבֵּר אֶת־חֲמֵשׁ הַיְרִיעֹת אַחַת אֶל־
39  אֶחָת וְחָמֵשׁ יְרִיעֹת חִבַּר אַחַת אֶל־אֶחָת׃ וַיַּעַשׂ לֻלְאֹת
40   תְּכֵלֶת עַל שְׂפַת הַיְרִיעָה הָאֶחָת מִקָּצָה בַּמַּחְבָּרֶת
41   כֵּן עָשָׂה בִּשְׂפַת הַיְרִיעָה הַקִּיצוֹנָה בַּמַּחְבֶּרֶת
42   הַשֵּׁנִית׃ חֲמִשִּׁים לֻלָאֹת עָשָׂה בַּיְרִיעָה הָאֶחָת
