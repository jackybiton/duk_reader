 1  וְאֶת־שֵׁשׁ הַיְרִיעֹת לְבָד וְכָפַלְתָּ אֶת־הַיְרִיעָה
 2   הַשִּׁשִּׁית אֶל־מוּל פְּנֵי הָאֹהֶל׃ וְעָשִׂיתָ חֲמִשִּׁים לֻלָאֹת
 3   עַל שְׂפַת הַיְרִיעָה הָאֶחָת הַקִּיצֹנָה בַּחֹבָרֶת וַחֲמִשִּׁים
 4   לֻלָאֹת עַל שְׂפַת הַיְרִיעָה הַחֹבֶרֶת הַשֵּׁנִית׃ וְעָשִׂיתָ
 5   קַרְסֵי נְחֹשֶׁת חֲמִשִּׁים וְהֵבֵאתָ אֶת־הַקְּרָסִים בַּלֻּלָאֹת
 6   וְחִבַּרְתָּ אֶת־הָאֹהֶל וְהָיָה אֶחָד׃ וְסֶרַח הָעֹדֵף בִּירִיעֹת
 7   הָאֹהֶל חֲצִי הַיְרִיעָה הָעֹדֶפֶת תִּסְרַח עַל אֲחֹרֵי
 8   הַמִּשְׁכָּן׃ וְהָאַמָּה מִזֶּה וְהָאַמָּה מִזֶּה בָּעֹדֵף בְּאֹרֶךְ
 9   יְרִיעֹת הָאֹהֶל יִהְיֶה סָרוּחַ עַל־צִדֵּי הַמִּשְׁכָּן מִזֶּה וּמִזֶּה
10   לְכַסֹּתוֹ׃ וְעָשִׂיתָ מִכְסֶה לָאֹהֶל עֹרֹת אֵילִם מְאָדָּמִים
11   וּמִכְסֵה עֹרֹת תְּחָשִׁים מִלְמָעְלָה׃ [פ]
12    וְעָשִׂיתָ אֶת־הַקְּרָשִׁים לַמִּשְׁכָּן עֲצֵי שִׁטִּים עֹמְדִים׃
13   עֶשֶׂר אַמּוֹת אֹרֶךְ הַקָּרֶשׁ וְאַמָּה וַחֲצִי הָאַמָּה רֹחַב
14   הַקֶּרֶשׁ הָאֶחָד׃ שְׁתֵּי יָדוֹת לַקֶּרֶשׁ הָאֶחָד מְשֻׁלָּבֹת
15   אִשָּׁה אֶל־אֲחֹתָהּ כֵּן תַּעֲשֶׂה לְכֹל קַרְשֵׁי הַמִּשְׁכָּן׃
16   וְעָשִׂיתָ אֶת־הַקְּרָשִׁים לַמִּשְׁכָּן עֶשְׂרִים קֶרֶשׁ לִפְאַת
17   נֶגְבָּה תֵימָנָה׃ וְאַרְבָּעִים אַדְנֵי־כֶסֶף תַּעֲשֶׂה תַּחַת
18   עֶשְׂרִים הַקָּרֶשׁ שְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד
19   לִשְׁתֵּי יְדֹתָיו וּשְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד לִשְׁתֵּי
20   יְדֹתָיו׃ וּלְצֶלַע הַמִּשְׁכָּן הַשֵּׁנִית לִפְאַת צָפוֹן עֶשְׂרִים
21   קָרֶשׁ׃ וְאַרְבָּעִים אַדְנֵיהֶם כָּסֶף שְׁנֵי אֲדָנִים תַּחַת
22   הַקֶּרֶשׁ הָאֶחָד וּשְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד׃
23   וּלְיַרְכְּתֵי הַמִּשְׁכָּן יָמָּה תַּעֲשֶׂה שִׁשָּׁה קְרָשִׁים׃ וּשְׁנֵי
24   קְרָשִׁים תַּעֲשֶׂה לִמְקֻצְעֹת הַמִּשְׁכָּן בַּיַּרְכָתָיִם׃ וְיִהְיוּ
25   תֹאֲמִם מִלְּמַטָּה וְיַחְדָּו יִהְיוּ תַמִּים עַל־רֹאשׁוֹ אֶל־
26  הַטַּבַּעַת הָאֶחָת כֵּן יִהְיֶה לִשְׁנֵיהֶם לִשְׁנֵי הַמִּקְצֹעֹת
27   יִהְיוּ׃ וְהָיוּ שְׁמֹנָה קְרָשִׁים וְאַדְנֵיהֶם כֶּסֶף שִׁשָּׁה עָשָׂר
28   אֲדָנִים שְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד וּשְׁנֵי אֲדָנִים
29   תַּחַת הַקֶּרֶשׁ הָאֶחָד׃ וְעָשִׂיתָ בְרִיחִם עֲצֵי שִׁטִּים
30   חֲמִשָּׁה לְקַרְשֵׁי צֶלַע־הַמִּשְׁכָּן הָאֶחָד׃ וַחֲמִשָּׁה
31   בְרִיחִם לְקַרְשֵׁי צֶלַע־הַמִּשְׁכָּן הַשֵּׁנִית וַחֲמִשָּׁה
32   בְרִיחִם לְקַרְשֵׁי צֶלַע הַמִּשְׁכָּן לַיַּרְכָתַיִם יָמָּה׃
33   וְהַבְּרִיחַ הַתִּיכֹן בְּתוֹךְ הַקְּרָשִׁים מַבְרִחַ מִן־הַקָּצֶה
34   אֶל־הַקָּצֶה׃ וְאֶת־הַקְּרָשִׁים תְּצַפֶּה זָהָב וְאֶת־טַבְּעֹתֵיהֶם
35   תַּעֲשֶׂה זָהָב בָּתִּים לַבְּרִיחִם וְצִפִּיתָ אֶת־הַבְּרִיחִם
36   זָהָב׃ וַהֲקֵמֹתָ אֶת־הַמִּשְׁכָּן כְּמִשְׁפָּטוֹ אֲשֶׁר הָרְאֵיתָ
37   בָּהָר׃ [ס]              וְעָשִׂיתָ פָרֹכֶת תְּכֵלֶת וְאַרְגָּמָן
38   וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מָשְׁזָר מַעֲשֵׂה חֹשֵׁב יַעֲשֶׂה אֹתָהּ
39   כְּרֻבִים׃ וְנָתַתָּה אֹתָהּ עַל־אַרְבָּעָה עַמּוּדֵי שִׁטִּים מְצֻפִּים
40   זָהָב וָוֵיהֶם זָהָב עַל־אַרְבָּעָה אַדְנֵי־כָסֶף׃ וְנָתַתָּה אֶת־
41  הַפָּרֹכֶת תַּחַת הַקְּרָסִים וְהֵבֵאתָ שָׁמָּה מִבֵּית לַפָּרֹכֶת
42   אֵת אֲרוֹן הָעֵדוּת וְהִבְדִּילָה הַפָּרֹכֶת לָכֶם בֵּין הַקֹּדֶשׁ
