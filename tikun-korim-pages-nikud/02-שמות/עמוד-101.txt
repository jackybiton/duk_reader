 1  וְשָׁב אֶל־הַמַּחֲנֶה וּמְשָׁרְתוֹ יְהוֹשֻׁעַ בִּן־נוּן נַעַר לֹא
 2   יָמִישׁ מִתּוֹךְ הָאֹהֶל׃ [פ]
 3    וַיֹּאמֶר מֹשֶׁה אֶל־יְהוָה רְאֵה אַתָּה אֹמֵר אֵלַי הַעַל
 4   אֶת־הָעָם הַזֶּה וְאַתָּה לֹא הוֹדַעְתַּנִי אֵת אֲשֶׁר־תִּשְׁלַח
 5   עִמִּי וְאַתָּה אָמַרְתָּ יְדַעְתִּיךָ בְשֵׁם וְגַם־מָצָאתָ חֵן
 6   בְּעֵינָי׃ וְעַתָּה אִם־נָא מָצָאתִי חֵן בְּעֵינֶיךָ הוֹדִעֵנִי נָא
 7   אֶת־דְּרָכֶךָ וְאֵדָעֲךָ לְמַעַן אֶמְצָא־חֵן בְּעֵינֶיךָ וּרְאֵה
 8   כִּי עַמְּךָ הַגּוֹי הַזֶּה׃ וַיֹּאמַר פָּנַי יֵלֵכוּ וַהֲנִחֹתִי לָךְ׃
 9   וַיֹּאמֶר אֵלָיו אִם־אֵין פָּנֶיךָ הֹלְכִים אַל־תַּעֲלֵנוּ מִזֶּה׃
10   וּבַמֶּה  יִוָּדַע אֵפוֹא כִּי־מָצָאתִי חֵן בְּעֵינֶיךָ אֲנִי וְעַמֶּךָ
11   הֲלוֹא בְּלֶכְתְּךָ עִמָּנוּ וְנִפְלִינוּ אֲנִי וְעַמְּךָ מִכָּל־הָעָם
12   אֲשֶׁר עַל־פְּנֵי הָאֲדָמָה׃ [פ]
13    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה גַּם אֶת־הַדָּבָר הַזֶּה אֲשֶׁר
14   דִּבַּרְתָּ אֶעֱשֶׂה כִּי־מָצָאתָ חֵן בְּעֵינַי וָאֵדָעֲךָ בְּשֵׁם׃
15   וַיֹּאמַר הַרְאֵנִי נָא אֶת־כְּבֹדֶךָ׃ וַיֹּאמֶר אֲנִי אַעֲבִיר כָּל־
16  טוּבִי עַל־פָּנֶיךָ וְקָרָאתִי בְשֵׁם יְהוָה לְפָנֶיךָ וְחַנֹּתִי
17   אֶת־אֲשֶׁר אָחֹן וְרִחַמְתִּי אֶת־אֲשֶׁר אֲרַחֵם׃ וַיֹּאמֶר
18   לֹא תוּכַל לִרְאֹת אֶת־פָּנָי כִּי לֹא־יִרְאַנִי הָאָדָם וָחָי׃
19   וַיֹּאמֶר יְהוָה הִנֵּה מָקוֹם אִתִּי וְנִצַּבְתָּ עַל־הַצּוּר׃ וְהָיָה
20   בַּעֲבֹר כְּבֹדִי וְשַׂמְתִּיךָ בְּנִקְרַת הַצּוּר וְשַׂכֹּתִי כַפִּי
21   עָלֶיךָ עַד־עָבְרִי׃ וַהֲסִרֹתִי אֶת־כַּפִּי וְרָאִיתָ אֶת־אֲחֹרָי
22   וּפָנַי לֹא יֵרָאוּ׃ [פ]
23    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה פְּסָל־לְךָ שְׁנֵי־לֻחֹת אֲבָנִים
24   כָּרִאשֹׁנִים וְכָתַבְתִּי עַל־הַלֻּחֹת אֶת־הַדְּבָרִים אֲשֶׁר הָיוּ
25   עַל־הַלֻּחֹת הָרִאשֹׁנִים אֲשֶׁר שִׁבַּרְתָּ׃ וֶהְיֵה נָכוֹן לַבֹּקֶר
26   וְעָלִיתָ בַבֹּקֶר אֶל־הַר סִינַי וְנִצַּבְתָּ לִי שָׁם עַל־רֹאשׁ
27   הָהָר׃ וְאִישׁ לֹא־יַעֲלֶה עִמָּךְ וְגַם־אִישׁ אַל־יֵרָא בְּכָל־
28  הָהָר גַּם־הַצֹּאן וְהַבָּקָר אַל־יִרְעוּ אֶל־מוּל הָהָר הַהוּא׃
29   וַיִּפְסֹל שְׁנֵי־לֻחֹת אֲבָנִים כָּרִאשֹׁנִים וַיַּשְׁכֵּם מֹשֶׁה
30   בַבֹּקֶר וַיַּעַל אֶל־הַר סִינַי כַּאֲשֶׁר צִוָּה יְהוָה אֹתוֹ וַיִּקַּח
31   בְּיָדוֹ שְׁנֵי לֻחֹת אֲבָנִים׃ וַיֵּרֶד יְהוָה בֶּעָנָן וַיִּתְיַצֵּב עִמּוֹ
32   שָׁם וַיִּקְרָא בְשֵׁם יְהוָה׃ וַיַּעֲבֹר יְהוָה  עַל־פָּנָיו וַיִּקְרָא
33   יְהוָה  יְהוָה אֵל רַחוּם וְחַנּוּן אֶרֶךְ אַפַּיִם וְרַב־חֶסֶד
34   וֶאֱמֶת׃ נֹצֵר חֶסֶד לָאֲלָפִים נֹשֵׂא עָו‍ֹן וָפֶשַׁע וְחַטָּאָה
35   וְנַקֵּה לֹא יְנַקֶּה פֹּקֵד  עֲו‍ֹן אָבוֹת עַל־בָּנִים וְעַל־בְּנֵי בָנִים
36   עַל־שִׁלֵּשִׁים וְעַל־רִבֵּעִים׃ וַיְמַהֵר מֹשֶׁה וַיִּקֹּד אַרְצָה
37   וַיִּשְׁתָּחוּ׃ וַיֹּאמֶר אִם־נָא מָצָאתִי חֵן בְּעֵינֶיךָ אֲדֹנָי יֵלֶךְ־נָא
38   אֲדֹנָי בְּקִרְבֵּנוּ כִּי עַם־קְשֵׁה־עֹרֶף הוּא וְסָלַחְתָּ לַעֲו‍ֹנֵנוּ
39   וּלְחַטָּאתֵנוּ וּנְחַלְתָּנוּ׃  וַיֹּאמֶר הִנֵּה אָנֹכִי כֹּרֵת בְּרִית
40   נֶגֶד כָּל־עַמְּךָ אֶעֱשֶׂה נִפְלָאֹת אֲשֶׁר לֹא־נִבְרְאוּ בְכָל־
41  הָאָרֶץ וּבְכָל־הַגּוֹיִם וְרָאָה כָל־הָעָם אֲשֶׁר־אַתָּה בְקִרְבּוֹ
42   אֶת־מַעֲשֵׂה יְהוָה כִּי־נוֹרָא הוּא אֲשֶׁר אֲנִי עֹשֶׂה עִמָּךְ׃
