 1  וַיִּרְדֹּף אַחֲרֵי בְּנֵי יִשְׂרָאֵל וּבְנֵי יִשְׂרָאֵל יֹצְאִים בְּיָד
 2   רָמָה׃♢  וַיִּרְדְּפוּ מִצְרַיִם אַחֲרֵיהֶם וַיַּשִּׂיגוּ אוֹתָם חֹנִים
 3   עַל־הַיָּם כָּל־סוּס רֶכֶב פַּרְעֹה וּפָרָשָׁיו וְחֵילוֹ עַל־
 4  פִּי הַחִירֹת לִפְנֵי בַּעַל צְפֹן׃ וּפַרְעֹה הִקְרִיב וַיִּשְׂאוּ
 5   בְנֵי־יִשְׂרָאֵל אֶת־עֵינֵיהֶם וְהִנֵּה מִצְרַיִם  נֹסֵעַ אַחֲרֵיהֶם
 6   וַיִּירְאוּ מְאֹד וַיִּצְעֲקוּ בְנֵי־יִשְׂרָאֵל אֶל־יְהוָה׃ וַיֹּאמְרוּ
 7   אֶל־מֹשֶׁה הֲמִבְּלִי אֵין־קְבָרִים בְּמִצְרַיִם לְקַחְתָּנוּ
 8   לָמוּת בַּמִּדְבָּר מַה־זֹּאת עָשִׂיתָ לָּנוּ לְהוֹצִיאָנוּ
 9   מִמִּצְרָיִם׃ הֲלֹא־זֶה הַדָּבָר אֲשֶׁר דִּבַּרְנוּ אֵלֶיךָ
10   בְמִצְרַיִם לֵאמֹר חֲדַל מִמֶּנּוּ וְנַעַבְדָה אֶת־מִצְרָיִם כִּי
11   טוֹב לָנוּ עֲבֹד אֶת־מִצְרַיִם מִמֻּתֵנוּ בַּמִּדְבָּר׃ וַיֹּאמֶר
12   מֹשֶׁה אֶל־הָעָם אַל־תִּירָאוּ הִתְיַצְּבוּ וּרְאוּ אֶת־יְשׁוּעַת
13   יְהוָה אֲשֶׁר־יַעֲשֶׂה לָכֶם הַיּוֹם כִּי אֲשֶׁר רְאִיתֶם אֶת־
14  מִצְרַיִם הַיּוֹם לֹא תֹסִפוּ לִרְאֹתָם עוֹד עַד־עוֹלָם׃ יְהוָה
15   יִלָּחֵם לָכֶם וְאַתֶּם תַּחֲרִשׁוּן׃ [פ]
16    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה מַה־תִּצְעַק אֵלָי דַּבֵּר אֶל־
17  בְּנֵי־יִשְׂרָאֵל וְיִסָּעוּ׃ וְאַתָּה הָרֵם אֶת־מַטְּךָ וּנְטֵה אֶת־
18  יָדְךָ עַל־הַיָּם וּבְקָעֵהוּ וְיָבֹאוּ בְנֵי־יִשְׂרָאֵל בְּתוֹךְ
19   הַיָּם בַּיַּבָּשָׁה׃ וַאֲנִי הִנְנִי מְחַזֵּק אֶת־לֵב מִצְרַיִם וְיָבֹאוּ
20   אַחֲרֵיהֶם וְאִכָּבְדָה בְּפַרְעֹה וּבְכָל־חֵילוֹ בְּרִכְבּוֹ
21   וּבְפָרָשָׁיו׃ וְיָדְעוּ מִצְרַיִם כִּי־אֲנִי יְהוָה בְּהִכָּבְדִי
22   בְּפַרְעֹה בְּרִכְבּוֹ וּבְפָרָשָׁיו׃ וַיִּסַּע מַלְאַךְ הָאֱלֹהִים
23   הַהֹלֵךְ לִפְנֵי מַחֲנֵה יִשְׂרָאֵל וַיֵּלֶךְ מֵאַחֲרֵיהֶם וַיִּסַּע
24   עַמּוּד הֶעָנָן מִפְּנֵיהֶם וַיַּעֲמֹד מֵאַחֲרֵיהֶם׃ וַיָּבֹא בֵּין
25   מַחֲנֵה מִצְרַיִם וּבֵין מַחֲנֵה יִשְׂרָאֵל וַיְהִי הֶעָנָן וְהַחֹשֶׁךְ
26   וַיָּאֶר אֶת־הַלָּיְלָה וְלֹא־קָרַב זֶה אֶל־זֶה כָּל־הַלָּיְלָה׃
27   וַיֵּט מֹשֶׁה אֶת־יָדוֹ עַל־הַיָּם וַיּוֹלֶךְ יְהוָה  אֶת־הַיָּם
28   בְּרוּחַ קָדִים עַזָּה כָּל־הַלַּיְלָה וַיָּשֶׂם אֶת־הַיָּם לֶחָרָבָה
29   וַיִּבָּקְעוּ הַמָּיִם׃ וַיָּבֹאוּ בְנֵי־יִשְׂרָאֵל בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה
30   וְהַמַּיִם לָהֶם חוֹמָה מִימִינָם וּמִשְּׂמֹאלָם׃ וַיִּרְדְּפוּ
31   מִצְרַיִם וַיָּבֹאוּ אַחֲרֵיהֶם כֹּל סוּס פַּרְעֹה רִכְבּוֹ
32   וּפָרָשָׁיו אֶל־תּוֹךְ הַיָּם׃ וַיְהִי בְּאַשְׁמֹרֶת הַבֹּקֶר וַיַּשְׁקֵף
33   יְהוָה אֶל־מַחֲנֵה מִצְרַיִם בְּעַמּוּד אֵשׁ וְעָנָן וַיָּהָם אֵת
34   מַחֲנֵה מִצְרָיִם׃ וַיָּסַר אֵת אֹפַן מַרְכְּבֹתָיו וַיְנַהֲגֵהוּ
35   בִּכְבֵדֻת וַיֹּאמֶר מִצְרַיִם אָנוּסָה מִפְּנֵי יִשְׂרָאֵל כִּי
36   יְהוָה נִלְחָם לָהֶם בְּמִצְרָיִם׃ [פ]
37    וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה נְטֵה אֶת־יָדְךָ עַל־הַיָּם
38   וְיָשֻׁבוּ הַמַּיִם עַל־מִצְרַיִם עַל־רִכְבּוֹ וְעַל־פָּרָשָׁיו׃
39   וַיֵּט מֹשֶׁה אֶת־יָדוֹ עַל־הַיָּם וַיָּשָׁב הַיָּם לִפְנוֹת
40   בֹּקֶר לְאֵיתָנוֹ וּמִצְרַיִם נָסִים לִקְרָאתוֹ וַיְנַעֵר
41   יְהוָה אֶת־מִצְרַיִם בְּתוֹךְ הַיָּם׃ וַיָּשֻׁבוּ הַמַּיִם וַיְכַסּוּ
42   אֶת־הָרֶכֶב וְאֶת־הַפָּרָשִׁים לְכֹל חֵיל פַּרְעֹה
