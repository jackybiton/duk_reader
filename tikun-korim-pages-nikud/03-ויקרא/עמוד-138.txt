 1  וּלְבִתּוֹ וּלְאָחִיו׃ וְלַאֲחֹתוֹ הַבְּתוּלָה הַקְּרוֹבָה אֵלָיו
 2   אֲשֶׁר לֹא־הָיְתָה לְאִישׁ לָהּ יִטַּמָּא׃ לֹא יִטַּמָּא בַּעַל
 3   בְּעַמָּיו לְהֵחַלּוֹ׃ לֹא־יִקְרְחוּ קָרְחָה בְּרֹאשָׁם וּפְאַת
 4   זְקָנָם לֹא יְגַלֵּחוּ וּבִבְשָׂרָם לֹא יִשְׂרְטוּ שָׂרָטֶת׃
 5   קְדֹשִׁים יִהְיוּ לֵאלֹהֵיהֶם וְלֹא יְחַלְּלוּ שֵׁם אֱלֹהֵיהֶם
 6   כִּי אֶת־אִשֵּׁי יְהוָה לֶחֶם אֱלֹהֵיהֶם הֵם מַקְרִיבִם
 7   וְהָיוּ קֹדֶשׁ׃♢ אִשָּׁה זֹנָה וַחֲלָלָה לֹא יִקָּחוּ וְאִשָּׁה
 8   גְּרוּשָׁה מֵאִישָׁהּ לֹא יִקָּחוּ כִּי־קָדֹשׁ הוּא לֵאלֹהָיו׃
 9   וְקִדַּשְׁתּוֹ כִּי־אֶת־לֶחֶם אֱלֹהֶיךָ הוּא מַקְרִיב קָדֹשׁ
10   יִהְיֶה־לָּךְ כִּי קָדוֹשׁ אֲנִי יְהוָה מְקַדִּשְׁכֶם׃ וּבַת אִישׁ
11   כֹּהֵן כִּי תֵחֵל לִזְנוֹת אֶת־אָבִיהָ הִיא מְחַלֶּלֶת בָּאֵשׁ
12   תִּשָּׂרֵף׃ [ס]             וְהַכֹּהֵן הַגָּדוֹל מֵאֶחָיו
13   אֲשֶׁר־יוּצַק עַל־רֹאשׁוֹ  שֶׁמֶן הַמִּשְׁחָה וּמִלֵּא אֶת־יָדוֹ
14   לִלְבֹּשׁ אֶת־הַבְּגָדִים אֶת־רֹאשׁוֹ לֹא יִפְרָע וּבְגָדָיו
15   לֹא יִפְרֹם׃ וְעַל כָּל־נַפְשֹׁת מֵת לֹא יָבֹא לְאָבִיו
16   וּלְאִמּוֹ לֹא יִטַּמָּא׃ וּמִן־הַמִּקְדָּשׁ לֹא יֵצֵא וְלֹא יְחַלֵּל
17   אֵת מִקְדַּשׁ אֱלֹהָיו כִּי נֵזֶר שֶׁמֶן מִשְׁחַת אֱלֹהָיו
18   עָלָיו אֲנִי יְהוָה׃♢ וְהוּא אִשָּׁה בִבְתוּלֶיהָ יִקָּח׃ אַלְמָנָה
19   וּגְרוּשָׁה וַחֲלָלָה זֹנָה אֶת־אֵלֶּה לֹא יִקָּח כִּי אִם־
20  בְּתוּלָה מֵעַמָּיו יִקַּח אִשָּׁה׃ וְלֹא־יְחַלֵּל זַרְעוֹ בְּעַמָּיו
21   כִּי אֲנִי יְהוָה מְקַדְּשׁוֹ׃♢ [ס]              וַיְדַבֵּר
22   יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־אַהֲרֹן לֵאמֹר
23   אִישׁ מִזַּרְעֲךָ לְדֹרֹתָם אֲשֶׁר יִהְיֶה בוֹ מוּם לֹא
24   יִקְרַב לְהַקְרִיב לֶחֶם אֱלֹהָיו׃ כִּי כָל־אִישׁ אֲשֶׁר־
25  בּוֹ מוּם לֹא יִקְרָב אִישׁ עִוֵּר אוֹ פִסֵּחַ אוֹ חָרֻם אוֹ
26   שָׂרוּעַ׃ אוֹ אִישׁ אֲשֶׁר־יִהְיֶה בוֹ שֶׁבֶר רָגֶל אוֹ
27   שֶׁבֶר יָד׃ אוֹ־גִבֵּן אוֹ־דַק אוֹ תְּבַלֻּל בְּעֵינוֹ אוֹ גָרָב
28   אוֹ יַלֶּפֶת אוֹ מְרוֹחַ אָשֶׁךְ׃ כָּל־אִישׁ אֲשֶׁר־בּוֹ מוּם
29   מִזֶּרַע אַהֲרֹן הַכֹּהֵן לֹא יִגַּשׁ לְהַקְרִיב אֶת־אִשֵּׁי
30   יְהוָה מוּם בּוֹ אֵת לֶחֶם אֱלֹהָיו לֹא יִגַּשׁ לְהַקְרִיב׃
31   לֶחֶם אֱלֹהָיו מִקָּדְשֵׁי הַקֳּדָשִׁים וּמִן־הַקֳּדָשִׁים
32   יֹאכֵל׃ אַךְ אֶל־הַפָּרֹכֶת לֹא יָבֹא וְאֶל־הַמִּזְבֵּחַ לֹא
33   יִגַּשׁ כִּי־מוּם בּוֹ וְלֹא יְחַלֵּל אֶת־מִקְדָּשַׁי כִּי אֲנִי
34   יְהוָה מְקַדְּשָׁם׃ וַיְדַבֵּר מֹשֶׁה אֶל־אַהֲרֹן וְאֶל־בָּנָיו
35   וְאֶל־כָּל־בְּנֵי יִשְׂרָאֵל׃ [פ]
36   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־אַהֲרֹן
37   וְאֶל־בָּנָיו וְיִנָּזְרוּ מִקָּדְשֵׁי בְנֵי־יִשְׂרָאֵל וְלֹא יְחַלְּלוּ
38   אֶת־שֵׁם קָדְשִׁי אֲשֶׁר הֵם מַקְדִּשִׁים לִי אֲנִי יְהוָה׃
39   אֱמֹר אֲלֵהֶם לְדֹרֹתֵיכֶם כָּל־אִישׁ  אֲשֶׁר־יִקְרַב
40   מִכָּל־זַרְעֲכֶם אֶל־הַקֳּדָשִׁים אֲשֶׁר יַקְדִּישׁוּ בְנֵי־
41  יִשְׂרָאֵל לַיהוָה וְטֻמְאָתוֹ עָלָיו וְנִכְרְתָה הַנֶּפֶשׁ
42   הַהִוא מִלְּפָנַי אֲנִי יְהוָה׃ אִישׁ אִישׁ מִזֶּרַע אַהֲרֹן
