 1  וְאֶת־עֹלָתָם לִפְנֵי יְהוָה וַתִּקְרֶאנָה אֹתִי כָּאֵלֶּה
 2   וְאָכַלְתִּי חַטָּאת הַיּוֹם הַיִּיטַב בְּעֵינֵי יְהוָה׃ וַיִּשְׁמַע
 3   מֹשֶׁה וַיִּיטַב בְּעֵינָיו׃ [פ]
 4    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר אֲלֵהֶם׃
 5   דַּבְּרוּ אֶל־בְּנֵי יִשְׂרָאֵל לֵאמֹר זֹאת הַחַיָּה אֲשֶׁר
 6   תֹּאכְלוּ מִכָּל־הַבְּהֵמָה אֲשֶׁר עַל־הָאָרֶץ׃ כֹּל  מַפְרֶסֶת
 7   פַּרְסָה וְשֹׁסַעַת שֶׁסַע פְּרָסֹת מַעֲלַת גֵּרָה בַּבְּהֵמָה
 8   אֹתָהּ תֹּאכֵלוּ׃ אַךְ אֶת־זֶה לֹא תֹאכְלוּ מִמַּעֲלֵי הַגֵּרָה
 9   וּמִמַּפְרִסֵי הַפַּרְסָה אֶת־הַגָּמָל כִּי־מַעֲלֵה גֵרָה הוּא
10   וּפַרְסָה אֵינֶנּוּ מַפְרִיס טָמֵא הוּא לָכֶם׃ וְאֶת־הַשָּׁפָן
11   כִּי־מַעֲלֵה גֵרָה הוּא וּפַרְסָה לֹא יַפְרִיס טָמֵא הוּא
12   לָכֶם׃ וְאֶת־הָאַרְנֶבֶת כִּי־מַעֲלַת גֵּרָה הִוא וּפַרְסָה לֹא
13   הִפְרִיסָה טְמֵאָה הִוא לָכֶם׃ וְאֶת־הַחֲזִיר כִּי־מַפְרִיס
14   פַּרְסָה הוּא וְשֹׁסַע שֶׁסַע פַּרְסָה וְהוּא גֵּרָה לֹא־יִגָּר
15   טָמֵא הוּא לָכֶם׃ מִבְּשָׂרָם לֹא תֹאכֵלוּ וּבְנִבְלָתָם לֹא
16   תִגָּעוּ טְמֵאִים הֵם לָכֶם׃ אֶת־זֶה תֹּאכְלוּ מִכֹּל אֲשֶׁר
17   בַּמָּיִם כֹּל אֲשֶׁר־לוֹ סְנַפִּיר וְקַשְׂקֶשֶׂת בַּמַּיִם בַּיַּמִּים
18   וּבַנְּחָלִים אֹתָם תֹּאכֵלוּ׃ וְכֹל אֲשֶׁר אֵין־לוֹ סְנַפִּיר
19   וְקַשְׂקֶשֶׂת בַּיַּמִּים וּבַנְּחָלִים מִכֹּל שֶׁרֶץ הַמַּיִם וּמִכֹּל
20   נֶפֶשׁ הַחַיָּה אֲשֶׁר בַּמָּיִם שֶׁקֶץ הֵם לָכֶם׃ וְשֶׁקֶץ יִהְיוּ
21   לָכֶם מִבְּשָׂרָם לֹא תֹאכֵלוּ וְאֶת־נִבְלָתָם תְּשַׁקֵּצוּ׃
22   כֹּל אֲשֶׁר אֵין־לוֹ סְנַפִּיר וְקַשְׂקֶשֶׂת בַּמָּיִם שֶׁקֶץ
23   הוּא לָכֶם׃ וְאֶת־אֵלֶּה תְּשַׁקְּצוּ מִן־הָעוֹף לֹא יֵאָכְלוּ
24   שֶׁקֶץ הֵם אֶת־הַנֶּשֶׁר וְאֶת־הַפֶּרֶס וְאֵת הָעָזְנִיָּה׃ וְאֶת־
25  הַדָּאָה וְאֶת־הָאַיָּה לְמִינָהּ׃ אֵת כָּל־עֹרֵב לְמִינוֹ׃ וְאֵת
26   בַּת הַיַּעֲנָה וְאֶת־הַתַּחְמָס וְאֶת־הַשָּׁחַף וְאֶת־הַנֵּץ
27   לְמִינֵהוּ׃ וְאֶת־הַכּוֹס וְאֶת־הַשָּׁלָךְ וְאֶת־הַיַּנְשׁוּף׃ וְאֶת־
28  הַתִּנְשֶׁמֶת וְאֶת־הַקָּאָת וְאֶת־הָרָחָם׃ וְאֵת הַחֲסִידָה
29   הָאֲנָפָה לְמִינָהּ וְאֶת־הַדּוּכִיפַת וְאֶת־הָעֲטַלֵּף׃ כֹּל
30   שֶׁרֶץ הָעוֹף הַהֹלֵךְ עַל־אַרְבַּע שֶׁקֶץ הוּא לָכֶם׃ אַךְ
31   אֶת־זֶה תֹּאכְלוּ מִכֹּל שֶׁרֶץ הָעוֹף הַהֹלֵךְ עַל־אַרְבַּע
32   אֲשֶׁר־לוֹ כְרָעַיִם מִמַּעַל לְרַגְלָיו לְנַתֵּר בָּהֵן עַל־
33  הָאָרֶץ׃ אֶת־אֵלֶּה מֵהֶם תֹּאכֵלוּ אֶת־הָאַרְבֶּה לְמִינוֹ
34   וְאֶת־הַסָּלְעָם לְמִינֵהוּ וְאֶת־הַחַרְגֹּל לְמִינֵהוּ וְאֶת־הֶחָגָב
35   לְמִינֵהוּ׃ וְכֹל שֶׁרֶץ הָעוֹף אֲשֶׁר־לוֹ אַרְבַּע רַגְלָיִם
36   שֶׁקֶץ הוּא לָכֶם׃ וּלְאֵלֶּה תִּטַּמָּאוּ כָּל־הַנֹּגֵעַ בְּנִבְלָתָם
37   יִטְמָא עַד־הָעָרֶב׃ וְכָל־הַנֹּשֵׂא מִנִּבְלָתָם יְכַבֵּס בְּגָדָיו
38   וְטָמֵא עַד־הָעָרֶב׃ לְכָל־הַבְּהֵמָה אֲשֶׁר הִוא מַפְרֶסֶת
39   פַּרְסָה וְשֶׁסַע  אֵינֶנָּה שֹׁסַעַת וְגֵרָה אֵינֶנָּה מַעֲלָה
40   טְמֵאִים הֵם לָכֶם כָּל־הַנֹּגֵעַ בָּהֶם יִטְמָא׃ וְכֹל  הוֹלֵךְ
41   עַל־כַּפָּיו בְּכָל־הַחַיָּה הַהֹלֶכֶת עַל־אַרְבַּע טְמֵאִים
42   הֵם לָכֶם כָּל־הַנֹּגֵעַ בְּנִבְלָתָם יִטְמָא עַד־הָעָרֶב׃
