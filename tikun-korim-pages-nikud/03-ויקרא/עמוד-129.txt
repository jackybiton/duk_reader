 1  וְלָקַח הַכֹּהֵן מִדַּם הָאָשָׁם וְנָתַן עַל־תְּנוּךְ אֹזֶן־הַמִּטַּהֵר
 2   הַיְמָנִית וְעַל־בֹּהֶן יָדוֹ הַיְמָנִית וְעַל־בֹּהֶן רַגְלוֹ הַיְמָנִית׃
 3   וּמִן־הַשֶּׁמֶן יִצֹק הַכֹּהֵן עַל־כַּף הַכֹּהֵן הַשְּׂמָאלִית׃
 4   וְהִזָּה הַכֹּהֵן בְּאֶצְבָּעוֹ הַיְמָנִית מִן־הַשֶּׁמֶן אֲשֶׁר עַל־
 5  כַּפּוֹ הַשְּׂמָאלִית שֶׁבַע פְּעָמִים לִפְנֵי יְהוָה׃ וְנָתַן
 6   הַכֹּהֵן מִן־הַשֶּׁמֶן  אֲשֶׁר עַל־כַּפּוֹ עַל־תְּנוּךְ אֹזֶן
 7   הַמִּטַּהֵר הַיְמָנִית וְעַל־בֹּהֶן יָדוֹ הַיְמָנִית וְעַל־בֹּהֶן
 8   רַגְלוֹ הַיְמָנִית עַל־מְקוֹם דַּם הָאָשָׁם׃ וְהַנּוֹתָר מִן־
 9  הַשֶּׁמֶן אֲשֶׁר עַל־כַּף הַכֹּהֵן יִתֵּן עַל־רֹאשׁ הַמִּטַּהֵר
10   לְכַפֵּר עָלָיו לִפְנֵי יְהוָה׃ וְעָשָׂה אֶת־הָאֶחָד מִן־
11  הַתֹּרִים אוֹ מִן־בְּנֵי הַיּוֹנָה מֵאֲשֶׁר תַּשִּׂיג יָדוֹ׃ אֵת
12   אֲשֶׁר־תַּשִּׂיג יָדוֹ אֶת־הָאֶחָד חַטָּאת וְאֶת־הָאֶחָד
13   עֹלָה עַל־הַמִּנְחָה וְכִפֶּר הַכֹּהֵן עַל הַמִּטַּהֵר לִפְנֵי
14   יְהוָה׃ זֹאת תּוֹרַת אֲשֶׁר־בּוֹ נֶגַע צָרָעַת אֲשֶׁר לֹא־
15  תַשִּׂיג יָדוֹ בְּטָהֳרָתוֹ׃ [פ]
16    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ כִּי תָבֹאוּ
17   אֶל־אֶרֶץ כְּנַעַן אֲשֶׁר אֲנִי נֹתֵן לָכֶם לַאֲחֻזָּה וְנָתַתִּי
18   נֶגַע צָרַעַת בְּבֵית אֶרֶץ אֲחֻזַּתְכֶם׃ וּבָא אֲשֶׁר־לוֹ
19   הַבַּיִת וְהִגִּיד לַכֹּהֵן לֵאמֹר כְּנֶגַע נִרְאָה לִי בַּבָּיִת׃
20   וְצִוָּה הַכֹּהֵן וּפִנּוּ אֶת־הַבַּיִת בְּטֶרֶם יָבֹא הַכֹּהֵן
21   לִרְאוֹת אֶת־הַנֶּגַע וְלֹא יִטְמָא כָּל־אֲשֶׁר בַּבָּיִת וְאַחַר
22   כֵּן יָבֹא הַכֹּהֵן לִרְאוֹת אֶת־הַבָּיִת׃ וְרָאָה אֶת־הַנֶּגַע
23   וְהִנֵּה הַנֶּגַע בְּקִירֹת הַבַּיִת שְׁקַעֲרוּרֹת יְרַקְרַקֹּת אוֹ
24   אֲדַמְדַּמֹּת וּמַרְאֵיהֶן שָׁפָל מִן־הַקִּיר׃ וְיָצָא הַכֹּהֵן
25   מִן־הַבַּיִת אֶל־פֶּתַח הַבָּיִת וְהִסְגִּיר אֶת־הַבַּיִת שִׁבְעַת
26   יָמִים׃ וְשָׁב הַכֹּהֵן בַּיּוֹם הַשְּׁבִיעִי וְרָאָה וְהִנֵּה פָּשָׂה
27   הַנֶּגַע בְּקִירֹת הַבָּיִת׃ וְצִוָּה הַכֹּהֵן וְחִלְּצוּ אֶת־הָאֲבָנִים
28   אֲשֶׁר בָּהֵן הַנָּגַע וְהִשְׁלִיכוּ אֶתְהֶן אֶל־מִחוּץ לָעִיר
29   אֶל־מָקוֹם טָמֵא׃ וְאֶת־הַבַּיִת יַקְצִעַ מִבַּיִת סָבִיב
30   וְשָׁפְכוּ אֶת־הֶעָפָר אֲשֶׁר הִקְצוּ אֶל־מִחוּץ לָעִיר
31   אֶל־מָקוֹם טָמֵא׃ וְלָקְחוּ אֲבָנִים אֲחֵרוֹת וְהֵבִיאוּ אֶל־
32  תַּחַת הָאֲבָנִים וְעָפָר אַחֵר יִקַּח וְטָח אֶת־הַבָּיִת׃ וְאִם־
33  יָשׁוּב הַנֶּגַע וּפָרַח בַּבַּיִת אַחַר חִלֵּץ אֶת־הָאֲבָנִים
34   וְאַחֲרֵי הִקְצוֹת אֶת־הַבַּיִת וְאַחֲרֵי הִטּוֹחַ׃ וּבָא הַכֹּהֵן
35   וְרָאָה וְהִנֵּה פָּשָׂה הַנֶּגַע בַּבָּיִת צָרַעַת מַמְאֶרֶת הִוא
36   בַּבַּיִת טָמֵא הוּא׃ וְנָתַץ אֶת־הַבַּיִת אֶת־אֲבָנָיו וְאֶת־
37  עֵצָיו וְאֵת כָּל־עֲפַר הַבָּיִת וְהוֹצִיא אֶל־מִחוּץ לָעִיר
38   אֶל־מָקוֹם טָמֵא׃ וְהַבָּא אֶל־הַבַּיִת כָּל־יְמֵי הִסְגִּיר
39   אֹתוֹ יִטְמָא עַד־הָעָרֶב׃ וְהַשֹּׁכֵב בַּבַּיִת יְכַבֵּס אֶת־
40  בְּגָדָיו וְהָאֹכֵל בַּבַּיִת יְכַבֵּס אֶת־בְּגָדָיו׃ וְאִם־בֹּא יָבֹא
41   הַכֹּהֵן וְרָאָה וְהִנֵּה לֹא־פָשָׂה הַנֶּגַע בַּבַּיִת אַחֲרֵי הִטֹּחַ
42   אֶת־הַבָּיִת וְטִהַר הַכֹּהֵן אֶת־הַבַּיִת כִּי נִרְפָּא הַנָּגַע׃
