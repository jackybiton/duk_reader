 1  וְאִשָּׁה לֹא־תַעֲמֹד לִפְנֵי בְהֵמָה לְרִבְעָהּ תֶּבֶל הוּא׃
 2   אַל־תִּטַּמְּאוּ בְּכָל־אֵלֶּה כִּי בְכָל־אֵלֶּה נִטְמְאוּ הַגּוֹיִם
 3   אֲשֶׁר־אֲנִי מְשַׁלֵּחַ מִפְּנֵיכֶם׃ וַתִּטְמָא הָאָרֶץ וָאֶפְקֹד
 4   עֲו‍ֹנָהּ עָלֶיהָ וַתָּקִא הָאָרֶץ אֶת־יֹשְׁבֶיהָ׃ וּשְׁמַרְתֶּם
 5   אַתֶּם אֶת־חֻקֹּתַי וְאֶת־מִשְׁפָּטַי וְלֹא תַעֲשׂוּ מִכֹּל
 6   הַתּוֹעֵבֹת הָאֵלֶּה הָאֶזְרָח וְהַגֵּר הַגָּר בְּתוֹכְכֶם׃
 7   כִּי אֶת־כָּל־הַתּוֹעֵבֹת הָאֵל עָשׂוּ אַנְשֵׁי־הָאָרֶץ
 8   אֲשֶׁר לִפְנֵיכֶם וַתִּטְמָא הָאָרֶץ׃  וְלֹא־תָקִיא הָאָרֶץ
 9   אֶתְכֶם בְּטַמַּאֲכֶם אֹתָהּ כַּאֲשֶׁר קָאָה אֶת־הַגּוֹי
10   אֲשֶׁר לִפְנֵיכֶם׃ כִּי כָּל־אֲשֶׁר יַעֲשֶׂה מִכֹּל הַתּוֹעֵבֹת
11   הָאֵלֶּה וְנִכְרְתוּ הַנְּפָשׁוֹת הָעֹשֹׂת מִקֶּרֶב עַמָּם׃
12   וּשְׁמַרְתֶּם אֶת־מִשְׁמַרְתִּי לְבִלְתִּי עֲשׂוֹת מֵחֻקּוֹת
13   הַתּוֹעֵבֹת אֲשֶׁר נַעֲשׂוּ לִפְנֵיכֶם וְלֹא תִטַּמְּאוּ בָּהֶם
14   אֲנִי יְהוָה אֱלֹהֵיכֶם׃ [פ]
15  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־כָּל־עֲדַת
16   בְּנֵי־יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם קְדֹשִׁים תִּהְיוּ כִּי
17   קָדוֹשׁ אֲנִי יְהוָה אֱלֹהֵיכֶם׃ אִישׁ אִמּוֹ וְאָבִיו תִּירָאוּ
18   וְאֶת־שַׁבְּתֹתַי תִּשְׁמֹרוּ אֲנִי יְהוָה אֱלֹהֵיכֶם׃ אַל־תִּפְנוּ
19   אֶל־הָאֱלִילִם וֵאלֹהֵי מַסֵּכָה לֹא תַעֲשׂוּ לָכֶם אֲנִי
20   יְהוָה אֱלֹהֵיכֶם׃♢ וְכִי תִזְבְּחוּ זֶבַח שְׁלָמִים לַיהוָה
21   לִרְצֹנְכֶם תִּזְבָּחֻהוּ׃ בְּיוֹם זִבְחֲכֶם יֵאָכֵל וּמִמָּחֳרָת
22   וְהַנּוֹתָר עַד־יוֹם הַשְּׁלִישִׁי בָּאֵשׁ יִשָּׂרֵף׃ וְאִם הֵאָכֹל
23   יֵאָכֵל בַּיּוֹם הַשְּׁלִישִׁי פִּגּוּל הוּא לֹא יֵרָצֶה׃ וְאֹכְלָיו
24   עֲו‍ֹנוֹ יִשָּׂא כִּי־אֶת־קֹדֶשׁ יְהוָה חִלֵּל וְנִכְרְתָה הַנֶּפֶשׁ
25   הַהִוא מֵעַמֶּיהָ׃ וּבְקֻצְרְכֶם אֶת־קְצִיר אַרְצְכֶם לֹא
26   תְכַלֶּה פְּאַת שָׂדְךָ לִקְצֹר וְלֶקֶט קְצִירְךָ לֹא תְלַקֵּט׃
27   וְכַרְמְךָ לֹא תְעוֹלֵל וּפֶרֶט כַּרְמְךָ לֹא תְלַקֵּט לֶעָנִי
28   וְלַגֵּר תַּעֲזֹב אֹתָם אֲנִי יְהוָה אֱלֹהֵיכֶם׃♢ לֹא תִּגְנֹבוּ
29   וְלֹא־תְכַחֲשׁוּ וְלֹא־תְשַׁקְּרוּ אִישׁ בַּעֲמִיתוֹ׃ וְלֹא־
30  תִשָּׁבְעוּ בִשְׁמִי לַשָּׁקֶר וְחִלַּלְתָּ אֶת־שֵׁם אֱלֹהֶיךָ
31   אֲנִי יְהוָה׃ לֹא־תַעֲשֹׁק אֶת־רֵעֲךָ וְלֹא תִגְזֹל לֹא־תָלִין
32   פְּעֻלַּת שָׂכִיר אִתְּךָ עַד־בֹּקֶר׃ לֹא־תְקַלֵּל חֵרֵשׁ
33   וְלִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁל וְיָרֵאתָ מֵּאֱלֹהֶיךָ אֲנִי
34   יְהוָה׃♢  לֹא־תַעֲשׂוּ עָוֶל בַּמִּשְׁפָּט לֹא־תִשָּׂא פְנֵי־דָל
35   וְלֹא תֶהְדַּר פְּנֵי גָדוֹל בְּצֶדֶק תִּשְׁפֹּט עֲמִיתֶךָ׃ לֹא־
36  תֵלֵךְ רָכִיל בְּעַמֶּיךָ לֹא תַעֲמֹד עַל־דַּם רֵעֶךָ אֲנִי
37   יְהוָה׃ לֹא־תִשְׂנָא אֶת־אָחִיךָ בִּלְבָבֶךָ הוֹכֵחַ תּוֹכִיחַ
38   אֶת־עֲמִיתֶךָ וְלֹא־תִשָּׂא עָלָיו חֵטְא׃ לֹא־תִקֹּם וְלֹא־
39  תִטֹּר אֶת־בְּנֵי עַמֶּךָ וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ אֲנִי יְהוָה׃
40   אֶת־חֻקֹּתַי תִּשְׁמֹרוּ בְּהֶמְתְּךָ לֹא־תַרְבִּיעַ כִּלְאַיִם
41   שָׂדְךָ לֹא־תִזְרַע כִּלְאָיִם וּבֶגֶד כִּלְאַיִם שַׁעַטְנֵז לֹא
42   יַעֲלֶה עָלֶיךָ׃ וְאִישׁ כִּי־יִשְׁכַּב אֶת־אִשָּׁה שִׁכְבַת־זֶרַע
