 1  וְכִי־יִהְיֶה בַקָּרַחַת אוֹ בַגַּבַּחַת נֶגַע לָבָן אֲדַמְדָּם
 2   צָרַעַת פֹּרַחַת הִוא בְּקָרַחְתּוֹ אוֹ בְגַבַּחְתּוֹ׃ וְרָאָה אֹתוֹ
 3   הַכֹּהֵן וְהִנֵּה שְׂאֵת־הַנֶּגַע לְבָנָה אֲדַמְדֶּמֶת בְּקָרַחְתּוֹ
 4   אוֹ בְגַבַּחְתּוֹ כְּמַרְאֵה צָרַעַת עוֹר בָּשָׂר׃ אִישׁ־צָרוּעַ
 5   הוּא טָמֵא הוּא טַמֵּא יְטַמְּאֶנּוּ הַכֹּהֵן בְּרֹאשׁוֹ נִגְעוֹ׃
 6   וְהַצָּרוּעַ אֲשֶׁר־בּוֹ הַנֶּגַע בְּגָדָיו יִהְיוּ פְרֻמִים וְרֹאשׁוֹ
 7   יִהְיֶה פָרוּעַ וְעַל־שָׂפָם יַעְטֶה וְטָמֵא  טָמֵא יִקְרָא׃
 8   כָּל־יְמֵי אֲשֶׁר הַנֶּגַע בּוֹ יִטְמָא טָמֵא הוּא בָּדָד יֵשֵׁב
 9   מִחוּץ לַמַּחֲנֶה מוֹשָׁבוֹ׃ [ס]             וְהַבֶּגֶד
10   כִּי־יִהְיֶה בוֹ נֶגַע צָרָעַת בְּבֶגֶד צֶמֶר אוֹ בְּבֶגֶד
11   פִּשְׁתִּים׃ אוֹ בִשְׁתִי אוֹ בְעֵרֶב לַפִּשְׁתִּים וְלַצָּמֶר
12   אוֹ בְעוֹר אוֹ בְּכָל־מְלֶאכֶת עוֹר׃ וְהָיָה הַנֶּגַע יְרַקְרַק
13   אוֹ אֲדַמְדָּם בַּבֶּגֶד אוֹ בָעוֹר אוֹ־בַשְּׁתִי אוֹ־בָעֵרֶב
14   אוֹ בְכָל־כְּלִי־עוֹר נֶגַע צָרַעַת הוּא וְהָרְאָה אֶת־
15  הַכֹּהֵן׃ וְרָאָה הַכֹּהֵן אֶת־הַנָּגַע וְהִסְגִּיר אֶת־הַנֶּגַע
16   שִׁבְעַת יָמִים׃ וְרָאָה אֶת־הַנֶּגַע בַּיּוֹם הַשְּׁבִיעִי כִּי־
17  פָשָׂה הַנֶּגַע בַּבֶּגֶד אוֹ־בַשְּׁתִי אוֹ־בָעֵרֶב אוֹ בָעוֹר
18   לְכֹל אֲשֶׁר־יֵעָשֶׂה הָעוֹר לִמְלָאכָה צָרַעַת מַמְאֶרֶת
19   הַנֶּגַע טָמֵא הוּא׃ וְשָׂרַף אֶת־הַבֶּגֶד אוֹ אֶת־הַשְּׁתִי
20   אוֹ אֶת־הָעֵרֶב בַּצֶּמֶר אוֹ בַפִּשְׁתִּים אוֹ אֶת־כָּל־
21  כְּלִי הָעוֹר אֲשֶׁר־יִהְיֶה בוֹ הַנָּגַע כִּי־צָרַעַת מַמְאֶרֶת
22   הִוא בָּאֵשׁ תִּשָּׂרֵף׃ וְאִם יִרְאֶה הַכֹּהֵן וְהִנֵּה לֹא־
23  פָשָׂה הַנֶּגַע בַּבֶּגֶד אוֹ בַשְּׁתִי אוֹ בָעֵרֶב אוֹ בְּכָל־
24  כְּלִי־עוֹר׃ וְצִוָּה הַכֹּהֵן וְכִבְּסוּ אֵת אֲשֶׁר־בּוֹ הַנָּגַע
25   וְהִסְגִּירוֹ שִׁבְעַת־יָמִים שֵׁנִית׃  וְרָאָה הַכֹּהֵן אַחֲרֵי
26   הֻכַּבֵּס אֶת־הַנֶּגַע וְהִנֵּה לֹא־הָפַךְ הַנֶּגַע אֶת־עֵינוֹ
27   וְהַנֶּגַע לֹא־פָשָׂה טָמֵא הוּא בָּאֵשׁ תִּשְׂרְפֶנּוּ פְּחֶתֶת
28   הִוא בְּקָרַחְתּוֹ אוֹ בְגַבַּחְתּוֹ׃ וְאִם רָאָה הַכֹּהֵן וְהִנֵּה
29   כֵּהָה הַנֶּגַע אַחֲרֵי הֻכַּבֵּס אֹתוֹ וְקָרַע אֹתוֹ מִן־הַבֶּגֶד
30   אוֹ מִן־הָעוֹר אוֹ מִן־הַשְּׁתִי אוֹ מִן־הָעֵרֶב׃  וְאִם־תֵּרָאֶה
31   עוֹד בַּבֶּגֶד אוֹ־בַשְּׁתִי אוֹ־בָעֵרֶב אוֹ בְכָל־כְּלִי־עוֹר
32   פֹּרַחַת הִוא בָּאֵשׁ תִּשְׂרְפֶנּוּ אֵת אֲשֶׁר־בּוֹ הַנָּגַע׃
33   וְהַבֶּגֶד אוֹ־הַשְּׁתִי אוֹ־הָעֵרֶב אוֹ־כָל־כְּלִי הָעוֹר
34   אֲשֶׁר תְּכַבֵּס וְסָר מֵהֶם הַנָּגַע וְכֻבַּס שֵׁנִית וְטָהֵר׃
35   זֹאת תּוֹרַת נֶגַע־צָרַעַת בֶּגֶד הַצֶּמֶר  אוֹ הַפִּשְׁתִּים
36   אוֹ הַשְּׁתִי אוֹ הָעֵרֶב אוֹ כָּל־כְּלִי־עוֹר לְטַהֲרוֹ אוֹ
37   לְטַמְּאוֹ׃ [פ]
38  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ זֹאת תִּהְיֶה תּוֹרַת
39   הַמְּצֹרָע בְּיוֹם טָהֳרָתוֹ וְהוּבָא אֶל־הַכֹּהֵן׃ וְיָצָא הַכֹּהֵן
40   אֶל־מִחוּץ לַמַּחֲנֶה וְרָאָה הַכֹּהֵן וְהִנֵּה נִרְפָּא נֶגַע־
41  הַצָּרַעַת מִן־הַצָּרוּעַ׃ וְצִוָּה הַכֹּהֵן וְלָקַח לַמִּטַּהֵר שְׁתֵּי־
42  צִפֳּרִים חַיּוֹת טְהֹרוֹת וְעֵץ אֶרֶז וּשְׁנִי תוֹלַעַת וְאֵזֹב׃
