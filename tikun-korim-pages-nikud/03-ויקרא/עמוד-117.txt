 1  וְכִפֶּר עָלָיו הַכֹּהֵן לִפְנֵי יְהוָה וְנִסְלַח לוֹ עַל־אַחַת
 2   מִכֹּל אֲשֶׁר־יַעֲשֶׂה לְאַשְׁמָה בָהּ׃ [פ]
 3  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ צַו אֶת־אַהֲרֹן וְאֶת־בָּנָיו
 4   לֵאמֹר זֹאת תּוֹרַת הָעֹלָה הִוא הָעֹלָה עַל מוֹקְדָה עַל־
 5  הַמִּזְבֵּחַ כָּל־הַלַּיְלָה עַד־הַבֹּקֶר וְאֵשׁ הַמִּזְבֵּחַ תּוּקַד
 6   בּוֹ׃ וְלָבַשׁ הַכֹּהֵן מִדּוֹ בַד וּמִכְנְסֵי־בַד יִלְבַּשׁ עַל־
 7  בְּשָׂרוֹ וְהֵרִים אֶת־הַדֶּשֶׁן אֲשֶׁר תֹּאכַל הָאֵשׁ אֶת־
 8  הָעֹלָה עַל־הַמִּזְבֵּחַ וְשָׂמוֹ אֵצֶל הַמִּזְבֵּחַ׃♢ וּפָשַׁט אֶת־
 9  בְּגָדָיו וְלָבַשׁ בְּגָדִים אֲחֵרִים וְהוֹצִיא אֶת־הַדֶּשֶׁן אֶל־
10  מִחוּץ לַמַּחֲנֶה אֶל־מָקוֹם טָהוֹר׃ וְהָאֵשׁ עַל־הַמִּזְבֵּחַ
11   תּוּקַד־בּוֹ לֹא תִכְבֶּה וּבִעֵר עָלֶיהָ הַכֹּהֵן עֵצִים
12   בַּבֹּקֶר בַּבֹּקֶר וְעָרַךְ עָלֶיהָ הָעֹלָה וְהִקְטִיר עָלֶיהָ
13   חֶלְבֵי הַשְּׁלָמִים׃ אֵשׁ תָּמִיד תּוּקַד עַל־הַמִּזְבֵּחַ לֹא
14   תִכְבֶּה׃♢ [ס]             וְזֹאת תּוֹרַת הַמִּנְחָה
15   הַקְרֵב אֹתָהּ בְּנֵי־אַהֲרֹן לִפְנֵי יְהוָה אֶל־פְּנֵי הַמִּזְבֵּחַ׃
16   וְהֵרִים מִמֶּנּוּ בְּקֻמְצוֹ מִסֹּלֶת הַמִּנְחָה וּמִשַּׁמְנָהּ וְאֵת
17   כָּל־הַלְּבֹנָה אֲשֶׁר עַל־הַמִּנְחָה וְהִקְטִיר הַמִּזְבֵּחַ רֵיחַ
18   נִיחֹחַ אַזְכָּרָתָהּ לַיהוָה׃ וְהַנּוֹתֶרֶת מִמֶּנָּה יֹאכְלוּ אַהֲרֹן
19   וּבָנָיו מַצּוֹת תֵּאָכֵל בְּמָקוֹם קָדֹשׁ בַּחֲצַר אֹהֶל־מוֹעֵד
20   יֹאכְלוּהָ׃ לֹא תֵאָפֶה חָמֵץ חֶלְקָם נָתַתִּי אֹתָהּ מֵאִשָּׁי
21   קֹדֶשׁ קָדָשִׁים הִוא כַּחַטָּאת וְכָאָשָׁם׃ כָּל־זָכָר בִּבְנֵי
22   אַהֲרֹן יֹאכְלֶנָּה חָק־עוֹלָם לְדֹרֹתֵיכֶם מֵאִשֵּׁי יְהוָה
23   כֹּל אֲשֶׁר־יִגַּע בָּהֶם יִקְדָּשׁ׃♢ [פ]
24    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ זֶה קָרְבַּן אַהֲרֹן
25   וּבָנָיו אֲשֶׁר־יַקְרִיבוּ לַיהוָה בְּיוֹם הִמָּשַׁח אֹתוֹ
26   עֲשִׂירִת הָאֵפָה סֹלֶת מִנְחָה תָּמִיד מַחֲצִיתָהּ בַּבֹּקֶר
27   וּמַחֲצִיתָהּ בָּעָרֶב׃ עַל־מַחֲבַת בַּשֶּׁמֶן תֵּעָשֶׂה מֻרְבֶּכֶת
28   תְּבִיאֶנָּה תֻּפִינֵי מִנְחַת פִּתִּים תַּקְרִיב רֵיחַ־נִיחֹחַ
29   לַיהוָה׃ וְהַכֹּהֵן הַמָּשִׁיחַ תַּחְתָּיו מִבָּנָיו יַעֲשֶׂה אֹתָהּ
30   חָק־עוֹלָם לַיהוָה כָּלִיל תָּקְטָר׃ וְכָל־מִנְחַת כֹּהֵן
31   כָּלִיל תִּהְיֶה לֹא תֵאָכֵל׃ [פ]
32   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־אַהֲרֹן וְאֶל־
33  בָּנָיו לֵאמֹר זֹאת תּוֹרַת הַחַטָּאת בִּמְקוֹם אֲשֶׁר תִּשָּׁחֵט
34   הָעֹלָה תִּשָּׁחֵט הַחַטָּאת לִפְנֵי יְהוָה קֹדֶשׁ קָדָשִׁים
35   הִוא׃ הַכֹּהֵן הַמְחַטֵּא אֹתָהּ יֹאכְלֶנָּה בְּמָקוֹם קָדֹשׁ
36   תֵּאָכֵל בַּחֲצַר אֹהֶל מוֹעֵד׃ כֹּל אֲשֶׁר־יִגַּע בִּבְשָׂרָהּ
37   יִקְדָּשׁ וַאֲשֶׁר יִזֶּה מִדָּמָהּ עַל־הַבֶּגֶד אֲשֶׁר יִזֶּה עָלֶיהָ
38   תְּכַבֵּס בְּמָקוֹם קָדֹשׁ׃ וּכְלִי־חֶרֶשׂ אֲשֶׁר תְּבֻשַּׁל־בּוֹ
39   יִשָּׁבֵר וְאִם־בִּכְלִי נְחֹשֶׁת בֻּשָּׁלָה וּמֹרַק וְשֻׁטַּף בַּמָּיִם׃
40   כָּל־זָכָר בַּכֹּהֲנִים יֹאכַל אֹתָהּ קֹדֶשׁ קָדָשִׁים הִוא׃ וְכָל־
41  חַטָּאת אֲשֶׁר יוּבָא מִדָּמָהּ אֶל־אֹהֶל מוֹעֵד לְכַפֵּר
42   בַּקֹּדֶשׁ לֹא תֵאָכֵל בָּאֵשׁ תִּשָּׂרֵף׃ [פ]
