 1  וַיִּסְמֹךְ אַהֲרֹן וּבָנָיו אֶת־יְדֵיהֶם עַל־רֹאשׁ פַּר הַחַטָּאת׃
 2   וַיִּשְׁחָט וַיִּקַּח מֹשֶׁה אֶת־הַדָּם וַיִּתֵּן עַל־קַרְנוֹת הַמִּזְבֵּחַ
 3   סָבִיב בְּאֶצְבָּעוֹ וַיְחַטֵּא אֶת־הַמִּזְבֵּחַ וְאֶת־הַדָּם יָצַק
 4   אֶל־יְסוֹד הַמִּזְבֵּחַ וַיְקַדְּשֵׁהוּ לְכַפֵּר עָלָיו׃ וַיִּקַּח אֶת־
 5  כָּל־הַחֵלֶב אֲשֶׁר עַל־הַקֶּרֶב וְאֵת יֹתֶרֶת הַכָּבֵד וְאֶת־
 6  שְׁתֵּי הַכְּלָיֹת וְאֶת־חֶלְבְּהֶן וַיַּקְטֵר מֹשֶׁה הַמִּזְבֵּחָה׃
 7   וְאֶת־הַפָּר וְאֶת־עֹרוֹ וְאֶת־בְּשָׂרוֹ וְאֶת־פִּרְשׁוֹ שָׂרַף
 8   בָּאֵשׁ מִחוּץ לַמַּחֲנֶה כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃
 9   וַיַּקְרֵב אֵת אֵיל הָעֹלָה וַיִּסְמְכוּ אַהֲרֹן וּבָנָיו אֶת־יְדֵיהֶם
10   עַל־רֹאשׁ הָאָיִל׃ וַיִּשְׁחָט וַיִּזְרֹק מֹשֶׁה אֶת־הַדָּם עַל־
11  הַמִּזְבֵּחַ סָבִיב׃ וְאֶת־הָאַיִל נִתַּח לִנְתָחָיו וַיַּקְטֵר מֹשֶׁה
12   אֶת־הָרֹאשׁ וְאֶת־הַנְּתָחִים וְאֶת־הַפָּדֶר׃ וְאֶת־הַקֶּרֶב וְאֶת־
13  הַכְּרָעַיִם רָחַץ בַּמָּיִם וַיַּקְטֵר מֹשֶׁה אֶת־כָּל־הָאַיִל
14   הַמִּזְבֵּחָה עֹלָה הוּא לְרֵיחַ־נִיחֹחַ אִשֶּׁה הוּא לַיהוָה
15   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃  וַיַּקְרֵב אֶת־הָאַיִל הַשֵּׁנִי
16   אֵיל הַמִּלֻּאִים וַיִּסְמְכוּ אַהֲרֹן וּבָנָיו אֶת־יְדֵיהֶם עַל־
17  רֹאשׁ הָאָיִל׃ וַיִּשְׁחָט  וַיִּקַּח מֹשֶׁה מִדָּמוֹ וַיִּתֵּן עַל־תְּנוּךְ
18   אֹזֶן־אַהֲרֹן הַיְמָנִית וְעַל־בֹּהֶן יָדוֹ הַיְמָנִית וְעַל־בֹּהֶן
19   רַגְלוֹ הַיְמָנִית׃ וַיַּקְרֵב אֶת־בְּנֵי אַהֲרֹן וַיִּתֵּן מֹשֶׁה מִן־
20  הַדָּם עַל־תְּנוּךְ אָזְנָם הַיְמָנִית וְעַל־בֹּהֶן יָדָם הַיְמָנִית
21   וְעַל־בֹּהֶן רַגְלָם הַיְמָנִית וַיִּזְרֹק מֹשֶׁה אֶת־הַדָּם עַל־
22  הַמִּזְבֵּחַ סָבִיב׃ וַיִּקַּח אֶת־הַחֵלֶב וְאֶת־הָאַלְיָה וְאֶת־כָּל־
23  הַחֵלֶב אֲשֶׁר עַל־הַקֶּרֶב וְאֵת יֹתֶרֶת הַכָּבֵד וְאֶת־שְׁתֵּי
24   הַכְּלָיֹת וְאֶת־חֶלְבְּהֶן וְאֵת שׁוֹק הַיָּמִין׃ וּמִסַּל הַמַּצּוֹת
25   אֲשֶׁר  לִפְנֵי יְהוָה לָקַח חַלַּת מַצָּה אַחַת וְחַלַּת לֶחֶם
26   שֶׁמֶן אַחַת וְרָקִיק אֶחָד וַיָּשֶׂם עַל־הַחֲלָבִים וְעַל שׁוֹק
27   הַיָּמִין׃ וַיִּתֵּן אֶת־הַכֹּל עַל כַּפֵּי אַהֲרֹן וְעַל כַּפֵּי בָנָיו
28   וַיָּנֶף אֹתָם תְּנוּפָה לִפְנֵי יְהוָה׃ וַיִּקַּח מֹשֶׁה אֹתָם מֵעַל
29   כַּפֵּיהֶם וַיַּקְטֵר הַמִּזְבֵּחָה עַל־הָעֹלָה מִלֻּאִים הֵם לְרֵיחַ
30   נִיחֹחַ אִשֶּׁה הוּא לַיהוָה׃ וַיִּקַּח מֹשֶׁה אֶת־הֶחָזֶה וַיְנִיפֵהוּ
31   תְנוּפָה לִפְנֵי יְהוָה מֵאֵיל הַמִּלֻּאִים לְמֹשֶׁה הָיָה לְמָנָה
32   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃  וַיִּקַּח מֹשֶׁה מִשֶּׁמֶן
33   הַמִּשְׁחָה וּמִן־הַדָּם אֲשֶׁר עַל־הַמִּזְבֵּחַ וַיַּז עַל־אַהֲרֹן
34   עַל־בְּגָדָיו וְעַל־בָּנָיו וְעַל־בִּגְדֵי בָנָיו אִתּוֹ וַיְקַדֵּשׁ אֶת־
35  אַהֲרֹן אֶת־בְּגָדָיו וְאֶת־בָּנָיו וְאֶת־בִּגְדֵי בָנָיו אִתּוֹ׃ וַיֹּאמֶר
36   מֹשֶׁה אֶל־אַהֲרֹן וְאֶל־בָּנָיו בַּשְּׁלוּ אֶת־הַבָּשָׂר פֶּתַח
37   אֹהֶל מוֹעֵד וְשָׁם תֹּאכְלוּ אֹתוֹ וְאֶת־הַלֶּחֶם אֲשֶׁר בְּסַל
38   הַמִּלֻּאִים כַּאֲשֶׁר צִוֵּיתִי לֵאמֹר אַהֲרֹן וּבָנָיו יֹאכְלֻהוּ׃
39   וְהַנּוֹתָר בַּבָּשָׂר וּבַלָּחֶם בָּאֵשׁ תִּשְׂרֹפוּ׃  וּמִפֶּתַח אֹהֶל
40   מוֹעֵד לֹא תֵצְאוּ שִׁבְעַת יָמִים עַד יוֹם מְלֹאת יְמֵי
41   מִלֻּאֵיכֶם כִּי שִׁבְעַת יָמִים יְמַלֵּא אֶת־יֶדְכֶם׃ כַּאֲשֶׁר
42   עָשָׂה בַּיּוֹם הַזֶּה צִוָּה יְהוָה לַעֲשֹׂת לְכַפֵּר עֲלֵיכֶם׃
