 1  וְכִסָּהוּ בֶּעָפָר׃ כִּי־נֶפֶשׁ כָּל־בָּשָׂר דָּמוֹ בְנַפְשׁוֹ הוּא
 2   וָאֹמַר לִבְנֵי יִשְׂרָאֵל דַּם כָּל־בָּשָׂר לֹא תֹאכֵלוּ
 3   כִּי נֶפֶשׁ כָּל־בָּשָׂר דָּמוֹ הִוא כָּל־אֹכְלָיו יִכָּרֵת׃
 4   וְכָל־נֶפֶשׁ אֲשֶׁר תֹּאכַל נְבֵלָה וּטְרֵפָה בָּאֶזְרָח
 5   וּבַגֵּר וְכִבֶּס בְּגָדָיו וְרָחַץ בַּמַּיִם וְטָמֵא עַד־הָעֶרֶב
 6   וְטָהֵר׃ וְאִם לֹא יְכַבֵּס וּבְשָׂרוֹ לֹא יִרְחָץ וְנָשָׂא
 7   עֲו‍ֹנוֹ׃ [פ]
 8   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
 9   וְאָמַרְתָּ אֲלֵהֶם אֲנִי יְהוָה אֱלֹהֵיכֶם׃ כְּמַעֲשֵׂה אֶרֶץ־
10  מִצְרַיִם אֲשֶׁר יְשַׁבְתֶּם־בָּהּ לֹא תַעֲשׂוּ וּכְמַעֲשֵׂה
11   אֶרֶץ־כְּנַעַן אֲשֶׁר אֲנִי מֵבִיא אֶתְכֶם שָׁמָּה לֹא תַעֲשׂוּ
12   וּבְחֻקֹּתֵיהֶם לֹא תֵלֵכוּ׃ אֶת־מִשְׁפָּטַי תַּעֲשׂוּ וְאֶת־חֻקֹּתַי
13   תִּשְׁמְרוּ לָלֶכֶת בָּהֶם אֲנִי יְהוָה אֱלֹהֵיכֶם׃ וּשְׁמַרְתֶּם
14   אֶת־חֻקֹּתַי וְאֶת־מִשְׁפָּטַי אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם
15   וָחַי בָּהֶם אֲנִי יְהוָה׃ [ס]              אִישׁ
16   אִישׁ אֶל־כָּל־שְׁאֵר בְּשָׂרוֹ לֹא תִקְרְבוּ לְגַלּוֹת עֶרְוָה
17   אֲנִי יְהוָה׃ [ס]             עֶרְוַת
18   אָבִיךָ וְעֶרְוַת אִמְּךָ לֹא תְגַלֵּה אִמְּךָ הִוא לֹא תְגַלֶּה
19   עֶרְוָתָהּ׃ [ס]             עֶרְוַת אֵשֶׁת־אָבִיךָ לֹא
20   תְגַלֵּה עֶרְוַת אָבִיךָ הִוא׃ [ס]             עֶרְוַת
21   אֲחוֹתְךָ בַת־אָבִיךָ אוֹ בַת־אִמֶּךָ מוֹלֶדֶת בַּיִת אוֹ
22   מוֹלֶדֶת חוּץ לֹא תְגַלֶּה עֶרְוָתָן׃ [ס]             עֶרְוַת
23   בַּת־בִּנְךָ אוֹ בַת־בִּתְּךָ לֹא תְגַלֶּה עֶרְוָתָן כִּי עֶרְוָתְךָ
24   הֵנָּה׃ [ס]             עֶרְוַת בַּת־
25  אֵשֶׁת אָבִיךָ מוֹלֶדֶת אָבִיךָ אֲחוֹתְךָ הִוא לֹא תְגַלֶּה
26   עֶרְוָתָהּ׃ [ס]             עֶרְוַת אֲחוֹת־אָבִיךָ לֹא תְגַלֵּה
27   שְׁאֵר אָבִיךָ הִוא׃ [ס]             עֶרְוַת אֲחוֹת־אִמְּךָ
28   לֹא תְגַלֵּה כִּי־שְׁאֵר אִמְּךָ הִוא׃ [ס]             עֶרְוַת
29   אֲחִי־אָבִיךָ לֹא תְגַלֵּה אֶל־אִשְׁתּוֹ לֹא תִקְרָב דֹּדָתְךָ
30   הִוא׃ [ס]             עֶרְוַת
31   כַּלָּתְךָ לֹא תְגַלֵּה אֵשֶׁת בִּנְךָ הִוא לֹא תְגַלֶּה
32   עֶרְוָתָהּ׃ [ס]             עֶרְוַת אֵשֶׁת־אָחִיךָ לֹא תְגַלֵּה
33   עֶרְוַת אָחִיךָ הִוא׃ [ס]             עֶרְוַת
34   אִשָּׁה וּבִתָּהּ לֹא תְגַלֵּה אֶת־בַּת־בְּנָהּ וְאֶת־בַּת־
35  בִּתָּהּ לֹא תִקַּח לְגַלּוֹת עֶרְוָתָהּ שַׁאֲרָה הֵנָּה זִמָּה
36   הִוא׃ וְאִשָּׁה אֶל־אֲחֹתָהּ לֹא תִקָּח לִצְרֹר לְגַלּוֹת
37   עֶרְוָתָהּ עָלֶיהָ בְּחַיֶּיהָ׃ וְאֶל־אִשָּׁה בְּנִדַּת טֻמְאָתָהּ
38   לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ׃ וְאֶל־אֵשֶׁת עֲמִיתְךָ לֹא־
39  תִתֵּן שְׁכָבְתְּךָ לְזָרַע לְטָמְאָה־בָהּ׃ וּמִזַּרְעֲךָ לֹא־תִתֵּן
40   לְהַעֲבִיר לַמֹּלֶךְ וְלֹא תְחַלֵּל אֶת־שֵׁם אֱלֹהֶיךָ אֲנִי
41   יְהוָה׃  וְאֶת־זָכָר לֹא תִשְׁכַּב מִשְׁכְּבֵי אִשָּׁה תּוֹעֵבָה
42   הִוא׃ וּבְכָל־בְּהֵמָה לֹא־תִתֵּן שְׁכָבְתְּךָ לְטָמְאָה־בָהּ
