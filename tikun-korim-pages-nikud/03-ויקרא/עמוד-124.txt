 1  וְהַנֹּשֵׂא אֶת־נִבְלָתָם יְכַבֵּס בְּגָדָיו וְטָמֵא עַד־הָעָרֶב
 2   טְמֵאִים הֵמָּה לָכֶם׃ [ס]             וְזֶה לָכֶם
 3   הַטָּמֵא בַּשֶּׁרֶץ הַשֹּׁרֵץ עַל־הָאָרֶץ הַחֹלֶד וְהָעַכְבָּר
 4   וְהַצָּב לְמִינֵהוּ׃ וְהָאֲנָקָה וְהַכֹּחַ וְהַלְּטָאָה וְהַחֹמֶט
 5   וְהַתִּנְשָׁמֶת׃ אֵלֶּה הַטְּמֵאִים לָכֶם בְּכָל־הַשָּׁרֶץ כָּל־
 6  הַנֹּגֵעַ בָּהֶם בְּמֹתָם יִטְמָא עַד־הָעָרֶב׃ וְכֹל אֲשֶׁר־יִפֹּל־
 7  עָלָיו מֵהֶם  בְּמֹתָם יִטְמָא מִכָּל־כְּלִי־עֵץ אוֹ בֶגֶד אוֹ־
 8  עוֹר אוֹ שָׂק כָּל־כְּלִי אֲשֶׁר־יֵעָשֶׂה מְלָאכָה בָּהֶם
 9   בַּמַּיִם יוּבָא וְטָמֵא עַד־הָעֶרֶב וְטָהֵר׃  וְכָל־כְּלִי־חֶרֶשׂ
10   אֲשֶׁר־יִפֹּל מֵהֶם אֶל־תּוֹכוֹ כֹּל אֲשֶׁר בְּתוֹכוֹ יִטְמָא
11   וְאֹתוֹ תִשְׁבֹּרוּ׃ מִכָּל־הָאֹכֶל אֲשֶׁר יֵאָכֵל אֲשֶׁר יָבוֹא
12   עָלָיו מַיִם יִטְמָא וְכָל־מַשְׁקֶה אֲשֶׁר יִשָּׁתֶה בְּכָל־
13  כְּלִי יִטְמָא׃ וְכֹל אֲשֶׁר־יִפֹּל מִנִּבְלָתָם  עָלָיו יִטְמָא
14   תַּנּוּר וְכִירַיִם יֻתָּץ טְמֵאִים הֵם וּטְמֵאִים יִהְיוּ לָכֶם׃
15   אַךְ מַעְיָן וּבוֹר מִקְוֵה־מַיִם יִהְיֶה טָהוֹר וְנֹגֵעַ בְּנִבְלָתָם
16   יִטְמָא׃ וְכִי יִפֹּל מִנִּבְלָתָם עַל־כָּל־זֶרַע זֵרוּעַ אֲשֶׁר
17   יִזָּרֵעַ טָהוֹר הוּא׃ וְכִי יֻתַּן־מַיִם עַל־זֶרַע וְנָפַל מִנִּבְלָתָם
18   עָלָיו טָמֵא הוּא לָכֶם׃ [ס]             וְכִי יָמוּת
19   מִן־הַבְּהֵמָה אֲשֶׁר־הִיא לָכֶם לְאָכְלָה הַנֹּגֵעַ בְּנִבְלָתָהּ
20   יִטְמָא עַד־הָעָרֶב׃ וְהָאֹכֵל מִנִּבְלָתָהּ יְכַבֵּס בְּגָדָיו
21   וְטָמֵא עַד־הָעָרֶב וְהַנֹּשֵׂא אֶת־נִבְלָתָהּ יְכַבֵּס בְּגָדָיו
22   וְטָמֵא עַד־הָעָרֶב׃ וְכָל־הַשֶּׁרֶץ הַשֹּׁרֵץ עַל־הָאָרֶץ
23   שֶׁקֶץ הוּא לֹא יֵאָכֵל׃ כֹּל הוֹלֵךְ עַל־גָּחוֹן וְכֹל  הוֹלֵךְ
24   עַל־אַרְבַּע עַד כָּל־מַרְבֵּה רַגְלַיִם לְכָל־הַשֶּׁרֶץ
25   הַשֹּׁרֵץ עַל־הָאָרֶץ לֹא תֹאכְלוּם כִּי־שֶׁקֶץ הֵם׃ אַל־
26  תְּשַׁקְּצוּ אֶת־נַפְשֹׁתֵיכֶם בְּכָל־הַשֶּׁרֶץ הַשֹּׁרֵץ וְלֹא
27   תִטַּמְּאוּ בָּהֶם וְנִטְמֵתֶם בָּם׃ כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם
28   וְהִתְקַדִּשְׁתֶּם וִהְיִיתֶם קְדֹשִׁים כִּי קָדוֹשׁ אָנִי וְלֹא
29   תְטַמְּאוּ אֶת־נַפְשֹׁתֵיכֶם בְּכָל־הַשֶּׁרֶץ הָרֹמֵשׂ עַל־
30  הָאָרֶץ׃  כִּי  אֲנִי יְהוָה הַמַּעֲלֶה אֶתְכֶם מֵאֶרֶץ מִצְרַיִם
31   לִהְיֹת לָכֶם לֵאלֹהִים וִהְיִיתֶם קְדֹשִׁים כִּי קָדוֹשׁ
32   אָנִי׃ זֹאת תּוֹרַת הַבְּהֵמָה וְהָעוֹף וְכֹל נֶפֶשׁ הַחַיָּה
33   הָרֹמֶשֶׂת בַּמָּיִם וּלְכָל־נֶפֶשׁ הַשֹּׁרֶצֶת עַל־הָאָרֶץ׃
34   לְהַבְדִּיל בֵּין הַטָּמֵא וּבֵין הַטָּהֹר וּבֵין הַחַיָּה הַנֶּאֱכֶלֶת
35   וּבֵין הַחַיָּה אֲשֶׁר לֹא תֵאָכֵל׃ [פ]
36  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
37   לֵאמֹר אִשָּׁה כִּי תַזְרִיעַ וְיָלְדָה זָכָר וְטָמְאָה שִׁבְעַת
38   יָמִים כִּימֵי נִדַּת דְּו‍ֹתָהּ תִּטְמָא׃ וּבַיּוֹם הַשְּׁמִינִי
39   יִמּוֹל בְּשַׂר עָרְלָתוֹ׃ וּשְׁלֹשִׁים יוֹם וּשְׁלֹשֶׁת יָמִים
40   תֵּשֵׁב בִּדְמֵי טָהֳרָה בְּכָל־קֹדֶשׁ לֹא־תִגָּע וְאֶל־
41  הַמִּקְדָּשׁ לֹא תָבֹא עַד־מְלֹאת יְמֵי טָהֳרָהּ׃♢ וְאִם־
42  נְקֵבָה תֵלֵד וְטָמְאָה שְׁבֻעַיִם כְּנִדָּתָהּ וְשִׁשִּׁים יוֹם
