 1  וְהֵבִיא אֶת־אֲשָׁמוֹ לַיהוָה עַל חַטָּאתוֹ אֲשֶׁר חָטָא
 2   נְקֵבָה מִן־הַצֹּאן כִּשְׂבָּה אוֹ־שְׂעִירַת עִזִּים לְחַטָּאת
 3   וְכִפֶּר עָלָיו הַכֹּהֵן מֵחַטָּאתוֹ׃ וְאִם־לֹא תַגִּיעַ יָדוֹ דֵּי
 4   שֶׂה וְהֵבִיא אֶת־אֲשָׁמוֹ אֲשֶׁר חָטָא שְׁתֵּי תֹרִים אוֹ־
 5  שְׁנֵי בְנֵי־יוֹנָה לַיהוָה אֶחָד לְחַטָּאת וְאֶחָד לְעֹלָה׃
 6   וְהֵבִיא אֹתָם אֶל־הַכֹּהֵן וְהִקְרִיב אֶת־אֲשֶׁר לַחַטָּאת
 7   רִאשׁוֹנָה וּמָלַק אֶת־רֹאשׁוֹ מִמּוּל עָרְפּוֹ וְלֹא יַבְדִּיל׃
 8   וְהִזָּה מִדַּם הַחַטָּאת עַל־קִיר הַמִּזְבֵּחַ וְהַנִּשְׁאָר בַּדָּם
 9   יִמָּצֵה אֶל־יְסוֹד הַמִּזְבֵּחַ חַטָּאת הוּא׃ וְאֶת־הַשֵּׁנִי
10   יַעֲשֶׂה עֹלָה כַּמִּשְׁפָּט וְכִפֶּר עָלָיו הַכֹּהֵן מֵחַטָּאתוֹ
11   אֲשֶׁר־חָטָא וְנִסְלַח לוֹ׃ [ס]              וְאִם־
12  לֹא תַשִּׂיג יָדוֹ לִשְׁתֵּי תֹרִים אוֹ לִשְׁנֵי בְנֵי־יוֹנָה
13   וְהֵבִיא אֶת־קָרְבָּנוֹ אֲשֶׁר חָטָא עֲשִׂירִת הָאֵפָה סֹלֶת
14   לְחַטָּאת לֹא־יָשִׂים עָלֶיהָ שֶׁמֶן וְלֹא־יִתֵּן עָלֶיהָ לְבֹנָה
15   כִּי חַטָּאת הִוא׃ וֶהֱבִיאָהּ אֶל־הַכֹּהֵן וְקָמַץ הַכֹּהֵן
16   מִמֶּנָּה מְלוֹא קֻמְצוֹ אֶת־אַזְכָּרָתָהּ וְהִקְטִיר הַמִּזְבֵּחָה
17   עַל אִשֵּׁי יְהוָה חַטָּאת הִוא׃ וְכִפֶּר עָלָיו הַכֹּהֵן עַל־
18  חַטָּאתוֹ אֲשֶׁר־חָטָא מֵאַחַת מֵאֵלֶּה וְנִסְלַח לוֹ וְהָיְתָה
19   לַכֹּהֵן כַּמִּנְחָה׃ [ס]             וַיְדַבֵּר יְהוָה אֶל־
20  מֹשֶׁה לֵּאמֹר׃ נֶפֶשׁ כִּי־תִמְעֹל מַעַל וְחָטְאָה בִּשְׁגָגָה
21   מִקָּדְשֵׁי יְהוָה וְהֵבִיא אֶת־אֲשָׁמוֹ לַיהוָה אַיִל תָּמִים
22   מִן־הַצֹּאן בְּעֶרְכְּךָ כֶּסֶף־שְׁקָלִים בְּשֶׁקֶל־הַקֹּדֶשׁ
23   לְאָשָׁם׃ וְאֵת אֲשֶׁר חָטָא מִן־הַקֹּדֶשׁ יְשַׁלֵּם וְאֶת־
24  חֲמִישִׁתוֹ יוֹסֵף עָלָיו וְנָתַן אֹתוֹ לַכֹּהֵן וְהַכֹּהֵן יְכַפֵּר
25   עָלָיו בְּאֵיל הָאָשָׁם וְנִסְלַח לוֹ׃ [פ]
26   וְאִם־נֶפֶשׁ כִּי תֶחֱטָא וְעָשְׂתָה אַחַת מִכָּל־מִצְו‍ֹת
27   יְהוָה אֲשֶׁר לֹא תֵעָשֶׂינָה וְלֹא־יָדַע וְאָשֵׁם וְנָשָׂא
28   עֲו‍ֹנוֹ׃ וְהֵבִיא אַיִל תָּמִים מִן־הַצֹּאן בְּעֶרְכְּךָ לְאָשָׁם
29   אֶל־הַכֹּהֵן וְכִפֶּר עָלָיו הַכֹּהֵן עַל שִׁגְגָתוֹ אֲשֶׁר־
30  שָׁגָג וְהוּא לֹא־יָדַע וְנִסְלַח לוֹ׃ אָשָׁם הוּא אָשֹׁם
31   אָשַׁם לַיהוָה׃ [פ]
32   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ נֶפֶשׁ כִּי תֶחֱטָא וּמָעֲלָה
33   מַעַל בַּיהוָה וְכִחֵשׁ בַּעֲמִיתוֹ בְּפִקָּדוֹן אוֹ־בִתְשׂוּמֶת יָד
34   אוֹ בְגָזֵל אוֹ עָשַׁק אֶת־עֲמִיתוֹ׃ אוֹ־מָצָא אֲבֵדָה וְכִחֶשׁ
35   בָּהּ וְנִשְׁבַּע עַל־שָׁקֶר עַל־אַחַת מִכֹּל אֲשֶׁר־יַעֲשֶׂה
36   הָאָדָם לַחֲטֹא בָהֵנָּה׃ וְהָיָה כִּי־יֶחֱטָא וְאָשֵׁם וְהֵשִׁיב
37   אֶת־הַגְּזֵלָה אֲשֶׁר גָּזָל אוֹ אֶת־הָעֹשֶׁק אֲשֶׁר עָשָׁק
38   אוֹ אֶת־הַפִּקָּדוֹן אֲשֶׁר הָפְקַד אִתּוֹ אוֹ אֶת־הָאֲבֵדָה
39   אֲשֶׁר מָצָא׃  אוֹ מִכֹּל אֲשֶׁר־יִשָּׁבַע עָלָיו לַשֶּׁקֶר
40   וְשִׁלַּם אֹתוֹ בְּרֹאשׁוֹ וַחֲמִשִׁתָיו יֹסֵף עָלָיו לַאֲשֶׁר
41   הוּא לוֹ יִתְּנֶנּוּ בְּיוֹם אַשְׁמָתוֹ׃ וְאֶת־אֲשָׁמוֹ יָבִיא לַיהוָה
42   אַיִל תָּמִים מִן־הַצֹּאן בְּעֶרְכְּךָ לְאָשָׁם אֶל־הַכֹּהֵן׃
