 1  וְצִוָּה הַכֹּהֵן וְשָׁחַט אֶת־הַצִּפּוֹר הָאֶחָת אֶל־כְּלִי־חֶרֶשׂ
 2   עַל־מַיִם חַיִּים׃♢ אֶת־הַצִּפֹּר הַחַיָּה יִקַּח אֹתָהּ וְאֶת־עֵץ
 3   הָאֶרֶז וְאֶת־שְׁנִי הַתּוֹלַעַת וְאֶת־הָאֵזֹב וְטָבַל אוֹתָם
 4   וְאֵת  הַצִּפֹּר הַחַיָּה בְּדַם הַצִּפֹּר הַשְּׁחֻטָה עַל הַמַּיִם
 5   הַחַיִּים׃ וְהִזָּה עַל הַמִּטַּהֵר מִן־הַצָּרַעַת שֶׁבַע פְּעָמִים
 6   וְטִהֲרוֹ וְשִׁלַּח אֶת־הַצִּפֹּר הַחַיָּה עַל־פְּנֵי הַשָּׂדֶה׃
 7   וְכִבֶּס הַמִּטַּהֵר אֶת־בְּגָדָיו וְגִלַּח אֶת־כָּל־שְׂעָרוֹ וְרָחַץ
 8   בַּמַּיִם וְטָהֵר וְאַחַר יָבוֹא אֶל־הַמַּחֲנֶה וְיָשַׁב מִחוּץ
 9   לְאָהֳלוֹ שִׁבְעַת יָמִים׃ וְהָיָה בַיּוֹם הַשְּׁבִיעִי יְגַלַּח אֶת־
10  כָּל־שְׂעָרוֹ אֶת־רֹאשׁוֹ וְאֶת־זְקָנוֹ וְאֵת גַּבֹּת עֵינָיו וְאֶת־
11  כָּל־שְׂעָרוֹ יְגַלֵּחַ וְכִבֶּס אֶת־בְּגָדָיו וְרָחַץ אֶת־בְּשָׂרוֹ
12   בַּמַּיִם וְטָהֵר׃♢ וּבַיּוֹם הַשְּׁמִינִי יִקַּח שְׁנֵי־כְבָשִׂים
13   תְּמִימִם וְכַבְשָׂה אַחַת בַּת־שְׁנָתָהּ תְּמִימָה וּשְׁלֹשָׁה
14   עֶשְׂרֹנִים סֹלֶת מִנְחָה בְּלוּלָה בַשֶּׁמֶן וְלֹג אֶחָד שָׁמֶן׃
15   וְהֶעֱמִיד הַכֹּהֵן הַמְטַהֵר אֵת הָאִישׁ הַמִּטַּהֵר וְאֹתָם
16   לִפְנֵי יְהוָה פֶּתַח אֹהֶל מוֹעֵד׃ וְלָקַח הַכֹּהֵן אֶת־הַכֶּבֶשׂ
17   הָאֶחָד וְהִקְרִיב אֹתוֹ לְאָשָׁם וְאֶת־לֹג הַשָּׁמֶן וְהֵנִיף
18   אֹתָם תְּנוּפָה לִפְנֵי יְהוָה׃♢  וְשָׁחַט אֶת־הַכֶּבֶשׂ בִּמְקוֹם
19   אֲשֶׁר יִשְׁחַט אֶת־הַחַטָּאת וְאֶת־הָעֹלָה בִּמְקוֹם
20   הַקֹּדֶשׁ כִּי כַּחַטָּאת הָאָשָׁם הוּא לַכֹּהֵן קֹדֶשׁ
21   קָדָשִׁים הוּא׃ וְלָקַח הַכֹּהֵן מִדַּם הָאָשָׁם וְנָתַן הַכֹּהֵן
22   עַל־תְּנוּךְ אֹזֶן הַמִּטַּהֵר הַיְמָנִית וְעַל־בֹּהֶן יָדוֹ הַיְמָנִית
23   וְעַל־בֹּהֶן רַגְלוֹ הַיְמָנִית׃ וְלָקַח הַכֹּהֵן מִלֹּג הַשָּׁמֶן
24   וְיָצַק עַל־כַּף הַכֹּהֵן הַשְּׂמָאלִית׃ וְטָבַל הַכֹּהֵן אֶת־
25  אֶצְבָּעוֹ הַיְמָנִית מִן־הַשֶּׁמֶן אֲשֶׁר עַל־כַּפּוֹ הַשְּׂמָאלִית
26   וְהִזָּה מִן־הַשֶּׁמֶן בְּאֶצְבָּעוֹ שֶׁבַע פְּעָמִים לִפְנֵי יְהוָה׃
27   וּמִיֶּתֶר הַשֶּׁמֶן אֲשֶׁר עַל־כַּפּוֹ יִתֵּן הַכֹּהֵן עַל־תְּנוּךְ
28   אֹזֶן הַמִּטַּהֵר הַיְמָנִית וְעַל־בֹּהֶן יָדוֹ הַיְמָנִית וְעַל־בֹּהֶן
29   רַגְלוֹ הַיְמָנִית עַל דַּם הָאָשָׁם׃ וְהַנּוֹתָר בַּשֶּׁמֶן אֲשֶׁר
30   עַל־כַּף הַכֹּהֵן יִתֵּן עַל־רֹאשׁ הַמִּטַּהֵר וְכִפֶּר עָלָיו
31   הַכֹּהֵן לִפְנֵי יְהוָה׃ וְעָשָׂה הַכֹּהֵן אֶת־הַחַטָּאת וְכִפֶּר
32   עַל־הַמִּטַּהֵר מִטֻּמְאָתוֹ וְאַחַר יִשְׁחַט אֶת־הָעֹלָה׃
33   וְהֶעֱלָה הַכֹּהֵן אֶת־הָעֹלָה וְאֶת־הַמִּנְחָה הַמִּזְבֵּחָה
34   וְכִפֶּר עָלָיו הַכֹּהֵן וְטָהֵר׃ [ס]              וְאִם־
35  דַּל הוּא וְאֵין יָדוֹ מַשֶּׂגֶת וְלָקַח כֶּבֶשׂ אֶחָד אָשָׁם
36   לִתְנוּפָה לְכַפֵּר עָלָיו וְעִשָּׂרוֹן סֹלֶת אֶחָד בָּלוּל בַּשֶּׁמֶן
37   לְמִנְחָה וְלֹג שָׁמֶן׃ וּשְׁתֵּי תֹרִים אוֹ שְׁנֵי בְּנֵי
38   יוֹנָה אֲשֶׁר תַּשִּׂיג יָדוֹ וְהָיָה אֶחָד חַטָּאת וְהָאֶחָד
39   עֹלָה׃ וְהֵבִיא אֹתָם בַּיּוֹם הַשְּׁמִינִי לְטָהֳרָתוֹ אֶל־
40  הַכֹּהֵן אֶל־פֶּתַח אֹהֶל־מוֹעֵד לִפְנֵי יְהוָה׃ וְלָקַח
41   הַכֹּהֵן אֶת־כֶּבֶשׂ הָאָשָׁם וְאֶת־לֹג הַשָּׁמֶן וְהֵנִיף אֹתָם
42   הַכֹּהֵן תְּנוּפָה לִפְנֵי יְהוָה׃ וְשָׁחַט אֶת־כֶּבֶשׂ הָאָשָׁם
