 1  וַעֲשִׂיתֶם אֹתָם אֲנִי יְהוָה מְקַדִּשְׁכֶם׃ כִּי־אִישׁ אִישׁ
 2   אֲשֶׁר יְקַלֵּל אֶת־אָבִיו וְאֶת־אִמּוֹ מוֹת יוּמָת אָבִיו
 3   וְאִמּוֹ קִלֵּל דָּמָיו בּוֹ׃ וְאִישׁ אֲשֶׁר יִנְאַף אֶת־אֵשֶׁת
 4   אִישׁ אֲשֶׁר יִנְאַף אֶת־אֵשֶׁת רֵעֵהוּ מוֹת־יוּמַת הַנֹּאֵף
 5   וְהַנֹּאָפֶת׃ וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת־אֵשֶׁת אָבִיו עֶרְוַת
 6   אָבִיו גִּלָּה מוֹת־יוּמְתוּ שְׁנֵיהֶם דְּמֵיהֶם בָּם׃ וְאִישׁ
 7   אֲשֶׁר יִשְׁכַּב אֶת־כַּלָּתוֹ מוֹת יוּמְתוּ שְׁנֵיהֶם תֶּבֶל
 8   עָשׂוּ דְּמֵיהֶם בָּם׃ וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת־זָכָר
 9   מִשְׁכְּבֵי אִשָּׁה תּוֹעֵבָה עָשׂוּ שְׁנֵיהֶם מוֹת יוּמָתוּ
10   דְּמֵיהֶם בָּם׃ וְאִישׁ אֲשֶׁר יִקַּח אֶת־אִשָּׁה וְאֶת־אִמָּהּ
11   זִמָּה הִוא בָּאֵשׁ יִשְׂרְפוּ אֹתוֹ וְאֶתְהֶן וְלֹא־תִהְיֶה זִמָּה
12   בְּתוֹכְכֶם׃ וְאִישׁ אֲשֶׁר יִתֵּן שְׁכָבְתּוֹ בִּבְהֵמָה מוֹת
13   יוּמָת וְאֶת־הַבְּהֵמָה תַּהֲרֹגוּ׃ וְאִשָּׁה אֲשֶׁר תִּקְרַב אֶל־
14  כָּל־בְּהֵמָה לְרִבְעָה אֹתָהּ וְהָרַגְתָּ אֶת־הָאִשָּׁה וְאֶת־
15  הַבְּהֵמָה מוֹת יוּמָתוּ דְּמֵיהֶם בָּם׃ וְאִישׁ אֲשֶׁר־יִקַּח
16   אֶת־אֲחֹתוֹ בַּת־אָבִיו אוֹ בַת־אִמּוֹ וְרָאָה אֶת־עֶרְוָתָהּ
17   וְהִיא־תִרְאֶה אֶת־עֶרְוָתוֹ חֶסֶד הוּא וְנִכְרְתוּ לְעֵינֵי
18   בְּנֵי עַמָּם עֶרְוַת אֲחֹתוֹ גִּלָּה עֲו‍ֹנוֹ יִשָּׂא׃ וְאִישׁ אֲשֶׁר־
19  יִשְׁכַּב אֶת־אִשָּׁה דָּוָה וְגִלָּה אֶת־עֶרְוָתָהּ אֶת־מְקֹרָהּ
20   הֶעֱרָה וְהִוא גִּלְּתָה אֶת־מְקוֹר דָּמֶיהָ וְנִכְרְתוּ שְׁנֵיהֶם
21   מִקֶּרֶב עַמָּם׃ וְעֶרְוַת אֲחוֹת אִמְּךָ וַאֲחוֹת אָבִיךָ לֹא
22   תְגַלֵּה כִּי אֶת־שְׁאֵרוֹ הֶעֱרָה עֲו‍ֹנָם יִשָּׂאוּ׃ וְאִישׁ אֲשֶׁר
23   יִשְׁכַּב אֶת־דֹּדָתוֹ עֶרְוַת דֹּדוֹ גִּלָּה חֶטְאָם יִשָּׂאוּ
24   עֲרִירִים יָמֻתוּ׃ וְאִישׁ אֲשֶׁר יִקַּח אֶת־אֵשֶׁת אָחִיו
25   נִדָּה הִוא עֶרְוַת אָחִיו גִּלָּה עֲרִירִים יִהְיוּ׃ וּשְׁמַרְתֶּם
26   אֶת־כָּל־חֻקֹּתַי וְאֶת־כָּל־מִשְׁפָּטַי וַעֲשִׂיתֶם אֹתָם וְלֹא־
27  תָקִיא אֶתְכֶם הָאָרֶץ אֲשֶׁר אֲנִי מֵבִיא אֶתְכֶם שָׁמָּה
28   לָשֶׁבֶת בָּהּ׃  וְלֹא תֵלְכוּ בְּחֻקֹּת הַגּוֹי אֲשֶׁר־אֲנִי
29   מְשַׁלֵּחַ מִפְּנֵיכֶם כִּי אֶת־כָּל־אֵלֶּה עָשׂוּ וָאָקֻץ בָּם׃
30   וָאֹמַר לָכֶם אַתֶּם תִּירְשׁוּ אֶת־אַדְמָתָם וַאֲנִי אֶתְּנֶנָּה
31   לָכֶם לָרֶשֶׁת אֹתָהּ אֶרֶץ זָבַת חָלָב וּדְבָשׁ אֲנִי
32   יְהוָה אֱלֹהֵיכֶם אֲשֶׁר־הִבְדַּלְתִּי אֶתְכֶם מִן־הָעַמִּים׃
33    וְהִבְדַּלְתֶּם בֵּין־הַבְּהֵמָה הַטְּהֹרָה לַטְּמֵאָה וּבֵין־הָעוֹף
34   הַטָּמֵא לַטָּהֹר וְלֹא־תְשַׁקְּצוּ אֶת־נַפְשֹׁתֵיכֶם בַּבְּהֵמָה
35   וּבָעוֹף וּבְכֹל אֲשֶׁר תִּרְמֹשׂ הָאֲדָמָה אֲשֶׁר־הִבְדַּלְתִּי
36   לָכֶם לְטַמֵּא׃ וִהְיִיתֶם לִי קְדֹשִׁים כִּי קָדוֹשׁ אֲנִי יְהוָה
37   וָאַבְדִּל אֶתְכֶם מִן־הָעַמִּים לִהְיוֹת לִי׃ וְאִישׁ אוֹ־אִשָּׁה
38   כִּי־יִהְיֶה בָהֶם אוֹב אוֹ יִדְּעֹנִי מוֹת יוּמָתוּ בָּאֶבֶן יִרְגְּמוּ
39   אֹתָם דְּמֵיהֶם בָּם׃ [פ]
40  וַיֹּאמֶר יְהוָה אֶל־מֹשֶׁה אֱמֹר אֶל־הַכֹּהֲנִים בְּנֵי
41   אַהֲרֹן וְאָמַרְתָּ אֲלֵהֶם לְנֶפֶשׁ לֹא־יִטַּמָּא בְּעַמָּיו׃
42   כִּי אִם־לִשְׁאֵרוֹ הַקָּרֹב אֵלָיו לְאִמּוֹ וּלְאָבִיו וְלִבְנוֹ
