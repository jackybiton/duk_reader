 1  וְכִי־תִזְבְּחוּ זֶבַח־תּוֹדָה לַיהוָה לִרְצֹנְכֶם תִּזְבָּחוּ׃ בַּיּוֹם
 2   הַהוּא יֵאָכֵל לֹא־תוֹתִירוּ מִמֶּנּוּ עַד־בֹּקֶר אֲנִי יְהוָה׃
 3   וּשְׁמַרְתֶּם מִצְו‍ֹתַי וַעֲשִׂיתֶם אֹתָם אֲנִי יְהוָה׃ וְלֹא
 4   תְחַלְּלוּ אֶת־שֵׁם קָדְשִׁי וְנִקְדַּשְׁתִּי בְּתוֹךְ בְּנֵי יִשְׂרָאֵל
 5   אֲנִי יְהוָה מְקַדִּשְׁכֶם׃ הַמּוֹצִיא אֶתְכֶם מֵאֶרֶץ מִצְרַיִם
 6   לִהְיוֹת לָכֶם לֵאלֹהִים אֲנִי יְהוָה׃ [פ]
 7    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
 8   וְאָמַרְתָּ אֲלֵהֶם מוֹעֲדֵי יְהוָה אֲשֶׁר־תִּקְרְאוּ אֹתָם
 9   מִקְרָאֵי קֹדֶשׁ אֵלֶּה הֵם מוֹעֲדָי׃ שֵׁשֶׁת יָמִים תֵּעָשֶׂה
10   מְלָאכָה וּבַיּוֹם הַשְּׁבִיעִי שַׁבַּת שַׁבָּתוֹן מִקְרָא־
11  קֹדֶשׁ כָּל־מְלָאכָה לֹא תַעֲשׂוּ שַׁבָּת הִוא לַיהוָה
12   בְּכֹל מוֹשְׁבֹתֵיכֶם׃ [פ]
13   אֵלֶּה מוֹעֲדֵי יְהוָה מִקְרָאֵי קֹדֶשׁ אֲשֶׁר־תִּקְרְאוּ אֹתָם
14   בְּמוֹעֲדָם׃ בַּחֹדֶשׁ הָרִאשׁוֹן בְּאַרְבָּעָה עָשָׂר לַחֹדֶשׁ
15   בֵּין הָעַרְבָּיִם פֶּסַח לַיהוָה׃ וּבַחֲמִשָּׁה עָשָׂר יוֹם
16   לַחֹדֶשׁ הַזֶּה חַג הַמַּצּוֹת לַיהוָה שִׁבְעַת יָמִים מַצּוֹת
17   תֹּאכֵלוּ׃ בַּיּוֹם הָרִאשׁוֹן מִקְרָא־קֹדֶשׁ יִהְיֶה לָכֶם כָּל־
18  מְלֶאכֶת עֲבֹדָה לֹא תַעֲשׂוּ׃ וְהִקְרַבְתֶּם אִשֶּׁה לַיהוָה
19   שִׁבְעַת יָמִים בַּיּוֹם הַשְּׁבִיעִי מִקְרָא־קֹדֶשׁ כָּל־
20  מְלֶאכֶת עֲבֹדָה לֹא תַעֲשׂוּ׃ [פ]
21   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
22   וְאָמַרְתָּ אֲלֵהֶם כִּי־תָבֹאוּ אֶל־הָאָרֶץ אֲשֶׁר אֲנִי
23   נֹתֵן לָכֶם וּקְצַרְתֶּם אֶת־קְצִירָהּ וַהֲבֵאתֶם אֶת־
24  עֹמֶר רֵאשִׁית קְצִירְכֶם אֶל־הַכֹּהֵן׃ וְהֵנִיף אֶת־
25  הָעֹמֶר לִפְנֵי יְהוָה לִרְצֹנְכֶם מִמָּחֳרַת הַשַּׁבָּת יְנִיפֶנּוּ
26   הַכֹּהֵן׃ וַעֲשִׂיתֶם בְּיוֹם הֲנִיפְכֶם אֶת־הָעֹמֶר כֶּבֶשׂ
27   תָּמִים בֶּן־שְׁנָתוֹ לְעֹלָה לַיהוָה׃ וּמִנְחָתוֹ שְׁנֵי
28   עֶשְׂרֹנִים סֹלֶת בְּלוּלָה בַשֶּׁמֶן אִשֶּׁה לַיהוָה רֵיחַ
29   נִיחֹחַ וְנִסְכֹּה יַיִן רְבִיעִת הַהִין׃ וְלֶחֶם וְקָלִי וְכַרְמֶל
30   לֹא תֹאכְלוּ עַד־עֶצֶם הַיּוֹם הַזֶּה עַד הֲבִיאֲכֶם
31   אֶת־קָרְבַּן אֱלֹהֵיכֶם חֻקַּת עוֹלָם לְדֹרֹתֵיכֶם בְּכֹל
32   מֹשְׁבֹתֵיכֶם׃ [ס]             וּסְפַרְתֶּם לָכֶם
33   מִמָּחֳרַת הַשַּׁבָּת מִיּוֹם הֲבִיאֲכֶם אֶת־עֹמֶר הַתְּנוּפָה
34   שֶׁבַע שַׁבָּתוֹת תְּמִימֹת תִּהְיֶינָה׃ עַד מִמָּחֳרַת הַשַּׁבָּת
35   הַשְּׁבִיעִת תִּסְפְּרוּ חֲמִשִּׁים יוֹם וְהִקְרַבְתֶּם מִנְחָה
36   חֲדָשָׁה לַיהוָה׃ מִמּוֹשְׁבֹתֵיכֶם תָּבִיאּוּ  לֶחֶם תְּנוּפָה
37   שְׁתַּיִם שְׁנֵי עֶשְׂרֹנִים סֹלֶת תִּהְיֶינָה חָמֵץ תֵּאָפֶינָה
38   בִּכּוּרִים לַיהוָה׃ וְהִקְרַבְתֶּם עַל־הַלֶּחֶם שִׁבְעַת
39   כְּבָשִׂים תְּמִימִם בְּנֵי שָׁנָה וּפַר בֶּן־בָּקָר אֶחָד
40   וְאֵילִם שְׁנָיִם יִהְיוּ עֹלָה לַיהוָה וּמִנְחָתָם וְנִסְכֵּיהֶם
41   אִשֵּׁה רֵיחַ־נִיחֹחַ לַיהוָה׃ וַעֲשִׂיתֶם שְׂעִיר־עִזִּים אֶחָד
42   לְחַטָּאת וּשְׁנֵי כְבָשִׂים בְּנֵי שָׁנָה לְזֶבַח שְׁלָמִים׃
