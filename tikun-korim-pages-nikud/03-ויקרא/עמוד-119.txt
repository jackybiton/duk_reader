 1  וְלַבְּהֵמָה׃ כָּל־נֶפֶשׁ אֲשֶׁר־תֹּאכַל כָּל־דָּם וְנִכְרְתָה
 2   הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ׃ [פ]
 3   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
 4   לֵאמֹר הַמַּקְרִיב אֶת־זֶבַח שְׁלָמָיו לַיהוָה יָבִיא אֶת־
 5  קָרְבָּנוֹ לַיהוָה מִזֶּבַח שְׁלָמָיו׃ יָדָיו תְּבִיאֶינָה אֵת
 6   אִשֵּׁי יְהוָה אֶת־הַחֵלֶב עַל־הֶחָזֶה יְבִיאֶנּוּ אֵת הֶחָזֶה
 7   לְהָנִיף אֹתוֹ תְּנוּפָה לִפְנֵי יְהוָה׃ וְהִקְטִיר הַכֹּהֵן אֶת־
 8  הַחֵלֶב הַמִּזְבֵּחָה וְהָיָה הֶחָזֶה לְאַהֲרֹן וּלְבָנָיו׃ וְאֵת
 9   שׁוֹק הַיָּמִין תִּתְּנוּ תְרוּמָה לַכֹּהֵן מִזִּבְחֵי שַׁלְמֵיכֶם׃
10   הַמַּקְרִיב אֶת־דַּם הַשְּׁלָמִים וְאֶת־הַחֵלֶב מִבְּנֵי אַהֲרֹן
11   לוֹ תִהְיֶה שׁוֹק הַיָּמִין לְמָנָה׃ כִּי אֶת־חֲזֵה הַתְּנוּפָה
12   וְאֵת  שׁוֹק הַתְּרוּמָה לָקַחְתִּי מֵאֵת בְּנֵי־יִשְׂרָאֵל
13   מִזִּבְחֵי שַׁלְמֵיהֶם וָאֶתֵּן אֹתָם לְאַהֲרֹן הַכֹּהֵן וּלְבָנָיו
14   לְחָק־עוֹלָם מֵאֵת בְּנֵי יִשְׂרָאֵל׃ זֹאת מִשְׁחַת אַהֲרֹן
15   וּמִשְׁחַת בָּנָיו מֵאִשֵּׁי יְהוָה בְּיוֹם הִקְרִיב אֹתָם לְכַהֵן
16   לַיהוָה׃ אֲשֶׁר צִוָּה יְהוָה לָתֵת לָהֶם בְּיוֹם מָשְׁחוֹ
17   אֹתָם מֵאֵת בְּנֵי יִשְׂרָאֵל חֻקַּת עוֹלָם לְדֹרֹתָם׃ זֹאת
18   הַתּוֹרָה לָעֹלָה לַמִּנְחָה וְלַחַטָּאת וְלָאָשָׁם וְלַמִּלּוּאִים
19   וּלְזֶבַח הַשְּׁלָמִים׃ אֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה בְּהַר
20   סִינָי בְּיוֹם צַוּ‍ֹתוֹ אֶת־בְּנֵי יִשְׂרָאֵל לְהַקְרִיב אֶת־
21  קָרְבְּנֵיהֶם לַיהוָה בְּמִדְבַּר סִינָי׃ [פ]
22    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ קַח אֶת־אַהֲרֹן וְאֶת־
23  בָּנָיו אִתּוֹ וְאֵת הַבְּגָדִים וְאֵת שֶׁמֶן הַמִּשְׁחָה וְאֵת  פַּר
24   הַחַטָּאת וְאֵת שְׁנֵי הָאֵילִים וְאֵת סַל הַמַּצּוֹת׃ וְאֵת
25   כָּל־הָעֵדָה הַקְהֵל אֶל־פֶּתַח אֹהֶל מוֹעֵד׃ וַיַּעַשׂ מֹשֶׁה
26   כַּאֲשֶׁר צִוָּה יְהוָה אֹתוֹ וַתִּקָּהֵל הָעֵדָה אֶל־פֶּתַח אֹהֶל
27   מוֹעֵד׃ וַיֹּאמֶר מֹשֶׁה אֶל־הָעֵדָה זֶה הַדָּבָר אֲשֶׁר־צִוָּה
28   יְהוָה לַעֲשׂוֹת׃ וַיַּקְרֵב מֹשֶׁה אֶת־אַהֲרֹן וְאֶת־בָּנָיו
29   וַיִּרְחַץ אֹתָם בַּמָּיִם׃ וַיִּתֵּן עָלָיו אֶת־הַכֻּתֹּנֶת וַיַּחְגֹּר אֹתוֹ
30   בָּאַבְנֵט וַיַּלְבֵּשׁ אֹתוֹ אֶת־הַמְּעִיל וַיִּתֵּן עָלָיו אֶת־הָאֵפֹד
31   וַיַּחְגֹּר אֹתוֹ בְּחֵשֶׁב הָאֵפֹד וַיֶּאְפֹּד לוֹ בּוֹ׃ וַיָּשֶׂם עָלָיו
32   אֶת־הַחֹשֶׁן וַיִּתֵּן אֶל־הַחֹשֶׁן אֶת־הָאוּרִים וְאֶת־הַתֻּמִּים׃
33   וַיָּשֶׂם אֶת־הַמִּצְנֶפֶת עַל־רֹאשׁוֹ וַיָּשֶׂם עַל־הַמִּצְנֶפֶת
34   אֶל־מוּל פָּנָיו אֵת צִיץ הַזָּהָב נֵזֶר הַקֹּדֶשׁ כַּאֲשֶׁר
35   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ וַיִּקַּח מֹשֶׁה אֶת־שֶׁמֶן הַמִּשְׁחָה
36   וַיִּמְשַׁח אֶת־הַמִּשְׁכָּן וְאֶת־כָּל־אֲשֶׁר־בּוֹ וַיְקַדֵּשׁ אֹתָם׃
37   וַיַּז מִמֶּנּוּ עַל־הַמִּזְבֵּחַ שֶׁבַע פְּעָמִים וַיִּמְשַׁח אֶת־
38  הַמִּזְבֵּחַ וְאֶת־כָּל־כֵּלָיו וְאֶת־הַכִּיֹּר וְאֶת־כַּנּוֹ לְקַדְּשָׁם׃
39   וַיִּצֹק מִשֶּׁמֶן הַמִּשְׁחָה עַל רֹאשׁ אַהֲרֹן וַיִּמְשַׁח אֹתוֹ
40   לְקַדְּשׁוֹ׃ וַיַּקְרֵב מֹשֶׁה אֶת־בְּנֵי אַהֲרֹן וַיַּלְבִּשֵׁם
41   כֻּתֳּנֹת וַיַּחְגֹּר אֹתָם אַבְנֵט וַיַּחֲבֹשׁ לָהֶם מִגְבָּעוֹת
42   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃  וַיַּגֵּשׁ אֵת פַּר הַחַטָּאת
