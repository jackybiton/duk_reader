 1  וְתוֹשָׁבִים אַתֶּם עִמָּדִי׃ וּבְכֹל אֶרֶץ אֲחֻזַּתְכֶם גְּאֻלָּה
 2   תִּתְּנוּ לָאָרֶץ׃ [ס]              כִּי־יָמוּךְ אָחִיךָ
 3   וּמָכַר מֵאֲחֻזָּתוֹ וּבָא גֹאֲלוֹ הַקָּרֹב אֵלָיו וְגָאַל אֵת
 4   מִמְכַּר אָחִיו׃ וְאִישׁ כִּי לֹא יִהְיֶה־לּוֹ גֹּאֵל וְהִשִּׂיגָה
 5   יָדוֹ וּמָצָא כְּדֵי גְאֻלָּתוֹ׃ וְחִשַּׁב אֶת־שְׁנֵי מִמְכָּרוֹ
 6   וְהֵשִׁיב אֶת־הָעֹדֵף לָאִישׁ אֲשֶׁר מָכַר־לוֹ וְשָׁב
 7   לַאֲחֻזָּתוֹ׃ וְאִם לֹא־מָצְאָה יָדוֹ דֵּי הָשִׁיב לוֹ וְהָיָה
 8   מִמְכָּרוֹ בְּיַד הַקֹּנֶה אֹתוֹ עַד שְׁנַת הַיּוֹבֵל וְיָצָא
 9   בַּיֹּבֵל וְשָׁב לַאֲחֻזָּתוֹ׃ [ס]              וְאִישׁ
10   כִּי־יִמְכֹּר בֵּית־מוֹשַׁב עִיר חוֹמָה וְהָיְתָה גְּאֻלָּתוֹ
11   עַד־תֹּם שְׁנַת מִמְכָּרוֹ יָמִים תִּהְיֶה גְאֻלָּתוֹ׃ וְאִם
12   לֹא־יִגָּאֵל עַד־מְלֹאת לוֹ שָׁנָה תְמִימָה וְקָם הַבַּיִת
13   אֲשֶׁר־בָּעִיר אֲשֶׁר־לוֹ חֹמָה לַצְּמִיתֻת לַקֹּנֶה אֹתוֹ
14   לְדֹרֹתָיו לֹא יֵצֵא בַּיֹּבֵל׃ וּבָתֵּי הַחֲצֵרִים אֲשֶׁר
15   אֵין־לָהֶם חֹמָה סָבִיב עַל־שְׂדֵה הָאָרֶץ יֵחָשֵׁב
16   גְּאֻלָּה תִּהְיֶה־לּוֹ וּבַיֹּבֵל יֵצֵא׃ וְעָרֵי הַלְוִיִּם בָּתֵּי עָרֵי
17   אֲחֻזָּתָם גְּאֻלַּת עוֹלָם תִּהְיֶה לַלְוִיִּם׃ וַאֲשֶׁר יִגְאַל מִן־
18  הַלְוִיִּם וְיָצָא מִמְכַּר־בַּיִת וְעִיר אֲחֻזָּתוֹ בַּיֹּבֵל כִּי
19   בָתֵּי עָרֵי הַלְוִיִּם הִוא אֲחֻזָּתָם בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׃
20   וּשְׂדֵה מִגְרַשׁ עָרֵיהֶם לֹא יִמָּכֵר כִּי־אֲחֻזַּת עוֹלָם
21   הוּא לָהֶם׃ [ס]             וְכִי־
22  יָמוּךְ אָחִיךָ וּמָטָה יָדוֹ עִמָּךְ וְהֶחֱזַקְתָּ בּוֹ גֵּר
23   וְתוֹשָׁב וָחַי עִמָּךְ׃ אַל־תִּקַּח מֵאִתּוֹ נֶשֶׁךְ וְתַרְבִּית
24   וְיָרֵאתָ מֵאֱלֹהֶיךָ וְחֵי אָחִיךָ עִמָּךְ׃ אֶת־כַּסְפְּךָ לֹא־
25  תִתֵּן לוֹ בְּנֶשֶׁךְ וּבְמַרְבִּית לֹא־תִתֵּן אָכְלֶךָ׃ אֲנִי
26   יְהוָה אֱלֹהֵיכֶם אֲשֶׁר־הוֹצֵאתִי אֶתְכֶם מֵאֶרֶץ
27   מִצְרָיִם לָתֵת לָכֶם אֶת־אֶרֶץ כְּנַעַן לִהְיוֹת לָכֶם
28   לֵאלֹהִים׃ [ס]              וְכִי־
29  יָמוּךְ אָחִיךָ עִמָּךְ וְנִמְכַּר־לָךְ לֹא־תַעֲבֹד בּוֹ עֲבֹדַת
30   עָבֶד׃ כְּשָׂכִיר כְּתוֹשָׁב יִהְיֶה עִמָּךְ עַד־שְׁנַת
31   הַיֹּבֵל יַעֲבֹד עִמָּךְ׃ וְיָצָא מֵעִמָּךְ הוּא וּבָנָיו עִמּוֹ
32   וְשָׁב אֶל־מִשְׁפַּחְתּוֹ וְאֶל־אֲחֻזַּת אֲבֹתָיו יָשׁוּב׃ כִּי־
33  עֲבָדַי הֵם אֲשֶׁר־הוֹצֵאתִי אֹתָם מֵאֶרֶץ מִצְרָיִם
34   לֹא יִמָּכְרוּ מִמְכֶּרֶת עָבֶד׃ לֹא־תִרְדֶּה בוֹ בְּפָרֶךְ
35   וְיָרֵאתָ מֵאֱלֹהֶיךָ׃ וְעַבְדְּךָ וַאֲמָתְךָ אֲשֶׁר יִהְיוּ־לָךְ
36   מֵאֵת הַגּוֹיִם אֲשֶׁר סְבִיבֹתֵיכֶם מֵהֶם תִּקְנוּ עֶבֶד
37   וְאָמָה׃ וְגַם מִבְּנֵי הַתּוֹשָׁבִים הַגָּרִים עִמָּכֶם מֵהֶם
38   תִּקְנוּ וּמִמִּשְׁפַּחְתָּם אֲשֶׁר עִמָּכֶם אֲשֶׁר הוֹלִידוּ
39   בְּאַרְצְכֶם וְהָיוּ לָכֶם לַאֲחֻזָּה׃ וְהִתְנַחַלְתֶּם אֹתָם
40   לִבְנֵיכֶם אַחֲרֵיכֶם לָרֶשֶׁת אֲחֻזָּה לְעֹלָם בָּהֶם תַּעֲבֹדוּ
41   וּבְאַחֵיכֶם בְּנֵי־יִשְׂרָאֵל אִישׁ בְּאָחִיו לֹא־תִרְדֶּה בוֹ
42   בְּפָרֶךְ׃ [ס]              וְכִי תַשִּׂיג יַד גֵּר
