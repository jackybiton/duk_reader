 1  וַיָּשֶׂם אֶת־קְרָשָׁיו וַיִּתֵּן אֶת־בְּרִיחָיו וַיָּקֶם אֶת־עַמּוּדָיו׃
 2   וַיִּפְרֹשׂ אֶת־הָאֹהֶל עַל־הַמִּשְׁכָּן וַיָּשֶׂם אֶת־מִכְסֵה
 3   הָאֹהֶל עָלָיו מִלְמָעְלָה כַּאֲשֶׁר צִוָּה יְהוָה אֶת־
 4  מֹשֶׁה׃ [ס]             וַיִּקַּח
 5   וַיִּתֵּן אֶת־הָעֵדֻת אֶל־הָאָרֹן וַיָּשֶׂם אֶת־הַבַּדִּים עַל־
 6  הָאָרֹן וַיִּתֵּן אֶת־הַכַּפֹּרֶת עַל־הָאָרֹן מִלְמָעְלָה׃ וַיָּבֵא
 7   אֶת־הָאָרֹן אֶל־הַמִּשְׁכָּן וַיָּשֶׂם אֵת פָּרֹכֶת הַמָּסָךְ
 8   וַיָּסֶךְ עַל אֲרוֹן הָעֵדוּת כַּאֲשֶׁר צִוָּה יְהוָה אֶת־
 9  מֹשֶׁה׃ [ס]             וַיִּתֵּן אֶת־
10  הַשֻּׁלְחָן בְּאֹהֶל מוֹעֵד עַל יֶרֶךְ הַמִּשְׁכָּן צָפֹנָה מִחוּץ
11   לַפָּרֹכֶת׃ וַיַּעֲרֹךְ עָלָיו עֵרֶךְ לֶחֶם לִפְנֵי יְהוָה כַּאֲשֶׁר
12   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיָּשֶׂם
13   אֶת־הַמְּנֹרָה בְּאֹהֶל מוֹעֵד נֹכַח הַשֻּׁלְחָן עַל יֶרֶךְ
14   הַמִּשְׁכָּן נֶגְבָּה׃ וַיַּעַל הַנֵּרֹת לִפְנֵי יְהוָה כַּאֲשֶׁר צִוָּה
15   יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיָּשֶׂם
16   אֶת־מִזְבַּח הַזָּהָב בְּאֹהֶל מוֹעֵד לִפְנֵי הַפָּרֹכֶת׃
17   וַיַּקְטֵר עָלָיו קְטֹרֶת סַמִּים כַּאֲשֶׁר צִוָּה יְהוָה אֶת־
18  מֹשֶׁה׃ [ס]              וַיָּשֶׂם אֶת־מָסַךְ הַפֶּתַח
19   לַמִּשְׁכָּן׃ וְאֵת מִזְבַּח הָעֹלָה שָׂם פֶּתַח מִשְׁכַּן אֹהֶל־
20  מוֹעֵד וַיַּעַל עָלָיו אֶת־הָעֹלָה וְאֶת־הַמִּנְחָה כַּאֲשֶׁר
21   צִוָּה יְהוָה אֶת־מֹשֶׁה׃ [ס]             וַיָּשֶׂם
22   אֶת־הַכִּיֹּר בֵּין־אֹהֶל מוֹעֵד וּבֵין הַמִּזְבֵּחַ וַיִּתֵּן שָׁמָּה
23   מַיִם לְרָחְצָה׃ וְרָחֲצוּ מִמֶּנּוּ מֹשֶׁה וְאַהֲרֹן וּבָנָיו
24   אֶת־יְדֵיהֶם וְאֶת־רַגְלֵיהֶם׃ בְּבֹאָם אֶל־אֹהֶל מוֹעֵד
25   וּבְקָרְבָתָם אֶל־הַמִּזְבֵּחַ יִרְחָצוּ כַּאֲשֶׁר צִוָּה יְהוָה
26   אֶת־מֹשֶׁה׃ [ס]             וַיָּקֶם אֶת־הֶחָצֵר
27   סָבִיב לַמִּשְׁכָּן וְלַמִּזְבֵּחַ וַיִּתֵּן אֶת־מָסַךְ שַׁעַר הֶחָצֵר
28   וַיְכַל מֹשֶׁה אֶת־הַמְּלָאכָה׃ [פ]
29    וַיְכַס הֶעָנָן אֶת־אֹהֶל מוֹעֵד וּכְבוֹד יְהוָה מָלֵא אֶת־
30  הַמִּשְׁכָּן׃ וְלֹא־יָכֹל מֹשֶׁה לָבוֹא אֶל־אֹהֶל מוֹעֵד כִּי־
31  שָׁכַן עָלָיו הֶעָנָן וּכְבוֹד יְהוָה מָלֵא אֶת־הַמִּשְׁכָּן׃
32   וּבְהֵעָלוֹת הֶעָנָן מֵעַל הַמִּשְׁכָּן יִסְעוּ בְּנֵי יִשְׂרָאֵל
33   בְּכֹל מַסְעֵיהֶם׃ וְאִם־לֹא יֵעָלֶה הֶעָנָן וְלֹא יִסְעוּ עַד־יוֹם
34   הֵעָלֹתוֹ׃ כִּי עֲנַן יְהוָה עַל־הַמִּשְׁכָּן יוֹמָם וְאֵשׁ תִּהְיֶה
35   לַיְלָה בּוֹ לְעֵינֵי כָל־בֵּית־יִשְׂרָאֵל בְּכָל־מַסְעֵיהֶם׃

40  וַיִּקְרָא אֶל־מֹשֶׁה וַיְדַבֵּר יְהוָה אֵלָיו מֵאֹהֶל מוֹעֵד
41   לֵאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אָדָם
42   כִּי־יַקְרִיב מִכֶּם קָרְבָּן לַיהוָה מִן־הַבְּהֵמָה מִן־הַבָּקָר
