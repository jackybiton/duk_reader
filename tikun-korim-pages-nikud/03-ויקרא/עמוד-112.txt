 1  וּמִן־הַצֹּאן תַּקְרִיבוּ אֶת־קָרְבַּנְכֶם׃ אִם־עֹלָה קָרְבָּנוֹ
 2   מִן־הַבָּקָר זָכָר תָּמִים יַקְרִיבֶנּוּ אֶל־פֶּתַח אֹהֶל מוֹעֵד
 3   יַקְרִיב אֹתוֹ לִרְצֹנוֹ לִפְנֵי יְהוָה׃ וְסָמַךְ יָדוֹ עַל רֹאשׁ
 4   הָעֹלָה וְנִרְצָה לוֹ לְכַפֵּר עָלָיו׃♢ וְשָׁחַט אֶת־בֶּן הַבָּקָר
 5   לִפְנֵי יְהוָה וְהִקְרִיבוּ בְּנֵי אַהֲרֹן הַכֹּהֲנִים אֶת־הַדָּם
 6   וְזָרְקוּ אֶת־הַדָּם עַל־הַמִּזְבֵּחַ סָבִיב אֲשֶׁר־פֶּתַח אֹהֶל
 7   מוֹעֵד׃ וְהִפְשִׁיט אֶת־הָעֹלָה וְנִתַּח אֹתָהּ לִנְתָחֶיהָ׃ וְנָתְנוּ
 8   בְּנֵי אַהֲרֹן הַכֹּהֵן אֵשׁ עַל־הַמִּזְבֵּחַ וְעָרְכוּ עֵצִים עַל־
 9  הָאֵשׁ׃ וְעָרְכוּ בְּנֵי אַהֲרֹן הַכֹּהֲנִים אֵת הַנְּתָחִים אֶת־
10  הָרֹאשׁ וְאֶת־הַפָּדֶר עַל־הָעֵצִים אֲשֶׁר עַל־הָאֵשׁ
11   אֲשֶׁר עַל־הַמִּזְבֵּחַ׃ וְקִרְבּוֹ וּכְרָעָיו יִרְחַץ בַּמָּיִם
12   וְהִקְטִיר הַכֹּהֵן אֶת־הַכֹּל הַמִּזְבֵּחָה עֹלָה אִשֵּׁה רֵיחַ־
13  נִיחוֹחַ לַיהוָה׃♢ [ס]             וְאִם־מִן־הַצֹּאן
14   קָרְבָּנוֹ מִן־הַכְּשָׂבִים אוֹ מִן־הָעִזִּים לְעֹלָה זָכָר תָּמִים
15   יַקְרִיבֶנּוּ׃ וְשָׁחַט אֹתוֹ עַל יֶרֶךְ הַמִּזְבֵּחַ צָפֹנָה לִפְנֵי
16   יְהוָה וְזָרְקוּ בְּנֵי אַהֲרֹן הַכֹּהֲנִים אֶת־דָּמוֹ עַל־הַמִּזְבֵּחַ
17   סָבִיב׃ וְנִתַּח אֹתוֹ לִנְתָחָיו וְאֶת־רֹאשׁוֹ וְאֶת־פִּדְרוֹ
18   וְעָרַךְ הַכֹּהֵן אֹתָם עַל־הָעֵצִים אֲשֶׁר עַל־הָאֵשׁ
19   אֲשֶׁר עַל־הַמִּזְבֵּחַ׃ וְהַקֶּרֶב וְהַכְּרָעַיִם יִרְחַץ בַּמָּיִם
20   וְהִקְרִיב הַכֹּהֵן אֶת־הַכֹּל וְהִקְטִיר הַמִּזְבֵּחָה עֹלָה
21   הוּא אִשֵּׁה רֵיחַ נִיחֹחַ לַיהוָה׃♢ [פ]
22    וְאִם מִן־הָעוֹף עֹלָה קָרְבָּנוֹ לַיהוָה וְהִקְרִיב מִן־הַתֹּרִים
23   אוֹ מִן־בְּנֵי הַיּוֹנָה אֶת־קָרְבָּנוֹ׃ וְהִקְרִיבוֹ הַכֹּהֵן אֶל־
24  הַמִּזְבֵּחַ וּמָלַק אֶת־רֹאשׁוֹ וְהִקְטִיר הַמִּזְבֵּחָה וְנִמְצָה
25   דָמוֹ עַל קִיר הַמִּזְבֵּחַ׃ וְהֵסִיר אֶת־מֻרְאָתוֹ בְּנֹצָתָהּ
26   וְהִשְׁלִיךְ אֹתָהּ אֵצֶל הַמִּזְבֵּחַ קֵדְמָה אֶל־מְקוֹם הַדָּשֶׁן׃
27   וְשִׁסַּע אֹתוֹ בִכְנָפָיו לֹא יַבְדִּיל וְהִקְטִיר אֹתוֹ הַכֹּהֵן
28   הַמִּזְבֵּחָה עַל־הָעֵצִים אֲשֶׁר עַל־הָאֵשׁ עֹלָה הוּא
29   אִשֵּׁה רֵיחַ נִיחֹחַ לַיהוָה׃ [ס]             וְנֶפֶשׁ
30   כִּי־תַקְרִיב קָרְבַּן מִנְחָה לַיהוָה סֹלֶת יִהְיֶה קָרְבָּנוֹ
31   וְיָצַק עָלֶיהָ שֶׁמֶן וְנָתַן עָלֶיהָ לְבֹנָה׃ וֶהֱבִיאָהּ אֶל־בְּנֵי
32   אַהֲרֹן הַכֹּהֲנִים וְקָמַץ מִשָּׁם מְלֹא קֻמְצוֹ מִסָּלְתָּהּ
33   וּמִשַּׁמְנָהּ עַל כָּל־לְבֹנָתָהּ וְהִקְטִיר הַכֹּהֵן אֶת־
34  אַזְכָּרָתָהּ הַמִּזְבֵּחָה אִשֵּׁה רֵיחַ נִיחֹחַ לַיהוָה׃ וְהַנּוֹתֶרֶת
35   מִן־הַמִּנְחָה לְאַהֲרֹן וּלְבָנָיו קֹדֶשׁ קָדָשִׁים מֵאִשֵּׁי
36   יְהוָה׃ [ס]             וְכִי תַקְרִב קָרְבַּן מִנְחָה מַאֲפֵה
37   תַנּוּר סֹלֶת חַלּוֹת מַצֹּת בְּלוּלֹת בַּשֶּׁמֶן וּרְקִיקֵי
38   מַצּוֹת מְשֻׁחִים בַּשָּׁמֶן׃ [ס]             וְאִם־
39  מִנְחָה עַל־הַמַּחֲבַת קָרְבָּנֶךָ סֹלֶת בְּלוּלָה בַשֶּׁמֶן
40   מַצָּה תִהְיֶה׃ פָּתוֹת אֹתָהּ פִּתִּים וְיָצַקְתָּ עָלֶיהָ שָׁמֶן
41   מִנְחָה הִוא׃ [ס]              וְאִם־
42  מִנְחַת מַרְחֶשֶׁת קָרְבָּנֶךָ סֹלֶת בַּשֶּׁמֶן תֵּעָשֶׂה׃
