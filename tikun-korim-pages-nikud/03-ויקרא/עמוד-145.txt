 1  וְתוֹשָׁב עִמָּךְ וּמָךְ אָחִיךָ עִמּוֹ וְנִמְכַּר לְגֵר תּוֹשָׁב
 2   עִמָּךְ אוֹ לְעֵקֶר מִשְׁפַּחַת גֵּר׃ אַחֲרֵי נִמְכַּר גְּאֻלָּה תִּהְיֶה־
 3  לּוֹ אֶחָד מֵאֶחָיו יִגְאָלֶנּוּ׃ אוֹ־דֹדוֹ אוֹ בֶן־דֹּדוֹ יִגְאָלֶנּוּ אוֹ־
 4  מִשְּׁאֵר בְּשָׂרוֹ מִמִּשְׁפַּחְתּוֹ יִגְאָלֶנּוּ אוֹ־הִשִּׂיגָה יָדוֹ
 5   וְנִגְאָל׃ וְחִשַּׁב עִם־קֹנֵהוּ מִשְּׁנַת הִמָּכְרוֹ לוֹ עַד שְׁנַת
 6   הַיֹּבֵל וְהָיָה כֶּסֶף מִמְכָּרוֹ בְּמִסְפַּר שָׁנִים כִּימֵי שָׂכִיר
 7   יִהְיֶה עִמּוֹ׃ אִם־עוֹד רַבּוֹת בַּשָּׁנִים לְפִיהֶן יָשִׁיב גְּאֻלָּתוֹ
 8   מִכֶּסֶף מִקְנָתוֹ׃ וְאִם־מְעַט נִשְׁאַר בַּשָּׁנִים עַד־שְׁנַת
 9   הַיֹּבֵל וְחִשַּׁב־לוֹ כְּפִי שָׁנָיו יָשִׁיב אֶת־גְּאֻלָּתוֹ׃ כִּשְׂכִיר
10   שָׁנָה בְּשָׁנָה יִהְיֶה עִמּוֹ לֹא־יִרְדֶּנּוּ בְּפֶרֶךְ לְעֵינֶיךָ׃
11   וְאִם־לֹא יִגָּאֵל בְּאֵלֶּה וְיָצָא בִּשְׁנַת הַיֹּבֵל הוּא וּבָנָיו
12   עִמּוֹ׃  כִּי־לִי בְנֵי־יִשְׂרָאֵל עֲבָדִים עֲבָדַי הֵם אֲשֶׁר־
13  הוֹצֵאתִי אוֹתָם מֵאֶרֶץ מִצְרָיִם אֲנִי יְהוָה אֱלֹהֵיכֶם׃
14   לֹא־תַעֲשׂוּ לָכֶם אֱלִילִם וּפֶסֶל וּמַצֵּבָה לֹא־תָקִימוּ
15   לָכֶם וְאֶבֶן מַשְׂכִּית לֹא תִתְּנוּ בְּאַרְצְכֶם לְהִשְׁתַּחֲו‍ֹת
16   עָלֶיהָ כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם׃ אֶת־שַׁבְּתֹתַי תִּשְׁמֹרוּ
17   וּמִקְדָּשִׁי תִּירָאוּ אֲנִי יְהוָה׃ [פ]
18  אִם־בְּחֻקֹּתַי תֵּלֵכוּ וְאֶת־מִצְו‍ֹתַי תִּשְׁמְרוּ וַעֲשִׂיתֶם
19   אֹתָם׃ וְנָתַתִּי גִשְׁמֵיכֶם בְּעִתָּם וְנָתְנָה הָאָרֶץ יְבוּלָהּ
20   וְעֵץ הַשָּׂדֶה יִתֵּן פִּרְיוֹ׃ וְהִשִּׂיג לָכֶם דַּיִשׁ אֶת־בָּצִיר
21   וּבָצִיר יַשִּׂיג אֶת־זָרַע וַאֲכַלְתֶּם לַחְמְכֶם לָשֹׂבַע
22   וִישַׁבְתֶּם לָבֶטַח בְּאַרְצְכֶם׃♢  וְנָתַתִּי שָׁלוֹם בָּאָרֶץ
23   וּשְׁכַבְתֶּם וְאֵין מַחֲרִיד וְהִשְׁבַּתִּי חַיָּה רָעָה מִן־הָאָרֶץ
24   וְחֶרֶב לֹא־תַעֲבֹר בְּאַרְצְכֶם׃ וּרְדַפְתֶּם אֶת־אֹיְבֵיכֶם
25   וְנָפְלוּ לִפְנֵיכֶם לֶחָרֶב׃ וְרָדְפוּ מִכֶּם חֲמִשָּׁה מֵאָה
26   וּמֵאָה מִכֶּם רְבָבָה יִרְדֹּפוּ וְנָפְלוּ אֹיְבֵיכֶם לִפְנֵיכֶם
27   לֶחָרֶב׃ וּפָנִיתִי אֲלֵיכֶם וְהִפְרֵיתִי אֶתְכֶם וְהִרְבֵּיתִי
28   אֶתְכֶם וַהֲקִימֹתִי אֶת־בְּרִיתִי אִתְּכֶם׃♢  וַאֲכַלְתֶּם יָשָׁן
29   נוֹשָׁן וְיָשָׁן מִפְּנֵי חָדָשׁ תּוֹצִיאוּ׃ וְנָתַתִּי מִשְׁכָּנִי
30   בְּתוֹכְכֶם וְלֹא־תִגְעַל נַפְשִׁי אֶתְכֶם׃ וְהִתְהַלַּכְתִּי
31   בְּתוֹכְכֶם וְהָיִיתִי לָכֶם לֵאלֹהִים וְאַתֶּם תִּהְיוּ־לִי לְעָם׃
32   אֲנִי יְהוָה אֱלֹהֵיכֶם אֲשֶׁר הוֹצֵאתִי אֶתְכֶם מֵאֶרֶץ
33   מִצְרַיִם מִהְיֹת לָהֶם עֲבָדִים וָאֶשְׁבֹּר מֹטֹת עֻלְּכֶם
34   וָאוֹלֵךְ אֶתְכֶם קוֹמְמִיּוּת׃♢ [פ]
35   וְאִם־לֹא תִשְׁמְעוּ לִי וְלֹא תַעֲשׂוּ אֵת כָּל־הַמִּצְו‍ֹת
36   הָאֵלֶּה׃ וְאִם־בְּחֻקֹּתַי תִּמְאָסוּ וְאִם אֶת־מִשְׁפָּטַי תִּגְעַל
37   נַפְשְׁכֶם לְבִלְתִּי עֲשׂוֹת אֶת־כָּל־מִצְו‍ֹתַי לְהַפְרְכֶם
38   אֶת־בְּרִיתִי׃ אַף־אֲנִי אֶעֱשֶׂה־זֹּאת לָכֶם וְהִפְקַדְתִּי
39   עֲלֵיכֶם בֶּהָלָה אֶת־הַשַּׁחֶפֶת וְאֶת־הַקַּדַּחַת מְכַלּוֹת
40   עֵינַיִם וּמְדִיבֹת נָפֶשׁ וּזְרַעְתֶּם לָרִיק זַרְעֲכֶם וַאֲכָלֻהוּ
41   אֹיְבֵיכֶם׃ וְנָתַתִּי פָנַי בָּכֶם וְנִגַּפְתֶּם לִפְנֵי אֹיְבֵיכֶם
42   וְרָדוּ בָכֶם שֹׂנְאֵיכֶם וְנַסְתֶּם וְאֵין־רֹדֵף אֶתְכֶם׃
