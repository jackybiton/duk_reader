 1  וּפֶתַח אֹהֶל מוֹעֵד תֵּשְׁבוּ יוֹמָם וָלַיְלָה שִׁבְעַת יָמִים
 2   וּשְׁמַרְתֶּם אֶת־מִשְׁמֶרֶת יְהוָה וְלֹא תָמוּתוּ כִּי־כֵן
 3   צֻוֵּיתִי׃ וַיַּעַשׂ אַהֲרֹן וּבָנָיו אֵת כָּל־הַדְּבָרִים אֲשֶׁר־
 4  צִוָּה יְהוָה בְּיַד־מֹשֶׁה׃ [ס]            וַיְהִי
 5   בַּיּוֹם הַשְּׁמִינִי קָרָא מֹשֶׁה לְאַהֲרֹן וּלְבָנָיו וּלְזִקְנֵי
 6   יִשְׂרָאֵל׃ וַיֹּאמֶר אֶל־אַהֲרֹן קַח־לְךָ עֵגֶל בֶּן־בָּקָר
 7   לְחַטָּאת וְאַיִל לְעֹלָה תְּמִימִם וְהַקְרֵב לִפְנֵי יְהוָה׃
 8   וְאֶל־בְּנֵי יִשְׂרָאֵל תְּדַבֵּר לֵאמֹר קְחוּ שְׂעִיר־עִזִּים
 9   לְחַטָּאת וְעֵגֶל וָכֶבֶשׂ בְּנֵי־שָׁנָה תְּמִימִם לְעֹלָה׃
10   וְשׁוֹר וָאַיִל לִשְׁלָמִים לִזְבֹּחַ לִפְנֵי יְהוָה וּמִנְחָה
11   בְּלוּלָה בַשָּׁמֶן כִּי הַיּוֹם יְהוָה נִרְאָה אֲלֵיכֶם׃ וַיִּקְחוּ
12   אֵת אֲשֶׁר צִוָּה מֹשֶׁה אֶל־פְּנֵי אֹהֶל מוֹעֵד וַיִּקְרְבוּ
13   כָּל־הָעֵדָה וַיַּעַמְדוּ לִפְנֵי יְהוָה׃ וַיֹּאמֶר מֹשֶׁה זֶה
14   הַדָּבָר אֲשֶׁר־צִוָּה יְהוָה תַּעֲשׂוּ וְיֵרָא אֲלֵיכֶם כְּבוֹד
15   יְהוָה׃♢ וַיֹּאמֶר מֹשֶׁה אֶל־אַהֲרֹן קְרַב אֶל־הַמִּזְבֵּחַ
16   וַעֲשֵׂה אֶת־חַטָּאתְךָ וְאֶת־עֹלָתֶךָ וְכַפֵּר בַּעַדְךָ וּבְעַד
17   הָעָם וַעֲשֵׂה אֶת־קָרְבַּן הָעָם וְכַפֵּר בַּעֲדָם כַּאֲשֶׁר
18   צִוָּה יְהוָה׃ וַיִּקְרַב אַהֲרֹן אֶל־הַמִּזְבֵּחַ וַיִּשְׁחַט אֶת־עֵגֶל
19   הַחַטָּאת אֲשֶׁר־לוֹ׃ וַיַּקְרִבוּ בְּנֵי אַהֲרֹן אֶת־הַדָּם אֵלָיו
20   וַיִּטְבֹּל אֶצְבָּעוֹ בַּדָּם וַיִּתֵּן עַל־קַרְנוֹת הַמִּזְבֵּחַ וְאֶת־
21  הַדָּם יָצַק אֶל־יְסוֹד הַמִּזְבֵּחַ׃ וְאֶת־הַחֵלֶב וְאֶת־הַכְּלָיֹת
22   וְאֶת־הַיֹּתֶרֶת מִן־הַכָּבֵד מִן־הַחַטָּאת הִקְטִיר הַמִּזְבֵּחָה
23   כַּאֲשֶׁר צִוָּה יְהוָה אֶת־מֹשֶׁה׃♢ וְאֶת־הַבָּשָׂר וְאֶת־הָעוֹר
24   שָׂרַף בָּאֵשׁ מִחוּץ לַמַּחֲנֶה׃ וַיִּשְׁחַט אֶת־הָעֹלָה
25   וַיַּמְצִאוּ בְּנֵי אַהֲרֹן אֵלָיו אֶת־הַדָּם וַיִּזְרְקֵהוּ עַל־
26  הַמִּזְבֵּחַ סָבִיב׃ וְאֶת־הָעֹלָה הִמְצִיאוּ אֵלָיו לִנְתָחֶיהָ
27   וְאֶת־הָרֹאשׁ וַיַּקְטֵר עַל־הַמִּזְבֵּחַ׃ וַיִּרְחַץ אֶת־הַקֶּרֶב
28   וְאֶת־הַכְּרָעָיִם וַיַּקְטֵר עַל־הָעֹלָה הַמִּזְבֵּחָה׃ וַיַּקְרֵב
29   אֵת קָרְבַּן הָעָם וַיִּקַּח אֶת־שְׂעִיר הַחַטָּאת אֲשֶׁר
30   לָעָם וַיִּשְׁחָטֵהוּ וַיְחַטְּאֵהוּ כָּרִאשׁוֹן׃ וַיַּקְרֵב אֶת־
31  הָעֹלָה וַיַּעֲשֶׂהָ כַּמִּשְׁפָּט׃♢  וַיַּקְרֵב אֶת־הַמִּנְחָה וַיְמַלֵּא
32   כַפּוֹ מִמֶּנָּה וַיַּקְטֵר עַל־הַמִּזְבֵּחַ מִלְּבַד עֹלַת הַבֹּקֶר׃
33   וַיִּשְׁחַט אֶת־הַשּׁוֹר וְאֶת־הָאַיִל זֶבַח הַשְּׁלָמִים אֲשֶׁר
34   לָעָם וַיַּמְצִאוּ בְּנֵי אַהֲרֹן אֶת־הַדָּם אֵלָיו וַיִּזְרְקֵהוּ עַל־
35  הַמִּזְבֵּחַ סָבִיב׃ וְאֶת־הַחֲלָבִים מִן־הַשּׁוֹר וּמִן־הָאַיִל
36   הָאַלְיָה וְהַמְכַסֶּה וְהַכְּלָיֹת וְיֹתֶרֶת הַכָּבֵד׃ וַיָּשִׂימוּ
37   אֶת־הַחֲלָבִים עַל־הֶחָזוֹת וַיַּקְטֵר הַחֲלָבִים הַמִּזְבֵּחָה׃
38   וְאֵת הֶחָזוֹת וְאֵת שׁוֹק הַיָּמִין הֵנִיף אַהֲרֹן תְּנוּפָה
39   לִפְנֵי יְהוָה כַּאֲשֶׁר צִוָּה מֹשֶׁה׃ וַיִּשָּׂא אַהֲרֹן אֶת־יָדָו
40   אֶל־הָעָם וַיְבָרְכֵם וַיֵּרֶד מֵעֲשֹׂת הַחַטָּאת וְהָעֹלָה
41   וְהַשְּׁלָמִים׃ וַיָּבֹא מֹשֶׁה וְאַהֲרֹן אֶל־אֹהֶל מוֹעֵד וַיֵּצְאוּ
42   וַיְבָרְכוּ אֶת־הָעָם וַיֵּרָא כְבוֹד־יְהוָה אֶל־כָּל־הָעָם׃
