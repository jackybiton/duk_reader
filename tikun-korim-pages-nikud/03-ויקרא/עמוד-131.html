1וְאִשָּׁהכִּי־תִהְיֶהזָבָהדָּםיִהְיֶהזֹבָהּבִּבְשָׂרָהּ
2שִׁבְעַתיָמִיםתִּהְיֶהבְנִדָּתָהּוְכָל־הַנֹּגֵעַבָּהּיִטְמָא
3עַד־הָעָרֶב׃וְכֹלאֲשֶׁרתִּשְׁכַּבעָלָיובְּנִדָּתָהּיִטְמָא
4וְכֹלאֲשֶׁר־תֵּשֵׁבעָלָיויִטְמָא׃וְכָל־הַנֹּגֵעַבְּמִשְׁכָּבָהּ
5יְכַבֵּסבְּגָדָיווְרָחַץבַּמַּיִםוְטָמֵאעַד־הָעָרֶב׃וְכָל־
6הַנֹּגֵעַבְּכָל־כְּלִיאֲשֶׁר־תֵּשֵׁבעָלָיויְכַבֵּסבְּגָדָיו
7וְרָחַץבַּמַּיִםוְטָמֵאעַד־הָעָרֶב׃וְאִםעַל־הַמִּשְׁכָּב
8הוּאאוֹעַל־הַכְּלִיאֲשֶׁר־הִואיֹשֶׁבֶת־עָלָיובְּנָגְעוֹ־בוֹ
9יִטְמָאעַד־הָעָרֶב׃וְאִםשָׁכֹביִשְׁכַּבאִישׁאֹתָהּ
10וּתְהִינִדָּתָהּעָלָיווְטָמֵאשִׁבְעַתיָמִיםוְכָל־הַמִּשְׁכָּב
11אֲשֶׁר־יִשְׁכַּב עָלָיו יִטְמָא׃וְאִשָּׁה
12כִּי־יָזוּבזוֹבדָּמָהּיָמִיםרַבִּיםבְּלֹאעֶת־נִדָּתָהּאוֹ
13כִי־תָזוּבעַל־נִדָּתָהּכָּל־יְמֵיזוֹבטֻמְאָתָהּכִּימֵינִדָּתָהּ
14תִּהְיֶהטְמֵאָההִוא׃כָּל־הַמִּשְׁכָּבאֲשֶׁר־תִּשְׁכַּב
15עָלָיוכָּל־יְמֵיזוֹבָהּכְּמִשְׁכַּבנִדָּתָהּיִהְיֶה־לָּהּוְכָל־
16הַכְּלִיאֲשֶׁרתֵּשֵׁבעָלָיוטָמֵאיִהְיֶהכְּטֻמְאַתנִדָּתָהּ׃
17וְכָל־הַנּוֹגֵעַבָּםיִטְמָאוְכִבֶּסבְּגָדָיווְרָחַץבַּמַּיִם
18וְטָמֵאעַד־הָעָרֶב׃וְאִם־טָהֲרָהמִזּוֹבָהּוְסָפְרָהלָּהּ
19שִׁבְעַתיָמִיםוְאַחַרתִּטְהָר׃וּבַיּוֹםהַשְּׁמִינִיתִּקַּח־
20לָהּשְׁתֵּיתֹרִיםאוֹשְׁנֵיבְּנֵייוֹנָהוְהֵבִיאָהאוֹתָם
21אֶל־הַכֹּהֵןאֶל־פֶּתַחאֹהֶלמוֹעֵד׃וְעָשָׂההַכֹּהֵן
22אֶת־הָאֶחָדחַטָּאתוְאֶת־הָאֶחָדעֹלָהוְכִפֶּרעָלֶיהָ
23הַכֹּהֵןלִפְנֵייְהוָהמִזּוֹבטֻמְאָתָהּ׃וְהִזַּרְתֶּםאֶת־בְּנֵי־
24יִשְׂרָאֵלמִטֻּמְאָתָםוְלֹאיָמֻתוּבְּטֻמְאָתָםבְּטַמְּאָם
25אֶת־מִשְׁכָּנִיאֲשֶׁרבְּתוֹכָם׃זֹאתתּוֹרַתהַזָּבוַאֲשֶׁר
26תֵּצֵאמִמֶּנּוּשִׁכְבַת־זֶרַעלְטָמְאָה־בָהּ׃וְהַדָּוָהבְּנִדָּתָהּ
27וְהַזָּבאֶת־זוֹבוֹלַזָּכָרוְלַנְּקֵבָהוּלְאִישׁאֲשֶׁריִשְׁכַּב
28עִם־טְמֵאָה׃
29וַיְדַבֵּריְהוָהאֶל־מֹשֶׁהאַחֲרֵימוֹתשְׁנֵיבְּנֵיאַהֲרֹן
30בְּקָרְבָתָםלִפְנֵי־יְהוָהוַיָּמֻתוּ׃וַיֹּאמֶריְהוָהאֶל־מֹשֶׁה
31דַּבֵּראֶל־אַהֲרֹןאָחִיךָוְאַל־יָבֹאבְכָל־עֵתאֶל־
32הַקֹּדֶשׁמִבֵּיתלַפָּרֹכֶתאֶל־פְּנֵיהַכַּפֹּרֶתאֲשֶׁר
33עַל־הָאָרֹןוְלֹאיָמוּתכִּיבֶּעָנָןאֵרָאֶהעַל־הַכַּפֹּרֶת׃
34בְּזֹאתיָבֹאאַהֲרֹןאֶל־הַקֹּדֶשׁבְּפַרבֶּן־בָּקָר
35לְחַטָּאתוְאַיִללְעֹלָה׃כְּתֹנֶת־בַּדקֹדֶשׁיִלְבָּשׁ
36וּמִכְנְסֵי־בַדיִהְיוּעַל־בְּשָׂרוֹוּבְאַבְנֵטבַּדיַחְגֹּר
37וּבְמִצְנֶפֶתבַּדיִצְנֹףבִּגְדֵי־קֹדֶשׁהֵםוְרָחַץבַּמַּיִם
38אֶת־בְּשָׂרוֹוּלְבֵשָׁם׃וּמֵאֵתעֲדַתבְּנֵייִשְׂרָאֵליִקַּח
39שְׁנֵי־שְׂעִירֵיעִזִּיםלְחַטָּאתוְאַיִלאֶחָדלְעֹלָה׃
40וְהִקְרִיבאַהֲרֹןאֶת־פַּרהַחַטָּאתאֲשֶׁר־לוֹוְכִפֶּר
