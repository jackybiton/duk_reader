 1  וְכִפֶּר עֲלֵהֶם הַכֹּהֵן וְנִסְלַח לָהֶם׃ וְהוֹצִיא אֶת־הַפָּר
 2   אֶל־מִחוּץ לַמַּחֲנֶה וְשָׂרַף אֹתוֹ כַּאֲשֶׁר שָׂרַף אֵת הַפָּר
 3   הָרִאשׁוֹן חַטַּאת הַקָּהָל הוּא׃ [פ]
 4   אֲשֶׁר נָשִׂיא יֶחֱטָא וְעָשָׂה אַחַת מִכָּל־מִצְו‍ֹת יְהוָה
 5   אֱלֹהָיו אֲשֶׁר לֹא־תֵעָשֶׂינָה בִּשְׁגָגָה וְאָשֵׁם׃ אוֹ־הוֹדַע
 6   אֵלָיו חַטָּאתוֹ אֲשֶׁר חָטָא בָּהּ וְהֵבִיא אֶת־קָרְבָּנוֹ
 7   שְׂעִיר עִזִּים זָכָר תָּמִים׃ וְסָמַךְ יָדוֹ עַל־רֹאשׁ הַשָּׂעִיר
 8   וְשָׁחַט אֹתוֹ בִּמְקוֹם אֲשֶׁר־יִשְׁחַט אֶת־הָעֹלָה לִפְנֵי
 9   יְהוָה חַטָּאת הוּא׃ וְלָקַח הַכֹּהֵן מִדַּם הַחַטָּאת בְּאֶצְבָּעוֹ
10   וְנָתַן עַל־קַרְנֹת מִזְבַּח הָעֹלָה וְאֶת־דָּמוֹ יִשְׁפֹּךְ אֶל־
11  יְסוֹד מִזְבַּח הָעֹלָה׃ וְאֶת־כָּל־חֶלְבּוֹ יַקְטִיר הַמִּזְבֵּחָה
12   כְּחֵלֶב זֶבַח הַשְּׁלָמִים וְכִפֶּר עָלָיו הַכֹּהֵן מֵחַטָּאתוֹ
13   וְנִסְלַח לוֹ׃ [פ]
14    וְאִם־נֶפֶשׁ אַחַת תֶּחֱטָא בִשְׁגָגָה מֵעַם הָאָרֶץ בַּעֲשֹׂתָהּ
15   אַחַת מִמִּצְו‍ֹת יְהוָה אֲשֶׁר לֹא־תֵעָשֶׂינָה וְאָשֵׁם׃ אוֹ
16   הוֹדַע אֵלָיו חַטָּאתוֹ אֲשֶׁר חָטָא וְהֵבִיא קָרְבָּנוֹ שְׂעִירַת
17   עִזִּים תְּמִימָה נְקֵבָה עַל־חַטָּאתוֹ אֲשֶׁר חָטָא׃ וְסָמַךְ אֶת־
18  יָדוֹ עַל רֹאשׁ הַחַטָּאת וְשָׁחַט אֶת־הַחַטָּאת בִּמְקוֹם
19   הָעֹלָה׃ וְלָקַח הַכֹּהֵן מִדָּמָהּ בְּאֶצְבָּעוֹ וְנָתַן עַל־קַרְנֹת
20   מִזְבַּח הָעֹלָה וְאֶת־כָּל־דָּמָהּ יִשְׁפֹּךְ אֶל־יְסוֹד הַמִּזְבֵּחַ׃
21   וְאֶת־כָּל־חֶלְבָּהּ יָסִיר כַּאֲשֶׁר הוּסַר חֵלֶב מֵעַל זֶבַח
22   הַשְּׁלָמִים וְהִקְטִיר הַכֹּהֵן הַמִּזְבֵּחָה לְרֵיחַ נִיחֹחַ לַיהוָה
23   וְכִפֶּר עָלָיו הַכֹּהֵן וְנִסְלַח לוֹ׃ [פ]
24   וְאִם־כֶּבֶשׂ יָבִיא קָרְבָּנוֹ לְחַטָּאת נְקֵבָה תְמִימָה
25   יְבִיאֶנָּה׃ וְסָמַךְ אֶת־יָדוֹ עַל רֹאשׁ הַחַטָּאת וְשָׁחַט
26   אֹתָהּ לְחַטָּאת בִּמְקוֹם אֲשֶׁר יִשְׁחַט אֶת־הָעֹלָה׃
27   וְלָקַח הַכֹּהֵן מִדַּם הַחַטָּאת בְּאֶצְבָּעוֹ וְנָתַן עַל־קַרְנֹת
28   מִזְבַּח הָעֹלָה וְאֶת־כָּל־דָּמָהּ יִשְׁפֹּךְ אֶל־יְסוֹד הַמִּזְבֵּחַ׃
29   וְאֶת־כָּל־חֶלְבָּהּ יָסִיר כַּאֲשֶׁר יוּסַר חֵלֶב־הַכֶּשֶׂב
30   מִזֶּבַח הַשְּׁלָמִים וְהִקְטִיר הַכֹּהֵן אֹתָם הַמִּזְבֵּחָה עַל
31   אִשֵּׁי יְהוָה וְכִפֶּר עָלָיו הַכֹּהֵן עַל־חַטָּאתוֹ אֲשֶׁר־
32  חָטָא וְנִסְלַח לוֹ׃ [פ]
33   וְנֶפֶשׁ כִּי־תֶחֱטָא וְשָׁמְעָה קוֹל אָלָה וְהוּא עֵד אוֹ
34   רָאָה אוֹ יָדָע אִם־לוֹא יַגִּיד וְנָשָׂא עֲו‍ֹנוֹ׃ אוֹ נֶפֶשׁ אֲשֶׁר
35   תִּגַּע בְּכָל־דָּבָר טָמֵא אוֹ בְנִבְלַת חַיָּה טְמֵאָה אוֹ
36   בְּנִבְלַת בְּהֵמָה טְמֵאָה אוֹ בְּנִבְלַת שֶׁרֶץ טָמֵא וְנֶעְלַם
37   מִמֶּנּוּ וְהוּא טָמֵא וְאָשֵׁם׃ אוֹ כִי יִגַּע בְּטֻמְאַת אָדָם
38   לְכֹל טֻמְאָתוֹ אֲשֶׁר יִטְמָא בָּהּ וְנֶעְלַם מִמֶּנּוּ וְהוּא יָדַע
39   וְאָשֵׁם׃ אוֹ נֶפֶשׁ כִּי תִשָּׁבַע לְבַטֵּא בִשְׂפָתַיִם לְהָרַע
40   אוֹ לְהֵיטִיב לְכֹל אֲשֶׁר יְבַטֵּא הָאָדָם בִּשְׁבֻעָה
41   וְנֶעְלַם מִמֶּנּוּ וְהוּא־יָדַע וְאָשֵׁם לְאַחַת מֵאֵלֶּה׃ וְהָיָה
42   כִי־יֶאְשַׁם לְאַחַת מֵאֵלֶּה וְהִתְוַדָּה אֲשֶׁר חָטָא עָלֶיהָ׃
