 1  שְׁנֵי הַשְּׂעִירִם גֹּרָלוֹת גּוֹרָל אֶחָד לַיהוָה וְגוֹרָל
 2   אֶחָד לַעֲזָאזֵל׃ וְהִקְרִיב אַהֲרֹן אֶת־הַשָּׂעִיר אֲשֶׁר
 3   עָלָה עָלָיו הַגּוֹרָל לַיהוָה וְעָשָׂהוּ חַטָּאת׃ וְהַשָּׂעִיר
 4   אֲשֶׁר עָלָה עָלָיו הַגּוֹרָל לַעֲזָאזֵל יָעֳמַד־חַי לִפְנֵי
 5   יְהוָה לְכַפֵּר עָלָיו לְשַׁלַּח אֹתוֹ לַעֲזָאזֵל הַמִּדְבָּרָה׃
 6   וְהִקְרִיב אַהֲרֹן אֶת־פַּר הַחַטָּאת אֲשֶׁר־לוֹ וְכִפֶּר
 7   בַּעֲדוֹ וּבְעַד בֵּיתוֹ וְשָׁחַט אֶת־פַּר הַחַטָּאת אֲשֶׁר־
 8  לוֹ׃♢ וְלָקַח מְלֹא־הַמַּחְתָּה גַּחֲלֵי־אֵשׁ מֵעַל הַמִּזְבֵּחַ
 9   מִלִּפְנֵי יְהוָה וּמְלֹא חָפְנָיו קְטֹרֶת סַמִּים דַּקָּה
10   וְהֵבִיא מִבֵּית לַפָּרֹכֶת׃ וְנָתַן אֶת־הַקְּטֹרֶת עַל־הָאֵשׁ
11   לִפְנֵי יְהוָה וְכִסָּה  עֲנַן הַקְּטֹרֶת אֶת־הַכַּפֹּרֶת אֲשֶׁר
12   עַל־הָעֵדוּת וְלֹא יָמוּת׃ וְלָקַח מִדַּם הַפָּר וְהִזָּה
13   בְאֶצְבָּעוֹ עַל־פְּנֵי הַכַּפֹּרֶת קֵדְמָה וְלִפְנֵי הַכַּפֹּרֶת
14   יַזֶּה שֶׁבַע־פְּעָמִים מִן־הַדָּם בְּאֶצְבָּעוֹ׃ וְשָׁחַט אֶת־
15  שְׂעִיר הַחַטָּאת אֲשֶׁר לָעָם וְהֵבִיא אֶת־דָּמוֹ אֶל־
16  מִבֵּית לַפָּרֹכֶת וְעָשָׂה אֶת־דָּמוֹ כַּאֲשֶׁר עָשָׂה לְדַם
17   הַפָּר וְהִזָּה אֹתוֹ עַל־הַכַּפֹּרֶת וְלִפְנֵי הַכַּפֹּרֶת׃ וְכִפֶּר
18   עַל־הַקֹּדֶשׁ מִטֻּמְאֹת בְּנֵי יִשְׂרָאֵל וּמִפִּשְׁעֵיהֶם לְכָל־
19  חַטֹּאתָם וְכֵן יַעֲשֶׂה לְאֹהֶל מוֹעֵד הַשֹּׁכֵן אִתָּם
20   בְּתוֹךְ טֻמְאֹתָם׃ וְכָל־אָדָם לֹא־יִהְיֶה  בְּאֹהֶל מוֹעֵד
21   בְּבֹאוֹ לְכַפֵּר בַּקֹּדֶשׁ עַד־צֵאתוֹ וְכִפֶּר בַּעֲדוֹ וּבְעַד
22   בֵּיתוֹ וּבְעַד כָּל־קְהַל יִשְׂרָאֵל׃♢   וְיָצָא אֶל־הַמִּזְבֵּחַ
23   אֲשֶׁר לִפְנֵי־יְהוָה וְכִפֶּר עָלָיו וְלָקַח מִדַּם הַפָּר
24   וּמִדַּם הַשָּׂעִיר וְנָתַן עַל־קַרְנוֹת הַמִּזְבֵּחַ סָבִיב׃ וְהִזָּה
25   עָלָיו מִן־הַדָּם בְּאֶצְבָּעוֹ שֶׁבַע פְּעָמִים וְטִהֲרוֹ וְקִדְּשׁוֹ
26   מִטֻּמְאֹת בְּנֵי יִשְׂרָאֵל׃ וְכִלָּה מִכַּפֵּר אֶת־הַקֹּדֶשׁ
27   וְאֶת־אֹהֶל מוֹעֵד וְאֶת־הַמִּזְבֵּחַ וְהִקְרִיב אֶת־הַשָּׂעִיר
28   הֶחָי׃ וְסָמַךְ אַהֲרֹן אֶת־שְׁתֵּי יָדָו עַל רֹאשׁ הַשָּׂעִיר
29   הַחַי וְהִתְוַדָּה עָלָיו אֶת־כָּל־עֲו‍ֹנֹת בְּנֵי יִשְׂרָאֵל וְאֶת־
30  כָּל־פִּשְׁעֵיהֶם לְכָל־חַטֹּאתָם וְנָתַן אֹתָם עַל־רֹאשׁ
31   הַשָּׂעִיר וְשִׁלַּח בְּיַד־אִישׁ עִתִּי הַמִּדְבָּרָה׃ וְנָשָׂא
32   הַשָּׂעִיר עָלָיו אֶת־כָּל־עֲו‍ֹנֹתָם אֶל־אֶרֶץ גְּזֵרָה וְשִׁלַּח
33   אֶת־הַשָּׂעִיר בַּמִּדְבָּר׃ וּבָא אַהֲרֹן אֶל־אֹהֶל מוֹעֵד
34   וּפָשַׁט אֶת־בִּגְדֵי הַבָּד אֲשֶׁר לָבַשׁ בְּבֹאוֹ אֶל־הַקֹּדֶשׁ
35   וְהִנִּיחָם שָׁם׃ וְרָחַץ אֶת־בְּשָׂרוֹ בַמַּיִם בְּמָקוֹם קָדוֹשׁ
36   וְלָבַשׁ אֶת־בְּגָדָיו וְיָצָא וְעָשָׂה אֶת־עֹלָתוֹ וְאֶת־עֹלַת
37   הָעָם וְכִפֶּר בַּעֲדוֹ וּבְעַד הָעָם׃  וְאֵת חֵלֶב הַחַטָּאת
38   יַקְטִיר הַמִּזְבֵּחָה׃ וְהַמְשַׁלֵּחַ אֶת־הַשָּׂעִיר לַעֲזָאזֵל
39   יְכַבֵּס בְּגָדָיו וְרָחַץ אֶת־בְּשָׂרוֹ בַּמָּיִם וְאַחֲרֵי־כֵן
40   יָבוֹא אֶל־הַמַּחֲנֶה׃ וְאֵת פַּר הַחַטָּאת וְאֵת  שְׂעִיר
41   הַחַטָּאת אֲשֶׁר הוּבָא אֶת־דָּמָם לְכַפֵּר בַּקֹּדֶשׁ
42   יוֹצִיא אֶל־מִחוּץ לַמַּחֲנֶה וְשָׂרְפוּ בָאֵשׁ אֶת־עֹרֹתָם
