 1  וְהִוא שִׁפְחָה נֶחֱרֶפֶת לְאִישׁ וְהָפְדֵּה לֹא נִפְדָּתָה
 2   אוֹ חֻפְשָׁה לֹא נִתַּן־לָהּ בִּקֹּרֶת תִּהְיֶה לֹא יוּמְתוּ כִּי־
 3  לֹא חֻפָּשָׁה׃ וְהֵבִיא אֶת־אֲשָׁמוֹ לַיהוָה אֶל־פֶּתַח
 4   אֹהֶל מוֹעֵד אֵיל אָשָׁם׃ וְכִפֶּר עָלָיו הַכֹּהֵן בְּאֵיל
 5   הָאָשָׁם לִפְנֵי יְהוָה עַל־חַטָּאתוֹ אֲשֶׁר חָטָא וְנִסְלַח
 6   לוֹ מֵחַטָּאתוֹ אֲשֶׁר חָטָא׃ [פ]
 7    וְכִי־תָבֹאוּ אֶל־הָאָרֶץ וּנְטַעְתֶּם כָּל־עֵץ מַאֲכָל
 8   וַעֲרַלְתֶּם עָרְלָתוֹ אֶת־פִּרְיוֹ שָׁלֹשׁ שָׁנִים יִהְיֶה לָכֶם
 9   עֲרֵלִים לֹא יֵאָכֵל׃ וּבַשָּׁנָה הָרְבִיעִת יִהְיֶה כָּל־פִּרְיוֹ
10   קֹדֶשׁ הִלּוּלִים לַיהוָה׃ וּבַשָּׁנָה הַחֲמִישִׁת תֹּאכְלוּ אֶת־
11  פִּרְיוֹ לְהוֹסִיף לָכֶם תְּבוּאָתוֹ אֲנִי יְהוָה אֱלֹהֵיכֶם׃ לֹא
12   תֹאכְלוּ עַל־הַדָּם לֹא תְנַחֲשׁוּ וְלֹא תְעוֹנֵנוּ׃ לֹא תַקִּפוּ
13   פְּאַת רֹאשְׁכֶם וְלֹא תַשְׁחִית אֵת פְּאַת זְקָנֶךָ׃ וְשֶׂרֶט
14   לָנֶפֶשׁ לֹא תִתְּנוּ בִּבְשַׂרְכֶם וּכְתֹבֶת קַעֲקַע לֹא תִתְּנוּ
15   בָּכֶם אֲנִי יְהוָה׃ אַל־תְּחַלֵּל אֶת־בִּתְּךָ לְהַזְנוֹתָהּ וְלֹא־
16  תִזְנֶה הָאָרֶץ וּמָלְאָה הָאָרֶץ זִמָּה׃ אֶת־שַׁבְּתֹתַי תִּשְׁמֹרוּ
17   וּמִקְדָּשִׁי תִּירָאוּ אֲנִי יְהוָה׃ אַל־תִּפְנוּ אֶל־הָאֹבֹת
18   וְאֶל־הַיִּדְּעֹנִים אַל־תְּבַקְשׁוּ לְטָמְאָה בָהֶם אֲנִי יְהוָה
19   אֱלֹהֵיכֶם׃ מִפְּנֵי שֵׂיבָה תָּקוּם וְהָדַרְתָּ פְּנֵי זָקֵן וְיָרֵאתָ
20   מֵּאֱלֹהֶיךָ אֲנִי יְהוָה׃ [ס]              וְכִי־יָגוּר
21   אִתְּךָ גֵּר בְּאַרְצְכֶם לֹא תוֹנוּ אֹתוֹ׃ כְּאֶזְרָח מִכֶּם יִהְיֶה
22   לָכֶם הַגֵּר  הַגָּר אִתְּכֶם וְאָהַבְתָּ לוֹ כָּמוֹךָ כִּי־גֵרִים
23   הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם אֲנִי יְהוָה אֱלֹהֵיכֶם׃ לֹא־תַעֲשׂוּ
24   עָוֶל בַּמִּשְׁפָּט בַּמִּדָּה בַּמִּשְׁקָל וּבַמְּשׂוּרָה׃ מֹאזְנֵי
25   צֶדֶק אַבְנֵי־צֶדֶק אֵיפַת צֶדֶק וְהִין צֶדֶק יִהְיֶה לָכֶם
26   אֲנִי יְהוָה אֱלֹהֵיכֶם אֲשֶׁר־הוֹצֵאתִי אֶתְכֶם מֵאֶרֶץ
27   מִצְרָיִם׃ וּשְׁמַרְתֶּם אֶת־כָּל־חֻקֹּתַי וְאֶת־כָּל־מִשְׁפָּטַי
28   וַעֲשִׂיתֶם אֹתָם אֲנִי יְהוָה׃ [פ]
29    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ וְאֶל־בְּנֵי יִשְׂרָאֵל
30   תֹּאמַר אִישׁ אִישׁ מִבְּנֵי יִשְׂרָאֵל וּמִן־הַגֵּר  הַגָּר
31   בְּיִשְׂרָאֵל אֲשֶׁר יִתֵּן מִזַּרְעוֹ לַמֹּלֶךְ מוֹת יוּמָת עַם
32   הָאָרֶץ יִרְגְּמֻהוּ בָאָבֶן׃ וַאֲנִי אֶתֵּן אֶת־פָּנַי בָּאִישׁ
33   הַהוּא וְהִכְרַתִּי אֹתוֹ מִקֶּרֶב עַמּוֹ כִּי מִזַּרְעוֹ נָתַן
34   לַמֹּלֶךְ לְמַעַן טַמֵּא אֶת־מִקְדָּשִׁי וּלְחַלֵּל אֶת־שֵׁם
35   קָדְשִׁי׃ וְאִם הַעְלֵם יַעְלִימוּ עַם הָאָרֶץ אֶת־עֵינֵיהֶם
36   מִן־הָאִישׁ הַהוּא בְּתִתּוֹ מִזַּרְעוֹ לַמֹּלֶךְ לְבִלְתִּי הָמִית
37   אֹתוֹ׃ וְשַׂמְתִּי אֲנִי אֶת־פָּנַי בָּאִישׁ הַהוּא וּבְמִשְׁפַּחְתּוֹ
38   וְהִכְרַתִּי אֹתוֹ וְאֵת  כָּל־הַזֹּנִים אַחֲרָיו לִזְנוֹת אַחֲרֵי
39   הַמֹּלֶךְ מִקֶּרֶב עַמָּם׃ וְהַנֶּפֶשׁ אֲשֶׁר תִּפְנֶה אֶל־הָאֹבֹת
40   וְאֶל־הַיִּדְּעֹנִים לִזְנֹת אַחֲרֵיהֶם וְנָתַתִּי אֶת־פָּנַי בַּנֶּפֶשׁ
41   הַהִוא וְהִכְרַתִּי אֹתוֹ מִקֶּרֶב עַמּוֹ׃ וְהִתְקַדִּשְׁתֶּם וִהְיִיתֶם
42   קְדֹשִׁים כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם׃  וּשְׁמַרְתֶּם אֶת־חֻקֹּתַי
