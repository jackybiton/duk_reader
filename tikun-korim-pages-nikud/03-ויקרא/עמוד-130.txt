 1  וְלָקַח לְחַטֵּא אֶת־הַבַּיִת שְׁתֵּי צִפֳּרִים וְעֵץ אֶרֶז
 2   וּשְׁנִי תוֹלַעַת וְאֵזֹב׃ וְשָׁחַט אֶת־הַצִּפֹּר הָאֶחָת אֶל־
 3  כְּלִי־חֶרֶשׂ עַל־מַיִם חַיִּים׃ וְלָקַח אֶת־עֵץ־הָאֶרֶז
 4   וְאֶת־הָאֵזֹב וְאֵת  שְׁנִי הַתּוֹלַעַת וְאֵת הַצִּפֹּר הַחַיָּה
 5   וְטָבַל אֹתָם בְּדַם הַצִּפֹּר הַשְּׁחוּטָה וּבַמַּיִם הַחַיִּים
 6   וְהִזָּה אֶל־הַבַּיִת שֶׁבַע פְּעָמִים׃ וְחִטֵּא אֶת־הַבַּיִת
 7   בְּדַם הַצִּפּוֹר וּבַמַּיִם הַחַיִּים וּבַצִּפֹּר הַחַיָּה וּבְעֵץ
 8   הָאֶרֶז וּבָאֵזֹב וּבִשְׁנִי הַתּוֹלָעַת׃ וְשִׁלַּח אֶת־הַצִּפֹּר
 9   הַחַיָּה אֶל־מִחוּץ לָעִיר אֶל־פְּנֵי הַשָּׂדֶה וְכִפֶּר
10   עַל־הַבַּיִת וְטָהֵר׃  זֹאת הַתּוֹרָה לְכָל־נֶגַע הַצָּרַעַת
11   וְלַנָּתֶק׃ וּלְצָרַעַת הַבֶּגֶד וְלַבָּיִת׃ וְלַשְׂאֵת וְלַסַּפַּחַת
12   וְלַבֶּהָרֶת׃ לְהוֹרֹת בְּיוֹם הַטָּמֵא וּבְיוֹם הַטָּהֹר זֹאת
13   תּוֹרַת הַצָּרָעַת׃ [פ]
14   וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה וְאֶל־אַהֲרֹן לֵאמֹר׃ דַּבְּרוּ
15   אֶל־בְּנֵי יִשְׂרָאֵל וַאֲמַרְתֶּם אֲלֵהֶם אִישׁ אִישׁ כִּי
16   יִהְיֶה זָב מִבְּשָׂרוֹ זוֹבוֹ טָמֵא הוּא׃ וְזֹאת תִּהְיֶה טֻמְאָתוֹ
17   בְּזוֹבוֹ רָר בְּשָׂרוֹ אֶת־זוֹבוֹ אוֹ־הֶחְתִּים בְּשָׂרוֹ מִזּוֹבוֹ
18   טֻמְאָתוֹ הִוא׃ כָּל־הַמִּשְׁכָּב אֲשֶׁר יִשְׁכַּב עָלָיו הַזָּב
19   יִטְמָא וְכָל־הַכְּלִי אֲשֶׁר־יֵשֵׁב עָלָיו יִטְמָא׃ וְאִישׁ
20   אֲשֶׁר יִגַּע בְּמִשְׁכָּבוֹ יְכַבֵּס בְּגָדָיו וְרָחַץ בַּמַּיִם
21   וְטָמֵא עַד־הָעָרֶב׃ וְהַיֹּשֵׁב עַל־הַכְּלִי אֲשֶׁר־יֵשֵׁב
22   עָלָיו הַזָּב יְכַבֵּס בְּגָדָיו וְרָחַץ בַּמַּיִם וְטָמֵא עַד־
23  הָעָרֶב׃ וְהַנֹּגֵעַ בִּבְשַׂר הַזָּב יְכַבֵּס בְּגָדָיו וְרָחַץ בַּמַּיִם
24   וְטָמֵא עַד־הָעָרֶב׃ וְכִי־יָרֹק הַזָּב בַּטָּהוֹר וְכִבֶּס בְּגָדָיו
25   וְרָחַץ בַּמַּיִם וְטָמֵא עַד־הָעָרֶב׃ וְכָל־הַמֶּרְכָּב אֲשֶׁר
26   יִרְכַּב עָלָיו הַזָּב יִטְמָא׃ וְכָל־הַנֹּגֵעַ בְּכֹל אֲשֶׁר יִהְיֶה
27   תַחְתָּיו יִטְמָא עַד־הָעָרֶב וְהַנּוֹשֵׂא אוֹתָם יְכַבֵּס בְּגָדָיו
28   וְרָחַץ בַּמַּיִם וְטָמֵא עַד־הָעָרֶב׃ וְכֹל אֲשֶׁר יִגַּע־בּוֹ
29   הַזָּב וְיָדָיו לֹא־שָׁטַף בַּמָּיִם וְכִבֶּס בְּגָדָיו וְרָחַץ בַּמַּיִם
30   וְטָמֵא עַד־הָעָרֶב׃ וּכְלִי־חֶרֶשׂ אֲשֶׁר־יִגַּע־בּוֹ הַזָּב
31   יִשָּׁבֵר וְכָל־כְּלִי־עֵץ יִשָּׁטֵף בַּמָּיִם׃ וְכִי־יִטְהַר הַזָּב
32   מִזּוֹבוֹ וְסָפַר לוֹ שִׁבְעַת יָמִים לְטָהֳרָתוֹ וְכִבֶּס בְּגָדָיו
33   וְרָחַץ בְּשָׂרוֹ בְּמַיִם חַיִּים וְטָהֵר׃ וּבַיּוֹם הַשְּׁמִינִי יִקַּח־
34  לוֹ שְׁתֵּי תֹרִים אוֹ שְׁנֵי בְּנֵי יוֹנָה וּבָא  לִפְנֵי יְהוָה
35   אֶל־פֶּתַח אֹהֶל מוֹעֵד וּנְתָנָם אֶל־הַכֹּהֵן׃ וְעָשָׂה אֹתָם
36   הַכֹּהֵן אֶחָד חַטָּאת וְהָאֶחָד עֹלָה וְכִפֶּר עָלָיו הַכֹּהֵן
37   לִפְנֵי יְהוָה מִזּוֹבוֹ׃ [ס]              וְאִישׁ כִּי־
38  תֵצֵא מִמֶּנּוּ שִׁכְבַת־זָרַע וְרָחַץ בַּמַּיִם אֶת־כָּל־בְּשָׂרוֹ
39   וְטָמֵא עַד־הָעָרֶב׃ וְכָל־בֶּגֶד וְכָל־עוֹר אֲשֶׁר־יִהְיֶה
40   עָלָיו שִׁכְבַת־זָרַע וְכֻבַּס בַּמַּיִם וְטָמֵא עַד־הָעָרֶב׃
41   וְאִשָּׁה אֲשֶׁר יִשְׁכַּב אִישׁ אֹתָהּ שִׁכְבַת־זָרַע וְרָחֲצוּ
42   בַמַּיִם וְטָמְאוּ עַד־הָעָרֶב׃ [פ]
