 1  וְזֹאת תּוֹרַת הָאָשָׁם קֹדֶשׁ קָדָשִׁים הוּא׃ בִּמְקוֹם
 2   אֲשֶׁר יִשְׁחֲטוּ אֶת־הָעֹלָה יִשְׁחֲטוּ אֶת־הָאָשָׁם וְאֶת־
 3  דָּמוֹ יִזְרֹק עַל־הַמִּזְבֵּחַ סָבִיב׃ וְאֵת כָּל־חֶלְבּוֹ יַקְרִיב
 4   מִמֶּנּוּ אֵת הָאַלְיָה וְאֶת־הַחֵלֶב הַמְכַסֶּה אֶת־הַקֶּרֶב׃
 5   וְאֵת שְׁתֵּי הַכְּלָיֹת וְאֶת־הַחֵלֶב אֲשֶׁר עֲלֵיהֶן אֲשֶׁר
 6   עַל־הַכְּסָלִים וְאֶת־הַיֹּתֶרֶת עַל־הַכָּבֵד עַל־הַכְּלָיֹת
 7   יְסִירֶנָּה׃ וְהִקְטִיר אֹתָם הַכֹּהֵן הַמִּזְבֵּחָה אִשֶּׁה לַיהוָה
 8   אָשָׁם הוּא׃ כָּל־זָכָר בַּכֹּהֲנִים יֹאכְלֶנּוּ בְּמָקוֹם קָדוֹשׁ
 9   יֵאָכֵל קֹדֶשׁ קָדָשִׁים הוּא׃ כַּחַטָּאת כָּאָשָׁם תּוֹרָה
10   אַחַת לָהֶם הַכֹּהֵן אֲשֶׁר יְכַפֶּר־בּוֹ לוֹ יִהְיֶה׃ וְהַכֹּהֵן
11   הַמַּקְרִיב אֶת־עֹלַת אִישׁ עוֹר הָעֹלָה אֲשֶׁר הִקְרִיב
12   לַכֹּהֵן לוֹ יִהְיֶה׃ וְכָל־מִנְחָה אֲשֶׁר תֵּאָפֶה בַּתַּנּוּר וְכָל־
13  נַעֲשָׂה בַמַּרְחֶשֶׁת וְעַל־מַחֲבַת לַכֹּהֵן הַמַּקְרִיב אֹתָהּ
14   לוֹ תִהְיֶה׃ וְכָל־מִנְחָה בְלוּלָה־בַשֶּׁמֶן וַחֲרֵבָה לְכָל־
15  בְּנֵי אַהֲרֹן תִּהְיֶה אִישׁ כְּאָחִיו׃ [פ]
16    וְזֹאת תּוֹרַת זֶבַח הַשְּׁלָמִים אֲשֶׁר יַקְרִיב לַיהוָה׃ אִם
17   עַל־תּוֹדָה יַקְרִיבֶנּוּ וְהִקְרִיב  עַל־זֶבַח הַתּוֹדָה חַלּוֹת
18   מַצּוֹת בְּלוּלֹת בַּשֶּׁמֶן וּרְקִיקֵי מַצּוֹת מְשֻׁחִים בַּשָּׁמֶן
19   וְסֹלֶת מֻרְבֶּכֶת חַלֹּת בְּלוּלֹת בַּשָּׁמֶן׃ עַל־חַלֹּת לֶחֶם
20   חָמֵץ יַקְרִיב קָרְבָּנוֹ עַל־זֶבַח תּוֹדַת שְׁלָמָיו׃ וְהִקְרִיב
21   מִמֶּנּוּ אֶחָד מִכָּל־קָרְבָּן תְּרוּמָה לַיהוָה לַכֹּהֵן הַזֹּרֵק
22   אֶת־דַּם הַשְּׁלָמִים לוֹ יִהְיֶה׃ וּבְשַׂר זֶבַח תּוֹדַת
23   שְׁלָמָיו בְּיוֹם קָרְבָּנוֹ יֵאָכֵל לֹא־יַנִּיחַ מִמֶּנּוּ עַד־בֹּקֶר׃
24   וְאִם־נֶדֶר  אוֹ נְדָבָה זֶבַח קָרְבָּנוֹ בְּיוֹם הַקְרִיבוֹ אֶת־
25  זִבְחוֹ יֵאָכֵל וּמִמָּחֳרָת וְהַנּוֹתָר מִמֶּנּוּ יֵאָכֵל׃ וְהַנּוֹתָר
26   מִבְּשַׂר הַזָּבַח בַּיּוֹם הַשְּׁלִישִׁי בָּאֵשׁ יִשָּׂרֵף׃ וְאִם
27   הֵאָכֹל יֵאָכֵל מִבְּשַׂר־זֶבַח שְׁלָמָיו בַּיּוֹם הַשְּׁלִישִׁי
28   לֹא יֵרָצֶה הַמַּקְרִיב אֹתוֹ לֹא יֵחָשֵׁב לוֹ פִּגּוּל יִהְיֶה
29   וְהַנֶּפֶשׁ הָאֹכֶלֶת מִמֶּנּוּ עֲו‍ֹנָהּ תִּשָּׂא׃ וְהַבָּשָׂר אֲשֶׁר־
30  יִגַּע בְּכָל־טָמֵא לֹא יֵאָכֵל בָּאֵשׁ יִשָּׂרֵף וְהַבָּשָׂר
31   כָּל־טָהוֹר יֹאכַל בָּשָׂר׃ וְהַנֶּפֶשׁ אֲשֶׁר־תֹּאכַל בָּשָׂר
32   מִזֶּבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה וְטֻמְאָתוֹ עָלָיו וְנִכְרְתָה
33   הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ׃ וְנֶפֶשׁ כִּי־תִגַּע בְּכָל־טָמֵא
34   בְּטֻמְאַת אָדָם אוֹ  בִּבְהֵמָה טְמֵאָה אוֹ בְּכָל־שֶׁקֶץ
35   טָמֵא וְאָכַל מִבְּשַׂר־זֶבַח הַשְּׁלָמִים אֲשֶׁר לַיהוָה
36   וְנִכְרְתָה הַנֶּפֶשׁ הַהִוא מֵעַמֶּיהָ׃ וַיְדַבֵּר יְהוָה אֶל־
37  מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל לֵאמֹר כָּל־חֵלֶב
38   שׁוֹר וְכֶשֶׂב וָעֵז לֹא תֹאכֵלוּ׃ וְחֵלֶב נְבֵלָה וְחֵלֶב
39   טְרֵפָה יֵעָשֶׂה לְכָל־מְלָאכָה וְאָכֹל לֹא תֹאכְלֻהוּ׃
40   כִּי כָּל־אֹכֵל חֵלֶב מִן־הַבְּהֵמָה אֲשֶׁר יַקְרִיב מִמֶּנָּה
41   אִשֶּׁה לַיהוָה וְנִכְרְתָה הַנֶּפֶשׁ הָאֹכֶלֶת מֵעַמֶּיהָ׃
42   וְכָל־דָּם לֹא תֹאכְלוּ בְּכֹל מוֹשְׁבֹתֵיכֶם לָעוֹף
