 1  וַיִּרְגְּמוּ אֹתוֹ אָבֶן וּבְנֵי־יִשְׂרָאֵל עָשׂוּ כַּאֲשֶׁר צִוָּה
 2   יְהוָה אֶת־מֹשֶׁה׃ [פ]
 3  וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה בְּהַר סִינַי לֵאמֹר׃ דַּבֵּר אֶל־
 4  בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם כִּי תָבֹאוּ אֶל־הָאָרֶץ
 5   אֲשֶׁר אֲנִי נֹתֵן לָכֶם וְשָׁבְתָה הָאָרֶץ שַׁבָּת לַיהוָה׃
 6   שֵׁשׁ שָׁנִים תִּזְרַע שָׂדֶךָ וְשֵׁשׁ שָׁנִים תִּזְמֹר כַּרְמֶךָ
 7   וְאָסַפְתָּ אֶת־תְּבוּאָתָהּ׃♢ וּבַשָּׁנָה הַשְּׁבִיעִת שַׁבַּת
 8   שַׁבָּתוֹן יִהְיֶה לָאָרֶץ שַׁבָּת לַיהוָה שָׂדְךָ לֹא תִזְרָע
 9   וְכַרְמְךָ לֹא תִזְמֹר׃ אֵת סְפִיחַ קְצִירְךָ לֹא תִקְצוֹר
10   וְאֶת־עִנְּבֵי נְזִירֶךָ לֹא תִבְצֹר שְׁנַת שַׁבָּתוֹן יִהְיֶה
11   לָאָרֶץ׃ וְהָיְתָה שַׁבַּת הָאָרֶץ לָכֶם לְאָכְלָה לְךָ
12   וּלְעַבְדְּךָ וְלַאֲמָתֶךָ וְלִשְׂכִירְךָ וּלְתוֹשָׁבְךָ הַגָּרִים
13   עִמָּךְ׃ וְלִבְהֶמְתְּךָ וְלַחַיָּה אֲשֶׁר בְּאַרְצֶךָ תִּהְיֶה כָל־
14  תְּבוּאָתָהּ לֶאֱכֹל׃♢ [ס]             וְסָפַרְתָּ
15   לְךָ שֶׁבַע שַׁבְּתֹת שָׁנִים שֶׁבַע שָׁנִים שֶׁבַע פְּעָמִים
16   וְהָיוּ לְךָ יְמֵי שֶׁבַע שַׁבְּתֹת הַשָּׁנִים תֵּשַׁע וְאַרְבָּעִים
17   שָׁנָה׃ וְהַעֲבַרְתָּ שׁוֹפַר תְּרוּעָה בַּחֹדֶשׁ הַשְּׁבִעִי
18   בֶּעָשׂוֹר לַחֹדֶשׁ בְּיוֹם הַכִּפֻּרִים תַּעֲבִירוּ שׁוֹפָר
19   בְּכָל־אַרְצְכֶם׃ וְקִדַּשְׁתֶּם אֵת שְׁנַת הַחֲמִשִּׁים
20   שָׁנָה וּקְרָאתֶם דְּרוֹר בָּאָרֶץ לְכָל־יֹשְׁבֶיהָ יוֹבֵל
21   הִוא תִּהְיֶה לָכֶם וְשַׁבְתֶּם אִישׁ אֶל־אֲחֻזָּתוֹ וְאִישׁ
22   אֶל־מִשְׁפַּחְתּוֹ תָּשֻׁבוּ׃ יוֹבֵל הִוא שְׁנַת הַחֲמִשִּׁים
23   שָׁנָה תִּהְיֶה לָכֶם לֹא תִזְרָעוּ וְלֹא תִקְצְרוּ אֶת־
24  סְפִיחֶיהָ וְלֹא תִבְצְרוּ אֶת־נְזִרֶיהָ׃ כִּי יוֹבֵל הִוא
25   קֹדֶשׁ תִּהְיֶה לָכֶם מִן־הַשָּׂדֶה תֹּאכְלוּ אֶת־תְּבוּאָתָהּ׃
26   בִּשְׁנַת הַיּוֹבֵל הַזֹּאת תָּשֻׁבוּ אִישׁ אֶל־אֲחֻזָּתוֹ׃♢  וְכִי־
27  תִמְכְּרוּ מִמְכָּר לַעֲמִיתֶךָ אוֹ קָנֹה מִיַּד עֲמִיתֶךָ אַל־
28  תּוֹנוּ אִישׁ אֶת־אָחִיו׃ בְּמִסְפַּר שָׁנִים אַחַר הַיּוֹבֵל
29   תִּקְנֶה מֵאֵת עֲמִיתֶךָ בְּמִסְפַּר שְׁנֵי־תְבוּאֹת יִמְכָּר־
30  לָךְ׃ לְפִי  רֹב הַשָּׁנִים תַּרְבֶּה מִקְנָתוֹ וּלְפִי מְעֹט
31   הַשָּׁנִים תַּמְעִיט מִקְנָתוֹ כִּי מִסְפַּר תְּבוּאֹת הוּא מֹכֵר
32   לָךְ׃ וְלֹא תוֹנוּ אִישׁ אֶת־עֲמִיתוֹ וְיָרֵאתָ מֵאֱלֹהֶיךָ
33   כִּי אֲנִי יְהוָה אֱלֹהֵיכֶם׃ וַעֲשִׂיתֶם אֶת־חֻקֹּתַי וְאֶת־
34  מִשְׁפָּטַי תִּשְׁמְרוּ וַעֲשִׂיתֶם אֹתָם וִישַׁבְתֶּם עַל־
35  הָאָרֶץ לָבֶטַח׃  וְנָתְנָה הָאָרֶץ פִּרְיָהּ וַאֲכַלְתֶּם לָשֹׂבַע
36   וִישַׁבְתֶּם לָבֶטַח עָלֶיהָ׃ וְכִי תֹאמְרוּ מַה־נֹּאכַל
37   בַּשָּׁנָה הַשְּׁבִיעִת הֵן לֹא נִזְרָע וְלֹא נֶאֱסֹף אֶת־
38  תְּבוּאָתֵנוּ׃ וְצִוִּיתִי אֶת־בִּרְכָתִי לָכֶם בַּשָּׁנָה הַשִּׁשִּׁית
39   וְעָשָׂת אֶת־הַתְּבוּאָה לִשְׁלֹשׁ הַשָּׁנִים׃ וּזְרַעְתֶּם אֵת
40   הַשָּׁנָה הַשְּׁמִינִת וַאֲכַלְתֶּם מִן־הַתְּבוּאָה יָשָׁן עַד
41   הַשָּׁנָה הַתְּשִׁיעִת עַד־בּוֹא תְּבוּאָתָהּ תֹּאכְלוּ יָשָׁן׃
42   וְהָאָרֶץ לֹא תִמָּכֵר לִצְמִתֻת כִּי־לִי הָאָרֶץ כִּי־גֵרִים
