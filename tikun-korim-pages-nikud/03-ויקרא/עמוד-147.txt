 1  וּבְיַעַן בְּמִשְׁפָּטַי מָאָסוּ וְאֶת־חֻקֹּתַי גָּעֲלָה נַפְשָׁם׃
 2   וְאַף־גַּם־זֹאת בִּהְיוֹתָם בְּאֶרֶץ אֹיְבֵיהֶם לֹא־מְאַסְתִּים
 3   וְלֹא־גְעַלְתִּים לְכַלֹּתָם לְהָפֵר בְּרִיתִי אִתָּם כִּי אֲנִי
 4   יְהוָה אֱלֹהֵיהֶם׃ וְזָכַרְתִּי לָהֶם בְּרִית רִאשֹׁנִים אֲשֶׁר
 5   הוֹצֵאתִי־אֹתָם מֵאֶרֶץ מִצְרַיִם לְעֵינֵי הַגּוֹיִם לִהְיוֹת
 6   לָהֶם לֵאלֹהִים אֲנִי יְהוָה׃ אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים
 7   וְהַתּוֹרֹת אֲשֶׁר נָתַן יְהוָה בֵּינוֹ וּבֵין בְּנֵי יִשְׂרָאֵל
 8   בְּהַר סִינַי בְּיַד־מֹשֶׁה׃ [פ]
 9    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל
10   וְאָמַרְתָּ אֲלֵהֶם אִישׁ כִּי יַפְלִא נֶדֶר בְּעֶרְכְּךָ נְפָשֹׁת
11   לַיהוָה׃ וְהָיָה עֶרְכְּךָ הַזָּכָר מִבֶּן עֶשְׂרִים שָׁנָה וְעַד
12   בֶּן־שִׁשִּׁים שָׁנָה וְהָיָה עֶרְכְּךָ חֲמִשִּׁים שֶׁקֶל כֶּסֶף
13   בְּשֶׁקֶל הַקֹּדֶשׁ׃ וְאִם־נְקֵבָה הִוא וְהָיָה עֶרְכְּךָ שְׁלֹשִׁים
14   שָׁקֶל׃ וְאִם מִבֶּן־חָמֵשׁ שָׁנִים וְעַד בֶּן־עֶשְׂרִים שָׁנָה
15   וְהָיָה עֶרְכְּךָ הַזָּכָר עֶשְׂרִים שְׁקָלִים וְלַנְּקֵבָה עֲשֶׂרֶת
16   שְׁקָלִים׃ וְאִם מִבֶּן־חֹדֶשׁ וְעַד בֶּן־חָמֵשׁ שָׁנִים וְהָיָה
17   עֶרְכְּךָ הַזָּכָר חֲמִשָּׁה שְׁקָלִים כָּסֶף וְלַנְּקֵבָה עֶרְכְּךָ
18   שְׁלֹשֶׁת שְׁקָלִים כָּסֶף׃ וְאִם מִבֶּן־שִׁשִּׁים שָׁנָה וָמַעְלָה
19   אִם־זָכָר וְהָיָה עֶרְכְּךָ חֲמִשָּׁה עָשָׂר שָׁקֶל וְלַנְּקֵבָה
20   עֲשָׂרָה שְׁקָלִים׃ וְאִם־מָךְ הוּא מֵעֶרְכֶּךָ וְהֶעֱמִידוֹ
21   לִפְנֵי הַכֹּהֵן וְהֶעֱרִיךְ אֹתוֹ הַכֹּהֵן עַל־פִּי אֲשֶׁר תַּשִּׂיג
22   יַד הַנֹּדֵר יַעֲרִיכֶנּוּ הַכֹּהֵן׃ [ס]             וְאִם־
23  בְּהֵמָה אֲשֶׁר יַקְרִיבוּ מִמֶּנָּה קָרְבָּן לַיהוָה כֹּל אֲשֶׁר
24   יִתֵּן מִמֶּנּוּ לַיהוָה יִהְיֶה־קֹּדֶשׁ׃ לֹא יַחֲלִיפֶנּוּ וְלֹא־יָמִיר
25   אֹתוֹ טוֹב בְּרָע אוֹ־רַע בְּטוֹב וְאִם־הָמֵר יָמִיר בְּהֵמָה
26   בִּבְהֵמָה וְהָיָה־הוּא וּתְמוּרָתוֹ יִהְיֶה־קֹּדֶשׁ׃ וְאִם כָּל־
27  בְּהֵמָה טְמֵאָה אֲשֶׁר לֹא־יַקְרִיבוּ מִמֶּנָּה קָרְבָּן לַיהוָה
28   וְהֶעֱמִיד אֶת־הַבְּהֵמָה לִפְנֵי הַכֹּהֵן׃ וְהֶעֱרִיךְ הַכֹּהֵן
29   אֹתָהּ בֵּין טוֹב וּבֵין רָע כְּעֶרְכְּךָ הַכֹּהֵן כֵּן יִהְיֶה׃ וְאִם־
30  גָּאֹל יִגְאָלֶנָּה וְיָסַף חֲמִישִׁתוֹ עַל־עֶרְכֶּךָ׃ וְאִישׁ כִּי־
31  יַקְדִּשׁ אֶת־בֵּיתוֹ קֹדֶשׁ לַיהוָה וְהֶעֱרִיכוֹ הַכֹּהֵן בֵּין
32   טוֹב וּבֵין רָע כַּאֲשֶׁר יַעֲרִיךְ אֹתוֹ הַכֹּהֵן כֵּן יָקוּם׃
33   וְאִם־הַמַּקְדִּישׁ יִגְאַל אֶת־בֵּיתוֹ וְיָסַף חֲמִישִׁית כֶּסֶף־
34  עֶרְכְּךָ עָלָיו וְהָיָה לוֹ׃  וְאִם  מִשְּׂדֵה אֲחֻזָּתוֹ יַקְדִּישׁ
35   אִישׁ לַיהוָה וְהָיָה עֶרְכְּךָ לְפִי זַרְעוֹ זֶרַע חֹמֶר
36   שְׂעֹרִים בַּחֲמִשִּׁים שֶׁקֶל כָּסֶף׃ אִם־מִשְּׁנַת הַיֹּבֵל
37   יַקְדִּישׁ שָׂדֵהוּ כְּעֶרְכְּךָ יָקוּם׃ וְאִם־אַחַר הַיֹּבֵל
38   יַקְדִּישׁ שָׂדֵהוּ וְחִשַּׁב־לוֹ הַכֹּהֵן אֶת־הַכֶּסֶף עַל־פִּי
39   הַשָּׁנִים הַנּוֹתָרֹת עַד שְׁנַת הַיֹּבֵל וְנִגְרַע מֵעֶרְכֶּךָ׃ וְאִם־
40  גָּאֹל יִגְאַל אֶת־הַשָּׂדֶה הַמַּקְדִּישׁ אֹתוֹ וְיָסַף חֲמִשִׁית
41   כֶּסֶף־עֶרְכְּךָ עָלָיו וְקָם לוֹ׃ וְאִם־לֹא יִגְאַל אֶת־הַשָּׂדֶה
42   וְאִם־מָכַר אֶת־הַשָּׂדֶה לְאִישׁ אַחֵר לֹא יִגָּאֵל עוֹד׃
