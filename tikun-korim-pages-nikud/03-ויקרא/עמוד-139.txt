 1  וְהוּא צָרוּעַ אוֹ זָב בַּקֳּדָשִׁים לֹא יֹאכַל עַד אֲשֶׁר
 2   יִטְהָר וְהַנֹּגֵעַ בְּכָל־טְמֵא־נֶפֶשׁ אוֹ אִישׁ אֲשֶׁר־תֵּצֵא
 3   מִמֶּנּוּ שִׁכְבַת־זָרַע׃ אוֹ־אִישׁ אֲשֶׁר יִגַּע בְּכָל־שֶׁרֶץ
 4   אֲשֶׁר יִטְמָא־לוֹ אוֹ בְאָדָם אֲשֶׁר יִטְמָא־לוֹ לְכֹל
 5   טֻמְאָתוֹ׃ נֶפֶשׁ אֲשֶׁר תִּגַּע־בּוֹ וְטָמְאָה עַד־הָעָרֶב וְלֹא
 6   יֹאכַל מִן־הַקֳּדָשִׁים כִּי אִם־רָחַץ בְּשָׂרוֹ בַּמָּיִם׃ וּבָא
 7   הַשֶּׁמֶשׁ וְטָהֵר וְאַחַר יֹאכַל מִן־הַקֳּדָשִׁים כִּי לַחְמוֹ
 8   הוּא׃ נְבֵלָה וּטְרֵפָה לֹא יֹאכַל לְטָמְאָה־בָהּ אֲנִי יְהוָה׃
 9   וְשָׁמְרוּ אֶת־מִשְׁמַרְתִּי וְלֹא־יִשְׂאוּ עָלָיו חֵטְא וּמֵתוּ
10   בוֹ כִּי יְחַלְּלֻהוּ אֲנִי יְהוָה מְקַדְּשָׁם׃ וְכָל־זָר לֹא־יֹאכַל
11   קֹדֶשׁ תּוֹשַׁב כֹּהֵן וְשָׂכִיר לֹא־יֹאכַל קֹדֶשׁ׃ וְכֹהֵן
12   כִּי־יִקְנֶה נֶפֶשׁ קִנְיַן כַּסְפּוֹ הוּא יֹאכַל בּוֹ וִילִיד בֵּיתוֹ
13   הֵם יֹאכְלוּ בְלַחְמוֹ׃ וּבַת־כֹּהֵן כִּי תִהְיֶה לְאִישׁ זָר
14   הִוא בִּתְרוּמַת הַקֳּדָשִׁים לֹא תֹאכֵל׃ וּבַת־כֹּהֵן כִּי
15   תִהְיֶה אַלְמָנָה וּגְרוּשָׁה וְזֶרַע אֵין לָהּ וְשָׁבָה אֶל־
16  בֵּית אָבִיהָ כִּנְעוּרֶיהָ מִלֶּחֶם אָבִיהָ תֹּאכֵל וְכָל־זָר
17   לֹא־יֹאכַל בּוֹ׃ וְאִישׁ כִּי־יֹאכַל קֹדֶשׁ בִּשְׁגָגָה וְיָסַף
18   חֲמִשִׁיתוֹ עָלָיו וְנָתַן לַכֹּהֵן אֶת־הַקֹּדֶשׁ׃ וְלֹא יְחַלְּלוּ
19   אֶת־קָדְשֵׁי בְּנֵי יִשְׂרָאֵל אֵת אֲשֶׁר־יָרִימוּ לַיהוָה׃
20   וְהִשִּׂיאוּ אוֹתָם עֲו‍ֹן אַשְׁמָה בְּאָכְלָם אֶת־קָדְשֵׁיהֶם
21   כִּי אֲנִי יְהוָה מְקַדְּשָׁם׃ [פ]
22    וַיְדַבֵּר יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ דַּבֵּר אֶל־אַהֲרֹן וְאֶל־
23  בָּנָיו וְאֶל כָּל־בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם אִישׁ
24   אִישׁ מִבֵּית יִשְׂרָאֵל וּמִן־הַגֵּר בְּיִשְׂרָאֵל אֲשֶׁר
25   יַקְרִיב קָרְבָּנוֹ לְכָל־נִדְרֵיהֶם וּלְכָל־נִדְבוֹתָם אֲשֶׁר־
26  יַקְרִיבוּ לַיהוָה לְעֹלָה׃ לִרְצֹנְכֶם תָּמִים זָכָר בַּבָּקָר
27   בַּכְּשָׂבִים וּבָעִזִּים׃ כֹּל אֲשֶׁר־בּוֹ מוּם לֹא תַקְרִיבוּ
28   כִּי־לֹא לְרָצוֹן יִהְיֶה לָכֶם׃ וְאִישׁ כִּי־יַקְרִיב זֶבַח־
29  שְׁלָמִים לַיהוָה לְפַלֵּא־נֶדֶר אוֹ לִנְדָבָה בַּבָּקָר
30   אוֹ בַצֹּאן תָּמִים יִהְיֶה לְרָצוֹן כָּל־מוּם לֹא יִהְיֶה־
31  בּוֹ׃ עַוֶּרֶת אוֹ שָׁבוּר אוֹ־חָרוּץ אוֹ־יַבֶּלֶת אוֹ גָרָב
32   אוֹ יַלֶּפֶת לֹא־תַקְרִיבוּ אֵלֶּה לַיהוָה וְאִשֶּׁה לֹא־
33  תִתְּנוּ מֵהֶם עַל־הַמִּזְבֵּחַ לַיהוָה׃ וְשׁוֹר וָשֶׂה שָׂרוּעַ
34   וְקָלוּט נְדָבָה תַּעֲשֶׂה אֹתוֹ וּלְנֵדֶר לֹא יֵרָצֶה׃ וּמָעוּךְ
35   וְכָתוּת וְנָתוּק וְכָרוּת לֹא תַקְרִיבוּ לַיהוָה וּבְאַרְצְכֶם
36   לֹא תַעֲשׂוּ׃ וּמִיַּד בֶּן־נֵכָר לֹא תַקְרִיבוּ אֶת־לֶחֶם
37   אֱלֹהֵיכֶם מִכָּל־אֵלֶּה כִּי מָשְׁחָתָם בָּהֶם מוּם בָּם
38   לֹא יֵרָצוּ לָכֶם׃ [ס]             וַיְדַבֵּר
39   יְהוָה אֶל־מֹשֶׁה לֵּאמֹר׃ שׁוֹר אוֹ־כֶשֶׂב אוֹ־עֵז
40   כִּי יִוָּלֵד וְהָיָה שִׁבְעַת יָמִים תַּחַת אִמּוֹ וּמִיּוֹם
41   הַשְּׁמִינִי וָהָלְאָה יֵרָצֶה לְקָרְבַּן אִשֶּׁה לַיהוָה׃ וְשׁוֹר
42   אוֹ־שֶׂה אֹתוֹ וְאֶת־בְּנוֹ לֹא תִשְׁחֲטוּ בְּיוֹם אֶחָד׃
